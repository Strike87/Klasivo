   307→  /// Handle notification tap navigation using GoRouter.

   312→
   313→    switch (relatedType) {
   314→      case 'conversation':
   315→        if (relatedId.isNotEmpty) {
   316→          router.go('/inbox/messages/$relatedId');
   317→        } else {
   318→          router.go('/inbox/messages');
   319→        }

   322→        if (relatedId.isNotEmpty) {
   323→          router.go('/teacher/exams/$relatedId');

   325→        break;
   326→      case 'assignment':
   327→        if (relatedId.isNotEmpty) {
   328→          router.go('/teacher/assignments/$relatedId');

   338→          router.go('/inbox/announcements');
   339→        }
   340→        break;

   343→    }

   413→class AuthChangeNotifier extends ChangeNotifier {
   414→  StreamSubscription<User?>? _firebaseSub;
   415→  StreamSubscription? _hiveSub;
   416→
   417→  AuthChangeNotifier() {
   418→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   419→      (_) => notifyListeners(),
   420→      onError: (_) => notifyListeners(),
   421→    );
   422→    _startHiveWatch();
   423→  }
   424→
   425→  void _startHiveWatch() {
   426→    // Use Hive's built-in watch() instead of polling every 500ms
   427→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   428→      notifyListeners();
   429→    });
   430→  }
   431→
   432→  @override
   433→  void dispose() {
   434→    _firebaseSub?.cancel();
   435→    _hiveSub?.cancel();
   436→    super.dispose();
   437→  }
   438→}
   439→
   440→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   441→  final notifier = AuthChangeNotifier();
   442→  ref.onDispose(() => notifier.dispose());
   443→  return notifier;
   444→});
   445→
   446→// ─── GoRouter with Auth Guards & v1.7 Complete Navigation ─────────────────────
   447→
   448→final routerProvider = Provider<GoRouter>((ref) {
   449→  final notifier = ref.watch(authChangeNotifierProvider);
   450→
   451→  return GoRouter(
   452→    initialLocation: '/',
   453→    debugLogDiagnostics: true,
   454→    observers: [FirebaseAnalyticsService.observer],
   455→    refreshListenable: notifier,
   456→    redirect: (context, state) {
   457→      final box = Hive.box(AppConstants.authBox);
   458→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   459→      final userRole = box.get('userRole', defaultValue: '') as String;
   460→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: true) as bool;
   461→      final mustChangePassword = box.get('mustChangePassword', defaultValue: false) as bool;
   462→      final isOnChangePassword = state.matchedLocation == '/change-password';
   463→
   464→      // Force redirect to change password if required
   465→      if (isLoggedIn && mustChangePassword && !isOnChangePassword) {
   466→        return '/change-password';
   467→      }
   468→      // Prevent navigating away from change password if forced
   469→      if (isLoggedIn && mustChangePassword && isOnChangePassword) {
   470→        return null; // Allow staying on change password
   471→      }
   472→      // If no longer required, redirect away from change password
   473→      if (isLoggedIn && !mustChangePassword && isOnChangePassword) {
   474→        // Management roles → /dashboard; student → /student; parent → /parent
   475→        if (KlasivoRole.managementRoles.contains(userRole)) return '/dashboard';
   476→        if (userRole == KlasivoRole.student) return '/student';
   477→        if (userRole == KlasivoRole.parent) return '/parent';
   478→        return '/dashboard'; // Fallback for unknown roles
   479→      }
   480→
   481→      final isOnSplash = state.matchedLocation == '/';
   482→      final isOnAuth = state.matchedLocation.startsWith('/auth');
   483→      final isOnWelcome = state.matchedLocation == '/welcome';
   484→      final isOnDashboard = state.matchedLocation.startsWith('/dashboard') ||
   485→          state.matchedLocation.startsWith('/academic') ||
   486→          state.matchedLocation.startsWith('/people') ||
   487→          state.matchedLocation.startsWith('/inbox') ||
   488→          state.matchedLocation.startsWith('/settings') ||
   489→          state.matchedLocation.startsWith('/teacher') ||
   490→          state.matchedLocation.startsWith('/lms');
   491→      final isOnStudent = state.matchedLocation.startsWith('/student');
   492→      final isOnParent = state.matchedLocation.startsWith('/parent') ||
   493→          state.matchedLocation.startsWith('/auth/parent');
   494→
   495→      // Splash → redirect based on auth state
   496→      if (isOnSplash) {
   497→        if (isLoggedIn && userRole.isNotEmpty) {
   498→          if (userRole == KlasivoRole.owner && !hasCompletedSetup) {
   499→            return '/welcome';
   500→          }
   501→          if (KlasivoRole.managementRoles.contains(userRole)) {
   502→            return '/dashboard';
   503→          }
   504→          if (userRole == KlasivoRole.student) return '/student';
   505→          if (userRole == KlasivoRole.parent) return '/parent';
   506→        }
   507→        return '/auth';
   508→      }
   509→
   510→      // Welcome screen — only accessible to logged-in owners who haven't completed setup
   511→      if (isOnWelcome) {
   512→        if (!isLoggedIn) return '/auth';
   513→        if (userRole != KlasivoRole.owner) return '/dashboard';
   514→        if (hasCompletedSetup) return '/dashboard';
   515→        return null;
   516→      }
   517→
   518→      // Auth screens → redirect if already logged in
   519→      if (isOnAuth) {
   520→        if (isLoggedIn && userRole.isNotEmpty) {
   521→          if (userRole == KlasivoRole.owner && !hasCompletedSetup) {
   522→            return '/welcome';
   523→          }
   524→          if (KlasivoRole.managementRoles.contains(userRole)) {
   525→            return '/dashboard';
   526→          }
   527→          if (userRole == KlasivoRole.student) return '/student';
   528→          if (userRole == KlasivoRole.parent) return '/parent';
   529→        }
   530→        return null;
   531→      }
   532→
   533→      // Protected screens → require login
