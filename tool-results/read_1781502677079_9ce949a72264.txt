    17→    17→ *   POST /v1/storage/upload-url     → Generate signed upload URL
    18→    18→ *   POST /v1/analytics/event        → Record server-side analytics event
    19→    19→ *   GET  /v1/admin/users            → List users in organization
    20→    20→ *   GET  /v1/admin/schools          → List organizations
    21→    21→ *   GET  /v1/admin/reports/summary  → Organization summary stats
    22→    22→ *   GET  /v1/docs                   → OpenAPI/Swagger JSON
    23→    23→ *
    24→    24→ * Security:
    25→    25→ *   - Bearer token auth (Firebase ID token) on all mutation endpoints

    27→    27→ *   - Rate limiting at Cloudflare level

    80→    80→// ─── Auth middleware ────────────────────────────────────────────
    81→    81→
    82→    82→async function verifyAuthToken(
    83→    83→  req: Request,
    84→    84→  res: Response,
    85→    85→  next: NextFunction,
    86→    86→): Promise<void> {
    87→    87→  const authHeader = req.headers.authorization;
    88→    88→  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    89→    89→    res.status(401).json({
    90→    90→      error: 'Missing or invalid Authorization header. Use: Bearer <Firebase ID token>',
    91→    91→    });
    92→    92→    return;
    93→    93→  }
    94→    94→
    95→    95→  const idToken = authHeader.split('Bearer ')[1];
    96→    96→  if (!idToken) {
    97→    97→    res.status(401).json({ error: 'Empty bearer token.' });
    98→    98→    return;
    99→    99→  }
   100→   100→
   101→   101→  try {
   102→   102→    const decodedToken = await admin.auth().verifyIdToken(idToken);
   103→   103→    req.user = decodedToken;
   104→   104→
   105→   105→    const userDoc = await admin
   106→   106→      .firestore()
   107→   107→      .collection('users')
   108→   108→      .doc(decodedToken.uid)
   109→   109→      .get();
   110→   110→    if (userDoc.exists) {
   111→   111→      req.userRole = userDoc.data()?.['role'] as string | undefined;
   112→   112→      req.userOrgId = userDoc.data()?.['organizationId'] as string | undefined;
   113→   113→      req.userName = userDoc.data()?.['fullName'] as string | undefined;
   114→   114→    }
   115→   115→
   116→   116→    next();
   117→   117→  } catch (err: unknown) {
   118→   118→    const msg = err instanceof Error ? err.message : String(err);
   119→   119→    console.error(`Token verification failed: ${msg}`);
   120→   120→    res.status(401).json({ error: 'Invalid or expired token.' });
   121→   121→  }
   122→   122→}

   126→   126→function requireAdmin(req: Request, res: Response, next: NextFunction): void {
   127→   127→  if (!req.user) {
   128→   128→    res.status(401).json({ error: 'Authentication required.' });
   129→   129→    return;
   130→   130→  }
   131→   131→  if (!['teacher', 'owner', 'admin'].includes(req.userRole ?? '')) {
   132→   132→    res.status(403).json({ error: 'Admin access required.' });
   133→   133→    return;
   134→   134→  }
   135→   135→  next();
   136→   136→}

   138→   138→// ─── Audit logging middleware ───────────────────────────────────
   139→   139→
   140→   140→async function auditLog(
   141→   141→  req: Request,
   142→   142→  _res: Response,
   143→   143→  next: NextFunction,
   144→   144→): Promise<void> {
   145→   145→  // Only audit mutation requests from authenticated users
   146→   146→  if (req.method === 'GET' || !req.user) {
   147→   147→    next();
   148→   148→    return;
   149→   149→  }
   150→   150→
   151→   151→  try {
   152→   152→    const db = admin.firestore();
   153→   153→    await db.collection('audit_logs').add({
   154→   154→      actorId: req.user.uid,                                             // legacy — remove in Phase 3
   155→   155→      actorName: req.userName ?? req.user.email ?? req.user.uid,         // legacy — remove in Phase 3
   156→   156→      actorRole: req.userRole ?? 'unknown',                              // legacy — keep for backward compat
   157→   157→      performedBy: req.user.uid,                                         // canonical actor field
   158→   158→      performedByRole: req.userRole ?? 'unknown',                        // Phase 1: add
   159→   159→      performedByOrgId: req.userOrgId ?? null,                           // Phase 1: add
   160→   160→      organizationId: req.userOrgId ?? null,
   161→   161→      action: `${req.method} ${req.path}`,
   162→   162→      resource: req.path,
   163→   163→      requestBody: _sanitizeForAudit(req.body),
   164→   164→      ipAddress: req.ip ?? req.headers['x-forwarded-for'] ?? 'unknown',
   165→   165→      userAgent: req.headers['user-agent'] ?? 'unknown',
   166→   166→      timestamp: admin.firestore.FieldValue.serverTimestamp(),
   167→   167→    });
   168→   168→  } catch (err: unknown) {
   169→   169→    // Audit logging failure should never block the request
   170→   170→    console.warn(`Audit log write failed: ${err instanceof Error ? err.message : String(err)}`);
   171→   171→  }
   172→   172→
   173→   173→  next();
   174→   174→}

   176→   176→/** Remove sensitive fields from request body before logging */
   177→   177→function _sanitizeForAudit(body: Record<string, unknown> | undefined): Record<string, unknown> {
   178→   178→  if (!body || typeof body !== 'object') return {};
   179→   179→  const sanitized = { ...body };
   180→   180→  const sensitiveKeys = ['password', 'token', 'secret', 'apiKey', 'creditCard', 'cvv'];
   181→   181→  for (const key of Object.keys(sanitized)) {
   182→   182→    if (sensitiveKeys.some((sk) => key.toLowerCase().includes(sk.toLowerCase()))) {
   183→   183→      sanitized[key] = '[REDACTED]';
   184→   184→    }
   185→   185→  }
   186→   186→  return sanitized;
   187→   187→}

   269→   269→app.post(
   270→   270→  '/v1/livekit/token',
   271→   271→  verifyAuthToken,
   272→   272→  async (req: Request, res: Response) => {
   273→   273→    initSentry();
   274→   274→    const scope = (req as any).sentryScope as Sentry.Scope | undefined;
   275→   275→    scope?.setTag('service', 'livekit');
   276→   276→    scope?.setTag('endpoint', '/v1/livekit/token');
   277→   277→
   278→   278→    if (!req.user) {
   279→   279→      res.status(401).json({ error: 'Authentication required.' });
   280→   280→      return;
   281→   281→    }
   282→   282→
   283→   283→    const { roomId, displayName } = req.body ?? {};
   284→   284→    // roomName/isTeacher: REMOVED — derived from room document and caller claims
   285→   285→
   286→   286→    if (!roomId || typeof roomId !== 'string') {
   287→   287→      res.status(400).json({ error: 'roomId is required and must be a string.' });
   288→   288→      return;
   289→   289→    }
   290→   290→
   291→   291→    try {
   292→   292→      // ── Load room document ──────────────────────────────────
   293→   293→      const roomDoc = await admin.firestore().collection('livekit_rooms').doc(roomId).get();
   294→   294→
   295→   295→      if (!roomDoc.exists) {
   296→   296→        res.status(404).json({ error: 'Room not found.' });
   297→   297→        return;
   298→   298→      }
   299→   299→
   300→   300→      const roomData = roomDoc.data()!;
   301→   301→      const roomName = roomData['name'] as string;
   302→   302→      if (!roomName) {
   303→   303→        res.status(500).json({ error: 'Room document is missing a name field.' });
   304→   304→        return;
   305→   305→      }
   306→   306→
   307→   307→      // Room name sanitization
   308→   308→      const sanitizedRoomName = roomName.replace(/[^a-zA-Z0-9_-]/g, '');
   309→   309→      if (sanitizedRoomName.length === 0 || sanitizedRoomName.length > 128) {
   310→   310→        res.status(500).json({ error: 'Invalid room name in document.' });
   311→   311→        return;
   312→   312→      }
   313→   313→
   314→   314→      // ── Org boundary check ──────────────────────────────────
   315→   315→      const roomOrgId = roomData['organizationId'] as string;
   316→   316→      if (!verifyOrgBoundary(req.userOrgId ?? '', roomOrgId, req.userRole ?? '')) {
   317→   317→        await _logTokenDenied(req.user!.uid, req.userRole ?? 'unknown', req.userOrgId ?? '', roomId, 'org_boundary', 'Cross-org access denied');
   318→   318→        Sentry.captureMessage(`Cross-org LiveKit token attempt (API): caller=${req.userOrgId} room=${roomOrgId}`);
   319→   319→        res.status(403).json({ error: 'You can only join rooms in your organization.' });
   320→   320→        return;
   321→   321→      }
   322→   322→
   323→   323→      // ── Scope authorization ────────────────────────────────
   324→   324→      const scopeAccessLevel = (req.user?.token?.['scopeAccessLevel'] as string) || '';
   325→   325→      const callerScope: Record<string, string[]> = {};
   326→   326→
   327→   327→      // Load caller's scope arrays from user doc (already loaded by verifyAuthToken middleware)
   328→   328→      const callerDoc = await admin.firestore().collection('users').doc(req.user!.uid).get();
   329→   329→      if (callerDoc.exists) {
   330→   330→        const callerData = callerDoc.data()!;
   331→   331→        callerScope['campusIds'] = callerData['campusIds'] as string[] || [];
   332→   332→        callerScope['stageIds'] = callerData['stageIds'] as string[] || [];
   333→   333→        callerScope['classIds'] = callerData['classIds'] as string[] || [];
   334→   334→        callerScope['subjectIds'] = callerData['subjectIds'] as string[] || [];
   335→   335→        callerScope['studentIds'] = callerData['studentIds'] as string[] || [];
   336→   336→      }
   337→   337→
   338→   338→      const scopeResult = verifyScopeAuthorization(scopeAccessLevel, callerScope, roomData as Record<string, unknown>);
   339→   339→      if (!scopeResult.authorized) {
   340→   340→        await _logTokenDenied(req.user!.uid, req.userRole ?? 'unknown', req.userOrgId ?? '', roomId, scopeResult.reason ?? 'unknown', scopeResult.message ?? 'Scope authorization failed');
   341→   341→        Sentry.captureMessage(`LiveKit scope denial (API): uid=${req.user!.uid} room=${roomId} reason=${scopeResult.reason}`);
   342→   342→        res.status(403).json({ error: scopeResult.message || 'You are not authorized to access this room.' });
   343→   343→        return;
   344→   344→      }
   345→   345→
   346→   346→      // ── Server-side role determination ──────────────────────
   347→   347→      const isTeacherOrAbove = LIVEKIT_ADMIN_ROLES.includes(req.userRole as any);
   348→   348→
   349→   349→      const { AccessToken } = await import('livekit-server-sdk');
   350→   350→
   351→   351→      const token = new AccessToken(
   352→   352→        LIVEKIT_API_KEY.value(),
   353→   353→        LIVEKIT_API_SECRET.value(),
   354→   354→        { identity: req.user.uid, name: displayName || req.user.uid },
   355→   355→      );
   356→   356→
   357→   357→      token.addGrant({
   358→   358→        room: sanitizedRoomName,
   359→   359→        roomJoin: true,
   360→   360→        canPublish: true,
   361→   361→        canSubscribe: true,
   362→   362→        canPublishData: true,
   363→   363→        roomAdmin: isTeacherOrAbove,
   364→   364→      });
   365→   365→
   366→   366→      const jwt = await token.toJwt();
   367→   367→
   368→   368→      Sentry.addBreadcrumb({
   369→   369→        category: 'livekit',
   370→   370→        message: `Token generated for room ${sanitizedRoomName}, roomAdmin=${isTeacherOrAbove}`,
   371→   371→        level: 'info',
   372→   372→      });
   373→   373→
   374→   374→      res.json({ token: jwt });
   375→   375→    } catch (err: unknown) {
   376→   376→      const msg = err instanceof Error ? err.message : String(err);
   377→   377→      console.error(`LiveKit token generation failed: ${msg}`);
   378→   378→      Sentry.captureException(err);
   379→   379→      res.status(500).json({ error: 'Failed to generate token.' });
   380→   380→    }
   381→   381→  },

   597→   597→ * POST /v1/storage/upload-url
   598→   598→ *
   599→   599→ * Generates a signed URL for the client to upload a file directly to
   600→   600→ * Firebase Storage. The client never needs storage credentials.
   601→   601→ *
   602→   602→ * Request body:
   603→   603→ *   { "filePath": "organizations/abc123/logo.png", "contentType": "image/png" }
   604→   604→ *
   605→   605→ * Response:
   606→   606→ *   { "uploadUrl": "https://storage.googleapis.com/...", "expiresAt": "..." }
   607→   607→ */
   608→   608→app.post(
   609→   609→  '/v1/storage/upload-url',
   610→   610→  verifyAuthToken,
   611→   611→  async (req: Request, res: Response) => {
   612→   612→    initSentry();
   613→   613→    const scope = (req as any).sentryScope as Sentry.Scope | undefined;
   614→   614→    scope?.setTag('service', 'storage');
   615→   615→    scope?.setTag('endpoint', '/v1/storage/upload-url');
   616→   616→
   617→   617→    const { filePath, contentType } = req.body ?? {};
   618→   618→
   619→   619→    if (!filePath || typeof filePath !== 'string') {
   620→   620→      res.status(400).json({ error: 'filePath is required.' });
   621→   621→      return;
   622→   622→    }
   623→   623→
   624→   624→    // Validate path doesn't escape allowed prefixes
   625→   625→    const allowedPrefixes = [
   626→   626→      `users/${req.user!.uid}/`,
   627→   627→      `organizations/${req.userOrgId ?? ''}/`,
   628→   628→      'exams/',
   629→   629→      'materials/',
   630→   630→      'submissions/',
   631→   631→    ];
   632→   632→
   633→   633→    const isAllowed = allowedPrefixes.some((prefix) => filePath.startsWith(prefix));
   634→   634→    if (!isAllowed) {
   635→   635→      res.status(403).json({
   636→   636→        error: 'File path must start with an allowed prefix.',
   637→   637→        allowedPrefixes: ['users/{uid}/', 'organizations/{orgId}/', 'exams/', 'materials/', 'submissions/'],
   638→   638→      });
   639→   639→      return;
   640→   640→    }
   641→   641→
   642→   642→    // Prevent path traversal
   643→   643→    if (filePath.includes('..') || filePath.includes('//')) {
   644→   644→      res.status(400).json({ error: 'Invalid file path.' });
   645→   645→      return;
   646→   646→    }
   647→   647→
   648→   648→    try {
   649→   649→      const bucket = admin.storage().bucket();
   650→   650→      const file = bucket.file(filePath);
   651→   651→
   652→   652→      const expiresInMinutes = 15;
   653→   653→      const expiresAt = new Date(Date.now() + expiresInMinutes * 60 * 1000);
   654→   654→
   655→   655→      const [uploadUrl] = await file.getSignedUrl({
   656→   656→        action: 'write',
   657→   657→        expires: expiresAt,
   658→   658→        contentType: contentType || 'application/octet-stream',
   659→   659→      });
   660→   660→
   661→   661→      // Also generate a read URL
   662→   662→      const [downloadUrl] = await file.getSignedUrl({
   663→   663→        action: 'read',
   664→   664→        expires: expiresAt,
   665→   665→      });
   666→   666→
   667→   667→      res.json({
   668→   668→        uploadUrl,
   669→   669→        downloadUrl,
   670→   670→        filePath,
   671→   671→        expiresAt: expiresAt.toISOString(),
   672→   672→      });
   673→   673→    } catch (err: unknown) {
   674→   674→      const msg = err instanceof Error ? err.message : String(err);
   675→   675→      console.error(`Signed URL generation failed: ${msg}`);
   676→   676→      Sentry.captureException(err);
   677→   677→      res.status(500).json({ error: 'Failed to generate upload URL.' });
   678→   678→    }
   679→   679→  },
