   390→   390→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   391→   391→      notifyListeners();
   392→   392→    });
   393→   393→  }

   395→   395→  @override
   396→   396→  void dispose() {
   397→   397→    _firebaseSub?.cancel();
   398→   398→    _hiveSub?.cancel();
   399→   399→    super.dispose();
   400→   400→  }
   401→   401→}

   411→   411→/// Custom [NavigatorObserver] that logs route changes as Sentry breadcrumbs
   412→   412→/// for full navigation traceability in error reports.
   413→   413→class SentryNavigationObserver extends NavigatorObserver {
   414→   414→  @override
   415→   415→  void didPush(Route<dynamic> route, Route<dynamic>? previousRoute) {
   416→   416→    _logNavigation('push', route, previousRoute);
   417→   417→  }
   418→   418→
   419→   419→  @override
   420→   420→  void didPop(Route<dynamic> route, Route<dynamic>? previousRoute) {
   421→   421→    _logNavigation('pop', route, previousRoute);
   422→   422→  }
   423→   423→
   424→   424→  @override
   425→   425→  void didReplace({Route<dynamic>? newRoute, Route<dynamic>? oldRoute}) {
   426→   426→    if (newRoute != null) {
   427→   427→      KlasivoSentry.breadcrumb.navigation(
   428→   428→        newRoute.settings.name ?? newRoute.runtimeType.toString(),
   429→   429→        action: 'replace',
   430→   430→        data: {
   431→   431→          if (oldRoute?.settings.name != null)
   432→   432→            'from': oldRoute!.settings.name!,
   433→   433→        },
   434→   434→      );
   435→   435→    }
   436→   436→  }
   437→   437→
   438→   438→  void _logNavigation(
   439→   439→    String action,
   440→   440→    Route<dynamic> route,
   441→   441→    Route<dynamic>? previousRoute,
   442→   442→  ) {
   443→   443→    final name = route.settings.name ?? route.runtimeType.toString();
   444→   444→    KlasivoSentry.breadcrumb.navigation(name, action: action, data: {
   445→   445→      if (previousRoute?.settings.name != null)
   446→   446→        'from': previousRoute!.settings.name!,
   447→   447→    });
   448→   448→  }
