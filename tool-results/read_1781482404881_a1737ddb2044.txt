    26→  /// Hash a password using SHA-256
    27→  static String hashPassword(String password) {
    28→    final bytes = utf8.encode(password);
    29→    final digest = sha256.convert(bytes);
    30→    return digest.toString();
    31→  }

    95→          step: 'STEP_2_USER_DOC_CREATE',

    98→        // Read-back verification: confirm the document actually exists in Firestore.

   103→            .collection(AppConstants.usersCollection)
   104→            .doc(user.uid)
   105→            .get();

   108→            'STEP_2 USER DOC SET SUCCEEDED BUT READ-BACK FAILED — doc users/${user.uid} does not exist after .set()',

   132→      } finally {
   133→        await createUserDocSpan.finish();
   134→      }
   135→
   136→      Sentry.addBreadcrumb(Breadcrumb(
   137→        category: 'registration',
   138→        message: 'STEP_2_USER_DOC_CREATE_SUCCESS',
   139→        data: {'uid': user.uid},
   140→      ));

   162→          stackTrace: st,

   166→            scope.setTag('ownerId', user.uid);
   167→            scope.setTag('flow', 'owner_registration');
   168→            scope.setTag('step', 'STEP_3_ORG_CREATE');
   169→          },
   170→        );
   171→        rethrow;
   172→      } finally {
   173→        await createOrgSpan.finish();

   176→      Sentry.addBreadcrumb(Breadcrumb(
   177→        category: 'registration',
   178→        message: 'STEP_3_ORG_CREATE_SUCCESS',
   179→        data: {'orgId': orgId},
   180→      ));
   181→
   182→      // ── Step 4: Patch user doc with real organizationId ────────────────

   322→  /// Unified email+password login for owners, teachers, and parents.
   323→  Future<Map<String, dynamic>> loginWithEmail({
   324→    required String email,
   325→    required String password,
   326→  }) async {
   327→    final transaction = KlasivoSentry.transactions.loginFlow('email');
   328→
   329→    try {
   330→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   331→        'method': 'email',
   332→        'email': email,
   333→      });
   334→
   335→      final userCredential =
   336→          await FirebaseService.loginWithEmail(email, password);
   337→      final user = userCredential.user!;
   338→
   339→      KlasivoSentry.breadcrumb.auth('auth_user_authenticated', data: {
   340→        'uid': user.uid,
   341→        'method': 'email',
   342→      });
   343→
   344→      // Set Sentry user context immediately after auth
   345→      await KlasivoSentry.userContext.setUser(
   346→        uid: user.uid,
   347→        email: email,
   348→      );
   349→
   350→      final readUserSpan = transaction.startChild('read_user_doc');
   351→      final userDoc = await _firestore
   352→          .collection(AppConstants.usersCollection)
   353→          .doc(user.uid)
   354→          .get();
   355→      await readUserSpan.finish();
   356→
   357→      if (!userDoc.exists) {
   358→        KlasivoSentry.breadcrumb.auth('login_failed_user_doc_missing', data: {
   359→          'uid': user.uid,
   360→        });
   361→        await Sentry.captureMessage(
   362→          'Login failed: users/${user.uid} document does not exist',
   363→          level: SentryLevel.error,
   364→        );
   365→        throw Exception('User data not found. Please contact your administrator.');
   366→      }
   367→
   368→      final userData = userDoc.data()!;
   369→      final role = userData['role'] as String?;
   370→      final isActive = userData['isActive'] as bool? ?? true;
   371→
   372→      // Update Sentry context with role and org
   373→      await KlasivoSentry.userContext.setRole(role ?? 'unknown');
   374→      final orgId = userData['organizationId'] as String?;
   375→      if (orgId != null && orgId.isNotEmpty) {
   376→        await KlasivoSentry.userContext.setOrganizationId(orgId);
   377→      }
   378→
   379→      if (!isActive) {
   380→        await _auth.signOut();
   381→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   382→          'uid': user.uid,
   383→          'role': role,
   384→        });
   385→        throw Exception('Your account has been deactivated. Contact your administrator.');
   386→      }
   387→
   388→      // Students should NOT use email login — they use student code login
   389→      if (role == AppConstants.roleStudent) {
   390→        await _auth.signOut();
   391→        KlasivoSentry.breadcrumb.auth('login_failed_student_email_login', data: {
   392→          'uid': user.uid,
   393→        });
   394→        throw Exception('Students must login with their student code.');
   395→      }
   396→
   397→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   398→        'uid': user.uid,
   399→        'role': role ?? 'unknown',
   400→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   401→      });
   402→
   403→      transaction.status = const SpanStatus.ok();
   404→
   405→      return {
   406→        'id': user.uid,
   407→        'organizationId': userData['organizationId'] ?? '',
   408→        'role': role ?? AppConstants.roleTeacher,
   409→        'authProvider': AuthProviders.password,
   410→        'fullName': userData['fullName'] ?? 'User',
   411→        'email': userData['email'] ?? email,
   412→        'isEmailVerified': user.emailVerified,
   413→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   414→      };
   415→    } catch (e, st) {
   416→      transaction.status = const SpanStatus.internalError();
   417→      await Sentry.captureException(
   418→        e,
   419→        stackTrace: st,
   420→        withScope: (scope) {
   421→          scope.setTag('flow', 'email_login');
   422→          scope.setTag('step', 'login_with_email');
   423→        },
   424→      );
   425→      rethrow;
   426→    } finally {
   427→      await transaction.finish();
   428→    }
   429→  }

   431→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   432→
   433→  /// Student login using student code + password.
   434→  /// Internally maps to Firebase Auth for push notifications,
   435→  /// multi-device, password reset, security rules, and analytics.
   436→  Future<Map<String, dynamic>> loginStudent({
   437→    required String studentCode,
   438→    required String password,
   439→  }) async {
   440→    final transaction = KlasivoSentry.transactions.loginFlow('student');
   441→
   442→    try {
   443→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   444→        'method': 'student_code',
   445→      });
   446→
   447→      // Step 1: Find user by studentCode
   448→      final findUserSpan = transaction.startChild('find_user_by_code');
   449→      final snapshot = await _firestore
   450→          .collection(AppConstants.usersCollection)
   451→          .where('studentCode', isEqualTo: studentCode)
   452→          .where('role', isEqualTo: AppConstants.roleStudent)
   453→          .limit(1)
   454→          .get();
   455→      await findUserSpan.finish();
   456→
   457→      if (snapshot.docs.isEmpty) {
   458→        KlasivoSentry.breadcrumb.auth('login_failed_student_not_found', data: {
   459→          'method': 'student_code',
   460→        });
   461→        throw Exception('Student not found. Please check your student code.');
   462→      }
   463→
   464→      final userDoc = snapshot.docs.first;
   465→      final student = userDoc.data();
   466→
   467→      // Step 2: Verify password
   468→      final storedPasswordHash = student['passwordHash'] as String?;
   469→      final storedPlaintext = student['password'] as String?;
   470→      final inputHash = hashPassword(password);
   471→
   472→      bool passwordMatches = false;
   473→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   474→        passwordMatches = inputHash == storedPasswordHash;
   475→      } else if (storedPlaintext != null) {
   476→        passwordMatches = password == storedPlaintext;
   477→        // Migrate to hash on successful login
   478→        if (passwordMatches) {
   479→          await _firestore
   480→              .collection(AppConstants.usersCollection)
   481→              .doc(userDoc.id)
   482→              .update({'passwordHash': inputHash});
   483→        }
   484→      }
   485→
   486→      if (!passwordMatches) {
   487→        KlasivoSentry.breadcrumb.auth('login_failed_invalid_password', data: {
   488→          'method': 'student_code',
   489→          'userId': userDoc.id,
   490→        });
   491→        throw Exception('Invalid password. Please try again.');
   492→      }
   493→
   494→      final isActive = student['isActive'] as bool? ?? true;
   495→      if (!isActive) {
   496→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   497→          'userId': userDoc.id,
   498→          'role': 'student',
   499→        });
   500→        throw Exception('Your account has been deactivated.');
   501→      }
   502→
   503→      // Step 3: Sign in via Firebase Auth using the student's internal email
   504→      final internalEmail = student['authEmail'] as String?;
   505→      if (internalEmail != null && internalEmail.isNotEmpty) {
   506→        try {
   507→          await _auth.signInWithEmailAndPassword(
   508→            email: internalEmail,
   509→            password: password,
   510→          );
   511→          KlasivoSentry.breadcrumb.auth('student_auth_signed_in', data: {
   512→            'userId': userDoc.id,
   513→          });
   514→        } catch (authError) {
   515→          // Graceful fallback — still allow login even if Firebase Auth fails
   516→          KlasivoSentry.breadcrumb.auth('student_auth_signin_failed_fallback', data: {
   517→            'userId': userDoc.id,
   518→            'error': authError.toString().substring(0, (authError.toString().length).clamp(0, 100)),
   519→          });
   520→        }
   521→      }
   522→
   523→      // Set Sentry user context
   524→      await KlasivoSentry.userContext.setUser(
   525→        uid: userDoc.id,
   526→        email: internalEmail ?? '',
   527→        role: AppConstants.roleStudent,
   528→        organizationId: student['organizationId'] as String?,
   529→      );
   530→
   531→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   532→        'uid': userDoc.id,
   533→        'role': 'student',
   534→        'method': 'student_code',
   535→      });
   536→
   537→      // Log doc ID audit — student uses Firestore doc ID (not auth UID)
   538→      KlasivoSentry.docIdAudit.logUserCreation(
   539→        flow: 'student_code_login',
   540→        collection: AppConstants.usersCollection,
   541→        docIdStrategy: userDoc.id.length > 28 ? 'auto_id' : 'uid',
   542→        actualDocId: userDoc.id,
   543→        authUid: _auth.currentUser?.uid,
   544→      );
   545→
   546→      transaction.status = const SpanStatus.ok();
   547→
   548→      return {
   549→        'id': userDoc.id,
   550→        'organizationId': student['organizationId'] ?? '',
   551→        'role': AppConstants.roleStudent,
   552→        'authProvider': AuthProviders.studentCode,
   553→        'fullName': student['fullName'] ?? 'Student',
   554→        'studentCode': student['studentCode'],
   555→        'classId': student['classId'],
   556→        'hasCompletedSetup': true,
   557→      };
   558→    } catch (e, st) {
   559→      transaction.status = const SpanStatus.internalError();
   560→      await Sentry.captureException(
   561→        e,
   562→        stackTrace: st,
   563→        withScope: (scope) {
   564→          scope.setTag('flow', 'student_login');
   565→          scope.setTag('step', 'login_student');
   566→          scope.setTag('method', 'student_code');
   567→        },
   568→      );
   569→      rethrow;
   570→    } finally {
   571→      await transaction.finish();
   572→    }
   573→  }

   577→  Future<Map<String, dynamic>> registerTeacherWithInvite({
   578→    required String email,
   579→    required String password,
   580→    required String fullName,
   581→    required String inviteCode,
   582→  }) async {
   583→    final transaction = Sentry.startTransaction(
   584→      'teacher_registration',
   585→      'registration',
   586→    );
   587→
   588→    try {
   589→      // Validate invite code
   590→      final inviteService = InviteCodeService();
   591→      final codeData = await inviteService.validateInviteCode(inviteCode);
   592→
   593→      if (codeData == null) {
   594→        throw Exception('Invalid or expired invite code.');
   595→      }
   596→
   597→      if (codeData['type'] != AppConstants.inviteTypeTeacher) {
   598→        throw Exception('This invite code is not for teachers.');
   599→      }
   600→
   601→      final organizationId = codeData['organizationId'] as String;
   602→
   603→      // Create Firebase Auth account
   604→      final userCredential =
   605→          await FirebaseService.registerWithEmail(email, password);
   606→      final user = userCredential.user!;
   607→
   608→      Sentry.addBreadcrumb(Breadcrumb(
   609→        category: 'registration',
   610→        message: 'STEP_1_AUTH_USER_CREATED',
   611→        data: {'uid': user.uid, 'email': email, 'role': 'teacher'},

  1210→        hasCompletedSetup = false; // Needs to name workspace
  1211→      } else if (role == AppConstants.roleParent) {
  1212→        organizationId = null; // Set when parent links a child
  1213→        hasCompletedSetup = false; // Needs to link child
  1214→      }
  1215→
  1216→      // For non-owner roles, write the user doc normally (owner doc already created above)
  1217→      if (role != AppConstants.roleOwner) {
  1218→        Sentry.addBreadcrumb(Breadcrumb(
  1219→          category: 'registration',
  1220→          message: 'STEP_2_USER_DOC_CREATE_START',
  1221→          data: {'role': role},
  1222→        ));
  1223→
  1224→        try {
  1225→          await SentryFirestoreHelper.docSet(
  1226→            collection: AppConstants.usersCollection,
  1227→            docId: user.uid,
  1228→            data: {
  1229→              'organizationId': organizationId,
  1230→              'role': role,
  1231→              'authProvider': AuthProviders.google,
  1232→              'fullName': fullName,
  1233→              'email': email,
  1234→              'photoUrl': user.photoURL,
  1235→              'phoneNumber': null,
  1236→              'isActive': true,
  1237→              'isEmailVerified': isEmailVerified,
  1238→              'hasCompletedSetup': hasCompletedSetup,
  1239→              'createdAt': FieldValue.serverTimestamp(),
  1240→              'updatedAt': FieldValue.serverTimestamp(),
  1241→            },
  1242→            flow: flowName,
  1243→            step: 'STEP_2_USER_DOC_CREATE',
  1244→          );
  1245→
  1246→          // Read-back verification: confirm the document actually exists
  1247→          final verifyDoc = await _firestore
  1248→              .collection(AppConstants.usersCollection)
  1249→              .doc(user.uid)
  1250→              .get();
  1251→          if (!verifyDoc.exists) {
  1252→            await Sentry.captureMessage(
  1253→              'STEP_2 USER DOC SET SUCCEEDED BUT READ-BACK FAILED — doc users/${user.uid} does not exist (role=$role)',
  1254→              level: SentryLevel.error,
  1255→            );
  1256→          }
  1257→
  1258→          KlasivoSentry.docIdAudit.logUserCreation(
  1259→            flow: flowName,
  1260→            collection: AppConstants.usersCollection,
  1261→            docIdStrategy: 'uid',
  1262→            actualDocId: user.uid,
  1263→            authUid: user.uid,
  1264→          );
  1265→        } catch (e, st) {
  1266→          rethrow;
  1267→        }
  1268→      }
  1269→
  1270→      // Set Sentry org/role context after all writes
  1271→      await Sentry.configureScope((scope) {
  1272→        scope.setTag('role', role);
  1273→        if (organizationId != null) {
  1274→          scope.setTag('organizationId', organizationId);
  1275→        }
  1276→      });
  1277→
  1278→      transaction.status = const SpanStatus.ok();
  1279→
  1280→      return {
  1281→        'id': user.uid,
  1282→        'organizationId': organizationId,
  1283→        'role': role,
  1284→        'authProvider': AuthProviders.google,
  1285→        'fullName': fullName,
  1286→        'email': email,
  1287→        'isEmailVerified': isEmailVerified,
  1288→        'hasCompletedSetup': hasCompletedSetup,
  1289→      };
  1290→    } catch (e) {
  1291→      transaction.status = const SpanStatus.internalError();
  1292→      rethrow;
  1293→    } finally {
  1294→      await transaction.finish();
  1295→    }
  1296→  }

  1398→  Future<bool> needsSetup() async {

  1400→      final user = _auth.currentUser;
  1401→      if (user == null) return false;
  1402→
  1403→      final userDoc = await _firestore
  1404→          .collection(AppConstants.usersCollection)
  1405→          .doc(user.uid)
  1406→          .get();
  1407→
  1408→      if (!userDoc.exists) return false;
  1409→
  1410→      final data = userDoc.data()!;
  1411→      final role = data['role'] as String?;
  1412→      final hasCompletedSetup = data['hasCompletedSetup'] as bool? ?? true;
  1413→
  1414→      // Owners need workspace naming, parents need child linking
  1415→      if ((role == AppConstants.roleOwner || role == AppConstants.roleParent) &&
  1416→          !hasCompletedSetup) {
  1417→        return true;
  1418→      }
  1419→
  1420→      return false;
  1421→    } catch (e) {
  1422→      return false;
  1423→    }
