   413→   413→class AuthChangeNotifier extends ChangeNotifier {
   414→   414→  StreamSubscription<User?>? _firebaseSub;
   415→   415→  StreamSubscription? _hiveSub;
   416→   416→
   417→   417→  AuthChangeNotifier() {
   418→   418→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   419→   419→      (_) => notifyListeners(),
   420→   420→      onError: (_) => notifyListeners(),
   421→   421→    );
   422→   422→    _startHiveWatch();
   423→   423→  }
   424→   424→
   425→   425→  void _startHiveWatch() {
   426→   426→    // Use Hive's built-in watch() instead of polling every 500ms
   427→   427→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   428→   428→      notifyListeners();
   429→   429→    });
   430→   430→  }
   431→   431→
   432→   432→  @override
   433→   433→  void dispose() {
   434→   434→    _firebaseSub?.cancel();
   435→   435→    _hiveSub?.cancel();
   436→   436→    super.dispose();
   437→   437→  }
   438→   438→}
   439→   439→
   440→   440→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   441→   441→  final notifier = AuthChangeNotifier();
   442→   442→  ref.onDispose(() => notifier.dispose());
   443→   443→  return notifier;
   444→   444→});
