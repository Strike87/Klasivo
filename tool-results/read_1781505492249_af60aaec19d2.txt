    30→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    31→
    32→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    33→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    34→class _SensitiveFields {
    35→  static const Set<String> keys = {
    36→    'password',
    37→    'passwordHash',
    38→    'confirmPassword',
    39→    'newPassword',
    40→    'accessToken',
    41→    'refreshToken',
    42→    'idToken',
    43→    'otp',
    44→    'otpCode',
    45→    'inviteCode',
    46→    'secret',
    47→    'apiKey',
    48→    'authEmail', // Internal student auth email — PII
    49→    'studentCode', // Sensitive — could be used to log in
    50→  };
    51→
    52→  /// Returns true if the key contains any sensitive substring.
    53→  static bool isSensitive(String key) {
    54→    final lower = key.toLowerCase();
    55→    for (final s in keys) {
    56→      if (lower.contains(s.toLowerCase())) return true;
    57→    }
    58→    return false;
    59→  }
    60→
    61→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    62→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    63→    return data.map((key, value) {
    64→      if (isSensitive(key)) {
    65→        return MapEntry(key, '[REDACTED]');
    66→      }
    67→      // Recursively sanitize nested maps
    68→      if (value is Map<String, dynamic>) {
    69→        return MapEntry(key, sanitize(value));
    70→      }
    71→      return MapEntry(key, value);
    72→    });
    73→  }
    74→
    75→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    76→  static String sanitizeString(String input) {
    77→    // Redact Bearer tokens
    78→    var result = input.replaceAll(
    79→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    80→      'Bearer [REDACTED]',
    81→    );
    82→    // Redact long hex strings (likely tokens)
    83→    result = result.replaceAll(
    84→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    85→      '[REDACTED_TOKEN]',
    86→    );
    87→    return result;
    88→  }
    89→}

   263→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception

   265→///
   266→/// Every Firestore exception will:
   267→///   1. Reach Sentry

   275→  /// Wrapped `.set()` with full observability.
   276→  static Future<void> docSet({
   277→    required String collection,
   278→    required String docId,
   279→    required Map<String, dynamic> data,
   280→    String? flow,
   281→    String? step,

   287→      collection: collection,
   288→      docId: docId,

   301→        e,
   302→        st,
   303→        collection: collection,
   304→        operation: 'set',
   305→        docId: docId,
   306→        flow: flow,
   307→        step: step,

   459→  /// Capture a Firestore exception with full context tags.
   460→  static Future<void> _captureFirestoreException(
   461→    Object exception,
   462→    StackTrace stackTrace, {
   463→    required String collection,
   464→    required String operation,
   465→    String? docId,
   466→    String? flow,
   467→    String? step,
   468→    Map<String, dynamic>? dataPreview,
   469→  }) async {
   470→    // Determine if this is a permission-denied error
   471→    String? errorCode;
   472→    if (exception is FirebaseException) {
   473→      errorCode = exception.code;
   474→    }
   475→
   476→    await Sentry.captureException(
   477→      exception,
   478→      stackTrace: stackTrace,
   479→      withScope: (scope) {
   480→        scope.setTag('collection', collection);
   481→        scope.setTag('operation', operation);
   482→        if (docId != null) scope.setTag('documentId', docId);
   483→        if (flow != null) scope.setTag('flow', flow);
   484→        if (step != null) scope.setTag('step', step);
   485→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   486→
   487→        // Attach user role and org from current Sentry context
   488→        final currentUser = FirebaseAuth.instance.currentUser;
   489→        if (currentUser != null) {
   490→          scope.setTag('auth_uid', currentUser.uid);
   491→        }
   492→
   493→        if (dataPreview != null && dataPreview.isNotEmpty) {
   494→          scope.setExtra('data_preview', dataPreview);
   495→        }
   496→      },
   497→    );
   498→
   499→    // ── Dual-report to Crashlytics ──
   500→    KlasivoCrashlytics.recordError(
   501→      exception,
   502→      stackTrace,
   503→      reason: 'Firestore $operation on $collection${docId != null ? '/$docId' : ''} failed '
   504→          '(flow=$flow, step=$step, firebase_code=$errorCode)',
   505→    );
   506→    KlasivoCrashlytics.setCustomKey('last_firestore_error', '$collection/$operation');
   507→    if (errorCode != null) KlasivoCrashlytics.setCustomKey('firebase_code', errorCode);
   508→  }
   509→}

   511→// ─── Guarded Async Operation ─────────────────────────────────────────────────
   512→
   513→/// Runs an async operation with automatic Sentry breadcrumb, exception capture,
   514→/// and metadata attachment. This is the recommended way to wrap any async
   515→/// operation that should never fail silently.
   516→///
   517→/// Example:
   518→///   final result = await KlasivoSentry.runGuarded(
   519→///     'create_exam',
   520→///     () => examService.createExam(...),
   521→///     category: 'firestore',
   522→///   );
   523→class KlasivoSentryGuard {
   524→  /// Execute [operation] inside a Sentry-guarded wrapper.
   525→  ///
   526→  /// - Automatically adds a breadcrumb before execution.
   527→  /// - On success: adds a success breadcrumb.
   528→  /// - On failure: captures the exception with scope tags and rethrows.
   529→  /// - The [category] defaults to the operation name but can be overridden.
   530→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   531→  ///   and the exception scope.
   532→  static Future<T> runGuarded<T>(
   533→    String operationName,
   534→    Future<T> Function() operation, {
   535→    String? category,
   536→    Map<String, String>? tags,
   537→    Map<String, dynamic>? data,
   538→  }) async {
   539→    final cat = category ?? 'operation';
   540→
   541→    Sentry.addBreadcrumb(Breadcrumb(
   542→      category: cat,
   543→      message: '${operationName}_started',
   544→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   545→      level: SentryLevel.info,
   546→    ));
   547→
   548→    try {
   549→      final result = await operation();
   550→
   551→      Sentry.addBreadcrumb(Breadcrumb(
   552→        category: cat,
   553→        message: '${operationName}_success',
   554→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   555→        level: SentryLevel.info,
   556→      ));
   557→
   558→      return result;
   559→    } catch (e, st) {
   560→      Sentry.addBreadcrumb(Breadcrumb(
   561→        category: cat,
   562→        message: '${operationName}_failed',
   563→        data: _SensitiveFields.sanitize({
   564→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   565→          ...?data,
   566→        }),
   567→        level: SentryLevel.error,
   568→      ));
   569→
   570→      await Sentry.captureException(
   571→        e,
   572→        stackTrace: st,
   573→        withScope: (scope) {
   574→          scope.setTag('operation', operationName);
   575→          tags?.forEach(scope.setTag);
   576→          if (data != null) {
   577→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   578→          }
   579→        },
   580→      );
   581→
   582→      // ── Dual-report to Crashlytics ──
   583→      KlasivoCrashlytics.recordError(e, st, reason: 'runGuarded: $operationName failed');
   584→      if (tags != null) {
   585→        tags.forEach(KlasivoCrashlytics.setCustomKey);
   586→      }
   587→
   588→      rethrow;
   589→    }
   590→  }
   591→
   592→  /// Execute [operation] with a Sentry transaction span.
   593→  ///
   594→  /// Creates a child span under the given [transaction], measures duration,
   595→  /// and sets the span status based on success/failure.
   596→  static Future<T> runWithSpan<T>(
   597→    ISentrySpan transaction,
   598→    String spanName,
   599→    Future<T> Function() operation, {
   600→    Map<String, String>? tags,
   601→  }) async {
   602→    final span = transaction.startChild(spanName);
   603→
   604→    try {
   605→      final result = await operation();
   606→      span.status = const SpanStatus.ok();
   607→      return result;
   608→    } catch (e, st) {
   609→      span.status = const SpanStatus.internalError();
   610→
   611→      await Sentry.captureException(
   612→        e,
   613→        stackTrace: st,
   614→        withScope: (scope) {
   615→          scope.setTag('span', spanName);
   616→          tags?.forEach(scope.setTag);
   617→        },
   618→      );
   619→
   620→      rethrow;
   621→    } finally {
   622→      await span.finish();
   623→    }
   624→  }
   625→}

   627→// ─── User Context Management ─────────────────────────────────────────────────
   628→
   629→/// Manages Sentry user context — attached on sign-in, updated on role/org
   630→/// changes, cleared on logout.
   631→///
   632→/// Attaches:
   633→///   - uid
   634→///   - email
   635→///   - role (tag)

   772→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   773→
   774→/// Documents the doc ID strategy for every user creation path.
   775→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   776→///
   777→/// | Path                  | Collection | Doc ID     | Status    |
   778→/// |-----------------------|-----------|------------|-----------|
   779→/// | Owner (email)         | users     | user.uid   | OK        |
   780→/// | Owner (Google)        | users     | user.uid   | OK        |
   781→/// | Teacher (invite)      | users     | user.uid   | OK        |
   782→/// | Teacher (Google)      | users     | user.uid   | OK        |
   783→/// | Parent (email)        | users     | user.uid   | OK        |
   784→/// | Parent (Google)       | users     | user.uid   | OK        |
   785→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   786→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   787→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   788→///
   789→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   790→/// `.doc(user.uid)`, which means:
   791→///   1. The doc ID doesn't match the auth UID
   792→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   793→///   3. QR enrollment is 100% broken in production
   794→class SentryDocIdAudit {
   795→  /// Log a doc ID audit breadcrumb for any user creation path.
   796→  static void logUserCreation({
   797→    required String flow,
   798→    required String collection,
   799→    required String docIdStrategy, // 'uid' or 'auto_id'
   800→    required String actualDocId,
   801→    String? authUid,
   802→  }) {
   803→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   804→    final isAutoId = docIdStrategy == 'auto_id';
   805→
   806→    Sentry.addBreadcrumb(Breadcrumb(
   807→      category: 'doc_id_audit',
   808→      message: 'user_doc_created',
   809→      data: _SensitiveFields.sanitize({
   810→        'flow': flow,
   811→        'collection': collection,
   812→        'docIdStrategy': docIdStrategy,
   813→        'actualDocId': actualDocId,
   814→        if (authUid != null) 'authUid': authUid,
   815→        'idMismatch': isMismatch,
   816→        'autoIdUsed': isAutoId,
   817→        'brokenByRules': isAutoId, // auto-id always fails security rules
   818→      }),
   819→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   820→    ));
   821→
   822→    if (isMismatch || isAutoId) {
   823→      Sentry.captureMessage(
   824→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   825→        '(docId=$actualDocId, authUid=$authUid). '
   826→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   827→        level: SentryLevel.warning,
   828→      );
   829→    }
   830→  }
   831→}

   925→// ─── Centralized Event Sanitizer ────────────────────────────────────────────
   926→
   927→/// Global Sentry event sanitizer used by the `beforeSend` callback in
   928→/// `main.dart`. Replaces the duplicate `_SentrySanitizer` class that was
   929→/// previously defined in main.dart.
   930→///
   931→/// Strips sensitive data from breadcrumbs, extras, and request headers.
   932→class KlasivoSentrySanitizer {
   933→  KlasivoSentrySanitizer._();
   934→
   935→  /// Sanitize a [SentryEvent] — returns the same event with sensitive data
   936→  /// replaced by '[REDACTED]'.
   937→  static SentryEvent sanitizeEvent(SentryEvent event) {
   938→    // Strip sensitive data from breadcrumbs
   939→    if (event.breadcrumbs != null) {
   940→      for (int i = 0; i < event.breadcrumbs!.length; i++) {
   941→        final crumb = event.breadcrumbs![i];
   942→        if (crumb.data != null) {
   943→          final sanitized = <String, dynamic>{};
   944→          crumb.data!.forEach((key, value) {
   945→            sanitized[key] = _SensitiveFields.isSensitive(key)
   946→                ? '[REDACTED]'
   947→                : value;
   948→          });
   949→          event.breadcrumbs![i] = Breadcrumb(
   950→            category: crumb.category,
   951→            message: crumb.message,
   952→            data: sanitized,
   953→            level: crumb.level,
   954→            timestamp: crumb.timestamp,
   955→            type: crumb.type,
   956→          );
   957→        }
   958→      }
   959→    }
   960→
   961→    // Strip sensitive data from extra fields
   962→    if (event.extra != null) {
   963→      final sanitized = <String, dynamic>{};
   964→      event.extra!.forEach((key, value) {
   965→        sanitized[key] = _SensitiveFields.isSensitive(key)
   966→            ? '[REDACTED]'
   967→            : value;
   968→      });
   969→      event.extra = sanitized;
   970→    }
   971→
   972→    // Strip sensitive data from request headers
   973→    if (event.request?.headers != null) {
   974→      final sanitized = <String, String>{};
   975→      event.request!.headers!.forEach((key, value) {
   976→        final lower = key.toLowerCase();
   977→        if (lower.contains('authorization') ||
   978→            lower.contains('cookie') ||
   979→            lower.contains('token')) {
   980→          sanitized[key] = '[REDACTED]';
   981→        } else {
   982→          sanitized[key] = value;
   983→        }
   984→      });
   985→      event.request = SentryRequest(headers: sanitized);
   986→    }
   987→
   988→    return event;
   989→  }
   990→}

  1058→// ─── Unified Dual-Reporting Facade ────────────────────────────────────────────
  1059→
  1060→/// **THE** recommended way to report errors in Klasivo.
  1061→/// Sends to BOTH Sentry AND Crashlytics in a single call.
  1062→///
  1063→/// This is the belt-and-suspenders pattern: if one platform has an outage,
  1064→/// the other still captures the error.
  1065→///
  1066→/// Usage:
  1067→///   await KlasivoObservability.reportError(
  1068→///     e,
  1069→///     st,
  1070→///     reason: 'Owner registration failed at STEP_2',
  1071→///     tags: {'flow': 'owner_registration', 'step': 'STEP_2'},
  1072→///   );
  1073→///
  1074→///   await KlasivoObservability.reportMessage(
  1075→///     'User doc missing after .set()',
  1076→///     level: SentryLevel.error,
  1077→///     tags: {'collection': 'users'},
  1078→///   );
  1079→class KlasivoObservability {
  1080→  KlasivoObservability._();
  1081→
  1082→  /// Report an error to BOTH Sentry and Crashlytics.
  1083→  ///
  1084→  /// - [exception] — the error object
  1085→  /// - [stackTrace] — the stack trace
  1086→  /// - [reason] — human-readable description (appears in Crashlytics "Reason")
  1087→  /// - [tags] — key-value pairs attached to Sentry scope + Crashlytics custom keys
  1088→  /// - [fatal] — marks as fatal in Crashlytics
  1089→  /// - [extras] — extra data attached to Sentry scope (not sent to Crashlytics)
  1090→  static Future<void> reportError(
  1091→    Object exception,
  1092→    StackTrace stackTrace, {
  1093→    String? reason,
  1094→    Map<String, String>? tags,
  1095→    Map<String, dynamic>? extras,
  1096→    bool fatal = false,
  1097→  }) async {
  1098→    // ── Sentry ──
  1099→    await Sentry.captureException(
  1100→      exception,
  1101→      stackTrace: stackTrace,
  1102→      withScope: (scope) {
  1103→        tags?.forEach(scope.setTag);
  1104→        if (extras != null) {
  1105→          extras.forEach((key, value) {
  1106→            scope.setExtra(
  1107→              key,
  1108→              _SensitiveFields.isSensitive(key) ? '[REDACTED]' : value,
  1109→            );
  1110→          });
  1111→        }
  1112→        if (reason != null) {
  1113→          scope.setTag(
  1114→              'reason', reason.length > 200 ? reason.substring(0, 200) : reason);
  1115→        }
  1116→      },
  1117→    );
  1118→
  1119→    // ── Crashlytics ──
  1120→    KlasivoCrashlytics.recordError(
  1121→      exception,
  1122→      stackTrace,
  1123→      reason: reason,
  1124→      fatal: fatal,
  1125→    );
  1126→
  1127→    // Mirror tags as Crashlytics custom keys for issue clustering
  1128→    if (tags != null) {
  1129→      tags.forEach(KlasivoCrashlytics.setCustomKey);
  1130→    }
  1131→  }
  1132→
  1133→  /// Report a message to BOTH Sentry and Crashlytics.
  1134→  ///
  1135→  /// - Sentry: captured as a message event with level
  1136→  /// - Crashlytics: logged via `log()` for crash breadcrumb context
  1137→  static Future<void> reportMessage(
  1138→    String message, {
  1139→    SentryLevel level = SentryLevel.info,
  1140→    Map<String, String>? tags,
  1141→  }) async {
  1142→    // ── Sentry ──
  1143→    await Sentry.captureMessage(message, level: level);
  1144→
  1145→    // ── Crashlytics ──
  1146→    final prefix = level == SentryLevel.error
  1147→        ? 'ERROR'
  1148→        : level == SentryLevel.warning
  1149→            ? 'WARN'
  1150→            : 'INFO';
  1151→    KlasivoCrashlytics.log('[$prefix] $message');
  1152→
  1153→    if (tags != null) {
  1154→      tags.forEach(KlasivoCrashlytics.setCustomKey);
  1155→    }
  1156→  }
  1157→
  1158→  /// Add a breadcrumb to BOTH Sentry and Crashlytics.
  1159→  ///
  1160→  /// Sentry: full structured breadcrumb
  1161→  /// Crashlytics: log message for crash context
  1162→  static void addBreadcrumb(
  1163→    String message, {
  1164→    String category = 'app',
  1165→    Map<String, dynamic>? data,
  1166→    SentryLevel level = SentryLevel.info,
  1167→  }) {
  1168→    // ── Sentry ──
  1169→    Sentry.addBreadcrumb(Breadcrumb(
  1170→      category: category,
  1171→      message: message,
  1172→      data: data != null ? _SensitiveFields.sanitize(data) : null,
  1173→      level: level,
  1174→    ));
  1175→
  1176→    // ── Crashlytics ──
  1177→    final dataStr = data != null && data.isNotEmpty
  1178→        ? ' | ${data.entries.map((e) => '${e.key}=${e.value}').join(',')}'
  1179→        : '';
  1180→    KlasivoCrashlytics.log('[$category] $message$dataStr');
  1181→  }
  1182→
  1183→  /// Convenience: report error with standard registration step tags.
  1184→  static Future<void> reportRegistrationError(
  1185→    Object exception,
  1186→    StackTrace stackTrace, {
  1187→    required String flow,
  1188→    required String step,
  1189→    String? role,
  1190→    String? uid,
  1191→    String? reason,
  1192→  }) async {
  1193→    await reportError(
  1194→      exception,
  1195→      stackTrace,
  1196→      reason: reason ?? '$flow failed at $step',
  1197→      tags: {
  1198→        'flow': flow,
  1199→        'step': step,
  1200→        if (role != null) 'role': role,
  1201→        if (uid != null) 'authUid': uid,
  1202→      },
  1203→    );
  1204→  }
