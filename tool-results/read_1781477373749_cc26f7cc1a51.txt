    24→  /// Hash a password using SHA-256
    25→  static String hashPassword(String password) {
    26→    final bytes = utf8.encode(password);
    27→    final digest = sha256.convert(bytes);
    28→    return digest.toString();
    29→  }

   156→  /// Unified email+password login for owners, teachers, and parents.
   157→  Future<Map<String, dynamic>> loginWithEmail({
   158→    required String email,
   159→    required String password,
   160→  }) async {
   161→    try {
   162→      final userCredential =
   163→          await FirebaseService.loginWithEmail(email, password);
   164→      final user = userCredential.user!;
   165→
   166→      final userDoc = await _firestore
   167→          .collection(AppConstants.usersCollection)
   168→          .doc(user.uid)
   169→          .get();
   170→
   171→      if (!userDoc.exists) {
   172→        throw Exception('User data not found. Please contact your administrator.');
   173→      }
   174→
   175→      final userData = userDoc.data()!;
   176→      final role = userData['role'] as String?;
   177→      final isActive = userData['isActive'] as bool? ?? true;
   178→
   179→      if (!isActive) {
   180→        await _auth.signOut();
   181→        throw Exception('Your account has been deactivated. Contact your administrator.');
   182→      }
   183→
   184→      // Students should NOT use email login — they use student code login
   185→      if (role == AppConstants.roleStudent) {
   186→        await _auth.signOut();
   187→        throw Exception('Students must login with their student code.');
   188→      }
   189→
   190→      return {
   191→        'id': user.uid,
   192→        'organizationId': userData['organizationId'] ?? '',
   193→        'role': role ?? AppConstants.roleTeacher,
   194→        'authProvider': AuthProviders.password,
   195→        'fullName': userData['fullName'] ?? 'User',
   196→        'email': userData['email'] ?? email,
   197→        'isEmailVerified': user.emailVerified,
   198→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   199→      };
   200→    } catch (e) {
   201→      rethrow;
   202→    }
   203→  }

   205→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   206→
   207→  /// Student login using student code + password.
   208→  /// Internally maps to Firebase Auth for push notifications,
   209→  /// multi-device, password reset, security rules, and analytics.
   210→  Future<Map<String, dynamic>> loginStudent({
   211→    required String studentCode,
   212→    required String password,
   213→  }) async {
   214→    try {
   215→      // Step 1: Find user by studentCode
   216→      final snapshot = await _firestore
   217→          .collection(AppConstants.usersCollection)
   218→          .where('studentCode', isEqualTo: studentCode)
   219→          .where('role', isEqualTo: AppConstants.roleStudent)
   220→          .limit(1)
   221→          .get();
   222→
   223→      if (snapshot.docs.isEmpty) {
   224→        throw Exception('Student not found. Please check your student code.');
   225→      }
   226→
   227→      final userDoc = snapshot.docs.first;
   228→      final student = userDoc.data();
   229→
   230→      // Step 2: Verify password
   231→      final storedPasswordHash = student['passwordHash'] as String?;
   232→      final storedPlaintext = student['password'] as String?;
   233→      final inputHash = hashPassword(password);
   234→
   235→      bool passwordMatches = false;
   236→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   237→        passwordMatches = inputHash == storedPasswordHash;
   238→      } else if (storedPlaintext != null) {
   239→        passwordMatches = password == storedPlaintext;
   240→        // Migrate to hash on successful login
   241→        if (passwordMatches) {
   242→          await _firestore
   243→              .collection(AppConstants.usersCollection)
   244→              .doc(userDoc.id)
   245→              .update({'passwordHash': inputHash});
   246→        }
   247→      }
   248→
   249→      if (!passwordMatches) {
   250→        throw Exception('Invalid password. Please try again.');
   251→      }
   252→
   253→      final isActive = student['isActive'] as bool? ?? true;
   254→      if (!isActive) {
   255→        throw Exception('Your account has been deactivated.');
   256→      }
   257→
   258→      // Step 3: Sign in via Firebase Auth using the student's internal email
   259→      final internalEmail = student['authEmail'] as String?;
   260→      if (internalEmail != null && internalEmail.isNotEmpty) {
   261→        try {
   262→          await _auth.signInWithEmailAndPassword(
   263→            email: internalEmail,
   264→            password: password,
   265→          );
   266→        } catch (authError) {
   267→          // Graceful fallback — still allow login even if Firebase Auth fails
   268→        }
   269→      }
   270→
   271→      return {
   272→        'id': userDoc.id,
   273→        'organizationId': student['organizationId'] ?? '',
   274→        'role': AppConstants.roleStudent,
   275→        'authProvider': AuthProviders.studentCode,
   276→        'fullName': student['fullName'] ?? 'Student',
   277→        'studentCode': student['studentCode'],
   278→        'classId': student['classId'],
   279→        'hasCompletedSetup': true,
   280→      };
   281→    } catch (e) {
   282→      rethrow;
   283→    }
   284→  }

   536→  ///   - owner: auto-creates organization (hasCompletedSetup=false)

   538→  ///   - parent: creates parent account (hasCompletedSetup=false, needs to link child)

   547→
   548→  Future<Map<String, dynamic>> _signInWithGoogle({
   549→    String? expectedRole,
   550→    bool isNewUser = false,
   551→  }) async {
   552→    try {
   553→      final googleUser = await GoogleSignIn().signIn();
   554→      if (googleUser == null) {
   555→        throw Exception('Google sign-in was cancelled.');
   556→      }
   557→
   558→      final googleAuth = await googleUser.authentication;
   559→      final credential = GoogleAuthProvider.credential(
   560→        accessToken: googleAuth.accessToken,
   561→        idToken: googleAuth.idToken,
   562→      );
   563→
   564→      final userCredential = await _auth.signInWithCredential(credential);
   565→      final user = userCredential.user!;
   566→
   567→      // Check if user document already exists
   568→      final userDoc = await _firestore
   569→          .collection(AppConstants.usersCollection)
   570→          .doc(user.uid)
   571→          .get();
   572→
   573→      if (userDoc.exists) {
   574→        // Existing user — log them in
   575→        final userData = userDoc.data()!;
   576→        final role = userData['role'] as String?;
   577→        final isActive = userData['isActive'] as bool? ?? true;
   578→
   579→        if (!isActive) {
   580→          await _auth.signOut();
   581→          throw Exception('Your account has been deactivated.');
   582→        }
   583→
   584→        // Students cannot use Google Sign-In
   585→        if (role == AppConstants.roleStudent) {
   586→          await _auth.signOut();
   587→          throw Exception('Students must login with their student code.');
   588→        }
   589→
   590→        return {
   591→          'id': user.uid,
   592→          'organizationId': userData['organizationId'] ?? '',
   593→          'role': role ?? expectedRole ?? AppConstants.roleOwner,
   594→          'authProvider': AuthProviders.google,
   595→          'fullName': userData['fullName'] ?? user.displayName ?? 'User',
   596→          'email': userData['email'] ?? user.email ?? '',
   597→          'isEmailVerified': user.emailVerified,
   598→          'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   599→        };
   600→      }
   601→
   602→      // New user — create account based on expected role
   603→      final fullName = user.displayName ?? 'User';
   604→      final email = user.email ?? '';
   605→      final role = expectedRole ?? AppConstants.roleOwner;
   606→      final isEmailVerified = user.emailVerified; // Google emails are pre-verified
   607→
   608→      String? organizationId;
   609→      bool hasCompletedSetup = true;
   610→
   611→      if (role == AppConstants.roleOwner) {
   612→        // Create user doc FIRST with placeholder orgId so isTeacherOrOwner() resolves
   613→        await _firestore
   614→            .collection(AppConstants.usersCollection)
   615→            .doc(user.uid)
   616→            .set({
   617→          'organizationId': '', // Placeholder — updated below
   618→          'role': AppConstants.roleOwner,
   619→          'authProvider': AuthProviders.google,
   620→          'fullName': fullName,
   621→          'email': email,
   622→          'photoUrl': user.photoURL,
   623→          'phoneNumber': null,
   624→          'isActive': true,
   625→          'isEmailVerified': isEmailVerified,
   626→          'hasCompletedSetup': false,
   627→          'createdAt': FieldValue.serverTimestamp(),
   628→          'updatedAt': FieldValue.serverTimestamp(),
   629→        });
   630→
   631→        // Now create the organization — rules can verify user is an owner
   632→        final orgService = OrganizationService();
   633→        organizationId = await orgService.createOrganization(
   634→          ownerId: user.uid,
   635→          name: "$fullName's Workspace",
   636→        );
   637→
   638→        // Patch the user doc with the real organizationId
   639→        await _firestore
   640→            .collection(AppConstants.usersCollection)
   641→            .doc(user.uid)
   642→            .update({
   643→          'organizationId': organizationId,
   644→          'updatedAt': FieldValue.serverTimestamp(),
   645→        });
   646→
   647→        hasCompletedSetup = false; // Needs to name workspace
   648→      } else if (role == AppConstants.roleParent) {
   649→        organizationId = null; // Set when parent links a child
   650→        hasCompletedSetup = false; // Needs to link child
   651→      }
   652→
   653→      // For non-owner roles, write the user doc normally (owner doc already created above)
   654→      if (role != AppConstants.roleOwner) {
   655→        await _firestore
   656→            .collection(AppConstants.usersCollection)
   657→            .doc(user.uid)
   658→            .set({
   659→          'organizationId': organizationId,
   660→          'role': role,
   661→          'authProvider': AuthProviders.google,
   662→          'fullName': fullName,
   663→          'email': email,
   664→          'photoUrl': user.photoURL,
   665→          'phoneNumber': null,
   666→          'isActive': true,
   667→          'isEmailVerified': isEmailVerified,
   668→          'hasCompletedSetup': hasCompletedSetup,
   669→          'createdAt': FieldValue.serverTimestamp(),
   670→          'updatedAt': FieldValue.serverTimestamp(),
   671→        });
   672→      }
   673→
   674→      return {
   675→        'id': user.uid,
   676→        'organizationId': organizationId,
   677→        'role': role,
   678→        'authProvider': AuthProviders.google,
   679→        'fullName': fullName,
   680→        'email': email,
   681→        'isEmailVerified': isEmailVerified,
   682→        'hasCompletedSetup': hasCompletedSetup,
   683→      };
   684→    } catch (e) {
   685→      rethrow;
   686→    }
   687→  }

   752→  Future<bool> needsSetup() async {
   753→    try {
   754→      final user = _auth.currentUser;
   755→      if (user == null) return false;
   756→
   757→      final userDoc = await _firestore
   758→          .collection(AppConstants.usersCollection)
   759→          .doc(user.uid)
   760→          .get();
   761→
   762→      if (!userDoc.exists) return false;
   763→
   764→      final data = userDoc.data()!;
   765→      final role = data['role'] as String?;
   766→      final hasCompletedSetup = data['hasCompletedSetup'] as bool? ?? true;
   767→
   768→      // Owners need workspace naming, parents need child linking
   769→      if ((role == AppConstants.roleOwner || role == AppConstants.roleParent) &&
   770→          !hasCompletedSetup) {
   771→        return true;
   772→      }
   773→
   774→      return false;
   775→    } catch (e) {
   776→      return false;
   777→    }
   778→  }
