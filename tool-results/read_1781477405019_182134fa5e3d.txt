    11→// ─── Background message handler (must be top-level function) ────────────────
    12→
    13→@pragma('vm:entry-point')
    14→Future<void> firebaseMessagingBackgroundHandler(RemoteMessage message) async {
    15→  // Initialize Firebase for background isolate
    16→  await Firebase.initializeApp();
    17→
    18→  // If this is a data-only message (no notification payload), show a local notification
    19→  // Notification+data messages are auto-displayed by the system
    20→  if (message.notification == null && message.data.isNotEmpty) {
    21→    final title = message.data['title'] as String? ?? 'Klasivo';
    22→    final body = message.data['body'] as String? ?? 'You have a new notification';
    23→    final channelId = message.data['channelId'] as String? ?? 'klasivo_channel';
    24→
    25→    final flutterLocalNotifications = FlutterLocalNotificationsPlugin();
    26→    const androidSettings = AndroidInitializationSettings('@mipmap/ic_launcher');
    27→    const initSettings = InitializationSettings(android: androidSettings);
    28→    await flutterLocalNotifications.initialize(initSettings);
    29→
    30→    final androidDetails = AndroidNotificationDetails(
    31→      channelId,
    32→      channelId == 'klasivo_messages'
    33→          ? 'Messages'
    34→          : channelId == 'klasivo_attendance'
    35→              ? 'Attendance'
    36→              : channelId == 'klasivo_scheduled'
    37→                  ? 'Exam Reminders'
    38→                  : 'Klasivo',
    39→      importance: Importance.high,
    40→      priority: Priority.high,
    41→    );
    42→    const iosDetails = DarwinNotificationDetails();
    43→    final details = NotificationDetails(android: androidDetails, iOS: iosDetails);
    44→
    45→    final id = DateTime.now().millisecondsSinceEpoch ~/ 1000;
    46→    await flutterLocalNotifications.show(id, title, body, details);
    47→  }
    48→}

   135→  static Future<void> _initializeFCM() async {
   136→    await requestPermissions();
   137→
   138→    _fcmToken = await _fcm.getToken();
   139→
   140→    // Persist FCM token to Firestore for server-side push
   141→    await _persistTokenToFirestore(_fcmToken);
   142→
   143→    _fcm.onTokenRefresh.listen((newToken) {
   144→      _fcmToken = newToken;
   145→      _persistTokenToFirestore(newToken);
   146→    });
   147→
   148→    FirebaseMessaging.onMessage.listen((RemoteMessage message) {
   149→      _handleForegroundMessage(message);
   150→    });
   151→
   152→    FirebaseMessaging.onMessageOpenedApp.listen((RemoteMessage message) {
   153→      _handleMessageOpenedApp(message);
   154→    });
   155→
   156→    final initialMessage = await _fcm.getInitialMessage();
   157→    if (initialMessage != null) {
   158→      _handleMessageOpenedApp(initialMessage);
   159→    }
   160→
   161→    FirebaseMessaging.onBackgroundMessage(firebaseMessagingBackgroundHandler);
   162→  }

   200→  static Future<void> _persistTokenToFirestore(String? token) async {
   201→    if (token == null) return;
   202→    try {
   203→      final user = FirebaseAuth.instance.currentUser;
   204→      if (user == null) return;
   205→
   206→      await FirebaseFirestore.instance
   207→          .collection(AppConstants.usersCollection)
   208→          .doc(user.uid)
   209→          .update({
   210→        'fcmToken': token,
   211→        'fcmTokenUpdatedAt': FieldValue.serverTimestamp(),
   212→      });
   213→    } catch (e) {
   214→      // Non-critical: token persistence failure shouldn't block the app
   215→      debugPrint('[NotificationService] Failed to persist FCM token: $e');
   216→    }
   217→  }

   221→  static Future<bool> requestPermissions() async {
   222→    final fcmSettings = await _fcm.requestPermission(
   223→      alert: true,
   224→      badge: true,
   225→      sound: true,
   226→      provisional: false,
   227→    );
   228→
   229→    final androidPlugin = _localNotifications.resolvePlatformSpecificImplementation<
   230→        AndroidFlutterLocalNotificationsPlugin>();
   231→    if (androidPlugin != null) {
   232→      return await androidPlugin.requestNotificationsPermission() ?? false;
   233→    }
   234→
   235→    final iosPlugin = _localNotifications.resolvePlatformSpecificImplementation<
   236→        IOSFlutterLocalNotificationsPlugin>();
   237→    if (iosPlugin != null) {
   238→      return await iosPlugin.requestPermissions(alert: true, badge: true, sound: true) ?? false;
   239→    }
   240→
   241→    return fcmSettings.authorizationStatus == AuthorizationStatus.authorized;
   242→  }

   268→  /// Subscribe the current user to all relevant FCM topics based on their role.
   269→  /// Call this after a successful login. Subscribes to:
   270→  /// - Organization topic (all users)
   271→  /// - Class topic (students + teachers assigned to a class)
   272→  static Future<void> subscribeUserToTopics({
   273→    required String userId,
   274→    required String role,
   275→    String? organizationId,
   276→    String? classId,
   277→  }) async {
   278→    try {
   279→      // Subscribe to organization topic
   280→      if (organizationId != null && organizationId.isNotEmpty) {
   281→        await subscribeToOrganization(organizationId);
   282→      }
   283→
   284→      // Subscribe to class topic
   285→      if (classId != null && classId.isNotEmpty) {
   286→        await subscribeToClass(classId);
   287→      }
   288→
   289→      // Teachers may be assigned to multiple classes
   290→      if (role == 'teacher' && organizationId != null) {
   291→        final assignmentsSnapshot = await _firestore
   292→            .collection(AppConstants.teacherAssignmentsCollection)
   293→            .where('teacherId', isEqualTo: userId)
   294→            .get();
   295→        for (final doc in assignmentsSnapshot.docs) {
   296→          final assignedClassId = doc.data()['classId'] as String?;
   297→          if (assignedClassId != null) {
   298→            await subscribeToClass(assignedClassId);
   299→          }
   300→        }
   301→      }
   302→
   303→      debugPrint('[NotificationService] Subscribed user $userId to FCM topics');
   304→    } catch (e) {
   305→      debugPrint('[NotificationService] Failed to subscribe to topics: $e');
   306→    }
   307→  }

   343→  /// Create a notification record in Firestore and show a local notification.
   344→  /// This is the central method for all in-app notifications.
   345→  static Future<String> createNotification({
   346→    required String userId,
   347→    required String type,
   348→    required String title,
   349→    required String body,
   350→    String? organizationId,
   351→    String? relatedId,
   352→    String? relatedType,
   353→    Map<String, dynamic>? data,
   354→    String localChannelId = 'klasivo_channel',
   355→  }) async {
   356→    try {
   357→      final docRef = await _firestore
   358→          .collection(AppConstants.notificationsCollection)
   359→          .add({
   360→        'userId': userId,
   361→        'type': type,
   362→        'title': title,
   363→        'body': body,
   364→        'organizationId': organizationId,
   365→        'relatedId': relatedId,
   366→        'relatedType': relatedType,
   367→        'data': data ?? {},
   368→        'isRead': false,
   369→        'createdAt': FieldValue.serverTimestamp(),
   370→      });
   371→
   372→      // Also show a local notification
   373→      await showNotification(
   374→        title: title,
   375→        body: body,
   376→        channelId: localChannelId,
   377→      );
   378→
   379→      return docRef.id;
   380→    } catch (e) {
   381→      rethrow;
   382→    }
   383→  }

   385→  /// Create notifications for multiple users (e.g., entire class).
   386→  static Future<void> createBulkNotifications({
   387→    required List<String> userIds,
   388→    required String type,
   389→    required String title,
   390→    required String body,

   394→    Map<String, dynamic>? data,
   395→  }) async {

   397→      final batch = _firestore.batch();
   398→
   399→      for (final userId in userIds) {
   400→        final docRef =
   401→            _firestore.collection(AppConstants.notificationsCollection).doc();
   402→        batch.set(docRef, {
   403→          'userId': userId,
   404→          'type': type,
   405→          'title': title,
   406→          'body': body,
   407→          'organizationId': organizationId,
   408→          'relatedId': relatedId,
   409→          'relatedType': relatedType,
   410→          'data': data ?? {},
   411→          'isRead': false,
   412→          'createdAt': FieldValue.serverTimestamp(),
   413→        });
   414→      }
   415→
   416→      await batch.commit();
   417→    } catch (e) {
   418→      rethrow;
   419→    }
   420→  }

   724→  // ─── Schedule Exam Reminders ─────────────────────────────────────────────
   725→
   726→  static Future<void> scheduleExamReminders({
   727→    required String examId,
   728→    required String examTitle,
   729→    required DateTime startDate,
   730→  }) async {
   731→    final now = DateTime.now();
   732→
   733→    final twentyFourHoursBefore = startDate.subtract(const Duration(hours: 24));
   734→    if (twentyFourHoursBefore.isAfter(now)) {
   735→      await _scheduleNotification(
   736→        id: (examId.hashCode & 0x7FFFFFFF) + 0,
   737→        title: 'Exam Reminder',
   738→        body: '"$examTitle" starts tomorrow',
   739→        scheduledDate: tz.TZDateTime.from(twentyFourHoursBefore, tz.local),
   740→      );
   741→    }
   742→
   743→    final oneHourBefore = startDate.subtract(const Duration(hours: 1));
   744→    if (oneHourBefore.isAfter(now)) {
   745→      await _scheduleNotification(
   746→        id: (examId.hashCode & 0x7FFFFFFF) + 1,
   747→        title: 'Exam Starting Soon',
   748→        body: '"$examTitle" starts in 1 hour',
   749→        scheduledDate: tz.TZDateTime.from(oneHourBefore, tz.local),
   750→      );
   751→    }
   752→
   753→    final fifteenMinBefore = startDate.subtract(const Duration(minutes: 15));
   754→    if (fifteenMinBefore.isAfter(now)) {
   755→      await _scheduleNotification(
   756→        id: (examId.hashCode & 0x7FFFFFFF) + 2,
   757→        title: 'Exam Starting Soon!',
   758→        body: '"$examTitle" starts in 15 minutes',
   759→        scheduledDate: tz.TZDateTime.from(fifteenMinBefore, tz.local),
   760→      );
   761→    }
   762→
   763→    if (startDate.isAfter(now)) {
   764→      await _scheduleNotification(
   765→        id: (examId.hashCode & 0x7FFFFFFF) + 3,
   766→        title: 'Exam Started!',
   767→        body: '"$examTitle" has started. Open the app to take it now!',
   768→        scheduledDate: tz.TZDateTime.from(startDate, tz.local),
   769→      );
   770→    }
   771→  }

   775→  static Future<void> cancelExamReminders(String examId) async {
   776→    await _localNotifications.cancel((examId.hashCode & 0x7FFFFFFF) + 0);

   788→  // ─── Schedule a Single Notification ──────────────────────────────────────
   789→
   790→  static Future<void> _scheduleNotification({
   791→    required int id,
   792→    required String title,
   793→    required String body,
   794→    required tz.TZDateTime scheduledDate,
   795→  }) async {
   796→    const androidDetails = AndroidNotificationDetails(
   797→      'klasivo_scheduled',
   798→      'Exam Reminders',
   799→      channelDescription: 'Scheduled notifications for exam reminders',
   800→      importance: Importance.high,
   801→      priority: Priority.high,
   802→    );
   803→
   804→    const iosDetails = DarwinNotificationDetails();
   805→
   806→    const details = NotificationDetails(
   807→      android: androidDetails,
   808→      iOS: iosDetails,
   809→    );
   810→
   811→    await _localNotifications.zonedSchedule(
   812→      id,
   813→      title,
   814→      body,
   815→      scheduledDate,
   816→      details,
   817→      androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
   818→      uiLocalNotificationDateInterpretation:
   819→          UILocalNotificationDateInterpretation.absoluteTime,
   820→    );
   821→  }
