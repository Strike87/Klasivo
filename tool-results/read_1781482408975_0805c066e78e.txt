    26→    26→  /// Hash a password using SHA-256
    27→    27→  static String hashPassword(String password) {
    28→    28→    final bytes = utf8.encode(password);
    29→    29→    final digest = sha256.convert(bytes);
    30→    30→    return digest.toString();
    31→    31→  }

   320→   320→  // ─── Teacher/Owner/Parent Login (Email + Password) ────────────────────────
   321→   321→
   322→   322→  /// Unified email+password login for owners, teachers, and parents.
   323→   323→  Future<Map<String, dynamic>> loginWithEmail({
   324→   324→    required String email,
   325→   325→    required String password,
   326→   326→  }) async {
   327→   327→    final transaction = KlasivoSentry.transactions.loginFlow('email');
   328→   328→
   329→   329→    try {
   330→   330→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   331→   331→        'method': 'email',
   332→   332→        'email': email,
   333→   333→      });
   334→   334→
   335→   335→      final userCredential =
   336→   336→          await FirebaseService.loginWithEmail(email, password);
   337→   337→      final user = userCredential.user!;
   338→   338→
   339→   339→      KlasivoSentry.breadcrumb.auth('auth_user_authenticated', data: {
   340→   340→        'uid': user.uid,
   341→   341→        'method': 'email',
   342→   342→      });
   343→   343→
   344→   344→      // Set Sentry user context immediately after auth
   345→   345→      await KlasivoSentry.userContext.setUser(
   346→   346→        uid: user.uid,
   347→   347→        email: email,
   348→   348→      );
   349→   349→
   350→   350→      final readUserSpan = transaction.startChild('read_user_doc');
   351→   351→      final userDoc = await _firestore
   352→   352→          .collection(AppConstants.usersCollection)
   353→   353→          .doc(user.uid)
   354→   354→          .get();
   355→   355→      await readUserSpan.finish();
   356→   356→
   357→   357→      if (!userDoc.exists) {
   358→   358→        KlasivoSentry.breadcrumb.auth('login_failed_user_doc_missing', data: {
   359→   359→          'uid': user.uid,
   360→   360→        });
   361→   361→        await Sentry.captureMessage(
   362→   362→          'Login failed: users/${user.uid} document does not exist',
   363→   363→          level: SentryLevel.error,
   364→   364→        );
   365→   365→        throw Exception('User data not found. Please contact your administrator.');
   366→   366→      }
   367→   367→
   368→   368→      final userData = userDoc.data()!;
   369→   369→      final role = userData['role'] as String?;
   370→   370→      final isActive = userData['isActive'] as bool? ?? true;
   371→   371→
   372→   372→      // Update Sentry context with role and org
   373→   373→      await KlasivoSentry.userContext.setRole(role ?? 'unknown');
   374→   374→      final orgId = userData['organizationId'] as String?;
   375→   375→      if (orgId != null && orgId.isNotEmpty) {
   376→   376→        await KlasivoSentry.userContext.setOrganizationId(orgId);
   377→   377→      }
   378→   378→
   379→   379→      if (!isActive) {
   380→   380→        await _auth.signOut();
   381→   381→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   382→   382→          'uid': user.uid,
   383→   383→          'role': role,
   384→   384→        });
   385→   385→        throw Exception('Your account has been deactivated. Contact your administrator.');
   386→   386→      }
   387→   387→
   388→   388→      // Students should NOT use email login — they use student code login
   389→   389→      if (role == AppConstants.roleStudent) {
   390→   390→        await _auth.signOut();
   391→   391→        KlasivoSentry.breadcrumb.auth('login_failed_student_email_login', data: {
   392→   392→          'uid': user.uid,
   393→   393→        });
   394→   394→        throw Exception('Students must login with their student code.');
   395→   395→      }
   396→   396→
   397→   397→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   398→   398→        'uid': user.uid,
   399→   399→        'role': role ?? 'unknown',
   400→   400→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   401→   401→      });
   402→   402→
   403→   403→      transaction.status = const SpanStatus.ok();
   404→   404→
   405→   405→      return {
   406→   406→        'id': user.uid,
   407→   407→        'organizationId': userData['organizationId'] ?? '',
   408→   408→        'role': role ?? AppConstants.roleTeacher,
   409→   409→        'authProvider': AuthProviders.password,
   410→   410→        'fullName': userData['fullName'] ?? 'User',
   411→   411→        'email': userData['email'] ?? email,
   412→   412→        'isEmailVerified': user.emailVerified,
   413→   413→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   414→   414→      };
   415→   415→    } catch (e, st) {
   416→   416→      transaction.status = const SpanStatus.internalError();
   417→   417→      await Sentry.captureException(
   418→   418→        e,
   419→   419→        stackTrace: st,
   420→   420→        withScope: (scope) {
   421→   421→          scope.setTag('flow', 'email_login');
   422→   422→          scope.setTag('step', 'login_with_email');
   423→   423→        },
   424→   424→      );
   425→   425→      rethrow;
   426→   426→    } finally {
   427→   427→      await transaction.finish();
   428→   428→    }
   429→   429→  }

   431→   431→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   432→   432→
   433→   433→  /// Student login using student code + password.
   434→   434→  /// Internally maps to Firebase Auth for push notifications,
   435→   435→  /// multi-device, password reset, security rules, and analytics.
   436→   436→  Future<Map<String, dynamic>> loginStudent({
   437→   437→    required String studentCode,
   438→   438→    required String password,
   439→   439→  }) async {
   440→   440→    final transaction = KlasivoSentry.transactions.loginFlow('student');
   441→   441→
   442→   442→    try {
   443→   443→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   444→   444→        'method': 'student_code',
   445→   445→      });
   446→   446→
   447→   447→      // Step 1: Find user by studentCode
   448→   448→      final findUserSpan = transaction.startChild('find_user_by_code');
   449→   449→      final snapshot = await _firestore
   450→   450→          .collection(AppConstants.usersCollection)
   451→   451→          .where('studentCode', isEqualTo: studentCode)
   452→   452→          .where('role', isEqualTo: AppConstants.roleStudent)
   453→   453→          .limit(1)
   454→   454→          .get();
   455→   455→      await findUserSpan.finish();
   456→   456→
   457→   457→      if (snapshot.docs.isEmpty) {
   458→   458→        KlasivoSentry.breadcrumb.auth('login_failed_student_not_found', data: {
   459→   459→          'method': 'student_code',
   460→   460→        });
   461→   461→        throw Exception('Student not found. Please check your student code.');
   462→   462→      }
   463→   463→
   464→   464→      final userDoc = snapshot.docs.first;
   465→   465→      final student = userDoc.data();
   466→   466→
   467→   467→      // Step 2: Verify password
   468→   468→      final storedPasswordHash = student['passwordHash'] as String?;
   469→   469→      final storedPlaintext = student['password'] as String?;
   470→   470→      final inputHash = hashPassword(password);
   471→   471→
   472→   472→      bool passwordMatches = false;
   473→   473→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   474→   474→        passwordMatches = inputHash == storedPasswordHash;
   475→   475→      } else if (storedPlaintext != null) {
   476→   476→        passwordMatches = password == storedPlaintext;
   477→   477→        // Migrate to hash on successful login
   478→   478→        if (passwordMatches) {
   479→   479→          await _firestore
   480→   480→              .collection(AppConstants.usersCollection)
   481→   481→              .doc(userDoc.id)
   482→   482→              .update({'passwordHash': inputHash});
   483→   483→        }
   484→   484→      }
   485→   485→
   486→   486→      if (!passwordMatches) {
   487→   487→        KlasivoSentry.breadcrumb.auth('login_failed_invalid_password', data: {
   488→   488→          'method': 'student_code',
   489→   489→          'userId': userDoc.id,
   490→   490→        });
   491→   491→        throw Exception('Invalid password. Please try again.');
   492→   492→      }
   493→   493→
   494→   494→      final isActive = student['isActive'] as bool? ?? true;
   495→   495→      if (!isActive) {
   496→   496→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   497→   497→          'userId': userDoc.id,
   498→   498→          'role': 'student',
   499→   499→        });
   500→   500→        throw Exception('Your account has been deactivated.');
   501→   501→      }
