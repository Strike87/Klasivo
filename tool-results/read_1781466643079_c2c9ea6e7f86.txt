     2→// KLASIVO — Scope Assignment Screen
     3→//
     4→// Visual tree selection for campus/stage/class hierarchy.
     5→// Shows which scope level is relevant for the user's role and allows
     6→// multi-select with a hierarchical tree view.
     7→//
     8→// Tree structure:
     9→//   Campus A
    10→//    ├─ Primary (Stage)
    11→//    │   ├─ Class 1A
    12→//    │   ├─ Class 1B
    13→//    └─ Secondary (Stage)
    14→//        ├─ Class 7A
    15→//        └─ Class 7B
    16→//
    17→// Calls the `assignScope` Cloud Function on save.
    18→// ═══════════════════════════════════════════════════════════════════════════════

    61→class _ScopeAssignmentScreenState extends ConsumerState<ScopeAssignmentScreen> {
    62→  late Set<String> _selectedCampusIds;
    63→  late Set<String> _selectedStageIds;
    64→  late Set<String> _selectedClassIds;
    65→  late Set<String> _selectedSubjectIds;
    66→  bool _loading = false;
    67→  Map<String, String> _idToNameCache = {};

    69→  @override
    70→  void initState() {
    71→    super.initState();
    72→    _selectedCampusIds = Set.from(widget.currentCampusIds);
    73→    _selectedStageIds = Set.from(widget.currentStageIds);
    74→    _selectedClassIds = Set.from(widget.currentClassIds);
    75→    _selectedSubjectIds = Set.from(widget.currentSubjectIds);
    76→  }
    77→
    78→  ScopeAccessLevel get _scopeLevel => scopeAccessLevelForRole(widget.currentRole);
    79→
    80→  bool get _showCampusPicker => _scopeLevel == ScopeAccessLevel.campus ||
    81→      _scopeLevel == ScopeAccessLevel.all;
    82→
    83→  bool get _showStagePicker => _scopeLevel == ScopeAccessLevel.stage ||
    84→      _scopeLevel == ScopeAccessLevel.campus;
    85→
    86→  bool get _showClassPicker => _scopeLevel == ScopeAccessLevel.class_;
    87→
    88→  @override
    89→  Widget build(BuildContext context) {

   190→
   191→  Widget _buildScopePicker() {
   192→    // Class-scoped roles see a visual tree (Campus → Stage → Class)
   193→    if (_showClassPicker) {
   194→      return _ClassScopeTree(
   195→        selectedStageIds: _selectedStageIds,
   196→        selectedClassIds: _selectedClassIds,
   197→        onStageToggle: _toggleStage,
   198→        onClassToggle: _toggleClass,
   199→        onStageNameResolved: _cacheName,
   200→        onClassNameResolved: _cacheName,
   201→      );
   202→    }
   203→
   204→    // Stage-scoped roles see stage list
   205→    if (_showStagePicker) {
   206→      return _StageScopeList(
   207→        selectedStageIds: _selectedStageIds,
   208→        onStageToggle: _toggleStage,
   209→        onNameResolved: _cacheName,
   210→      );
   211→    }
   212→
   213→    // Campus-scoped roles see campus list
   214→    if (_showCampusPicker) {
   215→      return _CampusScopeList(
   216→        selectedCampusIds: _selectedCampusIds,
   217→        onCampusToggle: _toggleCampus,
   218→        onNameResolved: _cacheName,
   219→      );
   220→    }
   221→
   222→    // All-access roles don't need scope
   223→    return const KlasivoEmptyState(
   224→      icon: Icons.check_circle_rounded,
   225→      title: 'No scope assignment needed',
   226→      subtitle: 'This role has full access within the organization',
   227→    );
   228→  }
   229→
   230→  void _cacheName(String id, String name) {
   231→    _idToNameCache[id] = name;
   232→  }
   233→
   234→  void _toggleCampus(String id) {
   235→    setState(() {
   236→      if (_selectedCampusIds.contains(id)) {
   237→        _selectedCampusIds.remove(id);
   238→      } else {
   239→        _selectedCampusIds.add(id);
   240→      }
   241→    });
   242→  }
   243→
   244→  void _toggleStage(String id) {
   245→    setState(() {
   246→      if (_selectedStageIds.contains(id)) {
   247→        _selectedStageIds.remove(id);
   248→      } else {
   249→        _selectedStageIds.add(id);
   250→      }
   251→    });
   252→  }
   253→
   254→  void _toggleClass(String id) {
   255→    setState(() {
   256→      if (_selectedClassIds.contains(id)) {
   257→        _selectedClassIds.remove(id);
   258→      } else {
   259→        _selectedClassIds.add(id);
   260→      }
   261→    });
   262→  }
   263→
   264→  Future<void> _save() async {
   265→    setState(() => _loading = true);
   266→
   267→    try {
   268→      final scopeData = <String, dynamic>{};
   269→
   270→      if (_showCampusPicker && _selectedCampusIds.isNotEmpty) {
   271→        scopeData['campusIds'] = _selectedCampusIds.toList();
   272→      }
   273→      if (_showStagePicker && _selectedStageIds.isNotEmpty) {
   274→        scopeData['stageIds'] = _selectedStageIds.toList();
   275→      }
   276→      if (_showClassPicker) {
   277→        if (_selectedClassIds.isNotEmpty) {
   278→          scopeData['classIds'] = _selectedClassIds.toList();
   279→        }
   280→        if (_selectedStageIds.isNotEmpty) {
   281→          scopeData['stageIds'] = _selectedStageIds.toList();
   282→        }
   283→      }
   284→
   285→      await ref.read(userManagementRepoProvider).assignScope(
   286→            targetUserId: widget.userId,
   287→            scopeData: scopeData,
   288→          );
   289→
   290→      if (mounted) {
   291→        KlasivoToast.show(
   292→          context,
   293→          message: 'Scope updated for ${widget.userName}',
   294→          type: KlasivoToastType.success,
   295→        );
   296→        Navigator.of(context).pop();
   297→      }
   298→    } catch (e) {
   299→      if (mounted) {
   300→        KlasivoToast.show(
   301→          context,
   302→          message: 'Failed to update scope: $e',
   303→          type: KlasivoToastType.error,
   304→        );
   305→      }
   306→    } finally {
   307→      if (mounted) setState(() => _loading = false);
   308→    }
   309→  }
   310→
   311→  String _scopeLabel(ScopeAccessLevel level) => switch (level) {
   312→        ScopeAccessLevel.all => 'All',
   313→        ScopeAccessLevel.campus => 'Campus',
   314→        ScopeAccessLevel.stage => 'Stage',
   315→        ScopeAccessLevel.class_ => 'Class',
   316→        ScopeAccessLevel.self => 'Self',
   317→        ScopeAccessLevel.linked => 'Linked',
   318→      };
   319→
   320→  String _scopeDescription(ScopeAccessLevel level) => switch (level) {
   321→        ScopeAccessLevel.all => 'Full access to all data in the organization',
   322→        ScopeAccessLevel.campus =>
   323→          'Can only access data within assigned campuses',
   324→        ScopeAccessLevel.stage =>
   325→          'Can only access data within assigned stages',
   326→        ScopeAccessLevel.class_ =>
   327→          'Can only access data within assigned classes and subjects',
   328→        ScopeAccessLevel.self => 'Can only view their own data',
   329→        ScopeAccessLevel.linked => 'Can only view linked children\'s data',
   330→      };
   331→}

   334→// VISUAL TREE — Campus → Stage → Class hierarchy for class-scoped roles
   335→// ═══════════════════════════════════════════════════════════════════════════════
   336→
   337→class _ClassScopeTree extends ConsumerWidget {
   338→  final Set<String> selectedStageIds;
   339→  final Set<String> selectedClassIds;
   340→  final void Function(String id) onStageToggle;
   341→  final void Function(String id) onClassToggle;
   342→  final void Function(String id, String name) onStageNameResolved;
   343→  final void Function(String id, String name) onClassNameResolved;
   344→
   345→  const _ClassScopeTree({
   346→    required this.selectedStageIds,
   347→    required this.selectedClassIds,
   348→    required this.onStageToggle,
   349→    required this.onClassToggle,
   350→    required this.onStageNameResolved,
   351→    required this.onClassNameResolved,
   352→  });
   353→
   354→  @override
   355→  Widget build(BuildContext context, WidgetRef ref) {
   356→    final treeAsync = ref.watch(scopeTreeProvider);
   357→
   358→    return treeAsync.when(
   359→      loading: () => const Center(child: KlasivoLoading()),
   360→      error: (e, _) => Center(child: Text('Error loading tree: $e')),
   361→      data: (tree) {
   362→        if (tree.isEmpty) {
   363→          return const KlasivoEmptyState(
   364→            icon: Icons.account_tree_rounded,
   365→            title: 'No organizational structure',
   366→            subtitle: 'Create stages and classes first',
   367→          );
   368→        }
   369→
   370→        return ListView(
   371→          padding: const EdgeInsets.symmetric(horizontal: 16),
   372→          children: tree
   373→              .expand((campus) => [
   374→                    // Campus header (non-selectable, just grouping)
   375→                    _CampusHeader(name: campus.name),
   376→                    // Stage → Class subtree
   377→                    ...campus.children.expand((stage) => [
   378→                          _StageTile(
   379→                            name: stage.name,
   380→                            id: stage.id,
   381→                            isSelected: selectedStageIds.contains(stage.id),
   382→                            onToggle: () {
   383→                              onStageNameResolved(stage.id, stage.name);
   384→                              onStageToggle(stage.id);
   385→                            },
   386→                            childCount: stage.children.length,
   387→                          ),
   388→                          // Class children
   389→                          ...stage.children.map((cls) => _ClassTile(
   390→                                name: cls.name,
   391→                                id: cls.id,
   392→                                isSelected:
   393→                                    selectedClassIds.contains(cls.id),
   394→                                onToggle: () {
   395→                                  onClassNameResolved(cls.id, cls.name);
   396→                                  onClassToggle(cls.id);
   397→                                },
   398→                                parentSelected:
   399→                                    selectedStageIds.contains(stage.id),
   400→                              )),
   401→                        ]),
   402→                  ])
   403→              .toList(),
   404→        );
   405→      },
   406→    );
   407→  }
   408→}
