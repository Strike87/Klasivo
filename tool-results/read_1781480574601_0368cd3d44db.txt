    25→    25→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    26→    26→
    27→    27→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    28→    28→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    29→    29→class _SensitiveFields {
    30→    30→  static const Set<String> keys = {
    31→    31→    'password',
    32→    32→    'passwordHash',
    33→    33→    'confirmPassword',
    34→    34→    'newPassword',
    35→    35→    'accessToken',
    36→    36→    'refreshToken',
    37→    37→    'idToken',
    38→    38→    'otp',
    39→    39→    'otpCode',
    40→    40→    'inviteCode',
    41→    41→    'secret',
    42→    42→    'apiKey',
    43→    43→    'authEmail', // Internal student auth email — PII
    44→    44→    'studentCode', // Sensitive — could be used to log in
    45→    45→  };
    46→    46→
    47→    47→  /// Returns true if the key contains any sensitive substring.
    48→    48→  static bool isSensitive(String key) {
    49→    49→    final lower = key.toLowerCase();
    50→    50→    for (final s in keys) {
    51→    51→      if (lower.contains(s.toLowerCase())) return true;
    52→    52→    }
    53→    53→    return false;
    54→    54→  }
    55→    55→
    56→    56→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    57→    57→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    58→    58→    return data.map((key, value) {
    59→    59→      if (isSensitive(key)) {
    60→    60→        return MapEntry(key, '[REDACTED]');
    61→    61→      }
    62→    62→      // Recursively sanitize nested maps
    63→    63→      if (value is Map<String, dynamic>) {
    64→    64→        return MapEntry(key, sanitize(value));
    65→    65→      }
    66→    66→      return MapEntry(key, value);
    67→    67→    });
    68→    68→  }
    69→    69→
    70→    70→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    71→    71→  static String sanitizeString(String input) {
    72→    72→    // Redact Bearer tokens
    73→    73→    var result = input.replaceAll(
    74→    74→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    75→    75→      'Bearer [REDACTED]',
    76→    76→    );
    77→    77→    // Redact long hex strings (likely tokens)
    78→    78→    result = result.replaceAll(
    79→    79→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    80→    80→      '[REDACTED_TOKEN]',
    81→    81→    );
    82→    82→    return result;
    83→    83→  }
    84→    84→}

   113→   113→      category: SentryCategories.auth,

   254→   254→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception
   255→   255→/// capture, and scope tags for every critical write.
   256→   256→///
   257→   257→/// Every Firestore exception will:
   258→   258→///   1. Reach Sentry

   261→   261→///   4. Include user role

   263→   263→class SentryFirestoreHelper {
   264→   264→  static final FirebaseFirestore _db = FirebaseFirestore.instance;
   265→   265→
   266→   266→  /// Wrapped `.set()` with full observability.
   267→   267→  static Future<void> docSet({
   268→   268→    required String collection,
   269→   269→    required String docId,
   270→   270→    required Map<String, dynamic> data,
   271→   271→    String? flow,
   272→   272→    String? step,
   273→   273→    SetOptions? options,
   274→   274→  }) async {
   275→   275→    final sanitizedData = _SensitiveFields.sanitize(data);
   276→   276→    KlasivoSentry.breadcrumb.firestore(
   277→   277→      'create',
   278→   278→      collection: collection,
   279→   279→      docId: docId,
   280→   280→      data: {'flow': flow, 'step': step},
   281→   281→    );
   282→   282→
   283→   283→    try {
   284→   284→      await _db.collection(collection).doc(docId).set(data, options);
   285→   285→      KlasivoSentry.breadcrumb.firestore(
   286→   286→        'create_success',
   287→   287→        collection: collection,
   288→   288→        docId: docId,
   289→   289→      );
   290→   290→    } catch (e, st) {
   291→   291→      await _captureFirestoreException(
   292→   292→        e,
   293→   293→        st,
   294→   294→        collection: collection,
   295→   295→        operation: 'set',
   296→   296→        docId: docId,
   297→   297→        flow: flow,
   298→   298→        step: step,
   299→   299→        dataPreview: sanitizedData,
   300→   300→      );
   301→   301→      rethrow;
   302→   302→    }
   303→   303→  }

   450→   450→  /// Capture a Firestore exception with full context tags.
   451→   451→  static Future<void> _captureFirestoreException(
   452→   452→    Object exception,
   453→   453→    StackTrace stackTrace, {
   454→   454→    required String collection,
   455→   455→    required String operation,
   456→   456→    String? docId,
   457→   457→    String? flow,
   458→   458→    String? step,
   459→   459→    Map<String, dynamic>? dataPreview,
   460→   460→  }) async {
   461→   461→    // Determine if this is a permission-denied error
   462→   462→    String? errorCode;
   463→   463→    if (exception is FirebaseException) {
   464→   464→      errorCode = exception.code;
   465→   465→    }
   466→   466→
   467→   467→    await Sentry.captureException(
   468→   468→      exception,
   469→   469→      stackTrace: stackTrace,
   470→   470→      withScope: (scope) {
   471→   471→        scope.setTag('collection', collection);
   472→   472→        scope.setTag('operation', operation);
   473→   473→        if (docId != null) scope.setTag('documentId', docId);
   474→   474→        if (flow != null) scope.setTag('flow', flow);
   475→   475→        if (step != null) scope.setTag('step', step);
   476→   476→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   477→   477→
   478→   478→        // Attach user role and org from current Sentry context
   479→   479→        final currentUser = FirebaseAuth.instance.currentUser;
   480→   480→        if (currentUser != null) {
   481→   481→          scope.setTag('auth_uid', currentUser.uid);
   482→   482→        }
   483→   483→
   484→   484→        if (dataPreview != null && dataPreview.isNotEmpty) {
   485→   485→          scope.setExtra('data_preview', dataPreview);
   486→   486→        }
   487→   487→      },
   488→   488→    );
   489→   489→  }
   490→   490→}

   492→   492→// ─── Guarded Async Operation ─────────────────────────────────────────────────
   493→   493→
   494→   494→/// Runs an async operation with automatic Sentry breadcrumb, exception capture,
   495→   495→/// and metadata attachment. This is the recommended way to wrap any async
   496→   496→/// operation that should never fail silently.
   497→   497→///
   498→   498→/// Example:
   499→   499→///   final result = await KlasivoSentry.runGuarded(
   500→   500→///     'create_exam',
   501→   501→///     () => examService.createExam(...),
   502→   502→///     category: 'firestore',
   503→   503→///   );
   504→   504→class KlasivoSentryGuard {
   505→   505→  /// Execute [operation] inside a Sentry-guarded wrapper.
   506→   506→  ///
   507→   507→  /// - Automatically adds a breadcrumb before execution.
   508→   508→  /// - On success: adds a success breadcrumb.
   509→   509→  /// - On failure: captures the exception with scope tags and rethrows.
   510→   510→  /// - The [category] defaults to the operation name but can be overridden.
   511→   511→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   512→   512→  ///   and the exception scope.
   513→   513→  static Future<T> runGuarded<T>(
   514→   514→    String operationName,
   515→   515→    Future<T> Function() operation, {
   516→   516→    String? category,
   517→   517→    Map<String, String>? tags,
   518→   518→    Map<String, dynamic>? data,
   519→   519→  }) async {
   520→   520→    final cat = category ?? 'operation';
   521→   521→
   522→   522→    Sentry.addBreadcrumb(Breadcrumb(
   523→   523→      category: cat,
   524→   524→      message: '${operationName}_started',
   525→   525→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   526→   526→      level: SentryLevel.info,
   527→   527→    ));
   528→   528→
   529→   529→    try {
   530→   530→      final result = await operation();
   531→   531→
   532→   532→      Sentry.addBreadcrumb(Breadcrumb(
   533→   533→        category: cat,
   534→   534→        message: '${operationName}_success',
   535→   535→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   536→   536→        level: SentryLevel.info,
   537→   537→      ));
   538→   538→
   539→   539→      return result;
   540→   540→    } catch (e, st) {
   541→   541→      Sentry.addBreadcrumb(Breadcrumb(
   542→   542→        category: cat,
   543→   543→        message: '${operationName}_failed',
   544→   544→        data: _SensitiveFields.sanitize({
   545→   545→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   546→   546→          ...?data,
   547→   547→        }),
   548→   548→        level: SentryLevel.error,
   549→   549→      ));
   550→   550→
   551→   551→      await Sentry.captureException(
   552→   552→        e,
   553→   553→        stackTrace: st,
   554→   554→        withScope: (scope) {
   555→   555→          scope.setTag('operation', operationName);
   556→   556→          tags?.forEach(scope.setTag);
   557→   557→          if (data != null) {
   558→   558→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   559→   559→          }
   560→   560→        },
   561→   561→      );
   562→   562→
   563→   563→      rethrow;
   564→   564→    }
   565→   565→  }
   566→   566→
   567→   567→  /// Execute [operation] with a Sentry transaction span.
   568→   568→  ///
   569→   569→  /// Creates a child span under the given [transaction], measures duration,
   570→   570→  /// and sets the span status based on success/failure.
   571→   571→  static Future<T> runWithSpan<T>(
   572→   572→    ISentrySpan transaction,
   573→   573→    String spanName,
   574→   574→    Future<T> Function() operation, {
   575→   575→    Map<String, String>? tags,
   576→   576→  }) async {
   577→   577→    final span = transaction.startChild(spanName);
   578→   578→
   579→   579→    try {
   580→   580→      final result = await operation();
   581→   581→      span.status = const SpanStatus.ok();
   582→   582→      return result;
   583→   583→    } catch (e, st) {
   584→   584→      span.status = const SpanStatus.internalError();
   585→   585→
   586→   586→      await Sentry.captureException(
   587→   587→        e,
   588→   588→        stackTrace: st,
   589→   589→        withScope: (scope) {
   590→   590→          scope.setTag('span', spanName);
   591→   591→          tags?.forEach(scope.setTag);
   592→   592→        },
   593→   593→      );
   594→   594→
   595→   595→      rethrow;
   596→   596→    } finally {
   597→   597→      await span.finish();
   598→   598→    }
   599→   599→  }
   600→   600→}

   604→   604→/// Manages Sentry user context — attached on sign-in, updated on role/org
   605→   605→/// changes, cleared on logout.
   606→   606→///
   607→   607→/// Attaches:
   608→   608→///   - uid
   609→   609→///   - email
   610→   610→///   - role (tag)

   657→   657→      scope.setTag('build_number', buildNumber);

   711→   711→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   712→   712→
   713→   713→/// Documents the doc ID strategy for every user creation path.
   714→   714→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   715→   715→///
   716→   716→/// | Path                  | Collection | Doc ID     | Status    |
   717→   717→/// |-----------------------|-----------|------------|-----------|
   718→   718→/// | Owner (email)         | users     | user.uid   | OK        |
   719→   719→/// | Owner (Google)        | users     | user.uid   | OK        |
   720→   720→/// | Teacher (invite)      | users     | user.uid   | OK        |
   721→   721→/// | Teacher (Google)      | users     | user.uid   | OK        |
   722→   722→/// | Parent (email)        | users     | user.uid   | OK        |
   723→   723→/// | Parent (Google)       | users     | user.uid   | OK        |
   724→   724→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   725→   725→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   726→   726→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   727→   727→///
   728→   728→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   729→   729→/// `.doc(user.uid)`, which means:
   730→   730→///   1. The doc ID doesn't match the auth UID
   731→   731→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   732→   732→///   3. QR enrollment is 100% broken in production
   733→   733→class SentryDocIdAudit {
   734→   734→  /// Log a doc ID audit breadcrumb for any user creation path.
   735→   735→  static void logUserCreation({
   736→   736→    required String flow,
   737→   737→    required String collection,
   738→   738→    required String docIdStrategy, // 'uid' or 'auto_id'
   739→   739→    required String actualDocId,
   740→   740→    String? authUid,
   741→   741→  }) {
   742→   742→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   743→   743→    final isAutoId = docIdStrategy == 'auto_id';
   744→   744→
   745→   745→    Sentry.addBreadcrumb(Breadcrumb(
   746→   746→      category: 'doc_id_audit',
   747→   747→      message: 'user_doc_created',
   748→   748→      data: _SensitiveFields.sanitize({
   749→   749→        'flow': flow,
   750→   750→        'collection': collection,
   751→   751→        'docIdStrategy': docIdStrategy,
   752→   752→        'actualDocId': actualDocId,
   753→   753→        if (authUid != null) 'authUid': authUid,
   754→   754→        'idMismatch': isMismatch,
   755→   755→        'autoIdUsed': isAutoId,
   756→   756→        'brokenByRules': isAutoId, // auto-id always fails security rules
   757→   757→      }),
   758→   758→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   759→   759→    ));
   760→   760→
   761→   761→    if (isMismatch || isAutoId) {
   762→   762→      Sentry.captureMessage(
   763→   763→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   764→   764→        '(docId=$actualDocId, authUid=$authUid). '
   765→   765→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   766→   766→        level: SentryLevel.warning,
   767→   767→      );
   768→   768→    }
   769→   769→  }
   770→   770→}

   774→   774→/// Top-level entry point for all Sentry observability in Klasivo.

   776→   776→/// Usage:

   779→   779→///   KlasivoSentry.runGuarded('create_exam', () => ...);

   783→   783→

   787→   787→  /// Firestore operation helper — use for all critical Firestore writes.
   788→   788→  static final firestore = SentryFirestoreHelper();

   790→   790→  /// Guarded operation runner — use to wrap any async operation.
   791→   791→  static final guard = KlasivoSentryGuard();
   792→   792→
   793→   793→  /// User context manager — use on sign-in, role change, org change, logout.
   794→   794→  static final userContext = SentryUserContext();

   796→   796→  /// Transaction factory — use to start named transactions for flows.
   797→   797→  static final transactions = SentryTransactions();
   798→   798→
   799→   799→  /// Doc ID audit — use when creating any user document.
   800→   800→  static final docIdAudit = SentryDocIdAudit();
   801→   801→
   802→   802→  /// Convenience method for runGuarded.
   803→   803→  static Future<T> runGuarded<T>(
   804→   804→    String operationName,
   805→   805→    Future<T> Function() operation, {
   806→   806→    String? category,
   807→   807→    Map<String, String>? tags,
   808→   808→    Map<String, dynamic>? data,

   819→   819→  /// Capture an exception with optional scope tags.
   820→   820→  static Future<SentryId> captureException(
   821→   821→    Object exception, {
   822→   822→    StackTrace? stackTrace,
   823→   823→    Map<String, String>? tags,
   824→   824→    Map<String, dynamic>? extras,
   825→   825→  }) {
   826→   826→    return Sentry.captureException(
   827→   827→      exception,
   828→   828→      stackTrace: stackTrace,
   829→   829→      withScope: (scope) {
   830→   830→        tags?.forEach(scope.setTag);
   831→   831→        extras?.forEach((key, value) {
   832→   832→          scope.setExtra(key, _SensitiveFields.isSensitive(key) ? '[REDACTED]' : value);
   833→   833→        });
   834→   834→      },
   835→   835→    );
   836→   836→  }
   837→   837→
   838→   838→  /// Capture a message with level.
   839→   839→  static Future<SentryId> captureMessage(
   840→   840→    String message, {
   841→   841→    SentryLevel level = SentryLevel.info,
   842→   842→    Map<String, String>? tags,
   843→   843→  }) {
   844→   844→    return Sentry.captureMessage(
   845→   845→      message,
   846→   846→      level: level,
   847→   847→    );
   848→   848→  }
