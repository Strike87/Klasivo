   237→// ─── Auth Change Notifier for GoRouter refresh ──────────────────────────────
   238→
   239→class AuthChangeNotifier extends ChangeNotifier {
   240→  StreamSubscription<User?>? _firebaseSub;
   241→  StreamSubscription? _hiveSub;
   242→
   243→  AuthChangeNotifier() {
   244→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   245→      (_) => notifyListeners(),
   246→      onError: (_) => notifyListeners(),
   247→    );
   248→    _startHiveWatch();
   249→  }
   250→
   251→  void _startHiveWatch() {
   252→    // Use Hive's built-in watch() instead of polling every 500ms
   253→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   254→      notifyListeners();
   255→    });
   256→  }
   257→
   258→  @override
   259→  void dispose() {
   260→    _firebaseSub?.cancel();
   261→    _hiveSub?.cancel();
   262→    super.dispose();
   263→  }
   264→}
   265→
   266→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   267→  final notifier = AuthChangeNotifier();
   268→  ref.onDispose(() => notifier.dispose());
   269→  return notifier;
   270→});
   271→
   272→// ─── GoRouter with Auth Guards & v1.7 Complete Navigation ─────────────────────
   273→
   274→final routerProvider = Provider<GoRouter>((ref) {
   275→  final notifier = ref.watch(authChangeNotifierProvider);
   276→
   277→  return GoRouter(
   278→    initialLocation: '/',
   279→    debugLogDiagnostics: true,
   280→    refreshListenable: notifier,
   281→    redirect: (context, state) {
   282→      final box = Hive.box(AppConstants.authBox);
   283→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   284→      final userRole = box.get('userRole', defaultValue: '') as String;
   285→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: false) as bool;
   286→      final mustChangePassword = box.get('mustChangePassword', defaultValue: false) as bool;
   287→      final isOnChangePassword = state.matchedLocation == '/change-password';
   288→
   289→      // Force redirect to change password if required
   290→      if (isLoggedIn && mustChangePassword && !isOnChangePassword) {
   291→        return '/change-password';
   292→      }
   293→      // Prevent navigating away from change password if forced
   294→      if (isLoggedIn && mustChangePassword && isOnChangePassword) {
   295→        return null; // Allow staying on change password
   296→      }
   297→      // If no longer required, redirect away from change password
   298→      if (isLoggedIn && !mustChangePassword && isOnChangePassword) {
   299→        // Management roles → /dashboard; student → /student; parent → /parent
   300→        if (KlasivoRole.managementRoles.contains(userRole)) return '/dashboard';
   301→        if (userRole == KlasivoRole.student) return '/student';
   302→        if (userRole == KlasivoRole.parent) return '/parent';
   303→        return '/dashboard'; // Fallback for unknown roles
   304→      }
   305→
   306→      final isOnSplash = state.matchedLocation == '/';
   307→      final isOnAuth = state.matchedLocation.startsWith('/auth');
   308→      final isOnWelcome = state.matchedLocation == '/welcome';
   309→      final isOnDashboard = state.matchedLocation.startsWith('/dashboard') ||
   310→          state.matchedLocation.startsWith('/academic') ||
   311→          state.matchedLocation.startsWith('/people') ||
   312→          state.matchedLocation.startsWith('/inbox') ||
   313→          state.matchedLocation.startsWith('/settings') ||
   314→          state.matchedLocation.startsWith('/teacher') ||
   315→          state.matchedLocation.startsWith('/lms');
   316→      final isOnStudent = state.matchedLocation.startsWith('/student');
   317→      final isOnParent = state.matchedLocation.startsWith('/parent') ||
   318→          state.matchedLocation.startsWith('/auth/parent');
   319→
   320→      // Splash → redirect based on auth state
   321→      if (isOnSplash) {
   322→        if (isLoggedIn && userRole.isNotEmpty) {
   323→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   324→            return '/welcome';
   325→          }
   326→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   327→            return '/dashboard';
   328→          }
   329→          if (userRole == AppConstants.roleStudent) return '/student';
   330→          if (userRole == AppConstants.roleParent) return '/parent';
   331→        }
   332→        return '/auth';
   333→      }
   334→
   335→      // Welcome screen — only accessible to logged-in owners who haven't completed setup
   336→      if (isOnWelcome) {
   337→        if (!isLoggedIn) return '/auth';
   338→        if (userRole != AppConstants.roleOwner) return '/dashboard';
   339→        if (hasCompletedSetup) return '/dashboard';
   340→        return null;
   341→      }
   342→
   343→      // Auth screens → redirect if already logged in
   344→      if (isOnAuth) {
   345→        if (isLoggedIn && userRole.isNotEmpty) {
   346→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   347→            return '/welcome';
   348→          }
   349→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   350→            return '/dashboard';
   351→          }
   352→          if (userRole == AppConstants.roleStudent) return '/student';
   353→          if (userRole == AppConstants.roleParent) return '/parent';
   354→        }
   355→        return null;
   356→      }
   357→
   358→      // Protected screens → require login
   359→      if (isOnDashboard || isOnStudent || isOnParent) {
   360→        if (!isLoggedIn) return '/auth';
   361→        if (userRole.isEmpty) return '/auth';
   362→
   363→        if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   364→          return '/welcome';
   365→        }
   366→
   367→        // ─── Feature Flag Gates ───────────────────────────────────────────
   368→        // LMS routes require lms flag
   369→        if (state.matchedLocation.startsWith('/lms')) {
   370→          final flagService = ref.read(featureFlagServiceProvider);
   371→          if (!flagService.isEnabled(FeatureFlags.lms)) {
   372→            return '/dashboard';
   373→          }
   374→        }
   375→
   376→        // Parent portal requires parentPortal flag
   377→        if (state.matchedLocation.startsWith('/parent') ||
   378→            state.matchedLocation.startsWith('/auth/parent')) {
   379→          final flagService = ref.read(featureFlagServiceProvider);
   380→          if (!flagService.isEnabled(FeatureFlags.parentPortal)) {
   381→            if (userRole == AppConstants.roleParent) return '/auth';
   382→            return '/dashboard';
   383→          }
   384→        }
   385→
   386→        if ((userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) &&
   387→            isOnDashboard) {
   388→          return null;
   389→        }
   390→        if (userRole == AppConstants.roleStudent && isOnStudent) {
   391→          return null;
   392→        }
   393→        if (userRole == AppConstants.roleParent && isOnParent) {
   394→          return null;
   395→        }
   396→
   397→        if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   398→          return '/dashboard';
   399→        }
   400→        if (userRole == AppConstants.roleStudent) return '/student';
   401→        if (userRole == AppConstants.roleParent) return '/parent';
   402→      }
   403→
   404→      return null;
   405→    },
