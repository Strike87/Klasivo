     1→/**
     2→ * Klasivo — API Gateway v1 (Express on Firebase Functions)
     3→ *
     4→ * Central backend gateway for api.klasivo.app.
     5→ * All sensitive operations route through here — the Flutter app never
     6→ * touches LiveKit secrets, Resend keys, or admin operations directly.
     7→ *
     8→ * Architecture:
     9→ *   Flutter App → api.klasivo.app/v1/* → Firebase Functions → LiveKit/Resend/Firebase
    10→ *
    11→ * Routes (v1):
    12→ *   GET  /v1/health                 → Health check + service status
    13→ *   POST /v1/livekit/token          → Generate LiveKit JWT
    14→ *   POST /v1/livekit/remove         → Remove participant from room
    15→ *   POST /v1/livekit/mute           → Mute a participant (teacher only)
    16→ *   POST /v1/livekit/endRoom        → End a live class room
    17→ *   POST /v1/storage/upload-url     → Generate signed upload URL
    18→ *   POST /v1/analytics/event        → Record server-side analytics event
    19→ *   GET  /v1/admin/users            → List users in organization
    20→ *   GET  /v1/admin/schools          → List organizations
    21→ *   GET  /v1/admin/reports/summary  → Organization summary stats
    22→ *   GET  /v1/docs                   → OpenAPI/Swagger JSON
    23→ *
    24→ * Security:
    25→ *   - Bearer token auth (Firebase ID token) on all mutation endpoints
    26→ *   - Admin endpoints require teacher/owner/admin role
    27→ *   - Rate limiting at Cloudflare level
    28→ *   - Security headers via Firebase Hosting config
    29→ *   - Audit logging for all important actions
    30→ */

    73→  });

    80→// ─── Auth middleware ────────────────────────────────────────────
    81→
    82→async function verifyAuthToken(
    83→  req: Request,
    84→  res: Response,
    85→  next: NextFunction,
    86→): Promise<void> {
    87→  const authHeader = req.headers.authorization;
    88→  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    89→    res.status(401).json({
    90→      error: 'Missing or invalid Authorization header. Use: Bearer <Firebase ID token>',
    91→    });
    92→    return;
    93→  }
    94→
    95→  const idToken = authHeader.split('Bearer ')[1];
    96→  if (!idToken) {
    97→    res.status(401).json({ error: 'Empty bearer token.' });
    98→    return;
    99→  }
   100→
   101→  try {
   102→    const decodedToken = await admin.auth().verifyIdToken(idToken);
   103→    req.user = decodedToken;
   104→
   105→    const userDoc = await admin
   106→      .firestore()
   107→      .collection('users')
   108→      .doc(decodedToken.uid)
   109→      .get();
   110→    if (userDoc.exists) {
   111→      req.userRole = userDoc.data()?.['role'] as string | undefined;
   112→      req.userOrgId = userDoc.data()?.['organizationId'] as string | undefined;
   113→      req.userName = userDoc.data()?.['fullName'] as string | undefined;
   114→    }
   115→
   116→    next();
   117→  } catch (err: unknown) {
   118→    const msg = err instanceof Error ? err.message : String(err);
   119→    console.error(`Token verification failed: ${msg}`);
   120→    res.status(401).json({ error: 'Invalid or expired token.' });
   121→  }
   122→}

   126→function requireAdmin(req: Request, res: Response, next: NextFunction): void {
   127→  if (!req.user) {
   128→    res.status(401).json({ error: 'Authentication required.' });
   129→    return;
   130→  }
   131→  if (!['teacher', 'owner', 'admin'].includes(req.userRole ?? '')) {
   132→    res.status(403).json({ error: 'Admin access required.' });
   133→    return;
   134→  }
   135→  next();
   136→}

   138→// ─── Audit logging middleware ───────────────────────────────────
   139→
   140→async function auditLog(
   141→  req: Request,
   142→  _res: Response,
   143→  next: NextFunction,
   144→): Promise<void> {
   145→  // Only audit mutation requests from authenticated users
   146→  if (req.method === 'GET' || !req.user) {
   147→    next();
   148→    return;
   149→  }
   150→
   151→  try {
   152→    const db = admin.firestore();
   153→    await db.collection('audit_logs').add({
   154→      actorId: req.user.uid,                                             // legacy — remove in Phase 3
   155→      actorName: req.userName ?? req.user.email ?? req.user.uid,         // legacy — remove in Phase 3
   156→      actorRole: req.userRole ?? 'unknown',                              // legacy — keep for backward compat
   157→      performedBy: req.user.uid,                                         // canonical actor field
   158→      performedByRole: req.userRole ?? 'unknown',                        // Phase 1: add
   159→      performedByOrgId: req.userOrgId ?? null,                           // Phase 1: add
   160→      organizationId: req.userOrgId ?? null,
   161→      action: `${req.method} ${req.path}`,
   162→      resource: req.path,
   163→      requestBody: _sanitizeForAudit(req.body),
   164→      ipAddress: req.ip ?? req.headers['x-forwarded-for'] ?? 'unknown',
   165→      userAgent: req.headers['user-agent'] ?? 'unknown',
   166→      timestamp: admin.firestore.FieldValue.serverTimestamp(),
   167→    });
   168→  } catch (err: unknown) {
   169→    // Audit logging failure should never block the request
   170→    console.warn(`Audit log write failed: ${err instanceof Error ? err.message : String(err)}`);
   171→  }
   172→
   173→  next();
   174→}
   175→
   176→/** Remove sensitive fields from request body before logging */
   177→function _sanitizeForAudit(body: Record<string, unknown> | undefined): Record<string, unknown> {
   178→  if (!body || typeof body !== 'object') return {};
   179→  const sanitized = { ...body };
   180→  const sensitiveKeys = ['password', 'token', 'secret', 'apiKey', 'creditCard', 'cvv'];
   181→  for (const key of Object.keys(sanitized)) {
   182→    if (sensitiveKeys.some((sk) => key.toLowerCase().includes(sk.toLowerCase()))) {
   183→      sanitized[key] = '[REDACTED]';
   184→    }
   185→  }
   186→  return sanitized;
   187→}

   194→/**

   198→async function _logTokenDenied(
   199→  uid: string,

   201→  orgId: string,

   203→  reason: string,
   204→  message: string,

   207→    await admin.firestore().collection('audit_logs').add({
   208→      organizationId: orgId,
   209→      performedBy: uid,

   211→      performedByOrgId: orgId,
   212→      action: 'livekit_token_denied',
   213→      targetType: 'livekit_room',
   214→      targetId: roomId,
   215→      metadata: { reason, message },
   216→      timestamp: admin.firestore.FieldValue.serverTimestamp(),
   217→    });
   218→  } catch {
   219→    console.warn('Failed to write livekit_token_denied audit event');
   220→  }
   221→}

   269→app.post(
   270→  '/v1/livekit/token',
   271→  verifyAuthToken,
   272→  async (req: Request, res: Response) => {
   273→    initSentry();
   274→    const scope = (req as any).sentryScope as Sentry.Scope | undefined;
   275→    scope?.setTag('service', 'livekit');
   276→    scope?.setTag('endpoint', '/v1/livekit/token');
   277→
   278→    if (!req.user) {
   279→      res.status(401).json({ error: 'Authentication required.' });
   280→      return;
   281→    }
   282→
   283→    const { roomId, displayName } = req.body ?? {};
   284→    // roomName/isTeacher: REMOVED — derived from room document and caller claims
   285→
   286→    if (!roomId || typeof roomId !== 'string') {
   287→      res.status(400).json({ error: 'roomId is required and must be a string.' });
   288→      return;
   289→    }
   290→
   291→    try {
   292→      // ── Load room document ──────────────────────────────────
   293→      const roomDoc = await admin.firestore().collection('livekit_rooms').doc(roomId).get();
   294→
   295→      if (!roomDoc.exists) {
   296→        res.status(404).json({ error: 'Room not found.' });
   297→        return;
   298→      }
   299→
   300→      const roomData = roomDoc.data()!;
   301→      const roomName = roomData['name'] as string;
   302→      if (!roomName) {
   303→        res.status(500).json({ error: 'Room document is missing a name field.' });
   304→        return;
   305→      }
   306→
   307→      // Room name sanitization
   308→      const sanitizedRoomName = roomName.replace(/[^a-zA-Z0-9_-]/g, '');
   309→      if (sanitizedRoomName.length === 0 || sanitizedRoomName.length > 128) {
   310→        res.status(500).json({ error: 'Invalid room name in document.' });
   311→        return;
   312→      }
   313→
   314→      // ── Org boundary check ──────────────────────────────────
   315→      const roomOrgId = roomData['organizationId'] as string;
   316→      if (!verifyOrgBoundary(req.userOrgId ?? '', roomOrgId, req.userRole ?? '')) {
   317→        await _logTokenDenied(req.user!.uid, req.userRole ?? 'unknown', req.userOrgId ?? '', roomId, 'org_boundary', 'Cross-org access denied');
   318→        Sentry.captureMessage(`Cross-org LiveKit token attempt (API): caller=${req.userOrgId} room=${roomOrgId}`);
   319→        res.status(403).json({ error: 'You can only join rooms in your organization.' });
   320→        return;
   321→      }
   322→
   323→      // ── Scope authorization ────────────────────────────────
   324→      const scopeAccessLevel = (req.user?.token?.['scopeAccessLevel'] as string) || '';
   325→      const callerScope: Record<string, string[]> = {};
   326→
   327→      // Load caller's scope arrays from user doc (already loaded by verifyAuthToken middleware)
   328→      const callerDoc = await admin.firestore().collection('users').doc(req.user!.uid).get();
   329→      if (callerDoc.exists) {
   330→        const callerData = callerDoc.data()!;
   331→        callerScope['campusIds'] = callerData['campusIds'] as string[] || [];
   332→        callerScope['stageIds'] = callerData['stageIds'] as string[] || [];
   333→        callerScope['classIds'] = callerData['classIds'] as string[] || [];
   334→        callerScope['subjectIds'] = callerData['subjectIds'] as string[] || [];
   335→        callerScope['studentIds'] = callerData['studentIds'] as string[] || [];
   336→      }
   337→
   338→      const scopeResult = verifyScopeAuthorization(scopeAccessLevel, callerScope, roomData as Record<string, unknown>);
   339→      if (!scopeResult.authorized) {
   340→        await _logTokenDenied(req.user!.uid, req.userRole ?? 'unknown', req.userOrgId ?? '', roomId, scopeResult.reason ?? 'unknown', scopeResult.message ?? 'Scope authorization failed');
   341→        Sentry.captureMessage(`LiveKit scope denial (API): uid=${req.user!.uid} room=${roomId} reason=${scopeResult.reason}`);
   342→        res.status(403).json({ error: scopeResult.message || 'You are not authorized to access this room.' });
   343→        return;
   344→      }
   345→
   346→      // ── Server-side role determination ──────────────────────
   347→      const isTeacherOrAbove = LIVEKIT_ADMIN_ROLES.includes(req.userRole as any);
   348→
   349→      const { AccessToken } = await import('livekit-server-sdk');
   350→
   351→      const token = new AccessToken(
   352→        LIVEKIT_API_KEY.value(),
   353→        LIVEKIT_API_SECRET.value(),
   354→        { identity: req.user.uid, name: displayName || req.user.uid },
   355→      );
   356→
   357→      token.addGrant({
   358→        room: sanitizedRoomName,
   359→        roomJoin: true,
   360→        canPublish: true,
   361→        canSubscribe: true,
   362→        canPublishData: true,
   363→        roomAdmin: isTeacherOrAbove,
   364→      });
   365→
   366→      const jwt = await token.toJwt();
   367→
   368→      Sentry.addBreadcrumb({
   369→        category: 'livekit',
   370→        message: `Token generated for room ${sanitizedRoomName}, roomAdmin=${isTeacherOrAbove}`,
   371→        level: 'info',
   372→      });
   373→
   374→      res.json({ token: jwt });
   375→    } catch (err: unknown) {
   376→      const msg = err instanceof Error ? err.message : String(err);
   377→      console.error(`LiveKit token generation failed: ${msg}`);
   378→      Sentry.captureException(err);
   379→      res.status(500).json({ error: 'Failed to generate token.' });
   380→    }
   381→  },
   382→);

   384→app.post(
   385→  '/v1/livekit/remove',
   386→  verifyAuthToken,
   387→  requireAdmin,
   388→  async (req: Request, res: Response) => {
   389→    initSentry();
   390→    const scope = (req as any).sentryScope as Sentry.Scope | undefined;
   391→    scope?.setTag('service', 'livekit');
   392→    scope?.setTag('endpoint', '/v1/livekit/remove');
   393→
   394→    const { roomName, participantIdentity, roomId } = req.body ?? {};
   395→
   396→    if (!roomName || !participantIdentity || !roomId) {
   397→      res.status(400).json({ error: 'roomName, participantIdentity, and roomId are required.' });
   398→      return;
   399→    }
   400→
   401→    try {
   402→      const { RoomServiceClient } = await import('livekit-server-sdk');
   403→      const db = admin.firestore();
   404→      const roomDoc = await db.collection('livekit_rooms').doc(roomId).get();
   405→
   406→      if (!roomDoc.exists) {
   407→        res.status(404).json({ error: 'Room not found.' });
   408→        return;
   409→      }
   410→
   411→      const roomOrgId = roomDoc.data()?.['organizationId'] as string;
   412→      if (roomOrgId !== req.userOrgId) {
   413→        res.status(403).json({ error: 'You can only remove participants from rooms in your organization.' });
   414→        return;
   415→      }
   416→
   417→      const livekitUrl =
   418→        (roomDoc.data()?.['metadata']?.['livekitUrl'] as string) ??
   419→        'https://klasivo.livekit.cloud';
   420→
   421→      const roomService = new RoomServiceClient(
   422→        livekitUrl.replace('wss://', 'https://'),
   423→        LIVEKIT_API_KEY.value(),
   424→        LIVEKIT_API_SECRET.value(),
   425→      );
   426→
   427→      try {
   428→        await roomService.removeParticipant(roomName, participantIdentity);
   429→      } catch (err: unknown) {
   430→        console.warn(`Failed to remove participant ${participantIdentity}: ${err instanceof Error ? err.message : String(err)}`);
   431→      }
   432→
   433→      // Update attendance
   434→      try {
   435→        await db.collection('livekit_rooms').doc(roomId).collection('attendance').doc(participantIdentity).update({
   436→          leftAt: admin.firestore.FieldValue.serverTimestamp(),
   437→          updatedAt: admin.firestore.FieldValue.serverTimestamp(),
   438→          removedBy: req.user!.uid,
   439→          removedAt: admin.firestore.FieldValue.serverTimestamp(),
   440→        });
   441→      } catch (err: unknown) {
   442→        console.warn(`Could not update attendance for ${participantIdentity}: ${err}`);
   443→      }
   444→
   445→      // Create in-app notification
   446→      try {
   447→        await db.collection('notifications').add({
   448→          userId: participantIdentity,
   449→          type: 'removed_from_class',
   450→          title: 'Removed from Class',
   451→          body: `You have been removed from the live class "${roomDoc.data()?.['name'] ?? roomName}".`,
   452→          data: { roomId, roomName },
   453→          read: false,
   454→          createdAt: admin.firestore.FieldValue.serverTimestamp(),
   455→        });
   456→      } catch (err: unknown) {
   457→        console.warn(`Could not create removal notification: ${err}`);
   458→      }
   459→
   460→      res.json({ success: true, removedIdentity: participantIdentity });
   461→    } catch (err: unknown) {
   462→      const msg = err instanceof Error ? err.message : String(err);
   463→      console.error(`Remove participant failed: ${msg}`);
   464→      Sentry.captureException(err);
   465→      res.status(500).json({ error: 'Failed to remove participant.' });
   466→    }
   467→  },

   594→// ═══════════════════════════════════════════════════════════════════
   595→
   596→/**
   597→ * POST /v1/storage/upload-url
   598→ *
   599→ * Generates a signed URL for the client to upload a file directly to
   600→ * Firebase Storage. The client never needs storage credentials.
   601→ *
   602→ * Request body:
   603→ *   { "filePath": "organizations/abc123/logo.png", "contentType": "image/png" }
   604→ *
   605→ * Response:
   606→ *   { "uploadUrl": "https://storage.googleapis.com/...", "expiresAt": "..." }
   607→ */
   608→app.post(
   609→  '/v1/storage/upload-url',
   610→  verifyAuthToken,
   611→  async (req: Request, res: Response) => {
   612→    initSentry();
   613→    const scope = (req as any).sentryScope as Sentry.Scope | undefined;
   614→    scope?.setTag('service', 'storage');
   615→    scope?.setTag('endpoint', '/v1/storage/upload-url');
   616→
   617→    const { filePath, contentType } = req.body ?? {};
   618→
   619→    if (!filePath || typeof filePath !== 'string') {
   620→      res.status(400).json({ error: 'filePath is required.' });
   621→      return;
   622→    }
   623→
   624→    // Validate path doesn't escape allowed prefixes
   625→    const allowedPrefixes = [
   626→      `users/${req.user!.uid}/`,
   627→      `organizations/${req.userOrgId ?? ''}/`,
   628→      'exams/',
   629→      'materials/',
   630→      'submissions/',
   631→    ];
   632→
   633→    const isAllowed = allowedPrefixes.some((prefix) => filePath.startsWith(prefix));
   634→    if (!isAllowed) {
   635→      res.status(403).json({
   636→        error: 'File path must start with an allowed prefix.',
   637→        allowedPrefixes: ['users/{uid}/', 'organizations/{orgId}/', 'exams/', 'materials/', 'submissions/'],
   638→      });
   639→      return;
   640→    }
   641→
   642→    // Prevent path traversal
   643→    if (filePath.includes('..') || filePath.includes('//')) {
   644→      res.status(400).json({ error: 'Invalid file path.' });
   645→      return;
   646→    }
   647→
   648→    try {
   649→      const bucket = admin.storage().bucket();
   650→      const file = bucket.file(filePath);
   651→
   652→      const expiresInMinutes = 15;
   653→      const expiresAt = new Date(Date.now() + expiresInMinutes * 60 * 1000);
   654→
   655→      const [uploadUrl] = await file.getSignedUrl({
   656→        action: 'write',
   657→        expires: expiresAt,
   658→        contentType: contentType || 'application/octet-stream',
   659→      });
   660→
   661→      // Also generate a read URL
   662→      const [downloadUrl] = await file.getSignedUrl({
   663→        action: 'read',
   664→        expires: expiresAt,
   665→      });
   666→
   667→      res.json({
   668→        uploadUrl,
   669→        downloadUrl,
   670→        filePath,
   671→        expiresAt: expiresAt.toISOString(),
   672→      });
   673→    } catch (err: unknown) {
   674→      const msg = err instanceof Error ? err.message : String(err);
   675→      console.error(`Signed URL generation failed: ${msg}`);
   676→      Sentry.captureException(err);
   677→      res.status(500).json({ error: 'Failed to generate upload URL.' });
   678→    }
   679→  },
