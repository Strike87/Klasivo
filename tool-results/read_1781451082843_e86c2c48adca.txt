     1→/**
     2→ * Klasivo — API Gateway v1 (Express on Firebase Functions)
     3→ *
     4→ * Central backend gateway for api.klasivo.app.
     5→ * All sensitive operations route through here — the Flutter app never
     6→ * touches LiveKit secrets, Resend keys, or admin operations directly.
     7→ *
     8→ * Architecture:
     9→ *   Flutter App → api.klasivo.app/v1/* → Firebase Functions → LiveKit/Resend/Firebase
    10→ *
    11→ * Routes (v1):
    12→ *   GET  /v1/health                 → Health check + service status
    13→ *   POST /v1/livekit/token          → Generate LiveKit JWT
    14→ *   POST /v1/livekit/remove         → Remove participant from room
    15→ *   POST /v1/livekit/mute           → Mute a participant (teacher only)
    16→ *   POST /v1/livekit/endRoom        → End a live class room
    17→ *   POST /v1/storage/upload-url     → Generate signed upload URL
    18→ *   POST /v1/analytics/event        → Record server-side analytics event
    19→ *   GET  /v1/admin/users            → List users in organization
    20→ *   GET  /v1/admin/schools          → List organizations
    21→ *   GET  /v1/admin/reports/summary  → Organization summary stats
    22→ *   GET  /v1/docs                   → OpenAPI/Swagger JSON
    23→ *
    24→ * Security:
    25→ *   - Bearer token auth (Firebase ID token) on all mutation endpoints
    26→ *   - Admin endpoints require teacher/owner/admin role
    27→ *   - Rate limiting at Cloudflare level
    28→ *   - Security headers via Firebase Hosting config
    29→ *   - Audit logging for all important actions
    30→ */

    65→// ─── Auth middleware ────────────────────────────────────────────
    66→
    67→async function verifyAuthToken(
    68→  req: Request,
    69→  res: Response,
    70→  next: NextFunction,
    71→): Promise<void> {
    72→  const authHeader = req.headers.authorization;
    73→  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    74→    res.status(401).json({
    75→      error: 'Missing or invalid Authorization header. Use: Bearer <Firebase ID token>',
    76→    });
    77→    return;
    78→  }
    79→
    80→  const idToken = authHeader.split('Bearer ')[1];
    81→  if (!idToken) {
    82→    res.status(401).json({ error: 'Empty bearer token.' });
    83→    return;
    84→  }
    85→
    86→  try {
    87→    const decodedToken = await admin.auth().verifyIdToken(idToken);
    88→    req.user = decodedToken;
    89→
    90→    const userDoc = await admin
    91→      .firestore()
    92→      .collection('users')
    93→      .doc(decodedToken.uid)
    94→      .get();
    95→    if (userDoc.exists) {
    96→      req.userRole = userDoc.data()?.['role'] as string | undefined;
    97→      req.userOrgId = userDoc.data()?.['organizationId'] as string | undefined;
    98→      req.userName = userDoc.data()?.['fullName'] as string | undefined;
    99→    }
   100→
   101→    next();
   102→  } catch (err: unknown) {
   103→    const msg = err instanceof Error ? err.message : String(err);
   104→    console.error(`Token verification failed: ${msg}`);
   105→    res.status(401).json({ error: 'Invalid or expired token.' });
   106→  }
   107→}

   111→function requireAdmin(req: Request, res: Response, next: NextFunction): void {
   112→  if (!req.user) {
   113→    res.status(401).json({ error: 'Authentication required.' });
   114→    return;
   115→  }
   116→  if (!['teacher', 'owner', 'admin'].includes(req.userRole ?? '')) {
   117→    res.status(403).json({ error: 'Admin access required.' });
   118→    return;
   119→  }
   120→  next();
   121→}

   123→// ─── Audit logging middleware ───────────────────────────────────
   124→
   125→async function auditLog(
   126→  req: Request,
   127→  _res: Response,
   128→  next: NextFunction,
   129→): Promise<void> {
   130→  // Only audit mutation requests from authenticated users
   131→  if (req.method === 'GET' || !req.user) {
   132→    next();
   133→    return;
   134→  }
   135→
   136→  try {
   137→    const db = admin.firestore();
   138→    await db.collection('audit_logs').add({
   139→      actorId: req.user.uid,                                             // legacy — remove in Phase 3
   140→      actorName: req.userName ?? req.user.email ?? req.user.uid,         // legacy — remove in Phase 3
   141→      actorRole: req.userRole ?? 'unknown',                              // legacy — keep for backward compat
   142→      performedBy: req.user.uid,                                         // canonical actor field
   143→      performedByRole: req.userRole ?? 'unknown',                        // Phase 1: add
   144→      performedByOrgId: req.userOrgId ?? null,                           // Phase 1: add
   145→      organizationId: req.userOrgId ?? null,
   146→      action: `${req.method} ${req.path}`,
   147→      resource: req.path,
   148→      requestBody: _sanitizeForAudit(req.body),
   149→      ipAddress: req.ip ?? req.headers['x-forwarded-for'] ?? 'unknown',
   150→      userAgent: req.headers['user-agent'] ?? 'unknown',
   151→      timestamp: admin.firestore.FieldValue.serverTimestamp(),
   152→    });
   153→  } catch (err: unknown) {
   154→    // Audit logging failure should never block the request
   155→    console.warn(`Audit log write failed: ${err instanceof Error ? err.message : String(err)}`);
   156→  }
   157→
   158→  next();
   159→}

   161→/** Remove sensitive fields from request body before logging */
   162→function _sanitizeForAudit(body: Record<string, unknown> | undefined): Record<string, unknown> {
   163→  if (!body || typeof body !== 'object') return {};
   164→  const sanitized = { ...body };
   165→  const sensitiveKeys = ['password', 'token', 'secret', 'apiKey', 'creditCard', 'cvv'];
   166→  for (const key of Object.keys(sanitized)) {
   167→    if (sensitiveKeys.some((sk) => key.toLowerCase().includes(sk.toLowerCase()))) {
   168→      sanitized[key] = '[REDACTED]';
   169→    }
   170→  }
   171→  return sanitized;
   172→}

   179→/**
   180→ * Write an audit event when a LiveKit token is denied via the API.

   184→  uid: string,

   186→  orgId: string,
   187→  roomId: string,

   193→      organizationId: orgId,

   196→      performedByOrgId: orgId,
   197→      action: 'livekit_token_denied',
   198→      targetType: 'livekit_room',
   199→      targetId: roomId,
   200→      metadata: { reason, message },
   201→      timestamp: admin.firestore.FieldValue.serverTimestamp(),
   202→    });
   203→  } catch {
   204→    console.warn('Failed to write livekit_token_denied audit event');
   205→  }
   206→}

   254→app.post(
   255→  '/v1/livekit/token',
   256→  verifyAuthToken,
   257→  async (req: Request, res: Response) => {
   258→    initSentry();
   259→    Sentry.setTag('service', 'livekit');
   260→    Sentry.setTag('endpoint', '/v1/livekit/token');
   261→
   262→    if (!req.user) {
   263→      res.status(401).json({ error: 'Authentication required.' });
   264→      return;
   265→    }
   266→
   267→    const { roomId, displayName } = req.body ?? {};
   268→    // roomName/isTeacher: REMOVED — derived from room document and caller claims
   269→
   270→    if (!roomId || typeof roomId !== 'string') {
   271→      res.status(400).json({ error: 'roomId is required and must be a string.' });
   272→      return;
   273→    }
   274→
   275→    try {
   276→      // ── Load room document ──────────────────────────────────
   277→      const roomDoc = await admin.firestore().collection('livekit_rooms').doc(roomId).get();
   278→
   279→      if (!roomDoc.exists) {
   280→        res.status(404).json({ error: 'Room not found.' });
   281→        return;
   282→      }
   283→
   284→      const roomData = roomDoc.data()!;
   285→      const roomName = roomData['name'] as string;
   286→      if (!roomName) {
   287→        res.status(500).json({ error: 'Room document is missing a name field.' });
   288→        return;
   289→      }
   290→
   291→      // Room name sanitization
   292→      const sanitizedRoomName = roomName.replace(/[^a-zA-Z0-9_-]/g, '');
   293→      if (sanitizedRoomName.length === 0 || sanitizedRoomName.length > 128) {
   294→        res.status(500).json({ error: 'Invalid room name in document.' });
   295→        return;
   296→      }
   297→
   298→      // ── Org boundary check ──────────────────────────────────
   299→      const roomOrgId = roomData['organizationId'] as string;
   300→      if (!verifyOrgBoundary(req.userOrgId ?? '', roomOrgId, req.userRole ?? '')) {
   301→        await _logTokenDenied(req.user!.uid, req.userRole ?? 'unknown', req.userOrgId ?? '', roomId, 'org_boundary', 'Cross-org access denied');
   302→        Sentry.captureMessage(`Cross-org LiveKit token attempt (API): caller=${req.userOrgId} room=${roomOrgId}`);
   303→        res.status(403).json({ error: 'You can only join rooms in your organization.' });
   304→        return;
   305→      }
   306→
   307→      // ── Scope authorization ────────────────────────────────
   308→      const scopeAccessLevel = (req.user?.token?.['scopeAccessLevel'] as string) || '';
   309→      const callerScope: Record<string, string[]> = {};
   310→
   311→      // Load caller's scope arrays from user doc (already loaded by verifyAuthToken middleware)
   312→      const callerDoc = await admin.firestore().collection('users').doc(req.user!.uid).get();
   313→      if (callerDoc.exists) {
   314→        const callerData = callerDoc.data()!;
   315→        callerScope['campusIds'] = callerData['campusIds'] as string[] || [];
   316→        callerScope['stageIds'] = callerData['stageIds'] as string[] || [];
   317→        callerScope['classIds'] = callerData['classIds'] as string[] || [];
   318→        callerScope['subjectIds'] = callerData['subjectIds'] as string[] || [];
   319→        callerScope['studentIds'] = callerData['studentIds'] as string[] || [];
   320→      }
   321→
   322→      const scopeResult = verifyScopeAuthorization(scopeAccessLevel, callerScope, roomData as Record<string, unknown>);
   323→      if (!scopeResult.authorized) {
   324→        await _logTokenDenied(req.user!.uid, req.userRole ?? 'unknown', req.userOrgId ?? '', roomId, scopeResult.reason ?? 'unknown', scopeResult.message ?? 'Scope authorization failed');
   325→        Sentry.captureMessage(`LiveKit scope denial (API): uid=${req.user!.uid} room=${roomId} reason=${scopeResult.reason}`);
   326→        res.status(403).json({ error: scopeResult.message || 'You are not authorized to access this room.' });
   327→        return;
   328→      }
   329→
   330→      // ── Server-side role determination ──────────────────────
   331→      const isTeacherOrAbove = LIVEKIT_ADMIN_ROLES.includes(req.userRole as any);
   332→
   333→      const { AccessToken } = await import('livekit-server-sdk');
   334→
   335→      const token = new AccessToken(
   336→        LIVEKIT_API_KEY.value(),
   337→        LIVEKIT_API_SECRET.value(),
   338→        { identity: req.user.uid, name: displayName || req.user.uid },
   339→      );
   340→
   341→      token.addGrant({
   342→        room: sanitizedRoomName,
   343→        roomJoin: true,
   344→        canPublish: true,
   345→        canSubscribe: true,
   346→        canPublishData: true,
   347→        roomAdmin: isTeacherOrAbove,
   348→      });
   349→
   350→      const jwt = await token.toJwt();
   351→
   352→      Sentry.addBreadcrumb({
   353→        category: 'livekit',
   354→        message: `Token generated for room ${sanitizedRoomName}, roomAdmin=${isTeacherOrAbove}`,
   355→        level: 'info',
   356→      });
   357→
   358→      res.json({ token: jwt });
   359→    } catch (err: unknown) {
   360→      const msg = err instanceof Error ? err.message : String(err);
   361→      console.error(`LiveKit token generation failed: ${msg}`);
   362→      Sentry.captureException(err);
   363→      res.status(500).json({ error: 'Failed to generate token.' });
   364→    }
   365→  },
   366→);

   368→app.post(
   369→  '/v1/livekit/remove',
   370→  verifyAuthToken,
   371→  requireAdmin,
   372→  async (req: Request, res: Response) => {
   373→    initSentry();
   374→    Sentry.setTag('service', 'livekit');
   375→    Sentry.setTag('endpoint', '/v1/livekit/remove');
   376→
   377→    const { roomName, participantIdentity, roomId } = req.body ?? {};
   378→
   379→    if (!roomName || !participantIdentity || !roomId) {
   380→      res.status(400).json({ error: 'roomName, participantIdentity, and roomId are required.' });

   390→        res.status(404).json({ error: 'Room not found.' });

   419→          leftAt: admin.firestore.FieldValue.serverTimestamp(),
   420→          updatedAt: admin.firestore.FieldValue.serverTimestamp(),
   421→          removedBy: req.user!.uid,
   422→          removedAt: admin.firestore.FieldValue.serverTimestamp(),
   423→        });
   424→      } catch (err: unknown) {
   425→        console.warn(`Could not update attendance for ${participantIdentity}: ${err}`);
   426→      }
   427→
   428→      // Create in-app notification
   429→      try {
   430→        await db.collection('notifications').add({
   431→          userId: participantIdentity,
   432→          type: 'removed_from_class',
   433→          title: 'Removed from Class',
   434→          body: `You have been removed from the live class "${roomDoc.data()?.['name'] ?? roomName}".`,
   435→          data: { roomId, roomName },
   436→          read: false,
   437→          createdAt: admin.firestore.FieldValue.serverTimestamp(),
   438→        });
   439→      } catch (err: unknown) {
   440→        console.warn(`Could not create removal notification: ${err}`);
   441→      }
   442→
   443→      res.json({ success: true, removedIdentity: participantIdentity });
   444→    } catch (err: unknown) {
   445→      const msg = err instanceof Error ? err.message : String(err);
   446→      console.error(`Remove participant failed: ${msg}`);
   447→      Sentry.captureException(err);
   448→      res.status(500).json({ error: 'Failed to remove participant.' });
   449→    }
   450→  },

   575→// ═══════════════════════════════════════════════════════════════════

   577→/**
   578→ * POST /v1/storage/upload-url
   579→ *
   580→ * Generates a signed URL for the client to upload a file directly to
   581→ * Firebase Storage. The client never needs storage credentials.
   582→ *
   583→ * Request body:
   584→ *   { "filePath": "organizations/abc123/logo.png", "contentType": "image/png" }
   585→ *
   586→ * Response:
   587→ *   { "uploadUrl": "https://storage.googleapis.com/...", "expiresAt": "..." }
   588→ */
   589→app.post(
   590→  '/v1/storage/upload-url',
   591→  verifyAuthToken,
   592→  async (req: Request, res: Response) => {
   593→    initSentry();
   594→    Sentry.setTag('service', 'storage');
   595→    Sentry.setTag('endpoint', '/v1/storage/upload-url');
   596→
   597→    const { filePath, contentType } = req.body ?? {};
   598→
   599→    if (!filePath || typeof filePath !== 'string') {
   600→      res.status(400).json({ error: 'filePath is required.' });
   601→      return;
   602→    }
   603→
   604→    // Validate path doesn't escape allowed prefixes
   605→    const allowedPrefixes = [
   606→      `users/${req.user!.uid}/`,
   607→      `organizations/${req.userOrgId ?? ''}/`,
   608→      'exams/',
   609→      'materials/',
   610→      'submissions/',
   611→    ];
   612→
   613→    const isAllowed = allowedPrefixes.some((prefix) => filePath.startsWith(prefix));
   614→    if (!isAllowed) {
   615→      res.status(403).json({
   616→        error: 'File path must start with an allowed prefix.',
   617→        allowedPrefixes: ['users/{uid}/', 'organizations/{orgId}/', 'exams/', 'materials/', 'submissions/'],
   618→      });
   619→      return;
   620→    }
   621→
   622→    // Prevent path traversal
   623→    if (filePath.includes('..') || filePath.includes('//')) {
   624→      res.status(400).json({ error: 'Invalid file path.' });
   625→      return;
   626→    }
   627→
   628→    try {
   629→      const bucket = admin.storage().bucket();
   630→      const file = bucket.file(filePath);
   631→
   632→      const expiresInMinutes = 15;
   633→      const expiresAt = new Date(Date.now() + expiresInMinutes * 60 * 1000);
   634→
   635→      const [uploadUrl] = await file.getSignedUrl({
   636→        action: 'write',
   637→        expires: expiresAt,
   638→        contentType: contentType || 'application/octet-stream',
   639→      });
   640→
   641→      // Also generate a read URL
   642→      const [downloadUrl] = await file.getSignedUrl({
   643→        action: 'read',
   644→        expires: expiresAt,
   645→      });
   646→
   647→      res.json({
   648→        uploadUrl,
   649→        downloadUrl,
   650→        filePath,
   651→        expiresAt: expiresAt.toISOString(),
   652→      });
   653→    } catch (err: unknown) {
   654→      const msg = err instanceof Error ? err.message : String(err);
   655→      console.error(`Signed URL generation failed: ${msg}`);
   656→      Sentry.captureException(err);
   657→      res.status(500).json({ error: 'Failed to generate upload URL.' });
   658→    }
   659→  },
