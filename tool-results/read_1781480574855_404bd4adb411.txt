    26→  /// Hash a password using SHA-256
    27→  static String hashPassword(String password) {
    28→    final bytes = utf8.encode(password);
    29→    final digest = sha256.convert(bytes);
    30→    return digest.toString();
    31→  }

   100→          stackTrace: st,

   302→  /// Unified email+password login for owners, teachers, and parents.
   303→  Future<Map<String, dynamic>> loginWithEmail({
   304→    required String email,
   305→    required String password,
   306→  }) async {
   307→    final transaction = KlasivoSentry.transactions.loginFlow('email');
   308→
   309→    try {
   310→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   311→        'method': 'email',
   312→        'email': email,
   313→      });
   314→
   315→      final userCredential =
   316→          await FirebaseService.loginWithEmail(email, password);
   317→      final user = userCredential.user!;
   318→
   319→      KlasivoSentry.breadcrumb.auth('auth_user_authenticated', data: {
   320→        'uid': user.uid,
   321→        'method': 'email',
   322→      });
   323→
   324→      // Set Sentry user context immediately after auth
   325→      await KlasivoSentry.userContext.setUser(
   326→        uid: user.uid,
   327→        email: email,
   328→      );
   329→
   330→      final readUserSpan = transaction.startChild('read_user_doc');
   331→      final userDoc = await _firestore
   332→          .collection(AppConstants.usersCollection)
   333→          .doc(user.uid)
   334→          .get();
   335→      await readUserSpan.finish();
   336→
   337→      if (!userDoc.exists) {
   338→        KlasivoSentry.breadcrumb.auth('login_failed_user_doc_missing', data: {
   339→          'uid': user.uid,
   340→        });
   341→        await Sentry.captureMessage(
   342→          'Login failed: users/${user.uid} document does not exist',
   343→          level: SentryLevel.error,
   344→        );
   345→        throw Exception('User data not found. Please contact your administrator.');
   346→      }
   347→
   348→      final userData = userDoc.data()!;
   349→      final role = userData['role'] as String?;
   350→      final isActive = userData['isActive'] as bool? ?? true;
   351→
   352→      // Update Sentry context with role and org
   353→      await KlasivoSentry.userContext.setRole(role ?? 'unknown');
   354→      final orgId = userData['organizationId'] as String?;
   355→      if (orgId != null && orgId.isNotEmpty) {
   356→        await KlasivoSentry.userContext.setOrganizationId(orgId);
   357→      }
   358→
   359→      if (!isActive) {
   360→        await _auth.signOut();
   361→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   362→          'uid': user.uid,
   363→          'role': role,
   364→        });
   365→        throw Exception('Your account has been deactivated. Contact your administrator.');
   366→      }
   367→
   368→      // Students should NOT use email login — they use student code login
   369→      if (role == AppConstants.roleStudent) {
   370→        await _auth.signOut();
   371→        KlasivoSentry.breadcrumb.auth('login_failed_student_email_login', data: {
   372→          'uid': user.uid,
   373→        });
   374→        throw Exception('Students must login with their student code.');
   375→      }
   376→
   377→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   378→        'uid': user.uid,
   379→        'role': role ?? 'unknown',
   380→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   381→      });
   382→
   383→      transaction.status = const SpanStatus.ok();
   384→
   385→      return {
   386→        'id': user.uid,
   387→        'organizationId': userData['organizationId'] ?? '',
   388→        'role': role ?? AppConstants.roleTeacher,
   389→        'authProvider': AuthProviders.password,
   390→        'fullName': userData['fullName'] ?? 'User',
   391→        'email': userData['email'] ?? email,
   392→        'isEmailVerified': user.emailVerified,
   393→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   394→      };
   395→    } catch (e, st) {
   396→      transaction.status = const SpanStatus.internalError();
   397→      await Sentry.captureException(
   398→        e,
   399→        stackTrace: st,
   400→        withScope: (scope) {
   401→          scope.setTag('flow', 'email_login');
   402→          scope.setTag('step', 'login_with_email');
   403→        },
   404→      );
   405→      rethrow;
   406→    } finally {
   407→      await transaction.finish();
   408→    }
   409→  }

   411→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   412→
   413→  /// Student login using student code + password.
   414→  /// Internally maps to Firebase Auth for push notifications,
   415→  /// multi-device, password reset, security rules, and analytics.
   416→  Future<Map<String, dynamic>> loginStudent({
   417→    required String studentCode,
   418→    required String password,
   419→  }) async {
   420→    final transaction = KlasivoSentry.transactions.loginFlow('student');
   421→
   422→    try {
   423→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   424→        'method': 'student_code',
   425→      });
   426→
   427→      // Step 1: Find user by studentCode
   428→      final findUserSpan = transaction.startChild('find_user_by_code');
   429→      final snapshot = await _firestore
   430→          .collection(AppConstants.usersCollection)
   431→          .where('studentCode', isEqualTo: studentCode)
   432→          .where('role', isEqualTo: AppConstants.roleStudent)
   433→          .limit(1)
   434→          .get();
   435→      await findUserSpan.finish();
   436→
   437→      if (snapshot.docs.isEmpty) {
   438→        KlasivoSentry.breadcrumb.auth('login_failed_student_not_found', data: {
   439→          'method': 'student_code',
   440→        });
   441→        throw Exception('Student not found. Please check your student code.');
   442→      }
   443→
   444→      final userDoc = snapshot.docs.first;
   445→      final student = userDoc.data();
   446→
   447→      // Step 2: Verify password
   448→      final storedPasswordHash = student['passwordHash'] as String?;
   449→      final storedPlaintext = student['password'] as String?;
   450→      final inputHash = hashPassword(password);
   451→
   452→      bool passwordMatches = false;
   453→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   454→        passwordMatches = inputHash == storedPasswordHash;
   455→      } else if (storedPlaintext != null) {
   456→        passwordMatches = password == storedPlaintext;
   457→        // Migrate to hash on successful login
   458→        if (passwordMatches) {
   459→          await _firestore
   460→              .collection(AppConstants.usersCollection)
   461→              .doc(userDoc.id)
   462→              .update({'passwordHash': inputHash});
   463→        }
   464→      }
   465→
   466→      if (!passwordMatches) {
   467→        KlasivoSentry.breadcrumb.auth('login_failed_invalid_password', data: {
   468→          'method': 'student_code',
   469→          'userId': userDoc.id,
   470→        });
   471→        throw Exception('Invalid password. Please try again.');
   472→      }
   473→
   474→      final isActive = student['isActive'] as bool? ?? true;
   475→      if (!isActive) {
   476→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   477→          'userId': userDoc.id,
   478→          'role': 'student',
   479→        });
   480→        throw Exception('Your account has been deactivated.');
   481→      }
   482→
   483→      // Step 3: Sign in via Firebase Auth using the student's internal email
   484→      final internalEmail = student['authEmail'] as String?;
   485→      if (internalEmail != null && internalEmail.isNotEmpty) {
   486→        try {
   487→          await _auth.signInWithEmailAndPassword(
   488→            email: internalEmail,
   489→            password: password,
   490→          );
   491→          KlasivoSentry.breadcrumb.auth('student_auth_signed_in', data: {
   492→            'userId': userDoc.id,
   493→          });
   494→        } catch (authError) {
   495→          // Graceful fallback — still allow login even if Firebase Auth fails
   496→          KlasivoSentry.breadcrumb.auth('student_auth_signin_failed_fallback', data: {
   497→            'userId': userDoc.id,
   498→            'error': authError.toString().substring(0, (authError.toString().length).clamp(0, 100)),
   499→          });
   500→        }
   501→      }
   502→
   503→      // Set Sentry user context
   504→      await KlasivoSentry.userContext.setUser(
   505→        uid: userDoc.id,
   506→        email: internalEmail ?? '',
   507→        role: AppConstants.roleStudent,
   508→        organizationId: student['organizationId'] as String?,
   509→      );
   510→
   511→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   512→        'uid': userDoc.id,
   513→        'role': 'student',
   514→        'method': 'student_code',
   515→      });
   516→
   517→      // Log doc ID audit — student uses Firestore doc ID (not auth UID)
   518→      KlasivoSentry.docIdAudit.logUserCreation(
   519→        flow: 'student_code_login',
   520→        collection: AppConstants.usersCollection,
   521→        docIdStrategy: userDoc.id.length > 28 ? 'auto_id' : 'uid',
   522→        actualDocId: userDoc.id,
   523→        authUid: _auth.currentUser?.uid,
   524→      );
   525→
   526→      transaction.status = const SpanStatus.ok();
   527→
   528→      return {
   529→        'id': userDoc.id,
   530→        'organizationId': student['organizationId'] ?? '',
   531→        'role': AppConstants.roleStudent,
   532→        'authProvider': AuthProviders.studentCode,
   533→        'fullName': student['fullName'] ?? 'Student',
   534→        'studentCode': student['studentCode'],
   535→        'classId': student['classId'],
   536→        'hasCompletedSetup': true,
   537→      };
   538→    } catch (e, st) {
   539→      transaction.status = const SpanStatus.internalError();
   540→      await Sentry.captureException(
   541→        e,
   542→        stackTrace: st,
   543→        withScope: (scope) {
   544→          scope.setTag('flow', 'student_login');
   545→          scope.setTag('step', 'login_student');
   546→          scope.setTag('method', 'student_code');
   547→        },
   548→      );
   549→      rethrow;
   550→    } finally {
   551→      await transaction.finish();
   552→    }
   553→  }

  1358→  Future<bool> needsSetup() async {
  1359→    try {
  1360→      final user = _auth.currentUser;
  1361→      if (user == null) return false;
  1362→
  1363→      final userDoc = await _firestore
  1364→          .collection(AppConstants.usersCollection)
  1365→          .doc(user.uid)
  1366→          .get();
  1367→
  1368→      if (!userDoc.exists) return false;
  1369→
  1370→      final data = userDoc.data()!;
  1371→      final role = data['role'] as String?;
  1372→      final hasCompletedSetup = data['hasCompletedSetup'] as bool? ?? true;
  1373→
  1374→      // Owners need workspace naming, parents need child linking
  1375→      if ((role == AppConstants.roleOwner || role == AppConstants.roleParent) &&
  1376→          !hasCompletedSetup) {
  1377→        return true;
  1378→      }
  1379→
  1380→      return false;
  1381→    } catch (e) {
  1382→      return false;
  1383→    }
  1384→  }
