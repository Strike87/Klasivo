    24→    24→  /// Hash a password using SHA-256

   156→   156→  /// Unified email+password login for owners, teachers, and parents.
   157→   157→  Future<Map<String, dynamic>> loginWithEmail({
   158→   158→    required String email,
   159→   159→    required String password,
   160→   160→  }) async {
   161→   161→    try {
   162→   162→      final userCredential =
   163→   163→          await FirebaseService.loginWithEmail(email, password);
   164→   164→      final user = userCredential.user!;
   165→   165→
   166→   166→      final userDoc = await _firestore
   167→   167→          .collection(AppConstants.usersCollection)
   168→   168→          .doc(user.uid)
   169→   169→          .get();
   170→   170→
   171→   171→      if (!userDoc.exists) {
   172→   172→        throw Exception('User data not found. Please contact your administrator.');
   173→   173→      }
   174→   174→
   175→   175→      final userData = userDoc.data()!;
   176→   176→      final role = userData['role'] as String?;
   177→   177→      final isActive = userData['isActive'] as bool? ?? true;
   178→   178→
   179→   179→      if (!isActive) {
   180→   180→        await _auth.signOut();
   181→   181→        throw Exception('Your account has been deactivated. Contact your administrator.');
   182→   182→      }
   183→   183→
   184→   184→      // Students should NOT use email login — they use student code login
   185→   185→      if (role == AppConstants.roleStudent) {
   186→   186→        await _auth.signOut();
   187→   187→        throw Exception('Students must login with their student code.');
   188→   188→      }
   189→   189→
   190→   190→      return {
   191→   191→        'id': user.uid,
   192→   192→        'organizationId': userData['organizationId'] ?? '',
   193→   193→        'role': role ?? AppConstants.roleTeacher,
   194→   194→        'authProvider': AuthProviders.password,
   195→   195→        'fullName': userData['fullName'] ?? 'User',
   196→   196→        'email': userData['email'] ?? email,
   197→   197→        'isEmailVerified': user.emailVerified,
   198→   198→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   199→   199→      };
   200→   200→    } catch (e) {
   201→   201→      rethrow;
   202→   202→    }
   203→   203→  }

   205→   205→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   206→   206→
   207→   207→  /// Student login using student code + password.
   208→   208→  /// Internally maps to Firebase Auth for push notifications,
   209→   209→  /// multi-device, password reset, security rules, and analytics.
   210→   210→  Future<Map<String, dynamic>> loginStudent({
   211→   211→    required String studentCode,
   212→   212→    required String password,
   213→   213→  }) async {
   214→   214→    try {
   215→   215→      // Step 1: Find user by studentCode
   216→   216→      final snapshot = await _firestore
   217→   217→          .collection(AppConstants.usersCollection)
   218→   218→          .where('studentCode', isEqualTo: studentCode)
   219→   219→          .where('role', isEqualTo: AppConstants.roleStudent)
   220→   220→          .limit(1)
   221→   221→          .get();
   222→   222→
   223→   223→      if (snapshot.docs.isEmpty) {
   224→   224→        throw Exception('Student not found. Please check your student code.');
   225→   225→      }
   226→   226→
   227→   227→      final userDoc = snapshot.docs.first;
   228→   228→      final student = userDoc.data();
   229→   229→
   230→   230→      // Step 2: Verify password
   231→   231→      final storedPasswordHash = student['passwordHash'] as String?;
   232→   232→      final storedPlaintext = student['password'] as String?;
   233→   233→      final inputHash = hashPassword(password);
   234→   234→
   235→   235→      bool passwordMatches = false;
   236→   236→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   237→   237→        passwordMatches = inputHash == storedPasswordHash;
   238→   238→      } else if (storedPlaintext != null) {
   239→   239→        passwordMatches = password == storedPlaintext;
   240→   240→        // Migrate to hash on successful login
   241→   241→        if (passwordMatches) {
   242→   242→          await _firestore
   243→   243→              .collection(AppConstants.usersCollection)
   244→   244→              .doc(userDoc.id)
   245→   245→              .update({'passwordHash': inputHash});
   246→   246→        }
   247→   247→      }
   248→   248→
   249→   249→      if (!passwordMatches) {
   250→   250→        throw Exception('Invalid password. Please try again.');
   251→   251→      }
   252→   252→
   253→   253→      final isActive = student['isActive'] as bool? ?? true;
   254→   254→      if (!isActive) {
   255→   255→        throw Exception('Your account has been deactivated.');
   256→   256→      }
   257→   257→
   258→   258→      // Step 3: Sign in via Firebase Auth using the student's internal email
   259→   259→      final internalEmail = student['authEmail'] as String?;
   260→   260→      if (internalEmail != null && internalEmail.isNotEmpty) {
   261→   261→        try {
   262→   262→          await _auth.signInWithEmailAndPassword(
   263→   263→            email: internalEmail,
   264→   264→            password: password,
   265→   265→          );
   266→   266→        } catch (authError) {
   267→   267→          // Graceful fallback — still allow login even if Firebase Auth fails
   268→   268→        }
   269→   269→      }
   270→   270→
   271→   271→      return {
   272→   272→        'id': userDoc.id,
   273→   273→        'organizationId': student['organizationId'] ?? '',
   274→   274→        'role': AppConstants.roleStudent,
   275→   275→        'authProvider': AuthProviders.studentCode,
   276→   276→        'fullName': student['fullName'] ?? 'Student',
   277→   277→        'studentCode': student['studentCode'],
   278→   278→        'classId': student['classId'],
   279→   279→        'hasCompletedSetup': true,
   280→   280→      };
   281→   281→    } catch (e) {
   282→   282→      rethrow;
   283→   283→    }
   284→   284→  }

   531→   531→  // ─── Google Sign-In (Unified — Owner / Teacher / Parent) ──────────────────
   532→   532→
   533→   533→  /// Sign in with Google. Works for owners, teachers, and parents.
   534→   534→  /// If the user already exists, logs them in.
   535→   535→  /// If the user is new, creates an account based on [expectedRole]:
   536→   536→  ///   - owner: auto-creates organization (hasCompletedSetup=false)
   537→   537→  ///   - teacher: needs invite code (handled separately via registerTeacherWithGoogle)
   538→   538→  ///   - parent: creates parent account (hasCompletedSetup=false, needs to link child)
   539→   539→  Future<Map<String, dynamic>> loginWithGoogle({
   540→   540→    String? expectedRole,
   541→   541→  }) async {
   542→   542→    return _signInWithGoogle(
   543→   543→      expectedRole: expectedRole,
   544→   544→      isNewUser: false,
   545→   545→    );
   546→   546→  }
   547→   547→
   548→   548→  Future<Map<String, dynamic>> _signInWithGoogle({
   549→   549→    String? expectedRole,
   550→   550→    bool isNewUser = false,
   551→   551→  }) async {
   552→   552→    try {
   553→   553→      final googleUser = await GoogleSignIn().signIn();
   554→   554→      if (googleUser == null) {
   555→   555→        throw Exception('Google sign-in was cancelled.');
   556→   556→      }
   557→   557→
   558→   558→      final googleAuth = await googleUser.authentication;
   559→   559→      final credential = GoogleAuthProvider.credential(
   560→   560→        accessToken: googleAuth.accessToken,
   561→   561→        idToken: googleAuth.idToken,
   562→   562→      );
   563→   563→
   564→   564→      final userCredential = await _auth.signInWithCredential(credential);
   565→   565→      final user = userCredential.user!;
   566→   566→
   567→   567→      // Check if user document already exists
   568→   568→      final userDoc = await _firestore
   569→   569→          .collection(AppConstants.usersCollection)
   570→   570→          .doc(user.uid)
   571→   571→          .get();
   572→   572→
   573→   573→      if (userDoc.exists) {
   574→   574→        // Existing user — log them in
   575→   575→        final userData = userDoc.data()!;
   576→   576→        final role = userData['role'] as String?;
   577→   577→        final isActive = userData['isActive'] as bool? ?? true;
   578→   578→
   579→   579→        if (!isActive) {
   580→   580→          await _auth.signOut();
   581→   581→          throw Exception('Your account has been deactivated.');
   582→   582→        }
   583→   583→
   584→   584→        // Students cannot use Google Sign-In
   585→   585→        if (role == AppConstants.roleStudent) {
   586→   586→          await _auth.signOut();
   587→   587→          throw Exception('Students must login with their student code.');
   588→   588→        }
   589→   589→
   590→   590→        return {
   591→   591→          'id': user.uid,
   592→   592→          'organizationId': userData['organizationId'] ?? '',
   593→   593→          'role': role ?? expectedRole ?? AppConstants.roleOwner,
   594→   594→          'authProvider': AuthProviders.google,
   595→   595→          'fullName': userData['fullName'] ?? user.displayName ?? 'User',
   596→   596→          'email': userData['email'] ?? user.email ?? '',
   597→   597→          'isEmailVerified': user.emailVerified,
   598→   598→          'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   599→   599→        };
   600→   600→      }
   601→   601→
   602→   602→      // New user — create account based on expected role
   603→   603→      final fullName = user.displayName ?? 'User';
   604→   604→      final email = user.email ?? '';
   605→   605→      final role = expectedRole ?? AppConstants.roleOwner;
   606→   606→      final isEmailVerified = user.emailVerified; // Google emails are pre-verified
   607→   607→
   608→   608→      String? organizationId;
   609→   609→      bool hasCompletedSetup = true;
   610→   610→
   611→   611→      if (role == AppConstants.roleOwner) {
   612→   612→        // Create user doc FIRST with placeholder orgId so isTeacherOrOwner() resolves
   613→   613→        await _firestore
   614→   614→            .collection(AppConstants.usersCollection)
   615→   615→            .doc(user.uid)
   616→   616→            .set({
   617→   617→          'organizationId': '', // Placeholder — updated below
   618→   618→          'role': AppConstants.roleOwner,
   619→   619→          'authProvider': AuthProviders.google,
   620→   620→          'fullName': fullName,
   621→   621→          'email': email,
   622→   622→          'photoUrl': user.photoURL,
   623→   623→          'phoneNumber': null,
   624→   624→          'isActive': true,
   625→   625→          'isEmailVerified': isEmailVerified,
   626→   626→          'hasCompletedSetup': false,
   627→   627→          'createdAt': FieldValue.serverTimestamp(),
   628→   628→          'updatedAt': FieldValue.serverTimestamp(),
   629→   629→        });
   630→   630→
   631→   631→        // Now create the organization — rules can verify user is an owner
   632→   632→        final orgService = OrganizationService();
   633→   633→        organizationId = await orgService.createOrganization(
   634→   634→          ownerId: user.uid,
   635→   635→          name: "$fullName's Workspace",
   636→   636→        );
   637→   637→
   638→   638→        // Patch the user doc with the real organizationId
   639→   639→        await _firestore
   640→   640→            .collection(AppConstants.usersCollection)
   641→   641→            .doc(user.uid)
   642→   642→            .update({
   643→   643→          'organizationId': organizationId,
   644→   644→          'updatedAt': FieldValue.serverTimestamp(),
   645→   645→        });
   646→   646→
   647→   647→        hasCompletedSetup = false; // Needs to name workspace
   648→   648→      } else if (role == AppConstants.roleParent) {
   649→   649→        organizationId = null; // Set when parent links a child
   650→   650→        hasCompletedSetup = false; // Needs to link child
   651→   651→      }
   652→   652→
   653→   653→      // For non-owner roles, write the user doc normally (owner doc already created above)
   654→   654→      if (role != AppConstants.roleOwner) {
   655→   655→        await _firestore
   656→   656→            .collection(AppConstants.usersCollection)
   657→   657→            .doc(user.uid)
   658→   658→            .set({
   659→   659→          'organizationId': organizationId,
   660→   660→          'role': role,
   661→   661→          'authProvider': AuthProviders.google,
   662→   662→          'fullName': fullName,
   663→   663→          'email': email,
   664→   664→          'photoUrl': user.photoURL,
   665→   665→          'phoneNumber': null,
   666→   666→          'isActive': true,
   667→   667→          'isEmailVerified': isEmailVerified,
   668→   668→          'hasCompletedSetup': hasCompletedSetup,
   669→   669→          'createdAt': FieldValue.serverTimestamp(),
   670→   670→          'updatedAt': FieldValue.serverTimestamp(),
   671→   671→        });
   672→   672→      }
   673→   673→
   674→   674→      return {
   675→   675→        'id': user.uid,
   676→   676→        'organizationId': organizationId,
   677→   677→        'role': role,
   678→   678→        'authProvider': AuthProviders.google,
   679→   679→        'fullName': fullName,
   680→   680→        'email': email,
   681→   681→        'isEmailVerified': isEmailVerified,
   682→   682→        'hasCompletedSetup': hasCompletedSetup,
   683→   683→      };
   684→   684→    } catch (e) {
   685→   685→      rethrow;
   686→   686→    }
   687→   687→  }
