   137→            sanitized[key] = _SentrySanitizer.isSensitive(key)
   138→                ? '[REDACTED]'
   139→                : value;
   140→          });
   141→          event.breadcrumbs![i] = Breadcrumb(
   142→            category: crumb.category,
   143→            message: crumb.message,
   144→            data: sanitized,
   145→            level: crumb.level,
   146→            timestamp: crumb.timestamp,
   147→            type: crumb.type,
   148→          );
   149→        }
   150→      }
   151→    }

   153→    // Strip sensitive data from extra fields
   154→    if (event.extra != null) {

   157→        sanitized[key] = _SentrySanitizer.isSensitive(key)
   158→            ? '[REDACTED]'
   159→            : value;
   160→      });
   161→      event.extra = sanitized;
   162→    }
   163→
   164→    // Strip sensitive data from request headers
   165→    if (event.request?.headers != null) {

   168→        final lower = key.toLowerCase();
   169→        if (lower.contains('authorization') ||

   172→          sanitized[key] = '[REDACTED]';
   173→        } else {
   174→          sanitized[key] = value;
   175→        }

   315→class _SentrySanitizer {
   316→  static const Set<String> _sensitiveKeys = {
   317→    'password', 'passwordHash', 'confirmPassword', 'newPassword',
   318→    'accessToken', 'refreshToken', 'idToken', 'otp', 'otpCode',
   319→    'inviteCode', 'secret', 'apiKey', 'authEmail', 'studentCode',
   320→  };
   321→
   322→  static bool isSensitive(String key) {
   323→    final lower = key.toLowerCase();
   324→    for (final s in _sensitiveKeys) {
   325→      if (lower.contains(s.toLowerCase())) return true;
   326→    }
   327→    return false;
   328→  }
   329→}

   417→class AuthChangeNotifier extends ChangeNotifier {
   418→  StreamSubscription<User?>? _firebaseSub;
   419→  StreamSubscription? _hiveSub;
   420→
   421→  AuthChangeNotifier() {
   422→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   423→      (_) => notifyListeners(),
   424→      onError: (_) => notifyListeners(),
   425→    );
   426→    _startHiveWatch();
   427→  }
   428→
   429→  void _startHiveWatch() {
   430→    // Use Hive's built-in watch() instead of polling every 500ms
   431→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   432→      notifyListeners();
   433→    });
   434→  }
   435→
   436→  @override
   437→  void dispose() {
   438→    _firebaseSub?.cancel();
   439→    _hiveSub?.cancel();
   440→    super.dispose();
   441→  }
   442→}

   444→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   445→  final notifier = AuthChangeNotifier();

   452→final routerProvider = Provider<GoRouter>((ref) {

   454→
   455→  return GoRouter(

   457→    debugLogDiagnostics: true,
   458→    refreshListenable: notifier,
   459→    // ── Navigation error builder — catches route errors and reports to Sentry ──
   460→    errorBuilder: (context, state) {
   461→      final error = state.error;
   462→      if (error != null) {
   463→        Sentry.captureException(
   464→          error,
   465→          withScope: (scope) {
   466→            scope.setTag('source', 'go_router');
   467→            scope.setTag('route', state.matchedLocation);
   468→            scope.setExtra('uri', state.uri.toString());
   469→          },
   470→        );
   471→        KlasivoSentry.breadcrumb.navigation(
   472→          state.matchedLocation,
   473→          action: 'error',
   474→          data: {'error': error.toString()},
   475→        );
   476→      }

   478→        body: Center(
   479→          child: Column(
   480→            mainAxisAlignment: MainAxisAlignment.center,
   481→            children: [
   482→              const Icon(Icons.error_outline, size: 48, color: KlasivoColors.error),

   484→              Text(
   485→                'Page not found',
   486→                style: KlasivoTypography.headlineMedium.copyWith(
   487→                  color: Theme.of(context).brightness == Brightness.dark
   488→                      ? KlasivoColors.darkTextPrimary
   489→                      : KlasivoColors.lightTextPrimary,
   490→                ),
   491→              ),
   492→              const SizedBox(height: KlasivoSpacing.sm),
   493→              Text(
   494→                'The page you are looking for does not exist.',
   495→                style: KlasivoTypography.bodyMedium.copyWith(
   496→                  color: Theme.of(context).brightness == Brightness.dark
   497→                      ? KlasivoColors.darkTextTertiary
   498→                      : KlasivoColors.lightTextTertiary,
   499→                ),
   500→                textAlign: TextAlign.center,
   501→              ),
   502→              const SizedBox(height: KlasivoSpacing.xl),
   503→              ElevatedButton(
   504→                onPressed: () => context.go('/'),
   505→                child: const Text('Go Home'),
   506→              ),
   507→            ],
   508→          ),
   509→        ),
   510→      );
   511→    },
   512→    redirect: (context, state) {
   513→      final box = Hive.box(AppConstants.authBox);
   514→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   515→      final userRole = box.get('userRole', defaultValue: '') as String;
   516→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: false) as bool;
   517→      final mustChangePassword = box.get('mustChangePassword', defaultValue: false) as bool;
   518→      final isOnChangePassword = state.matchedLocation == '/change-password';
   519→
   520→      // Force redirect to change password if required
   521→      if (isLoggedIn && mustChangePassword && !isOnChangePassword) {
   522→        return '/change-password';
   523→      }
   524→      // Prevent navigating away from change password if forced
   525→      if (isLoggedIn && mustChangePassword && isOnChangePassword) {
   526→        return null; // Allow staying on change password
   527→      }
   528→      // If no longer required, redirect away from change password
   529→      if (isLoggedIn && !mustChangePassword && isOnChangePassword) {
   530→        // Management roles → /dashboard; student → /student; parent → /parent
   531→        if (KlasivoRole.managementRoles.contains(userRole)) return '/dashboard';
   532→        if (userRole == KlasivoRole.student) return '/student';
   533→        if (userRole == KlasivoRole.parent) return '/parent';
   534→        return '/dashboard'; // Fallback for unknown roles
   535→      }
   536→
