    26→    26→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    27→    27→
    28→    28→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    29→    29→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    30→    30→class _SensitiveFields {
    31→    31→  static const Set<String> keys = {
    32→    32→    'password',
    33→    33→    'passwordHash',
    34→    34→    'confirmPassword',
    35→    35→    'newPassword',
    36→    36→    'accessToken',
    37→    37→    'refreshToken',
    38→    38→    'idToken',
    39→    39→    'otp',
    40→    40→    'otpCode',
    41→    41→    'inviteCode',
    42→    42→    'secret',
    43→    43→    'apiKey',
    44→    44→    'authEmail', // Internal student auth email — PII
    45→    45→    'studentCode', // Sensitive — could be used to log in
    46→    46→  };
    47→    47→
    48→    48→  /// Returns true if the key contains any sensitive substring.
    49→    49→  static bool isSensitive(String key) {
    50→    50→    final lower = key.toLowerCase();
    51→    51→    for (final s in keys) {
    52→    52→      if (lower.contains(s.toLowerCase())) return true;
    53→    53→    }
    54→    54→    return false;
    55→    55→  }
    56→    56→
    57→    57→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    58→    58→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    59→    59→    return data.map((key, value) {
    60→    60→      if (isSensitive(key)) {
    61→    61→        return MapEntry(key, '[REDACTED]');
    62→    62→      }
    63→    63→      // Recursively sanitize nested maps
    64→    64→      if (value is Map<String, dynamic>) {
    65→    65→        return MapEntry(key, sanitize(value));
    66→    66→      }
    67→    67→      return MapEntry(key, value);
    68→    68→    });
    69→    69→  }
    70→    70→
    71→    71→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    72→    72→  static String sanitizeString(String input) {
    73→    73→    // Redact Bearer tokens
    74→    74→    var result = input.replaceAll(
    75→    75→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    76→    76→      'Bearer [REDACTED]',
    77→    77→    );
    78→    78→    // Redact long hex strings (likely tokens)
    79→    79→    result = result.replaceAll(
    80→    80→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    81→    81→      '[REDACTED_TOKEN]',
    82→    82→    );
    83→    83→    return result;
    84→    84→  }
    85→    85→}

   105→   105→/// Fluent API for creating standardized Sentry breadcrumbs.
   106→   106→///

   113→   113→    Sentry.addBreadcrumb(Breadcrumb(
   114→   114→      category: SentryCategories.auth,
   115→   115→      message: message,
   116→   116→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   117→   117→      level: SentryLevel.info,
   118→   118→    ));
   119→   119→  }

   124→   124→      category: SentryCategories.registration,

   255→   255→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception
   256→   256→/// capture, and scope tags for every critical write.
   257→   257→///
   258→   258→/// Every Firestore exception will:
   259→   259→///   1. Reach Sentry

   264→   264→class SentryFirestoreHelper {

   266→   266→
   267→   267→  /// Wrapped `.set()` with full observability.
   268→   268→  static Future<void> docSet({
   269→   269→    required String collection,
   270→   270→    required String docId,
   271→   271→    required Map<String, dynamic> data,
   272→   272→    String? flow,
   273→   273→    String? step,
   274→   274→    SetOptions? options,
   275→   275→  }) async {
   276→   276→    final sanitizedData = _SensitiveFields.sanitize(data);
   277→   277→    KlasivoSentry.breadcrumb.firestore(
   278→   278→      'create',
   279→   279→      collection: collection,
   280→   280→      docId: docId,
   281→   281→      data: {'flow': flow, 'step': step},
   282→   282→    );
   283→   283→
   284→   284→    try {
   285→   285→      await _db.collection(collection).doc(docId).set(data, options);
   286→   286→      KlasivoSentry.breadcrumb.firestore(
   287→   287→        'create_success',
   288→   288→        collection: collection,
   289→   289→        docId: docId,
   290→   290→      );
   291→   291→    } catch (e, st) {
   292→   292→      await _captureFirestoreException(
   293→   293→        e,
   294→   294→        st,
   295→   295→        collection: collection,
   296→   296→        operation: 'set',
   297→   297→        docId: docId,
   298→   298→        flow: flow,
   299→   299→        step: step,
   300→   300→        dataPreview: sanitizedData,
   301→   301→      );
   302→   302→      rethrow;
   303→   303→    }
   304→   304→  }

   451→   451→  /// Capture a Firestore exception with full context tags.
   452→   452→  static Future<void> _captureFirestoreException(
   453→   453→    Object exception,
   454→   454→    StackTrace stackTrace, {
   455→   455→    required String collection,
   456→   456→    required String operation,
   457→   457→    String? docId,
   458→   458→    String? flow,
   459→   459→    String? step,
   460→   460→    Map<String, dynamic>? dataPreview,
   461→   461→  }) async {
   462→   462→    // Determine if this is a permission-denied error
   463→   463→    String? errorCode;
   464→   464→    if (exception is FirebaseException) {
   465→   465→      errorCode = exception.code;
   466→   466→    }
   467→   467→
   468→   468→    await Sentry.captureException(
   469→   469→      exception,
   470→   470→      stackTrace: stackTrace,
   471→   471→      withScope: (scope) {
   472→   472→        scope.setTag('collection', collection);
   473→   473→        scope.setTag('operation', operation);
   474→   474→        if (docId != null) scope.setTag('documentId', docId);
   475→   475→        if (flow != null) scope.setTag('flow', flow);
   476→   476→        if (step != null) scope.setTag('step', step);
   477→   477→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   478→   478→
   479→   479→        // Attach user role and org from current Sentry context
   480→   480→        final currentUser = FirebaseAuth.instance.currentUser;
   481→   481→        if (currentUser != null) {
   482→   482→          scope.setTag('auth_uid', currentUser.uid);
   483→   483→        }
   484→   484→
   485→   485→        if (dataPreview != null && dataPreview.isNotEmpty) {
   486→   486→          scope.setExtra('data_preview', dataPreview);
   487→   487→        }
   488→   488→      },
   489→   489→    );
   490→   490→  }
   491→   491→}

   493→   493→// ─── Guarded Async Operation ─────────────────────────────────────────────────
   494→   494→
   495→   495→/// Runs an async operation with automatic Sentry breadcrumb, exception capture,
   496→   496→/// and metadata attachment. This is the recommended way to wrap any async
   497→   497→/// operation that should never fail silently.
   498→   498→///
   499→   499→/// Example:
   500→   500→///   final result = await KlasivoSentry.runGuarded(
   501→   501→///     'create_exam',
   502→   502→///     () => examService.createExam(...),
   503→   503→///     category: 'firestore',
   504→   504→///   );
   505→   505→class KlasivoSentryGuard {
   506→   506→  /// Execute [operation] inside a Sentry-guarded wrapper.
   507→   507→  ///
   508→   508→  /// - Automatically adds a breadcrumb before execution.
   509→   509→  /// - On success: adds a success breadcrumb.
   510→   510→  /// - On failure: captures the exception with scope tags and rethrows.
   511→   511→  /// - The [category] defaults to the operation name but can be overridden.
   512→   512→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   513→   513→  ///   and the exception scope.
   514→   514→  static Future<T> runGuarded<T>(
   515→   515→    String operationName,
   516→   516→    Future<T> Function() operation, {
   517→   517→    String? category,
   518→   518→    Map<String, String>? tags,
   519→   519→    Map<String, dynamic>? data,
   520→   520→  }) async {
   521→   521→    final cat = category ?? 'operation';
   522→   522→
   523→   523→    Sentry.addBreadcrumb(Breadcrumb(
   524→   524→      category: cat,
   525→   525→      message: '${operationName}_started',
   526→   526→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   527→   527→      level: SentryLevel.info,
   528→   528→    ));
   529→   529→
   530→   530→    try {
   531→   531→      final result = await operation();
   532→   532→
   533→   533→      Sentry.addBreadcrumb(Breadcrumb(
   534→   534→        category: cat,
   535→   535→        message: '${operationName}_success',
   536→   536→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   537→   537→        level: SentryLevel.info,
   538→   538→      ));
   539→   539→
   540→   540→      return result;
   541→   541→    } catch (e, st) {
   542→   542→      Sentry.addBreadcrumb(Breadcrumb(
   543→   543→        category: cat,
   544→   544→        message: '${operationName}_failed',
   545→   545→        data: _SensitiveFields.sanitize({
   546→   546→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   547→   547→          ...?data,
   548→   548→        }),
   549→   549→        level: SentryLevel.error,
   550→   550→      ));
   551→   551→
   552→   552→      await Sentry.captureException(
   553→   553→        e,
   554→   554→        stackTrace: st,
   555→   555→        withScope: (scope) {
   556→   556→          scope.setTag('operation', operationName);
   557→   557→          tags?.forEach(scope.setTag);
   558→   558→          if (data != null) {
   559→   559→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   560→   560→          }
   561→   561→        },
   562→   562→      );
   563→   563→
   564→   564→      rethrow;
   565→   565→    }
   566→   566→  }
   567→   567→
   568→   568→  /// Execute [operation] with a Sentry transaction span.
   569→   569→  ///
   570→   570→  /// Creates a child span under the given [transaction], measures duration,
   571→   571→  /// and sets the span status based on success/failure.
   572→   572→  static Future<T> runWithSpan<T>(
   573→   573→    ISentrySpan transaction,
   574→   574→    String spanName,
   575→   575→    Future<T> Function() operation, {
   576→   576→    Map<String, String>? tags,
   577→   577→  }) async {
   578→   578→    final span = transaction.startChild(spanName);
   579→   579→
   580→   580→    try {
   581→   581→      final result = await operation();
   582→   582→      span.status = const SpanStatus.ok();
   583→   583→      return result;
   584→   584→    } catch (e, st) {
   585→   585→      span.status = const SpanStatus.internalError();
   586→   586→
   587→   587→      await Sentry.captureException(
   588→   588→        e,
   589→   589→        stackTrace: st,
   590→   590→        withScope: (scope) {
   591→   591→          scope.setTag('span', spanName);
   592→   592→          tags?.forEach(scope.setTag);
   593→   593→        },
   594→   594→      );
   595→   595→
   596→   596→      rethrow;
   597→   597→    } finally {
   598→   598→      await span.finish();
   599→   599→    }
   600→   600→  }
   601→   601→}

   712→   712→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   713→   713→
   714→   714→/// Documents the doc ID strategy for every user creation path.
   715→   715→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   716→   716→///
   717→   717→/// | Path                  | Collection | Doc ID     | Status    |
   718→   718→/// |-----------------------|-----------|------------|-----------|
   719→   719→/// | Owner (email)         | users     | user.uid   | OK        |
   720→   720→/// | Owner (Google)        | users     | user.uid   | OK        |
   721→   721→/// | Teacher (invite)      | users     | user.uid   | OK        |
   722→   722→/// | Teacher (Google)      | users     | user.uid   | OK        |
   723→   723→/// | Parent (email)        | users     | user.uid   | OK        |
   724→   724→/// | Parent (Google)       | users     | user.uid   | OK        |
   725→   725→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   726→   726→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   727→   727→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   728→   728→///
   729→   729→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   730→   730→/// `.doc(user.uid)`, which means:
   731→   731→///   1. The doc ID doesn't match the auth UID
   732→   732→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   733→   733→///   3. QR enrollment is 100% broken in production
   734→   734→class SentryDocIdAudit {
   735→   735→  /// Log a doc ID audit breadcrumb for any user creation path.
   736→   736→  static void logUserCreation({
   737→   737→    required String flow,
   738→   738→    required String collection,
   739→   739→    required String docIdStrategy, // 'uid' or 'auto_id'
   740→   740→    required String actualDocId,
   741→   741→    String? authUid,
   742→   742→  }) {
   743→   743→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   744→   744→    final isAutoId = docIdStrategy == 'auto_id';
   745→   745→
   746→   746→    Sentry.addBreadcrumb(Breadcrumb(
   747→   747→      category: 'doc_id_audit',
   748→   748→      message: 'user_doc_created',
   749→   749→      data: _SensitiveFields.sanitize({
   750→   750→        'flow': flow,
   751→   751→        'collection': collection,
   752→   752→        'docIdStrategy': docIdStrategy,
   753→   753→        'actualDocId': actualDocId,
   754→   754→        if (authUid != null) 'authUid': authUid,
   755→   755→        'idMismatch': isMismatch,
   756→   756→        'autoIdUsed': isAutoId,
   757→   757→        'brokenByRules': isAutoId, // auto-id always fails security rules
   758→   758→      }),
   759→   759→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   760→   760→    ));
   761→   761→
   762→   762→    if (isMismatch || isAutoId) {
   763→   763→      Sentry.captureMessage(
   764→   764→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   765→   765→        '(docId=$actualDocId, authUid=$authUid). '
   766→   766→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   767→   767→        level: SentryLevel.warning,
   768→   768→      );
   769→   769→    }
   770→   770→  }
   771→   771→}

   865→   865→// ─── Centralized Event Sanitizer ────────────────────────────────────────────
   866→   866→
   867→   867→/// Global Sentry event sanitizer used by the `beforeSend` callback in
   868→   868→/// `main.dart`. Replaces the duplicate `_SentrySanitizer` class that was
   869→   869→/// previously defined in main.dart.
   870→   870→///
   871→   871→/// Strips sensitive data from breadcrumbs, extras, and request headers.
   872→   872→class KlasivoSentrySanitizer {
   873→   873→  KlasivoSentrySanitizer._();
   874→   874→
   875→   875→  /// Sanitize a [SentryEvent] — returns the same event with sensitive data
   876→   876→  /// replaced by '[REDACTED]'.
   877→   877→  static SentryEvent sanitizeEvent(SentryEvent event) {
   878→   878→    // Strip sensitive data from breadcrumbs
   879→   879→    if (event.breadcrumbs != null) {
   880→   880→      for (int i = 0; i < event.breadcrumbs!.length; i++) {
   881→   881→        final crumb = event.breadcrumbs![i];
   882→   882→        if (crumb.data != null) {
   883→   883→          final sanitized = <String, dynamic>{};
   884→   884→          crumb.data!.forEach((key, value) {
   885→   885→            sanitized[key] = _SensitiveFields.isSensitive(key)
   886→   886→                ? '[REDACTED]'
   887→   887→                : value;
   888→   888→          });
   889→   889→          event.breadcrumbs![i] = Breadcrumb(
   890→   890→            category: crumb.category,
   891→   891→            message: crumb.message,
   892→   892→            data: sanitized,
   893→   893→            level: crumb.level,
   894→   894→            timestamp: crumb.timestamp,
   895→   895→            type: crumb.type,
   896→   896→          );
   897→   897→        }
   898→   898→      }
   899→   899→    }
   900→   900→
   901→   901→    // Strip sensitive data from extra fields
   902→   902→    if (event.extra != null) {
   903→   903→      final sanitized = <String, dynamic>{};
   904→   904→      event.extra!.forEach((key, value) {
   905→   905→        sanitized[key] = _SensitiveFields.isSensitive(key)
   906→   906→            ? '[REDACTED]'
   907→   907→            : value;
   908→   908→      });
   909→   909→      event.extra = sanitized;
   910→   910→    }
   911→   911→
   912→   912→    // Strip sensitive data from request headers
   913→   913→    if (event.request?.headers != null) {
   914→   914→      final sanitized = <String, String>{};
   915→   915→      event.request!.headers!.forEach((key, value) {
   916→   916→        final lower = key.toLowerCase();
   917→   917→        if (lower.contains('authorization') ||
   918→   918→            lower.contains('cookie') ||
   919→   919→            lower.contains('token')) {
   920→   920→          sanitized[key] = '[REDACTED]';
   921→   921→        } else {
   922→   922→          sanitized[key] = value;
   923→   923→        }
   924→   924→      });
   925→   925→      event.request = SentryRequest(headers: sanitized);
   926→   926→    }
   927→   927→
   928→   928→    return event;
   929→   929→  }
   930→   930→}
