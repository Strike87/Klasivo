    30→    30→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    31→    31→
    32→    32→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    33→    33→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    34→    34→class _SensitiveFields {
    35→    35→  static const Set<String> keys = {
    36→    36→    'password',
    37→    37→    'passwordHash',
    38→    38→    'confirmPassword',
    39→    39→    'newPassword',
    40→    40→    'accessToken',
    41→    41→    'refreshToken',
    42→    42→    'idToken',
    43→    43→    'otp',
    44→    44→    'otpCode',
    45→    45→    'inviteCode',
    46→    46→    'secret',
    47→    47→    'apiKey',
    48→    48→    'authEmail', // Internal student auth email — PII
    49→    49→    'studentCode', // Sensitive — could be used to log in
    50→    50→  };
    51→    51→
    52→    52→  /// Returns true if the key contains any sensitive substring.
    53→    53→  static bool isSensitive(String key) {
    54→    54→    final lower = key.toLowerCase();
    55→    55→    for (final s in keys) {
    56→    56→      if (lower.contains(s.toLowerCase())) return true;
    57→    57→    }
    58→    58→    return false;
    59→    59→  }
    60→    60→
    61→    61→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    62→    62→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    63→    63→    return data.map((key, value) {
    64→    64→      if (isSensitive(key)) {
    65→    65→        return MapEntry(key, '[REDACTED]');
    66→    66→      }
    67→    67→      // Recursively sanitize nested maps
    68→    68→      if (value is Map<String, dynamic>) {
    69→    69→        return MapEntry(key, sanitize(value));
    70→    70→      }
    71→    71→      return MapEntry(key, value);
    72→    72→    });
    73→    73→  }
    74→    74→
    75→    75→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    76→    76→  static String sanitizeString(String input) {
    77→    77→    // Redact Bearer tokens
    78→    78→    var result = input.replaceAll(
    79→    79→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    80→    80→      'Bearer [REDACTED]',
    81→    81→    );
    82→    82→    // Redact long hex strings (likely tokens)
    83→    83→    result = result.replaceAll(
    84→    84→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    85→    85→      '[REDACTED_TOKEN]',
    86→    86→    );
    87→    87→    return result;
    88→    88→  }
    89→    89→}

   263→   263→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception
   264→   264→/// capture, and scope tags for every critical write.
   265→   265→///
   266→   266→/// Every Firestore exception will:
   267→   267→///   1. Reach Sentry
   268→   268→///   2. Include collection name
   269→   269→///   3. Include document id
   270→   270→///   4. Include user role
   271→   271→///   5. Include organization id
   272→   272→class SentryFirestoreHelper {
   273→   273→  static final FirebaseFirestore _db = FirebaseFirestore.instance;
   274→   274→
   275→   275→  /// Wrapped `.set()` with full observability.
   276→   276→  static Future<void> docSet({
   277→   277→    required String collection,
   278→   278→    required String docId,
   279→   279→    required Map<String, dynamic> data,
   280→   280→    String? flow,
   281→   281→    String? step,
   282→   282→    SetOptions? options,
   283→   283→  }) async {
   284→   284→    final sanitizedData = _SensitiveFields.sanitize(data);
   285→   285→    KlasivoSentry.breadcrumb.firestore(
   286→   286→      'create',
   287→   287→      collection: collection,
   288→   288→      docId: docId,
   289→   289→      data: {'flow': flow, 'step': step},
   290→   290→    );
   291→   291→
   292→   292→    try {
   293→   293→      await _db.collection(collection).doc(docId).set(data, options);
   294→   294→      KlasivoSentry.breadcrumb.firestore(
   295→   295→        'create_success',
   296→   296→        collection: collection,
   297→   297→        docId: docId,
   298→   298→      );
   299→   299→    } catch (e, st) {
   300→   300→      await _captureFirestoreException(
   301→   301→        e,
   302→   302→        st,
   303→   303→        collection: collection,
   304→   304→        operation: 'set',
   305→   305→        docId: docId,
   306→   306→        flow: flow,
   307→   307→        step: step,
   308→   308→        dataPreview: sanitizedData,
   309→   309→      );
   310→   310→      rethrow;
   311→   311→    }
   312→   312→  }

   459→   459→  /// Capture a Firestore exception with full context tags.
   460→   460→  static Future<void> _captureFirestoreException(
   461→   461→    Object exception,
   462→   462→    StackTrace stackTrace, {
   463→   463→    required String collection,
   464→   464→    required String operation,
   465→   465→    String? docId,
   466→   466→    String? flow,
   467→   467→    String? step,
   468→   468→    Map<String, dynamic>? dataPreview,
   469→   469→  }) async {
   470→   470→    // Determine if this is a permission-denied error
   471→   471→    String? errorCode;
   472→   472→    if (exception is FirebaseException) {
   473→   473→      errorCode = exception.code;
   474→   474→    }
   475→   475→
   476→   476→    await Sentry.captureException(
   477→   477→      exception,
   478→   478→      stackTrace: stackTrace,
   479→   479→      withScope: (scope) {
   480→   480→        scope.setTag('collection', collection);
   481→   481→        scope.setTag('operation', operation);
   482→   482→        if (docId != null) scope.setTag('documentId', docId);
   483→   483→        if (flow != null) scope.setTag('flow', flow);
   484→   484→        if (step != null) scope.setTag('step', step);
   485→   485→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   486→   486→
   487→   487→        // Attach user role and org from current Sentry context
   488→   488→        final currentUser = FirebaseAuth.instance.currentUser;
   489→   489→        if (currentUser != null) {
   490→   490→          scope.setTag('auth_uid', currentUser.uid);
   491→   491→        }
   492→   492→
   493→   493→        if (dataPreview != null && dataPreview.isNotEmpty) {
   494→   494→          scope.setExtra('data_preview', dataPreview);
   495→   495→        }
   496→   496→      },
   497→   497→    );
   498→   498→
   499→   499→    // ── Dual-report to Crashlytics ──
   500→   500→    KlasivoCrashlytics.recordError(
   501→   501→      exception,
   502→   502→      stackTrace,
   503→   503→      reason: 'Firestore $operation on $collection${docId != null ? '/$docId' : ''} failed '
   504→   504→          '(flow=$flow, step=$step, firebase_code=$errorCode)',
   505→   505→    );
   506→   506→    KlasivoCrashlytics.setCustomKey('last_firestore_error', '$collection/$operation');
   507→   507→    if (errorCode != null) KlasivoCrashlytics.setCustomKey('firebase_code', errorCode);
   508→   508→  }
   509→   509→}

   516→   516→///
   517→   517→/// Example:
   518→   518→///   final result = await KlasivoSentry.runGuarded(
   519→   519→///     'create_exam',
   520→   520→///     () => examService.createExam(...),
   521→   521→///     category: 'firestore',
   522→   522→///   );
   523→   523→class KlasivoSentryGuard {
   524→   524→  /// Execute [operation] inside a Sentry-guarded wrapper.
   525→   525→  ///
   526→   526→  /// - Automatically adds a breadcrumb before execution.
   527→   527→  /// - On success: adds a success breadcrumb.
   528→   528→  /// - On failure: captures the exception with scope tags and rethrows.
   529→   529→  /// - The [category] defaults to the operation name but can be overridden.
   530→   530→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   531→   531→  ///   and the exception scope.
   532→   532→  static Future<T> runGuarded<T>(
   533→   533→    String operationName,
   534→   534→    Future<T> Function() operation, {
   535→   535→    String? category,
   536→   536→    Map<String, String>? tags,
   537→   537→    Map<String, dynamic>? data,
   538→   538→  }) async {
   539→   539→    final cat = category ?? 'operation';
   540→   540→
   541→   541→    Sentry.addBreadcrumb(Breadcrumb(
   542→   542→      category: cat,
   543→   543→      message: '${operationName}_started',
   544→   544→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   545→   545→      level: SentryLevel.info,
   546→   546→    ));
   547→   547→
   548→   548→    try {
   549→   549→      final result = await operation();
   550→   550→
   551→   551→      Sentry.addBreadcrumb(Breadcrumb(
   552→   552→        category: cat,
   553→   553→        message: '${operationName}_success',
   554→   554→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   555→   555→        level: SentryLevel.info,
   556→   556→      ));
   557→   557→
   558→   558→      return result;
   559→   559→    } catch (e, st) {
   560→   560→      Sentry.addBreadcrumb(Breadcrumb(
   561→   561→        category: cat,
   562→   562→        message: '${operationName}_failed',
   563→   563→        data: _SensitiveFields.sanitize({
   564→   564→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   565→   565→          ...?data,
   566→   566→        }),
   567→   567→        level: SentryLevel.error,
   568→   568→      ));
   569→   569→
   570→   570→      await Sentry.captureException(
   571→   571→        e,
   572→   572→        stackTrace: st,
   573→   573→        withScope: (scope) {
   574→   574→          scope.setTag('operation', operationName);
   575→   575→          tags?.forEach(scope.setTag);
   576→   576→          if (data != null) {
   577→   577→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   578→   578→          }
   579→   579→        },
   580→   580→      );
   581→   581→
   582→   582→      // ── Dual-report to Crashlytics ──
   583→   583→      KlasivoCrashlytics.recordError(e, st, reason: 'runGuarded: $operationName failed');
   584→   584→      if (tags != null) {
   585→   585→        tags.forEach(KlasivoCrashlytics.setCustomKey);
   586→   586→      }
   587→   587→
   588→   588→      rethrow;
   589→   589→    }
   590→   590→  }
   591→   591→
   592→   592→  /// Execute [operation] with a Sentry transaction span.
   593→   593→  ///
   594→   594→  /// Creates a child span under the given [transaction], measures duration,
   595→   595→  /// and sets the span status based on success/failure.
   596→   596→  static Future<T> runWithSpan<T>(
   597→   597→    ISentrySpan transaction,
   598→   598→    String spanName,
   599→   599→    Future<T> Function() operation, {
   600→   600→    Map<String, String>? tags,
   601→   601→  }) async {
   602→   602→    final span = transaction.startChild(spanName);
   603→   603→
   604→   604→    try {
   605→   605→      final result = await operation();
   606→   606→      span.status = const SpanStatus.ok();
   607→   607→      return result;
   608→   608→    } catch (e, st) {
   609→   609→      span.status = const SpanStatus.internalError();
   610→   610→
   611→   611→      await Sentry.captureException(
   612→   612→        e,
   613→   613→        stackTrace: st,
   614→   614→        withScope: (scope) {
   615→   615→          scope.setTag('span', spanName);
   616→   616→          tags?.forEach(scope.setTag);
   617→   617→        },
   618→   618→      );
   619→   619→
   620→   620→      rethrow;
   621→   621→    } finally {
   622→   622→      await span.finish();
   623→   623→    }
   624→   624→  }
   625→   625→}

   772→   772→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   773→   773→
   774→   774→/// Documents the doc ID strategy for every user creation path.
   775→   775→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   776→   776→///
   777→   777→/// | Path                  | Collection | Doc ID     | Status    |
   778→   778→/// |-----------------------|-----------|------------|-----------|
   779→   779→/// | Owner (email)         | users     | user.uid   | OK        |
   780→   780→/// | Owner (Google)        | users     | user.uid   | OK        |
   781→   781→/// | Teacher (invite)      | users     | user.uid   | OK        |
   782→   782→/// | Teacher (Google)      | users     | user.uid   | OK        |
   783→   783→/// | Parent (email)        | users     | user.uid   | OK        |
   784→   784→/// | Parent (Google)       | users     | user.uid   | OK        |
   785→   785→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   786→   786→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   787→   787→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   788→   788→///
   789→   789→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   790→   790→/// `.doc(user.uid)`, which means:
   791→   791→///   1. The doc ID doesn't match the auth UID
   792→   792→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   793→   793→///   3. QR enrollment is 100% broken in production
   794→   794→class SentryDocIdAudit {
   795→   795→  /// Log a doc ID audit breadcrumb for any user creation path.
   796→   796→  static void logUserCreation({
   797→   797→    required String flow,
   798→   798→    required String collection,
   799→   799→    required String docIdStrategy, // 'uid' or 'auto_id'
   800→   800→    required String actualDocId,
   801→   801→    String? authUid,
   802→   802→  }) {
   803→   803→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   804→   804→    final isAutoId = docIdStrategy == 'auto_id';
   805→   805→
   806→   806→    Sentry.addBreadcrumb(Breadcrumb(
   807→   807→      category: 'doc_id_audit',
   808→   808→      message: 'user_doc_created',
   809→   809→      data: _SensitiveFields.sanitize({
   810→   810→        'flow': flow,
   811→   811→        'collection': collection,
   812→   812→        'docIdStrategy': docIdStrategy,
   813→   813→        'actualDocId': actualDocId,
   814→   814→        if (authUid != null) 'authUid': authUid,
   815→   815→        'idMismatch': isMismatch,
   816→   816→        'autoIdUsed': isAutoId,
   817→   817→        'brokenByRules': isAutoId, // auto-id always fails security rules
   818→   818→      }),
   819→   819→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   820→   820→    ));
   821→   821→
   822→   822→    if (isMismatch || isAutoId) {
   823→   823→      Sentry.captureMessage(
   824→   824→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   825→   825→        '(docId=$actualDocId, authUid=$authUid). '
   826→   826→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   827→   827→        level: SentryLevel.warning,
   828→   828→      );
   829→   829→    }
   830→   830→  }
   831→   831→}

   925→   925→// ─── Centralized Event Sanitizer ────────────────────────────────────────────
   926→   926→
   927→   927→/// Global Sentry event sanitizer used by the `beforeSend` callback in
   928→   928→/// `main.dart`. Replaces the duplicate `_SentrySanitizer` class that was
   929→   929→/// previously defined in main.dart.
   930→   930→///
   931→   931→/// Strips sensitive data from breadcrumbs, extras, and request headers.
   932→   932→class KlasivoSentrySanitizer {
   933→   933→  KlasivoSentrySanitizer._();
   934→   934→
   935→   935→  /// Sanitize a [SentryEvent] — returns the same event with sensitive data
   936→   936→  /// replaced by '[REDACTED]'.
   937→   937→  static SentryEvent sanitizeEvent(SentryEvent event) {
   938→   938→    // Strip sensitive data from breadcrumbs
   939→   939→    if (event.breadcrumbs != null) {
   940→   940→      for (int i = 0; i < event.breadcrumbs!.length; i++) {
   941→   941→        final crumb = event.breadcrumbs![i];
   942→   942→        if (crumb.data != null) {
   943→   943→          final sanitized = <String, dynamic>{};
   944→   944→          crumb.data!.forEach((key, value) {
   945→   945→            sanitized[key] = _SensitiveFields.isSensitive(key)
   946→   946→                ? '[REDACTED]'
   947→   947→                : value;
   948→   948→          });
   949→   949→          event.breadcrumbs![i] = Breadcrumb(
   950→   950→            category: crumb.category,
   951→   951→            message: crumb.message,
   952→   952→            data: sanitized,
   953→   953→            level: crumb.level,
   954→   954→            timestamp: crumb.timestamp,
   955→   955→            type: crumb.type,
   956→   956→          );
   957→   957→        }
   958→   958→      }
   959→   959→    }
   960→   960→
   961→   961→    // Strip sensitive data from extra fields
   962→   962→    if (event.extra != null) {
   963→   963→      final sanitized = <String, dynamic>{};
   964→   964→      event.extra!.forEach((key, value) {
   965→   965→        sanitized[key] = _SensitiveFields.isSensitive(key)
   966→   966→            ? '[REDACTED]'
   967→   967→            : value;
   968→   968→      });
   969→   969→      event.extra = sanitized;
   970→   970→    }
   971→   971→
   972→   972→    // Strip sensitive data from request headers
   973→   973→    if (event.request?.headers != null) {
   974→   974→      final sanitized = <String, String>{};
   975→   975→      event.request!.headers!.forEach((key, value) {
   976→   976→        final lower = key.toLowerCase();
   977→   977→        if (lower.contains('authorization') ||
   978→   978→            lower.contains('cookie') ||
   979→   979→            lower.contains('token')) {
   980→   980→          sanitized[key] = '[REDACTED]';
   981→   981→        } else {
   982→   982→          sanitized[key] = value;
   983→   983→        }
   984→   984→      });
   985→   985→      event.request = SentryRequest(headers: sanitized);
   986→   986→    }
   987→   987→
   988→   988→    return event;
   989→   989→  }

  1058→  1058→// ─── Unified Dual-Reporting Facade ────────────────────────────────────────────
  1059→  1059→
  1060→  1060→/// **THE** recommended way to report errors in Klasivo.
  1061→  1061→/// Sends to BOTH Sentry AND Crashlytics in a single call.
  1062→  1062→///
  1063→  1063→/// This is the belt-and-suspenders pattern: if one platform has an outage,
  1064→  1064→/// the other still captures the error.
  1065→  1065→///
  1066→  1066→/// Usage:
  1067→  1067→///   await KlasivoObservability.reportError(
  1068→  1068→///     e,
  1069→  1069→///     st,
  1070→  1070→///     reason: 'Owner registration failed at STEP_2',
  1071→  1071→///     tags: {'flow': 'owner_registration', 'step': 'STEP_2'},
  1072→  1072→///   );
  1073→  1073→///
  1074→  1074→///   await KlasivoObservability.reportMessage(
  1075→  1075→///     'User doc missing after .set()',
  1076→  1076→///     level: SentryLevel.error,
  1077→  1077→///     tags: {'collection': 'users'},
  1078→  1078→///   );
  1079→  1079→class KlasivoObservability {
  1080→  1080→  KlasivoObservability._();
  1081→  1081→
  1082→  1082→  /// Report an error to BOTH Sentry and Crashlytics.
  1083→  1083→  ///
  1084→  1084→  /// - [exception] — the error object
  1085→  1085→  /// - [stackTrace] — the stack trace
  1086→  1086→  /// - [reason] — human-readable description (appears in Crashlytics "Reason")
  1087→  1087→  /// - [tags] — key-value pairs attached to Sentry scope + Crashlytics custom keys
  1088→  1088→  /// - [fatal] — marks as fatal in Crashlytics
  1089→  1089→  /// - [extras] — extra data attached to Sentry scope (not sent to Crashlytics)
  1090→  1090→  static Future<void> reportError(
  1091→  1091→    Object exception,
  1092→  1092→    StackTrace stackTrace, {
  1093→  1093→    String? reason,
  1094→  1094→    Map<String, String>? tags,
  1095→  1095→    Map<String, dynamic>? extras,
  1096→  1096→    bool fatal = false,
  1097→  1097→  }) async {
  1098→  1098→    // ── Sentry ──
  1099→  1099→    await Sentry.captureException(
  1100→  1100→      exception,
  1101→  1101→      stackTrace: stackTrace,
  1102→  1102→      withScope: (scope) {
  1103→  1103→        tags?.forEach(scope.setTag);
  1104→  1104→        if (extras != null) {
  1105→  1105→          extras.forEach((key, value) {
  1106→  1106→            scope.setExtra(
  1107→  1107→              key,
  1108→  1108→              _SensitiveFields.isSensitive(key) ? '[REDACTED]' : value,
  1109→  1109→            );
  1110→  1110→          });
  1111→  1111→        }
  1112→  1112→        if (reason != null) {
  1113→  1113→          scope.setTag(
  1114→  1114→              'reason', reason.length > 200 ? reason.substring(0, 200) : reason);
  1115→  1115→        }
  1116→  1116→      },
  1117→  1117→    );
  1118→  1118→
  1119→  1119→    // ── Crashlytics ──
  1120→  1120→    KlasivoCrashlytics.recordError(
  1121→  1121→      exception,
  1122→  1122→      stackTrace,
  1123→  1123→      reason: reason,
  1124→  1124→      fatal: fatal,
  1125→  1125→    );
  1126→  1126→
  1127→  1127→    // Mirror tags as Crashlytics custom keys for issue clustering
  1128→  1128→    if (tags != null) {
  1129→  1129→      tags.forEach(KlasivoCrashlytics.setCustomKey);
  1130→  1130→    }
  1131→  1131→  }
  1132→  1132→
  1133→  1133→  /// Report a message to BOTH Sentry and Crashlytics.
  1134→  1134→  ///
  1135→  1135→  /// - Sentry: captured as a message event with level
  1136→  1136→  /// - Crashlytics: logged via `log()` for crash breadcrumb context
  1137→  1137→  static Future<void> reportMessage(
  1138→  1138→    String message, {
  1139→  1139→    SentryLevel level = SentryLevel.info,
  1140→  1140→    Map<String, String>? tags,
  1141→  1141→  }) async {
  1142→  1142→    // ── Sentry ──
  1143→  1143→    await Sentry.captureMessage(message, level: level);
  1144→  1144→
  1145→  1145→    // ── Crashlytics ──
  1146→  1146→    final prefix = level == SentryLevel.error
  1147→  1147→        ? 'ERROR'
  1148→  1148→        : level == SentryLevel.warning
  1149→  1149→            ? 'WARN'
  1150→  1150→            : 'INFO';
  1151→  1151→    KlasivoCrashlytics.log('[$prefix] $message');
  1152→  1152→
  1153→  1153→    if (tags != null) {
  1154→  1154→      tags.forEach(KlasivoCrashlytics.setCustomKey);
  1155→  1155→    }
  1156→  1156→  }
  1157→  1157→
  1158→  1158→  /// Add a breadcrumb to BOTH Sentry and Crashlytics.
  1159→  1159→  ///
  1160→  1160→  /// Sentry: full structured breadcrumb
  1161→  1161→  /// Crashlytics: log message for crash context
  1162→  1162→  static void addBreadcrumb(
  1163→  1163→    String message, {
  1164→  1164→    String category = 'app',
  1165→  1165→    Map<String, dynamic>? data,
  1166→  1166→    SentryLevel level = SentryLevel.info,
  1167→  1167→  }) {
  1168→  1168→    // ── Sentry ──
  1169→  1169→    Sentry.addBreadcrumb(Breadcrumb(
  1170→  1170→      category: category,
  1171→  1171→      message: message,
  1172→  1172→      data: data != null ? _SensitiveFields.sanitize(data) : null,
  1173→  1173→      level: level,
  1174→  1174→    ));
  1175→  1175→
  1176→  1176→    // ── Crashlytics ──
  1177→  1177→    final dataStr = data != null && data.isNotEmpty
  1178→  1178→        ? ' | ${data.entries.map((e) => '${e.key}=${e.value}').join(',')}'
  1179→  1179→        : '';
  1180→  1180→    KlasivoCrashlytics.log('[$category] $message$dataStr');
  1181→  1181→  }
  1182→  1182→
  1183→  1183→  /// Convenience: report error with standard registration step tags.
  1184→  1184→  static Future<void> reportRegistrationError(
  1185→  1185→    Object exception,
  1186→  1186→    StackTrace stackTrace, {
  1187→  1187→    required String flow,
  1188→  1188→    required String step,
  1189→  1189→    String? role,
  1190→  1190→    String? uid,
  1191→  1191→    String? reason,
  1192→  1192→  }) async {
  1193→  1193→    await reportError(
  1194→  1194→      exception,
  1195→  1195→      stackTrace,
  1196→  1196→      reason: reason ?? '$flow failed at $step',
  1197→  1197→      tags: {
  1198→  1198→        'flow': flow,
  1199→  1199→        'step': step,
  1200→  1200→        if (role != null) 'role': role,
  1201→  1201→        if (uid != null) 'authUid': uid,
  1202→  1202→      },
  1203→  1203→    );
  1204→  1204→  }
