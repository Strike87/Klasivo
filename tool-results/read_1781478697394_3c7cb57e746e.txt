   253→   253→// ─── Auth Change Notifier for GoRouter refresh ──────────────────────────────
   254→   254→
   255→   255→class AuthChangeNotifier extends ChangeNotifier {
   256→   256→  StreamSubscription<User?>? _firebaseSub;
   257→   257→  StreamSubscription? _hiveSub;
   258→   258→
   259→   259→  AuthChangeNotifier() {
   260→   260→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   261→   261→      (_) => notifyListeners(),
   262→   262→      onError: (_) => notifyListeners(),
   263→   263→    );
   264→   264→    _startHiveWatch();
   265→   265→  }
   266→   266→
   267→   267→  void _startHiveWatch() {
   268→   268→    // Use Hive's built-in watch() instead of polling every 500ms
   269→   269→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   270→   270→      notifyListeners();
   271→   271→    });
   272→   272→  }
   273→   273→
   274→   274→  @override
   275→   275→  void dispose() {
   276→   276→    _firebaseSub?.cancel();
   277→   277→    _hiveSub?.cancel();
   278→   278→    super.dispose();
   279→   279→  }
   280→   280→}
   281→   281→
   282→   282→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   283→   283→  final notifier = AuthChangeNotifier();
   284→   284→  ref.onDispose(() => notifier.dispose());
   285→   285→  return notifier;
   286→   286→});

   288→   288→// ─── GoRouter with Auth Guards & v1.7 Complete Navigation ─────────────────────
   289→   289→
   290→   290→final routerProvider = Provider<GoRouter>((ref) {
   291→   291→  final notifier = ref.watch(authChangeNotifierProvider);
   292→   292→
   293→   293→  return GoRouter(
   294→   294→    initialLocation: '/',
   295→   295→    debugLogDiagnostics: true,
   296→   296→    refreshListenable: notifier,
   297→   297→    redirect: (context, state) {
   298→   298→      final box = Hive.box(AppConstants.authBox);
   299→   299→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   300→   300→      final userRole = box.get('userRole', defaultValue: '') as String;
   301→   301→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: false) as bool;
   302→   302→      final mustChangePassword = box.get('mustChangePassword', defaultValue: false) as bool;
   303→   303→      final isOnChangePassword = state.matchedLocation == '/change-password';
   304→   304→
   305→   305→      // Force redirect to change password if required
   306→   306→      if (isLoggedIn && mustChangePassword && !isOnChangePassword) {
   307→   307→        return '/change-password';
   308→   308→      }
   309→   309→      // Prevent navigating away from change password if forced
   310→   310→      if (isLoggedIn && mustChangePassword && isOnChangePassword) {
   311→   311→        return null; // Allow staying on change password
   312→   312→      }
   313→   313→      // If no longer required, redirect away from change password
   314→   314→      if (isLoggedIn && !mustChangePassword && isOnChangePassword) {
   315→   315→        // Management roles → /dashboard; student → /student; parent → /parent
   316→   316→        if (KlasivoRole.managementRoles.contains(userRole)) return '/dashboard';
   317→   317→        if (userRole == KlasivoRole.student) return '/student';
   318→   318→        if (userRole == KlasivoRole.parent) return '/parent';
   319→   319→        return '/dashboard'; // Fallback for unknown roles
   320→   320→      }
   321→   321→
   322→   322→      final isOnSplash = state.matchedLocation == '/';
   323→   323→      final isOnAuth = state.matchedLocation.startsWith('/auth');
   324→   324→      final isOnWelcome = state.matchedLocation == '/welcome';
   325→   325→      final isOnDashboard = state.matchedLocation.startsWith('/dashboard') ||
   326→   326→          state.matchedLocation.startsWith('/academic') ||
   327→   327→          state.matchedLocation.startsWith('/people') ||
   328→   328→          state.matchedLocation.startsWith('/inbox') ||
   329→   329→          state.matchedLocation.startsWith('/settings') ||
   330→   330→          state.matchedLocation.startsWith('/teacher') ||
   331→   331→          state.matchedLocation.startsWith('/lms');
   332→   332→      final isOnStudent = state.matchedLocation.startsWith('/student');
   333→   333→      final isOnParent = state.matchedLocation.startsWith('/parent') ||
   334→   334→          state.matchedLocation.startsWith('/auth/parent');
   335→   335→
   336→   336→      // Splash → redirect based on auth state
   337→   337→      if (isOnSplash) {
   338→   338→        if (isLoggedIn && userRole.isNotEmpty) {
   339→   339→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   340→   340→            return '/welcome';
   341→   341→          }
   342→   342→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   343→   343→            return '/dashboard';
   344→   344→          }
   345→   345→          if (userRole == AppConstants.roleStudent) return '/student';
   346→   346→          if (userRole == AppConstants.roleParent) return '/parent';
   347→   347→        }
   348→   348→        return '/auth';
   349→   349→      }
   350→   350→
   351→   351→      // Welcome screen — only accessible to logged-in owners who haven't completed setup
   352→   352→      if (isOnWelcome) {
   353→   353→        if (!isLoggedIn) return '/auth';
   354→   354→        if (userRole != AppConstants.roleOwner) return '/dashboard';
   355→   355→        if (hasCompletedSetup) return '/dashboard';
   356→   356→        return null;
   357→   357→      }
   358→   358→
   359→   359→      // Auth screens → redirect if already logged in
   360→   360→      if (isOnAuth) {
   361→   361→        if (isLoggedIn && userRole.isNotEmpty) {
   362→   362→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   363→   363→            return '/welcome';
   364→   364→          }
   365→   365→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   366→   366→            return '/dashboard';
   367→   367→          }
   368→   368→          if (userRole == AppConstants.roleStudent) return '/student';
   369→   369→          if (userRole == AppConstants.roleParent) return '/parent';
   370→   370→        }
   371→   371→        return null;
   372→   372→      }
   373→   373→
   374→   374→      // Protected screens → require login
   375→   375→      if (isOnDashboard || isOnStudent || isOnParent) {
   376→   376→        if (!isLoggedIn) return '/auth';
   377→   377→        if (userRole.isEmpty) return '/auth';
   378→   378→
   379→   379→        if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   380→   380→          return '/welcome';
   381→   381→        }
   382→   382→
   383→   383→        // ─── Feature Flag Gates ───────────────────────────────────────────
   384→   384→        // LMS routes require lms flag
   385→   385→        if (state.matchedLocation.startsWith('/lms')) {
   386→   386→          final flagService = ref.read(featureFlagServiceProvider);
   387→   387→          if (!flagService.isEnabled(FeatureFlags.lms)) {
   388→   388→            return '/dashboard';
   389→   389→          }
   390→   390→        }
   391→   391→
   392→   392→        // Parent portal requires parentPortal flag
   393→   393→        if (state.matchedLocation.startsWith('/parent') ||
   394→   394→            state.matchedLocation.startsWith('/auth/parent')) {
   395→   395→          final flagService = ref.read(featureFlagServiceProvider);
   396→   396→          if (!flagService.isEnabled(FeatureFlags.parentPortal)) {
   397→   397→            if (userRole == AppConstants.roleParent) return '/auth';
   398→   398→            return '/dashboard';
   399→   399→          }
   400→   400→        }
   401→   401→
   402→   402→        if ((userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) &&
   403→   403→            isOnDashboard) {
   404→   404→          return null;
   405→   405→        }
   406→   406→        if (userRole == AppConstants.roleStudent && isOnStudent) {
   407→   407→          return null;
   408→   408→        }
   409→   409→        if (userRole == AppConstants.roleParent && isOnParent) {
   410→   410→          return null;
   411→   411→        }
   412→   412→
   413→   413→        if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   414→   414→          return '/dashboard';
   415→   415→        }
   416→   416→        if (userRole == AppConstants.roleStudent) return '/student';
   417→   417→        if (userRole == AppConstants.roleParent) return '/parent';
   418→   418→      }
   419→   419→
   420→   420→      return null;
   421→   421→    },
   422→   422→    routes: [
   423→   423→      // ─── Splash ──────────────────────────────────────────────────────
   424→   424→      GoRoute(
   425→   425→        path: '/',
   426→   426→        builder: (context, state) => const SplashScreen(),
   427→   427→      ),
   428→   428→
   429→   429→      // ─── Auth Routes ─────────────────────────────────────────────────
   430→   430→      GoRoute(
   431→   431→        path: '/auth',
   432→   432→        builder: (context, state) => const RoleSelectionScreen(),
   433→   433→        routes: [
   434→   434→          GoRoute(
   435→   435→            path: 'teacher-login',
   436→   436→            builder: (context, state) => const TeacherLoginScreen(),
   437→   437→          ),
   438→   438→          GoRoute(
   439→   439→            path: 'teacher-register',
   440→   440→            builder: (context, state) => const TeacherRegistrationScreen(),
