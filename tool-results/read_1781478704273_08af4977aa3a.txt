    25→  /// Hash a password using SHA-256
    26→  static String hashPassword(String password) {
    27→    final bytes = utf8.encode(password);
    28→    final digest = sha256.convert(bytes);
    29→    return digest.toString();

   301→  /// Unified email+password login for owners, teachers, and parents.
   302→  Future<Map<String, dynamic>> loginWithEmail({
   303→    required String email,
   304→    required String password,
   305→  }) async {
   306→    try {
   307→      final userCredential =
   308→          await FirebaseService.loginWithEmail(email, password);
   309→      final user = userCredential.user!;
   310→
   311→      final userDoc = await _firestore
   312→          .collection(AppConstants.usersCollection)
   313→          .doc(user.uid)
   314→          .get();
   315→
   316→      if (!userDoc.exists) {
   317→        throw Exception('User data not found. Please contact your administrator.');
   318→      }
   319→
   320→      final userData = userDoc.data()!;
   321→      final role = userData['role'] as String?;
   322→      final isActive = userData['isActive'] as bool? ?? true;
   323→
   324→      if (!isActive) {
   325→        await _auth.signOut();
   326→        throw Exception('Your account has been deactivated. Contact your administrator.');
   327→      }
   328→
   329→      // Students should NOT use email login — they use student code login
   330→      if (role == AppConstants.roleStudent) {
   331→        await _auth.signOut();
   332→        throw Exception('Students must login with their student code.');
   333→      }
   334→
   335→      return {
   336→        'id': user.uid,
   337→        'organizationId': userData['organizationId'] ?? '',
   338→        'role': role ?? AppConstants.roleTeacher,
   339→        'authProvider': AuthProviders.password,
   340→        'fullName': userData['fullName'] ?? 'User',
   341→        'email': userData['email'] ?? email,
   342→        'isEmailVerified': user.emailVerified,
   343→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   344→      };
   345→    } catch (e) {
   346→      rethrow;
   347→    }
   348→  }

   350→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   351→
   352→  /// Student login using student code + password.
   353→  /// Internally maps to Firebase Auth for push notifications,
   354→  /// multi-device, password reset, security rules, and analytics.
   355→  Future<Map<String, dynamic>> loginStudent({
   356→    required String studentCode,
   357→    required String password,
   358→  }) async {
   359→    try {
   360→      // Step 1: Find user by studentCode
   361→      final snapshot = await _firestore
   362→          .collection(AppConstants.usersCollection)
   363→          .where('studentCode', isEqualTo: studentCode)
   364→          .where('role', isEqualTo: AppConstants.roleStudent)
   365→          .limit(1)
   366→          .get();
   367→
   368→      if (snapshot.docs.isEmpty) {
   369→        throw Exception('Student not found. Please check your student code.');
   370→      }
   371→
   372→      final userDoc = snapshot.docs.first;
   373→      final student = userDoc.data();
   374→
   375→      // Step 2: Verify password
   376→      final storedPasswordHash = student['passwordHash'] as String?;
   377→      final storedPlaintext = student['password'] as String?;
   378→      final inputHash = hashPassword(password);
   379→
   380→      bool passwordMatches = false;
   381→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   382→        passwordMatches = inputHash == storedPasswordHash;
   383→      } else if (storedPlaintext != null) {
   384→        passwordMatches = password == storedPlaintext;
   385→        // Migrate to hash on successful login
   386→        if (passwordMatches) {
   387→          await _firestore
   388→              .collection(AppConstants.usersCollection)
   389→              .doc(userDoc.id)
   390→              .update({'passwordHash': inputHash});
   391→        }
   392→      }
   393→
   394→      if (!passwordMatches) {
   395→        throw Exception('Invalid password. Please try again.');
   396→      }
   397→
   398→      final isActive = student['isActive'] as bool? ?? true;
   399→      if (!isActive) {
   400→        throw Exception('Your account has been deactivated.');
   401→      }
   402→
   403→      // Step 3: Sign in via Firebase Auth using the student's internal email
   404→      final internalEmail = student['authEmail'] as String?;
   405→      if (internalEmail != null && internalEmail.isNotEmpty) {
   406→        try {
   407→          await _auth.signInWithEmailAndPassword(
   408→            email: internalEmail,
   409→            password: password,
   410→          );
   411→        } catch (authError) {
   412→          // Graceful fallback — still allow login even if Firebase Auth fails
   413→        }
   414→      }
   415→
   416→      return {
   417→        'id': userDoc.id,
   418→        'organizationId': student['organizationId'] ?? '',
   419→        'role': AppConstants.roleStudent,
   420→        'authProvider': AuthProviders.studentCode,
   421→        'fullName': student['fullName'] ?? 'Student',
   422→        'studentCode': student['studentCode'],
   423→        'classId': student['classId'],
   424→        'hasCompletedSetup': true,
   425→      };
   426→    } catch (e) {
   427→      rethrow;
   428→    }
   429→  }

   562→  /// Register a teacher via Google Sign-In with an invite code.
   563→  Future<Map<String, dynamic>> registerTeacherWithGoogle({
   564→    required String inviteCode,
   565→  }) async {
   566→    try {
   567→      // Validate invite code first
   568→      final inviteService = InviteCodeService();
   569→      final codeData = await inviteService.validateInviteCode(inviteCode);
   570→
   571→      if (codeData == null) {
   572→        throw Exception('Invalid or expired invite code.');
   573→      }
   574→
   575→      if (codeData['type'] != AppConstants.inviteTypeTeacher) {
   576→        throw Exception('This invite code is not for teachers.');
   577→      }
   578→
   579→      final organizationId = codeData['organizationId'] as String;
   580→
   581→      // Trigger Google Sign-In
   582→      final googleUser = await GoogleSignIn().signIn();
   583→      if (googleUser == null) {
   584→        throw Exception('Google sign-in was cancelled.');
   585→      }
   586→
   587→      final googleAuth = await googleUser.authentication;
   588→      final credential = GoogleAuthProvider.credential(
   589→        accessToken: googleAuth.accessToken,
   590→        idToken: googleAuth.idToken,
   591→      );
   592→
   593→      final userCredential = await _auth.signInWithCredential(credential);
   594→      final user = userCredential.user!;
   595→
   596→      Sentry.addBreadcrumb(Breadcrumb(
   597→        category: 'registration',
   598→        message: 'STEP_1_AUTH_USER_CREATED',
   599→        data: {'uid': user.uid, 'email': user.email ?? '', 'role': 'teacher', 'authProvider': 'google'},
   600→      ));
   601→
   602→      await Sentry.configureScope((scope) {
   603→        scope.setUser(SentryUser(id: user.uid, email: user.email));
   604→        scope.setTag('role', 'teacher');
   605→        scope.setTag('organizationId', organizationId);
   606→      });
   607→
   608→      // Check if user document already exists
   609→      final userDoc = await _firestore
   610→          .collection(AppConstants.usersCollection)
   611→          .doc(user.uid)
   612→          .get();
   613→
   614→      if (userDoc.exists) {
   615→        Sentry.addBreadcrumb(Breadcrumb(
   616→          category: 'registration',
   617→          message: 'USER_DOC_ALREADY_EXISTS',
   618→          data: {'uid': user.uid, 'flow': 'teacher_google'},
   619→        ));
   620→        // Existing user — log them in
   621→        final userData = userDoc.data()!;
   622→        final isActive = userData['isActive'] as bool? ?? true;
   623→        if (!isActive) {
   624→          await _auth.signOut();
   625→          throw Exception('Your account has been deactivated.');
   626→        }
   627→        return {

   828→  /// Sign in with Google. Works for owners, teachers, and parents.
   829→  /// If the user already exists, logs them in.
   830→  /// If the user is new, creates an account based on [expectedRole]:
   831→  ///   - owner: auto-creates organization (hasCompletedSetup=false)
   832→  ///   - teacher: needs invite code (handled separately via registerTeacherWithGoogle)
   833→  ///   - parent: creates parent account (hasCompletedSetup=false, needs to link child)
   834→  Future<Map<String, dynamic>> loginWithGoogle({

   838→      expectedRole: expectedRole,
   839→      isNewUser: false,

   843→  Future<Map<String, dynamic>> _signInWithGoogle({
   844→    String? expectedRole,
   845→    bool isNewUser = false,
   846→  }) async {
   847→    final flowName = isNewUser ? '${expectedRole ?? 'owner'}_google_registration' : 'google_login';
   848→    final transaction = Sentry.startTransaction(flowName, 'registration');
   849→
   850→    try {
   851→      final googleUser = await GoogleSignIn().signIn();
   852→      if (googleUser == null) {
   853→        throw Exception('Google sign-in was cancelled.');
   854→      }
   855→
   856→      final googleAuth = await googleUser.authentication;
   857→      final credential = GoogleAuthProvider.credential(
   858→        accessToken: googleAuth.accessToken,
   859→        idToken: googleAuth.idToken,
   860→      );
   861→
   862→      final userCredential = await _auth.signInWithCredential(credential);
   863→      final user = userCredential.user!;
   864→
   865→      Sentry.addBreadcrumb(Breadcrumb(
   866→        category: 'registration',
   867→        message: 'STEP_1_AUTH_USER_CREATED',
   868→        data: {'uid': user.uid, 'email': user.email ?? '', 'role': expectedRole ?? 'unknown', 'authProvider': 'google'},
   869→      ));

   871→      await Sentry.configureScope((scope) {
   872→        scope.setUser(SentryUser(id: user.uid, email: user.email));
   873→      });
   874→
   875→      // Check if user document already exists
   876→      final userDoc = await _firestore
   877→          .collection(AppConstants.usersCollection)
   878→          .doc(user.uid)
   879→          .get();
   880→
   881→      if (userDoc.exists) {
   882→        Sentry.addBreadcrumb(Breadcrumb(
   883→          category: 'registration',
   884→          message: 'USER_DOC_ALREADY_EXISTS',
   885→          data: {'uid': user.uid, 'flow': flowName},
   886→        ));
   887→        // Existing user — log them in
   888→        final userData = userDoc.data()!;
   889→        final role = userData['role'] as String?;
   890→        final isActive = userData['isActive'] as bool? ?? true;
   891→
   892→        if (!isActive) {
   893→          await _auth.signOut();
   894→          throw Exception('Your account has been deactivated.');
   895→        }
   896→
   897→        // Students cannot use Google Sign-In
   898→        if (role == AppConstants.roleStudent) {
   899→          await _auth.signOut();
   900→          throw Exception('Students must login with their student code.');
   901→        }

   903→        transaction.status = const SpanStatus.ok();

   906→        return {
   907→          'id': user.uid,
   908→          'organizationId': userData['organizationId'] ?? '',

   930→          category: 'registration',
