    26→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    27→
    28→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    29→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    30→class _SensitiveFields {
    31→  static const Set<String> keys = {
    32→    'password',
    33→    'passwordHash',
    34→    'confirmPassword',
    35→    'newPassword',
    36→    'accessToken',
    37→    'refreshToken',
    38→    'idToken',
    39→    'otp',
    40→    'otpCode',
    41→    'inviteCode',
    42→    'secret',
    43→    'apiKey',
    44→    'authEmail', // Internal student auth email — PII
    45→    'studentCode', // Sensitive — could be used to log in
    46→  };
    47→
    48→  /// Returns true if the key contains any sensitive substring.
    49→  static bool isSensitive(String key) {
    50→    final lower = key.toLowerCase();
    51→    for (final s in keys) {
    52→      if (lower.contains(s.toLowerCase())) return true;
    53→    }
    54→    return false;
    55→  }
    56→
    57→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    58→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    59→    return data.map((key, value) {
    60→      if (isSensitive(key)) {
    61→        return MapEntry(key, '[REDACTED]');
    62→      }
    63→      // Recursively sanitize nested maps
    64→      if (value is Map<String, dynamic>) {
    65→        return MapEntry(key, sanitize(value));
    66→      }
    67→      return MapEntry(key, value);
    68→    });
    69→  }
    70→
    71→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    72→  static String sanitizeString(String input) {
    73→    // Redact Bearer tokens
    74→    var result = input.replaceAll(
    75→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    76→      'Bearer [REDACTED]',
    77→    );
    78→    // Redact long hex strings (likely tokens)
    79→    result = result.replaceAll(
    80→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    81→      '[REDACTED_TOKEN]',
    82→    );
    83→    return result;
    84→  }
    85→}

   253→// ─── Firestore Observability Helper ──────────────────────────────────────────

   255→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception
   256→/// capture, and scope tags for every critical write.
   257→///
   258→/// Every Firestore exception will:
   259→///   1. Reach Sentry
   260→///   2. Include collection name
   261→///   3. Include document id
   262→///   4. Include user role
   263→///   5. Include organization id
   264→class SentryFirestoreHelper {
   265→  static final FirebaseFirestore _db = FirebaseFirestore.instance;
   266→
   267→  /// Wrapped `.set()` with full observability.
   268→  static Future<void> docSet({
   269→    required String collection,
   270→    required String docId,
   271→    required Map<String, dynamic> data,
   272→    String? flow,
   273→    String? step,
   274→    SetOptions? options,
   275→  }) async {
   276→    final sanitizedData = _SensitiveFields.sanitize(data);
   277→    KlasivoSentry.breadcrumb.firestore(
   278→      'create',
   279→      collection: collection,
   280→      docId: docId,
   281→      data: {'flow': flow, 'step': step},
   282→    );
   283→
   284→    try {
   285→      await _db.collection(collection).doc(docId).set(data, options);
   286→      KlasivoSentry.breadcrumb.firestore(
   287→        'create_success',
   288→        collection: collection,
   289→        docId: docId,
   290→      );
   291→    } catch (e, st) {
   292→      await _captureFirestoreException(
   293→        e,
   294→        st,
   295→        collection: collection,
   296→        operation: 'set',
   297→        docId: docId,
   298→        flow: flow,
   299→        step: step,
   300→        dataPreview: sanitizedData,
   301→      );
   302→      rethrow;
   303→    }
   304→  }

   439→      await _captureFirestoreException(
   440→        e,
   441→        st,
   442→        collection: collection,
   443→        operation: 'transaction',
   444→        flow: flow,
   445→        step: step,
   446→      );

   448→    }

   451→  /// Capture a Firestore exception with full context tags.
   452→  static Future<void> _captureFirestoreException(
   453→    Object exception,
   454→    StackTrace stackTrace, {
   455→    required String collection,
   456→    required String operation,
   457→    String? docId,
   458→    String? flow,
   459→    String? step,
   460→    Map<String, dynamic>? dataPreview,
   461→  }) async {
   462→    // Determine if this is a permission-denied error
   463→    String? errorCode;
   464→    if (exception is FirebaseException) {
   465→      errorCode = exception.code;
   466→    }
   467→
   468→    await Sentry.captureException(
   469→      exception,
   470→      stackTrace: stackTrace,
   471→      withScope: (scope) {
   472→        scope.setTag('collection', collection);
   473→        scope.setTag('operation', operation);
   474→        if (docId != null) scope.setTag('documentId', docId);
   475→        if (flow != null) scope.setTag('flow', flow);
   476→        if (step != null) scope.setTag('step', step);
   477→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   478→
   479→        // Attach user role and org from current Sentry context
   480→        final currentUser = FirebaseAuth.instance.currentUser;
   481→        if (currentUser != null) {
   482→          scope.setTag('auth_uid', currentUser.uid);
   483→        }
   484→
   485→        if (dataPreview != null && dataPreview.isNotEmpty) {
   486→          scope.setExtra('data_preview', dataPreview);
   487→        }
   488→      },
   489→    );
   490→  }
   491→}

   493→// ─── Guarded Async Operation ─────────────────────────────────────────────────
   494→
   495→/// Runs an async operation with automatic Sentry breadcrumb, exception capture,
   496→/// and metadata attachment. This is the recommended way to wrap any async
   497→/// operation that should never fail silently.
   498→///
   499→/// Example:
   500→///   final result = await KlasivoSentry.runGuarded(
   501→///     'create_exam',
   502→///     () => examService.createExam(...),
   503→///     category: 'firestore',
   504→///   );
   505→class KlasivoSentryGuard {
   506→  /// Execute [operation] inside a Sentry-guarded wrapper.
   507→  ///
   508→  /// - Automatically adds a breadcrumb before execution.
   509→  /// - On success: adds a success breadcrumb.
   510→  /// - On failure: captures the exception with scope tags and rethrows.
   511→  /// - The [category] defaults to the operation name but can be overridden.
   512→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   513→  ///   and the exception scope.
   514→  static Future<T> runGuarded<T>(
   515→    String operationName,
   516→    Future<T> Function() operation, {
   517→    String? category,
   518→    Map<String, String>? tags,
   519→    Map<String, dynamic>? data,
   520→  }) async {
   521→    final cat = category ?? 'operation';
   522→
   523→    Sentry.addBreadcrumb(Breadcrumb(
   524→      category: cat,
   525→      message: '${operationName}_started',
   526→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   527→      level: SentryLevel.info,
   528→    ));
   529→
   530→    try {
   531→      final result = await operation();
   532→
   533→      Sentry.addBreadcrumb(Breadcrumb(
   534→        category: cat,
   535→        message: '${operationName}_success',
   536→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   537→        level: SentryLevel.info,
   538→      ));
   539→
   540→      return result;
   541→    } catch (e, st) {
   542→      Sentry.addBreadcrumb(Breadcrumb(
   543→        category: cat,
   544→        message: '${operationName}_failed',
   545→        data: _SensitiveFields.sanitize({
   546→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   547→          ...?data,
   548→        }),
   549→        level: SentryLevel.error,
   550→      ));
   551→
   552→      await Sentry.captureException(
   553→        e,
   554→        stackTrace: st,
   555→        withScope: (scope) {
   556→          scope.setTag('operation', operationName);
   557→          tags?.forEach(scope.setTag);
   558→          if (data != null) {
   559→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   560→          }
   561→        },
   562→      );
   563→
   564→      rethrow;
   565→    }
   566→  }
   567→
   568→  /// Execute [operation] with a Sentry transaction span.
   569→  ///
   570→  /// Creates a child span under the given [transaction], measures duration,
   571→  /// and sets the span status based on success/failure.
   572→  static Future<T> runWithSpan<T>(
   573→    ISentrySpan transaction,
   574→    String spanName,
   575→    Future<T> Function() operation, {
   576→    Map<String, String>? tags,
   577→  }) async {
   578→    final span = transaction.startChild(spanName);
   579→
   580→    try {
   581→      final result = await operation();
   582→      span.status = const SpanStatus.ok();
   583→      return result;
   584→    } catch (e, st) {
   585→      span.status = const SpanStatus.internalError();
   586→
   587→      await Sentry.captureException(
   588→        e,
   589→        stackTrace: st,
   590→        withScope: (scope) {
   591→          scope.setTag('span', spanName);
   592→          tags?.forEach(scope.setTag);
   593→        },
   594→      );
   595→
   596→      rethrow;
   597→    } finally {
   598→      await span.finish();
   599→    }
   600→  }
   601→}

   605→/// Manages Sentry user context — attached on sign-in, updated on role/org
   606→/// changes, cleared on logout.
   607→///
   608→/// Attaches:
   609→///   - uid
   610→///   - email
   611→///   - role (tag)

   615→class SentryUserContext {
   616→  /// Set user context after authentication.
   617→  static Future<void> setUser({
   618→    required String uid,
   619→    required String email,
   620→    String? role,
   621→    String? organizationId,
   622→  }) async {
   623→    await Sentry.configureScope((scope) {
   624→      scope.setUser(SentryUser(
   625→        id: uid,
   626→        email: email,
   627→      ));
   628→      if (role != null) {
   629→        scope.setTag('role', role);
   630→      }
   631→      if (organizationId != null && organizationId.isNotEmpty) {
   632→        scope.setTag('organizationId', organizationId);
   633→      }
   634→    });
   635→  }

   641→    });

   647→      scope.setTag('organizationId', orgId);

   651→  /// Set app version and build number tags (called once at init).
   652→  static Future<void> setAppVersion({
   653→    required String version,
   654→    required String buildNumber,

   712→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   713→
   714→/// Documents the doc ID strategy for every user creation path.
   715→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   716→///
   717→/// | Path                  | Collection | Doc ID     | Status    |
   718→/// |-----------------------|-----------|------------|-----------|
   719→/// | Owner (email)         | users     | user.uid   | OK        |
   720→/// | Owner (Google)        | users     | user.uid   | OK        |
   721→/// | Teacher (invite)      | users     | user.uid   | OK        |
   722→/// | Teacher (Google)      | users     | user.uid   | OK        |
   723→/// | Parent (email)        | users     | user.uid   | OK        |
   724→/// | Parent (Google)       | users     | user.uid   | OK        |
   725→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   726→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   727→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   728→///
   729→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   730→/// `.doc(user.uid)`, which means:
   731→///   1. The doc ID doesn't match the auth UID
   732→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   733→///   3. QR enrollment is 100% broken in production
   734→class SentryDocIdAudit {
   735→  /// Log a doc ID audit breadcrumb for any user creation path.
   736→  static void logUserCreation({
   737→    required String flow,
   738→    required String collection,
   739→    required String docIdStrategy, // 'uid' or 'auto_id'
   740→    required String actualDocId,
   741→    String? authUid,
   742→  }) {
   743→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   744→    final isAutoId = docIdStrategy == 'auto_id';
   745→
   746→    Sentry.addBreadcrumb(Breadcrumb(
   747→      category: 'doc_id_audit',
   748→      message: 'user_doc_created',
   749→      data: _SensitiveFields.sanitize({
   750→        'flow': flow,
   751→        'collection': collection,
   752→        'docIdStrategy': docIdStrategy,
   753→        'actualDocId': actualDocId,
   754→        if (authUid != null) 'authUid': authUid,
   755→        'idMismatch': isMismatch,
   756→        'autoIdUsed': isAutoId,
   757→        'brokenByRules': isAutoId, // auto-id always fails security rules
   758→      }),
   759→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   760→    ));
   761→
   762→    if (isMismatch || isAutoId) {
   763→      Sentry.captureMessage(
   764→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   765→        '(docId=$actualDocId, authUid=$authUid). '
   766→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   767→        level: SentryLevel.warning,
   768→      );
   769→    }
   770→  }
   771→}

   775→/// Top-level entry point for all Sentry observability in Klasivo.

   777→/// Usage:
   778→///   KlasivoSentry.breadcrumb.auth('login_started');

   785→  /// Breadcrumb builder — use for all breadcrumb creation.
   786→  static final breadcrumb = SentryBreadcrumbBuilder();

   788→  /// Firestore operation helper — use for all critical Firestore writes.
   789→  static final firestore = SentryFirestoreHelper();

   791→  /// Guarded operation runner — use to wrap any async operation.

   797→  /// Transaction factory — use to start named transactions for flows.
   798→  static final transactions = SentryTransactions();
   799→
   800→  /// Doc ID audit — use when creating any user document.
   801→  static final docIdAudit = SentryDocIdAudit();
   802→
   803→  /// Convenience method for runGuarded.
   804→  static Future<T> runGuarded<T>(
   805→    String operationName,
   806→    Future<T> Function() operation, {
   807→    String? category,
   808→    Map<String, String>? tags,
   809→    Map<String, dynamic>? data,
   810→  }) {

   812→      operationName,

   820→  /// Capture an exception with optional scope tags.
   821→  static Future<SentryId> captureException(
   822→    Object exception, {
   823→    StackTrace? stackTrace,
   824→    Map<String, String>? tags,
   825→    Map<String, dynamic>? extras,
   826→  }) {
   827→    return Sentry.captureException(
   828→      exception,
   829→      stackTrace: stackTrace,
   830→      withScope: (scope) {
   831→        tags?.forEach(scope.setTag);
   832→        extras?.forEach((key, value) {
   833→          scope.setExtra(key, _SensitiveFields.isSensitive(key) ? '[REDACTED]' : value);
   834→        });
   835→      },
   836→    );
   837→  }

   840→  static Future<SentryId> captureMessage(
   841→    String message, {
   842→    SentryLevel level = SentryLevel.info,
   843→    Map<String, String>? tags,

   845→    return Sentry.captureMessage(
   846→      message,
   847→      level: level,
   848→    );

   865→// ─── Centralized Event Sanitizer ────────────────────────────────────────────
   866→
   867→/// Global Sentry event sanitizer used by the `beforeSend` callback in
   868→/// `main.dart`. Replaces the duplicate `_SentrySanitizer` class that was
   869→/// previously defined in main.dart.
   870→///
   871→/// Strips sensitive data from breadcrumbs, extras, and request headers.
   872→class KlasivoSentrySanitizer {
   873→  KlasivoSentrySanitizer._();
   874→
   875→  /// Sanitize a [SentryEvent] — returns the same event with sensitive data
   876→  /// replaced by '[REDACTED]'.
   877→  static SentryEvent sanitizeEvent(SentryEvent event) {
   878→    // Strip sensitive data from breadcrumbs
   879→    if (event.breadcrumbs != null) {
   880→      for (int i = 0; i < event.breadcrumbs!.length; i++) {
   881→        final crumb = event.breadcrumbs![i];
   882→        if (crumb.data != null) {
   883→          final sanitized = <String, dynamic>{};
   884→          crumb.data!.forEach((key, value) {
   885→            sanitized[key] = _SensitiveFields.isSensitive(key)
   886→                ? '[REDACTED]'
   887→                : value;
   888→          });
   889→          event.breadcrumbs![i] = Breadcrumb(
   890→            category: crumb.category,
   891→            message: crumb.message,
   892→            data: sanitized,
   893→            level: crumb.level,
   894→            timestamp: crumb.timestamp,
   895→            type: crumb.type,
   896→          );
   897→        }
   898→      }
   899→    }
   900→
   901→    // Strip sensitive data from extra fields
   902→    if (event.extra != null) {
   903→      final sanitized = <String, dynamic>{};
   904→      event.extra!.forEach((key, value) {
   905→        sanitized[key] = _SensitiveFields.isSensitive(key)
   906→            ? '[REDACTED]'
   907→            : value;
   908→      });
   909→      event.extra = sanitized;
   910→    }
   911→
   912→    // Strip sensitive data from request headers
   913→    if (event.request?.headers != null) {
   914→      final sanitized = <String, String>{};
   915→      event.request!.headers!.forEach((key, value) {
   916→        final lower = key.toLowerCase();
   917→        if (lower.contains('authorization') ||
   918→            lower.contains('cookie') ||
   919→            lower.contains('token')) {
   920→          sanitized[key] = '[REDACTED]';
   921→        } else {
   922→          sanitized[key] = value;
   923→        }
   924→      });
   925→      event.request = SentryRequest(headers: sanitized);
   926→    }
   927→
   928→    return event;
   929→  }
   930→}
