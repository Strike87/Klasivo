    27→  /// Hash a password using SHA-256
    28→  static String hashPassword(String password) {
    29→    final bytes = utf8.encode(password);
    30→    final digest = sha256.convert(bytes);
    31→    return digest.toString();
    32→  }

   104→            .collection(AppConstants.usersCollection)
   105→            .doc(user.uid)

   137→      Sentry.addBreadcrumb(Breadcrumb(
   138→        category: 'registration',
   139→        message: 'STEP_2_USER_DOC_CREATE_SUCCESS',

   141→      ));

   168→            scope.setTag('flow', 'owner_registration');
   169→            scope.setTag('step', 'STEP_3_ORG_CREATE');

   172→        rethrow;
   173→      } finally {

   178→        category: 'registration',
   179→        message: 'STEP_3_ORG_CREATE_SUCCESS',
   180→        data: {'orgId': orgId},
   181→      ));
   182→
   183→      // ── Step 4: Patch user doc with real organizationId ────────────────

   216→      // Set Sentry org context after creation

   323→  /// Unified email+password login for owners, teachers, and parents.
   324→  Future<Map<String, dynamic>> loginWithEmail({
   325→    required String email,
   326→    required String password,
   327→  }) async {
   328→    final transaction = KlasivoSentry.transactions.loginFlow('email');
   329→
   330→    try {
   331→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   332→        'method': 'email',
   333→        'email': email,
   334→      });
   335→
   336→      final userCredential =
   337→          await FirebaseService.loginWithEmail(email, password);
   338→      final user = userCredential.user!;
   339→
   340→      KlasivoSentry.breadcrumb.auth('auth_user_authenticated', data: {
   341→        'uid': user.uid,
   342→        'method': 'email',
   343→      });
   344→
   345→      // Set Sentry user context immediately after auth
   346→      await KlasivoSentry.userContext.setUser(
   347→        uid: user.uid,
   348→        email: email,
   349→      );
   350→
   351→      final readUserSpan = transaction.startChild('read_user_doc');
   352→      final userDoc = await _firestore
   353→          .collection(AppConstants.usersCollection)
   354→          .doc(user.uid)
   355→          .get();
   356→      await readUserSpan.finish();
   357→
   358→      if (!userDoc.exists) {
   359→        KlasivoSentry.breadcrumb.auth('login_failed_user_doc_missing', data: {
   360→          'uid': user.uid,
   361→        });
   362→        await Sentry.captureMessage(
   363→          'Login failed: users/${user.uid} document does not exist',
   364→          level: SentryLevel.error,
   365→        );
   366→        throw Exception('User data not found. Please contact your administrator.');
   367→      }
   368→
   369→      final userData = userDoc.data()!;
   370→      final role = userData['role'] as String?;
   371→      final isActive = userData['isActive'] as bool? ?? true;
   372→
   373→      // Update Sentry context with role and org
   374→      await KlasivoSentry.userContext.setRole(role ?? 'unknown');
   375→      final orgId = userData['organizationId'] as String?;
   376→      if (orgId != null && orgId.isNotEmpty) {
   377→        await KlasivoSentry.userContext.setOrganizationId(orgId);
   378→      }
   379→
   380→      if (!isActive) {
   381→        await _auth.signOut();
   382→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   383→          'uid': user.uid,
   384→          'role': role,
   385→        });
   386→        throw Exception('Your account has been deactivated. Contact your administrator.');
   387→      }
   388→
   389→      // Students should NOT use email login — they use student code login
   390→      if (role == AppConstants.roleStudent) {
   391→        await _auth.signOut();
   392→        KlasivoSentry.breadcrumb.auth('login_failed_student_email_login', data: {
   393→          'uid': user.uid,
   394→        });
   395→        throw Exception('Students must login with their student code.');
   396→      }
   397→
   398→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   399→        'uid': user.uid,
   400→        'role': role ?? 'unknown',
   401→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   402→      });
   403→
   404→      transaction.status = const SpanStatus.ok();
   405→
   406→      return {
   407→        'id': user.uid,
   408→        'organizationId': userData['organizationId'] ?? '',
   409→        'role': role ?? AppConstants.roleTeacher,
   410→        'authProvider': AuthProviders.password,
   411→        'fullName': userData['fullName'] ?? 'User',
   412→        'email': userData['email'] ?? email,
   413→        'isEmailVerified': user.emailVerified,
   414→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   415→      };
   416→    } catch (e, st) {
   417→      transaction.status = const SpanStatus.internalError();
   418→      await Sentry.captureException(
   419→        e,
   420→        stackTrace: st,
   421→        withScope: (scope) {
   422→          scope.setTag('flow', 'email_login');
   423→          scope.setTag('step', 'login_with_email');
   424→        },
   425→      );
   426→      rethrow;
   427→    } finally {
   428→      await transaction.finish();
   429→    }
   430→  }

   432→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   433→
   434→  /// Student login using student code + password.
   435→  /// Internally maps to Firebase Auth for push notifications,
   436→  /// multi-device, password reset, security rules, and analytics.
   437→  Future<Map<String, dynamic>> loginStudent({
   438→    required String studentCode,
   439→    required String password,
   440→  }) async {
   441→    final transaction = KlasivoSentry.transactions.loginFlow('student');
   442→
   443→    try {
   444→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   445→        'method': 'student_code',
   446→      });
   447→
   448→      // Step 1: Find user by studentCode
   449→      final findUserSpan = transaction.startChild('find_user_by_code');
   450→      final snapshot = await _firestore
   451→          .collection(AppConstants.usersCollection)
   452→          .where('studentCode', isEqualTo: studentCode)
   453→          .where('role', isEqualTo: AppConstants.roleStudent)
   454→          .limit(1)
   455→          .get();
   456→      await findUserSpan.finish();
   457→
   458→      if (snapshot.docs.isEmpty) {
   459→        KlasivoSentry.breadcrumb.auth('login_failed_student_not_found', data: {
   460→          'method': 'student_code',
   461→        });
   462→        throw Exception('Student not found. Please check your student code.');
   463→      }
   464→
   465→      final userDoc = snapshot.docs.first;
   466→      final student = userDoc.data();
   467→
   468→      // Step 2: Verify password
   469→      final storedPasswordHash = student['passwordHash'] as String?;
   470→      final storedPlaintext = student['password'] as String?;
   471→      final inputHash = hashPassword(password);
   472→
   473→      bool passwordMatches = false;
   474→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   475→        passwordMatches = inputHash == storedPasswordHash;
   476→      } else if (storedPlaintext != null) {
   477→        passwordMatches = password == storedPlaintext;
   478→        // Migrate to hash on successful login
   479→        if (passwordMatches) {
   480→          await _firestore
   481→              .collection(AppConstants.usersCollection)
   482→              .doc(userDoc.id)
   483→              .update({'passwordHash': inputHash});
   484→        }
   485→      }
   486→
   487→      if (!passwordMatches) {
   488→        KlasivoSentry.breadcrumb.auth('login_failed_invalid_password', data: {
   489→          'method': 'student_code',
   490→          'userId': userDoc.id,
   491→        });
   492→        throw Exception('Invalid password. Please try again.');
   493→      }
   494→
   495→      final isActive = student['isActive'] as bool? ?? true;
   496→      if (!isActive) {
   497→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   498→          'userId': userDoc.id,
   499→          'role': 'student',
   500→        });
   501→        throw Exception('Your account has been deactivated.');
   502→      }
   503→
   504→      // Step 3: Sign in via Firebase Auth using the student's internal email
   505→      final internalEmail = student['authEmail'] as String?;
   506→      if (internalEmail != null && internalEmail.isNotEmpty) {
   507→        try {
   508→          await _auth.signInWithEmailAndPassword(
   509→            email: internalEmail,
   510→            password: password,
   511→          );
   512→          KlasivoSentry.breadcrumb.auth('student_auth_signed_in', data: {
   513→            'userId': userDoc.id,
   514→          });
   515→        } catch (authError) {
   516→          // Graceful fallback — still allow login even if Firebase Auth fails
   517→          KlasivoSentry.breadcrumb.auth('student_auth_signin_failed_fallback', data: {
   518→            'userId': userDoc.id,
   519→            'error': authError.toString().substring(0, (authError.toString().length).clamp(0, 100)),
   520→          });
   521→        }
   522→      }
   523→
   524→      // Set Sentry user context
   525→      await KlasivoSentry.userContext.setUser(
   526→        uid: userDoc.id,
   527→        email: internalEmail ?? '',
   528→        role: AppConstants.roleStudent,
   529→        organizationId: student['organizationId'] as String?,
   530→      );
   531→
   532→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   533→        'uid': userDoc.id,
   534→        'role': 'student',
   535→        'method': 'student_code',
   536→      });
   537→
   538→      // Log doc ID audit — student uses Firestore doc ID (not auth UID)
   539→      KlasivoSentry.docIdAudit.logUserCreation(
   540→        flow: 'student_code_login',
   541→        collection: AppConstants.usersCollection,
   542→        docIdStrategy: userDoc.id.length > 28 ? 'auto_id' : 'uid',
   543→        actualDocId: userDoc.id,
   544→        authUid: _auth.currentUser?.uid,
   545→      );
   546→
   547→      transaction.status = const SpanStatus.ok();
   548→
   549→      return {
   550→        'id': userDoc.id,
   551→        'organizationId': student['organizationId'] ?? '',
   552→        'role': AppConstants.roleStudent,
   553→        'authProvider': AuthProviders.studentCode,
   554→        'fullName': student['fullName'] ?? 'Student',
   555→        'studentCode': student['studentCode'],
   556→        'classId': student['classId'],
   557→        'hasCompletedSetup': true,
   558→      };
   559→    } catch (e, st) {
   560→      transaction.status = const SpanStatus.internalError();
   561→      await Sentry.captureException(
   562→        e,
   563→        stackTrace: st,
   564→        withScope: (scope) {
   565→          scope.setTag('flow', 'student_login');
   566→          scope.setTag('step', 'login_student');
   567→          scope.setTag('method', 'student_code');
   568→        },
   569→      );
   570→      rethrow;
   571→    } finally {
   572→      await transaction.finish();
   573→    }
   574→  }

   578→  Future<Map<String, dynamic>> registerTeacherWithInvite({
   579→    required String email,
   580→    required String password,

   606→          await FirebaseService.registerWithEmail(email, password);

   608→

   610→        category: 'registration',

  1208→          data: {'uid': user.uid, 'organizationId': organizationId},
  1209→        ));
  1210→
  1211→        hasCompletedSetup = false; // Needs to name workspace
  1212→      } else if (role == AppConstants.roleParent) {
  1213→        organizationId = null; // Set when parent links a child
  1214→        hasCompletedSetup = false; // Needs to link child
  1215→      }
  1216→
  1217→      // For non-owner roles, write the user doc normally (owner doc already created above)
  1218→      if (role != AppConstants.roleOwner) {
  1219→        Sentry.addBreadcrumb(Breadcrumb(
  1220→          category: 'registration',
  1221→          message: 'STEP_2_USER_DOC_CREATE_START',
  1222→          data: {'role': role},
  1223→        ));
  1224→
  1225→        try {
  1226→          await SentryFirestoreHelper.docSet(
  1227→            collection: AppConstants.usersCollection,
  1228→            docId: user.uid,
  1229→            data: {
  1230→              'organizationId': organizationId,
  1231→              'role': role,
  1232→              'authProvider': AuthProviders.google,
  1233→              'fullName': fullName,
  1234→              'email': email,
  1235→              'photoUrl': user.photoURL,
  1236→              'phoneNumber': null,
  1237→              'isActive': true,
  1238→              'isEmailVerified': isEmailVerified,
  1239→              'hasCompletedSetup': hasCompletedSetup,
  1240→              'createdAt': FieldValue.serverTimestamp(),
  1241→              'updatedAt': FieldValue.serverTimestamp(),
  1242→            },
  1243→            flow: flowName,
  1244→            step: 'STEP_2_USER_DOC_CREATE',
  1245→          );
  1246→
  1247→          // Read-back verification: confirm the document actually exists
  1248→          final verifyDoc = await _firestore
  1249→              .collection(AppConstants.usersCollection)
  1250→              .doc(user.uid)
  1251→              .get();
  1252→          if (!verifyDoc.exists) {
  1253→            await Sentry.captureMessage(
  1254→              'STEP_2 USER DOC SET SUCCEEDED BUT READ-BACK FAILED — doc users/${user.uid} does not exist (role=$role)',
  1255→              level: SentryLevel.error,
  1256→            );
  1257→          }
  1258→
  1259→          KlasivoSentry.docIdAudit.logUserCreation(
  1260→            flow: flowName,
  1261→            collection: AppConstants.usersCollection,
  1262→            docIdStrategy: 'uid',
  1263→            actualDocId: user.uid,
  1264→            authUid: user.uid,
  1265→          );
  1266→        } catch (e, st) {
  1267→          rethrow;
  1268→        }
  1269→      }
  1270→
  1271→      // Set Sentry org/role context after all writes
  1272→      await Sentry.configureScope((scope) {
  1273→        scope.setTag('role', role);
  1274→        if (organizationId != null) {
  1275→          scope.setTag('organizationId', organizationId);
  1276→        }
  1277→      });
  1278→
  1279→      transaction.status = const SpanStatus.ok();
  1280→
  1281→      return {
  1282→        'id': user.uid,
  1283→        'organizationId': organizationId,
  1284→        'role': role,
  1285→        'authProvider': AuthProviders.google,
  1286→        'fullName': fullName,
  1287→        'email': email,
  1288→        'isEmailVerified': isEmailVerified,
  1289→        'hasCompletedSetup': hasCompletedSetup,
  1290→      };
  1291→    } catch (e) {
  1292→      transaction.status = const SpanStatus.internalError();
  1293→      rethrow;
  1294→    } finally {
  1295→      await transaction.finish();
  1296→    }
  1297→  }

  1426→
  1427→      final data = userDoc.data()!;
  1428→      final role = data['role'] as String?;
  1429→      final hasCompletedSetup = data['hasCompletedSetup'] as bool? ?? true;
  1430→
  1431→      // Owners need workspace naming, parents need child linking
  1432→      if ((role == AppConstants.roleOwner || role == AppConstants.roleParent) &&
  1433→          !hasCompletedSetup) {
  1434→        return true;
  1435→      }
  1436→
