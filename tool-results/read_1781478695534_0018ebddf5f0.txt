   126→
   127→    await FirebaseCrashlytics.instance.setCrashlyticsCollectionEnabled(
   128→      !kDebugMode,

   253→// ─── Auth Change Notifier for GoRouter refresh ──────────────────────────────
   254→
   255→class AuthChangeNotifier extends ChangeNotifier {
   256→  StreamSubscription<User?>? _firebaseSub;
   257→  StreamSubscription? _hiveSub;
   258→
   259→  AuthChangeNotifier() {
   260→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   261→      (_) => notifyListeners(),
   262→      onError: (_) => notifyListeners(),
   263→    );
   264→    _startHiveWatch();
   265→  }
   266→
   267→  void _startHiveWatch() {
   268→    // Use Hive's built-in watch() instead of polling every 500ms
   269→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   270→      notifyListeners();
   271→    });
   272→  }
   273→
   274→  @override
   275→  void dispose() {
   276→    _firebaseSub?.cancel();
   277→    _hiveSub?.cancel();
   278→    super.dispose();
   279→  }
   280→}
   281→
   282→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   283→  final notifier = AuthChangeNotifier();
   284→  ref.onDispose(() => notifier.dispose());
   285→  return notifier;
   286→});
   287→
   288→// ─── GoRouter with Auth Guards & v1.7 Complete Navigation ─────────────────────
   289→
   290→final routerProvider = Provider<GoRouter>((ref) {
   291→  final notifier = ref.watch(authChangeNotifierProvider);
   292→
   293→  return GoRouter(
   294→    initialLocation: '/',
   295→    debugLogDiagnostics: true,
   296→    refreshListenable: notifier,
   297→    redirect: (context, state) {
   298→      final box = Hive.box(AppConstants.authBox);
   299→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   300→      final userRole = box.get('userRole', defaultValue: '') as String;
   301→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: false) as bool;
   302→      final mustChangePassword = box.get('mustChangePassword', defaultValue: false) as bool;
   303→      final isOnChangePassword = state.matchedLocation == '/change-password';
   304→
   305→      // Force redirect to change password if required
   306→      if (isLoggedIn && mustChangePassword && !isOnChangePassword) {
   307→        return '/change-password';
   308→      }
   309→      // Prevent navigating away from change password if forced
   310→      if (isLoggedIn && mustChangePassword && isOnChangePassword) {
   311→        return null; // Allow staying on change password
   312→      }
   313→      // If no longer required, redirect away from change password
   314→      if (isLoggedIn && !mustChangePassword && isOnChangePassword) {
   315→        // Management roles → /dashboard; student → /student; parent → /parent
   316→        if (KlasivoRole.managementRoles.contains(userRole)) return '/dashboard';
   317→        if (userRole == KlasivoRole.student) return '/student';
   318→        if (userRole == KlasivoRole.parent) return '/parent';
   319→        return '/dashboard'; // Fallback for unknown roles
   320→      }
   321→
   322→      final isOnSplash = state.matchedLocation == '/';
   323→      final isOnAuth = state.matchedLocation.startsWith('/auth');
   324→      final isOnWelcome = state.matchedLocation == '/welcome';
   325→      final isOnDashboard = state.matchedLocation.startsWith('/dashboard') ||
   326→          state.matchedLocation.startsWith('/academic') ||
   327→          state.matchedLocation.startsWith('/people') ||
   328→          state.matchedLocation.startsWith('/inbox') ||
   329→          state.matchedLocation.startsWith('/settings') ||
   330→          state.matchedLocation.startsWith('/teacher') ||
   331→          state.matchedLocation.startsWith('/lms');
   332→      final isOnStudent = state.matchedLocation.startsWith('/student');
   333→      final isOnParent = state.matchedLocation.startsWith('/parent') ||
   334→          state.matchedLocation.startsWith('/auth/parent');
   335→
   336→      // Splash → redirect based on auth state
   337→      if (isOnSplash) {
   338→        if (isLoggedIn && userRole.isNotEmpty) {
   339→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   340→            return '/welcome';
   341→          }
   342→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   343→            return '/dashboard';
   344→          }
   345→          if (userRole == AppConstants.roleStudent) return '/student';
   346→          if (userRole == AppConstants.roleParent) return '/parent';
   347→        }
   348→        return '/auth';
   349→      }
   350→
   351→      // Welcome screen — only accessible to logged-in owners who haven't completed setup
   352→      if (isOnWelcome) {
   353→        if (!isLoggedIn) return '/auth';
   354→        if (userRole != AppConstants.roleOwner) return '/dashboard';
   355→        if (hasCompletedSetup) return '/dashboard';
   356→        return null;
   357→      }
   358→
   359→      // Auth screens → redirect if already logged in
   360→      if (isOnAuth) {
   361→        if (isLoggedIn && userRole.isNotEmpty) {
   362→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   363→            return '/welcome';
   364→          }
   365→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   366→            return '/dashboard';
   367→          }
   368→          if (userRole == AppConstants.roleStudent) return '/student';
   369→          if (userRole == AppConstants.roleParent) return '/parent';
   370→        }
   371→        return null;
   372→      }
   373→
   374→      // Protected screens → require login
   375→      if (isOnDashboard || isOnStudent || isOnParent) {
   376→        if (!isLoggedIn) return '/auth';
   377→        if (userRole.isEmpty) return '/auth';
   378→
   379→        if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   380→          return '/welcome';
   381→        }
   382→
   383→        // ─── Feature Flag Gates ───────────────────────────────────────────
   384→        // LMS routes require lms flag
   385→        if (state.matchedLocation.startsWith('/lms')) {
   386→          final flagService = ref.read(featureFlagServiceProvider);
   387→          if (!flagService.isEnabled(FeatureFlags.lms)) {
   388→            return '/dashboard';
   389→          }
   390→        }
   391→
   392→        // Parent portal requires parentPortal flag
   393→        if (state.matchedLocation.startsWith('/parent') ||
   394→            state.matchedLocation.startsWith('/auth/parent')) {
   395→          final flagService = ref.read(featureFlagServiceProvider);
   396→          if (!flagService.isEnabled(FeatureFlags.parentPortal)) {
   397→            if (userRole == AppConstants.roleParent) return '/auth';
   398→            return '/dashboard';
   399→          }
   400→        }
   401→
   402→        if ((userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) &&
   403→            isOnDashboard) {
   404→          return null;
   405→        }
   406→        if (userRole == AppConstants.roleStudent && isOnStudent) {
   407→          return null;
   408→        }
   409→        if (userRole == AppConstants.roleParent && isOnParent) {
   410→          return null;
   411→        }
   412→
   413→        if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   414→          return '/dashboard';
   415→        }
   416→        if (userRole == AppConstants.roleStudent) return '/student';
   417→        if (userRole == AppConstants.roleParent) return '/parent';
   418→      }
   419→
   420→      return null;
   421→    },
