    27→    27→  /// Hash a password using SHA-256
    28→    28→  static String hashPassword(String password) {
    29→    29→    final bytes = utf8.encode(password);
    30→    30→    final digest = sha256.convert(bytes);
    31→    31→    return digest.toString();

   321→   321→  // ─── Teacher/Owner/Parent Login (Email + Password) ────────────────────────
   322→   322→
   323→   323→  /// Unified email+password login for owners, teachers, and parents.
   324→   324→  Future<Map<String, dynamic>> loginWithEmail({
   325→   325→    required String email,
   326→   326→    required String password,
   327→   327→  }) async {
   328→   328→    final transaction = KlasivoSentry.transactions.loginFlow('email');
   329→   329→
   330→   330→    try {
   331→   331→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   332→   332→        'method': 'email',
   333→   333→        'email': email,
   334→   334→      });
   335→   335→
   336→   336→      final userCredential =
   337→   337→          await FirebaseService.loginWithEmail(email, password);
   338→   338→      final user = userCredential.user!;
   339→   339→
   340→   340→      KlasivoSentry.breadcrumb.auth('auth_user_authenticated', data: {
   341→   341→        'uid': user.uid,
   342→   342→        'method': 'email',
   343→   343→      });
   344→   344→
   345→   345→      // Set Sentry user context immediately after auth
   346→   346→      await KlasivoSentry.userContext.setUser(
   347→   347→        uid: user.uid,
   348→   348→        email: email,
   349→   349→      );
   350→   350→
   351→   351→      final readUserSpan = transaction.startChild('read_user_doc');
   352→   352→      final userDoc = await _firestore
   353→   353→          .collection(AppConstants.usersCollection)
   354→   354→          .doc(user.uid)
   355→   355→          .get();
   356→   356→      await readUserSpan.finish();
   357→   357→
   358→   358→      if (!userDoc.exists) {
   359→   359→        KlasivoSentry.breadcrumb.auth('login_failed_user_doc_missing', data: {
   360→   360→          'uid': user.uid,
   361→   361→        });
   362→   362→        await Sentry.captureMessage(
   363→   363→          'Login failed: users/${user.uid} document does not exist',
   364→   364→          level: SentryLevel.error,
   365→   365→        );
   366→   366→        throw Exception('User data not found. Please contact your administrator.');
   367→   367→      }
   368→   368→
   369→   369→      final userData = userDoc.data()!;
   370→   370→      final role = userData['role'] as String?;
   371→   371→      final isActive = userData['isActive'] as bool? ?? true;
   372→   372→
   373→   373→      // Update Sentry context with role and org
   374→   374→      await KlasivoSentry.userContext.setRole(role ?? 'unknown');
   375→   375→      final orgId = userData['organizationId'] as String?;
   376→   376→      if (orgId != null && orgId.isNotEmpty) {
   377→   377→        await KlasivoSentry.userContext.setOrganizationId(orgId);
   378→   378→      }
   379→   379→
   380→   380→      if (!isActive) {
   381→   381→        await _auth.signOut();
   382→   382→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   383→   383→          'uid': user.uid,
   384→   384→          'role': role,
   385→   385→        });
   386→   386→        throw Exception('Your account has been deactivated. Contact your administrator.');
   387→   387→      }
   388→   388→
   389→   389→      // Students should NOT use email login — they use student code login
   390→   390→      if (role == AppConstants.roleStudent) {
   391→   391→        await _auth.signOut();
   392→   392→        KlasivoSentry.breadcrumb.auth('login_failed_student_email_login', data: {
   393→   393→          'uid': user.uid,
   394→   394→        });
   395→   395→        throw Exception('Students must login with their student code.');
   396→   396→      }
   397→   397→
   398→   398→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   399→   399→        'uid': user.uid,
   400→   400→        'role': role ?? 'unknown',
   401→   401→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   402→   402→      });
   403→   403→
   404→   404→      transaction.status = const SpanStatus.ok();
   405→   405→
   406→   406→      return {
   407→   407→        'id': user.uid,
   408→   408→        'organizationId': userData['organizationId'] ?? '',
   409→   409→        'role': role ?? AppConstants.roleTeacher,
   410→   410→        'authProvider': AuthProviders.password,
   411→   411→        'fullName': userData['fullName'] ?? 'User',
   412→   412→        'email': userData['email'] ?? email,
   413→   413→        'isEmailVerified': user.emailVerified,
   414→   414→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   415→   415→      };
   416→   416→    } catch (e, st) {
   417→   417→      transaction.status = const SpanStatus.internalError();
   418→   418→      await Sentry.captureException(
   419→   419→        e,
   420→   420→        stackTrace: st,
   421→   421→        withScope: (scope) {
   422→   422→          scope.setTag('flow', 'email_login');
   423→   423→          scope.setTag('step', 'login_with_email');
   424→   424→        },
   425→   425→      );
   426→   426→      rethrow;
   427→   427→    } finally {
   428→   428→      await transaction.finish();
   429→   429→    }
   430→   430→  }

   432→   432→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   433→   433→
   434→   434→  /// Student login using student code + password.
   435→   435→  /// Internally maps to Firebase Auth for push notifications,
   436→   436→  /// multi-device, password reset, security rules, and analytics.
   437→   437→  Future<Map<String, dynamic>> loginStudent({
   438→   438→    required String studentCode,
   439→   439→    required String password,
   440→   440→  }) async {
   441→   441→    final transaction = KlasivoSentry.transactions.loginFlow('student');
   442→   442→
   443→   443→    try {
   444→   444→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   445→   445→        'method': 'student_code',
   446→   446→      });
   447→   447→
   448→   448→      // Step 1: Find user by studentCode
   449→   449→      final findUserSpan = transaction.startChild('find_user_by_code');
   450→   450→      final snapshot = await _firestore
   451→   451→          .collection(AppConstants.usersCollection)
   452→   452→          .where('studentCode', isEqualTo: studentCode)
   453→   453→          .where('role', isEqualTo: AppConstants.roleStudent)
   454→   454→          .limit(1)
   455→   455→          .get();
   456→   456→      await findUserSpan.finish();
   457→   457→
   458→   458→      if (snapshot.docs.isEmpty) {
   459→   459→        KlasivoSentry.breadcrumb.auth('login_failed_student_not_found', data: {
   460→   460→          'method': 'student_code',
   461→   461→        });
   462→   462→        throw Exception('Student not found. Please check your student code.');
   463→   463→      }
   464→   464→
   465→   465→      final userDoc = snapshot.docs.first;
   466→   466→      final student = userDoc.data();
   467→   467→
   468→   468→      // Step 2: Verify password
   469→   469→      final storedPasswordHash = student['passwordHash'] as String?;
   470→   470→      final storedPlaintext = student['password'] as String?;
   471→   471→      final inputHash = hashPassword(password);
   472→   472→
   473→   473→      bool passwordMatches = false;
   474→   474→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   475→   475→        passwordMatches = inputHash == storedPasswordHash;
   476→   476→      } else if (storedPlaintext != null) {
   477→   477→        passwordMatches = password == storedPlaintext;
   478→   478→        // Migrate to hash on successful login
   479→   479→        if (passwordMatches) {
   480→   480→          await _firestore
   481→   481→              .collection(AppConstants.usersCollection)
   482→   482→              .doc(userDoc.id)
   483→   483→              .update({'passwordHash': inputHash});
   484→   484→        }
   485→   485→      }
   486→   486→
   487→   487→      if (!passwordMatches) {
   488→   488→        KlasivoSentry.breadcrumb.auth('login_failed_invalid_password', data: {
   489→   489→          'method': 'student_code',
   490→   490→          'userId': userDoc.id,
   491→   491→        });
   492→   492→        throw Exception('Invalid password. Please try again.');
   493→   493→      }
   494→   494→
   495→   495→      final isActive = student['isActive'] as bool? ?? true;
   496→   496→      if (!isActive) {
   497→   497→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   498→   498→          'userId': userDoc.id,
   499→   499→          'role': 'student',
   500→   500→        });
