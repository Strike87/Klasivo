    33→  /// Hash a password using SHA-256
    34→  static String hashPassword(String password) {
    35→    final bytes = utf8.encode(password);
    36→    final digest = sha256.convert(bytes);
    37→    return digest.toString();
    38→  }

    45→  Future<Map<String, dynamic>> registerOwner({

   165→  /// Unified email+password login for owners, teachers, and parents.
   166→  Future<Map<String, dynamic>> loginWithEmail({
   167→    required String email,
   168→    required String password,
   169→  }) async {
   170→    try {
   171→      final userCredential =
   172→          await FirebaseService.loginWithEmail(email, password);
   173→      final user = userCredential.user!;
   174→
   175→      final userDoc = await _firestore
   176→          .collection(AppConstants.usersCollection)
   177→          .doc(user.uid)
   178→          .get();
   179→
   180→      if (!userDoc.exists) {
   181→        throw Exception('User data not found. Please contact your administrator.');
   182→      }
   183→
   184→      final userData = userDoc.data()!;
   185→      final role = userData['role'] as String?;
   186→      final isActive = userData['isActive'] as bool? ?? true;
   187→
   188→      if (!isActive) {
   189→        await _auth.signOut();
   190→        throw Exception('Your account has been deactivated. Contact your administrator.');
   191→      }
   192→
   193→      // Students should NOT use email login — they use student code login
   194→      if (role == KlasivoRole.student) {
   195→        await _auth.signOut();
   196→        throw Exception('Students must login with their student code.');
   197→      }
   198→
   199→      return {
   200→        'id': user.uid,
   201→        'organizationId': userData['organizationId'] ?? '',
   202→        'role': role ?? KlasivoRole.teacher,
   203→        'authProvider': AuthProviders.password,
   204→        'fullName': userData['fullName'] ?? 'User',
   205→        'email': userData['email'] ?? email,
   206→        'isEmailVerified': user.emailVerified,
   207→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   208→      };
   209→    } catch (e) {
   210→      rethrow;
   211→    }
   212→  }

   214→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   215→
   216→  /// Student login using student code + password.
   217→  /// Internally maps to Firebase Auth for push notifications,
   218→  /// multi-device, password reset, security rules, and analytics.
   219→  Future<Map<String, dynamic>> loginStudent({
   220→    required String code,
   221→    required String password,
   222→  }) async {
   223→    try {
   224→      // Step 1: Find user by studentCode
   225→      final snapshot = await _firestore
   226→          .collection(AppConstants.usersCollection)
   227→          .where('studentCode', isEqualTo: code)
   228→          .where('role', isEqualTo: KlasivoRole.student)
   229→          .limit(1)
   230→          .get();
   231→
   232→      if (snapshot.docs.isEmpty) {
   233→        throw Exception('Student not found. Please check your student code.');
   234→      }
   235→
   236→      final userDoc = snapshot.docs.first;
   237→      final student = userDoc.data();
   238→
   239→      // Step 2: Verify password
   240→      final storedPasswordHash = student['passwordHash'] as String?;
   241→      final storedPlaintext = student['password'] as String?;
   242→      final inputHash = hashPassword(password);
   243→
   244→      bool passwordMatches = false;
   245→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   246→        passwordMatches = inputHash == storedPasswordHash;
   247→      } else if (storedPlaintext != null) {
   248→        passwordMatches = password == storedPlaintext;
   249→        // Migrate to hash on successful login
   250→        if (passwordMatches) {
   251→          await _firestore
   252→              .collection(AppConstants.usersCollection)
   253→              .doc(userDoc.id)
   254→              .update({'passwordHash': inputHash});
   255→        }
   256→      }
   257→
   258→      if (!passwordMatches) {
   259→        throw Exception('Invalid password. Please try again.');
   260→      }
   261→
   262→      final isActive = student['isActive'] as bool? ?? true;
   263→      if (!isActive) {
   264→        throw Exception('Your account has been deactivated.');
   265→      }
   266→
   267→      // Step 3: Sign in via Firebase Auth using the student's internal email
   268→      final internalEmail = student['authEmail'] as String?;
   269→      if (internalEmail != null && internalEmail.isNotEmpty) {
   270→        try {
   271→          await _auth.signInWithEmailAndPassword(
   272→            email: internalEmail,
   273→            password: password,
   274→          );
   275→        } catch (authError) {
   276→          // Graceful fallback — still allow login even if Firebase Auth fails
   277→        }
   278→      }
   279→
   280→      return {
   281→        'id': userDoc.id,
   282→        'organizationId': student['organizationId'] ?? '',
   283→        'role': KlasivoRole.student,
   284→        'authProvider': AuthProviders.studentCode,
   285→        'fullName': student['fullName'] ?? 'Student',
   286→        'studentCode': student['studentCode'],
   287→        'classId': student['classId'],
   288→        'hasCompletedSetup': true,
   289→      };
   290→    } catch (e) {
   291→      rethrow;
   292→    }
   293→  }

   543→  /// Sign in with Google. Works for owners, teachers, and parents.
   544→  /// If the user already exists, logs them in.

   546→  ///   - owner: auto-creates organization (hasCompletedSetup=false)

   548→  ///   - parent: creates parent account (hasCompletedSetup=false, needs to link child)
   549→  Future<Map<String, dynamic>> loginWithGoogle({
   550→    String? expectedRole,

   554→      isNewUser: false,

   557→
   558→  Future<Map<String, dynamic>> _signInWithGoogle({
   559→    String? expectedRole,
   560→    bool isNewUser = false,
   561→  }) async {
   562→    try {
   563→      final googleUser = await GoogleSignIn().signIn();
   564→      if (googleUser == null) {
   565→        throw Exception('Google sign-in was cancelled.');
   566→      }
   567→
   568→      final googleAuth = await googleUser.authentication;
   569→      final credential = GoogleAuthProvider.credential(
   570→        accessToken: googleAuth.accessToken,
   571→        idToken: googleAuth.idToken,
   572→      );
   573→
   574→      final userCredential = await _auth.signInWithCredential(credential);
   575→      final user = userCredential.user!;
   576→
   577→      // Check if user document already exists
   578→      final userDoc = await _firestore
   579→          .collection(AppConstants.usersCollection)
   580→          .doc(user.uid)
   581→          .get();
   582→
   583→      if (userDoc.exists) {
   584→        // Existing user — log them in
   585→        final userData = userDoc.data()!;
   586→        final role = userData['role'] as String?;
   587→        final isActive = userData['isActive'] as bool? ?? true;
   588→
   589→        if (!isActive) {
   590→          await _auth.signOut();
   591→          throw Exception('Your account has been deactivated.');
   592→        }
   593→
   594→        // Students cannot use Google Sign-In
   595→        if (role == KlasivoRole.student) {
   596→          await _auth.signOut();
   597→          throw Exception('Students must login with their student code.');
   598→        }
   599→
   600→        return {
   601→          'id': user.uid,
   602→          'organizationId': userData['organizationId'] ?? '',
   603→          'role': role ?? expectedRole ?? KlasivoRole.owner,
   604→          'authProvider': AuthProviders.google,
   605→          'fullName': userData['fullName'] ?? user.displayName ?? 'User',
   606→          'email': userData['email'] ?? user.email ?? '',
   607→          'isEmailVerified': user.emailVerified,
   608→          'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   609→        };
   610→      }
   611→
   612→      // New user — create account based on expected role
   613→      final fullName = user.displayName ?? 'User';
   614→      final email = user.email ?? '';
   615→      final role = expectedRole ?? KlasivoRole.owner;
   616→      final isEmailVerified = user.emailVerified; // Google emails are pre-verified
   617→
   618→      String? organizationId;
   619→      bool hasCompletedSetup = true;
   620→
   621→      if (role == KlasivoRole.owner) {
   622→        // Create user doc FIRST with placeholder orgId so isTeacherOrOwner() resolves
   623→        await _firestore
   624→            .collection(AppConstants.usersCollection)
   625→            .doc(user.uid)
   626→            .set({
   627→          'organizationId': '', // Placeholder — updated below
   628→          'role': KlasivoRole.owner,
   629→          'authProvider': AuthProviders.google,
   630→          'fullName': fullName,
   631→          'email': email,
   632→          'photoUrl': user.photoURL,
   633→          'phoneNumber': null,
   634→          'isActive': true,
   635→          'isEmailVerified': isEmailVerified,
   636→          'hasCompletedSetup': false,
   637→          'createdAt': FieldValue.serverTimestamp(),
   638→          'updatedAt': FieldValue.serverTimestamp(),
   639→        });
   640→
   641→        // Now create the organization — rules can verify user is an owner
   642→        final orgService = OrganizationService();
   643→        organizationId = await orgService.createOrganization(
   644→          ownerId: user.uid,
   645→          name: "$fullName's Workspace",
   646→        );
   647→
   648→        // Patch the user doc with the real organizationId
   649→        await _firestore
   650→            .collection(AppConstants.usersCollection)
   651→            .doc(user.uid)
   652→            .update({
   653→          'organizationId': organizationId,
   654→          'updatedAt': FieldValue.serverTimestamp(),
   655→        });
   656→
   657→        hasCompletedSetup = false; // Needs to name workspace
   658→      } else if (role == KlasivoRole.parent) {
   659→        organizationId = null; // Set when parent links a child
   660→        hasCompletedSetup = false; // Needs to link child
   661→      }
   662→
   663→      // For non-owner roles, write the user doc normally (owner doc already created above)
   664→      if (role != KlasivoRole.owner) {
   665→        await _firestore
   666→            .collection(AppConstants.usersCollection)
   667→            .doc(user.uid)
   668→            .set({
   669→          'organizationId': organizationId,
   670→          'role': role,
   671→          'authProvider': AuthProviders.google,
   672→          'fullName': fullName,
   673→          'email': email,
   674→          'photoUrl': user.photoURL,
   675→          'phoneNumber': null,
   676→          'isActive': true,
   677→          'isEmailVerified': isEmailVerified,
   678→          'hasCompletedSetup': hasCompletedSetup,
   679→          'createdAt': FieldValue.serverTimestamp(),
   680→          'updatedAt': FieldValue.serverTimestamp(),
   681→        });
   682→      }
   683→
   684→      return {
   685→        'id': user.uid,
   686→        'organizationId': organizationId,
   687→        'role': role,
   688→        'authProvider': AuthProviders.google,
   689→        'fullName': fullName,
   690→        'email': email,
   691→        'isEmailVerified': isEmailVerified,
   692→        'hasCompletedSetup': hasCompletedSetup,
   693→      };
   694→    } catch (e) {
   695→      rethrow;
   696→    }
   697→  }

   762→  Future<bool> needsSetup() async {
   763→    try {
   764→      final user = _auth.currentUser;
   765→      if (user == null) return false;
   766→
   767→      final userDoc = await _firestore
   768→          .collection(AppConstants.usersCollection)
   769→          .doc(user.uid)
   770→          .get();
   771→
   772→      if (!userDoc.exists) return false;
   773→
   774→      final data = userDoc.data()!;
   775→      final role = data['role'] as String?;
   776→      final hasCompletedSetup = data['hasCompletedSetup'] as bool? ?? true;
   777→
   778→      // Owners need workspace naming, parents need child linking
   779→      if ((role == KlasivoRole.owner || role == KlasivoRole.parent) &&
   780→          !hasCompletedSetup) {
   781→        return true;
   782→      }
   783→
   784→      return false;
   785→    } catch (e) {
   786→      return false;
   787→    }
   788→  }
