   301→   301→  /// Unified email+password login for owners, teachers, and parents.
   302→   302→  Future<Map<String, dynamic>> loginWithEmail({
   303→   303→    required String email,
   304→   304→    required String password,
   305→   305→  }) async {
   306→   306→    try {
   307→   307→      final userCredential =
   308→   308→          await FirebaseService.loginWithEmail(email, password);
   309→   309→      final user = userCredential.user!;
   310→   310→
   311→   311→      final userDoc = await _firestore
   312→   312→          .collection(AppConstants.usersCollection)
   313→   313→          .doc(user.uid)
   314→   314→          .get();
   315→   315→
   316→   316→      if (!userDoc.exists) {
   317→   317→        throw Exception('User data not found. Please contact your administrator.');
   318→   318→      }
   319→   319→
   320→   320→      final userData = userDoc.data()!;
   321→   321→      final role = userData['role'] as String?;
   322→   322→      final isActive = userData['isActive'] as bool? ?? true;
   323→   323→
   324→   324→      if (!isActive) {
   325→   325→        await _auth.signOut();
   326→   326→        throw Exception('Your account has been deactivated. Contact your administrator.');
   327→   327→      }
   328→   328→
   329→   329→      // Students should NOT use email login — they use student code login
   330→   330→      if (role == AppConstants.roleStudent) {
   331→   331→        await _auth.signOut();
   332→   332→        throw Exception('Students must login with their student code.');
   333→   333→      }
   334→   334→
   335→   335→      return {
   336→   336→        'id': user.uid,
   337→   337→        'organizationId': userData['organizationId'] ?? '',
   338→   338→        'role': role ?? AppConstants.roleTeacher,
   339→   339→        'authProvider': AuthProviders.password,
   340→   340→        'fullName': userData['fullName'] ?? 'User',
   341→   341→        'email': userData['email'] ?? email,
   342→   342→        'isEmailVerified': user.emailVerified,
   343→   343→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   344→   344→      };
   345→   345→    } catch (e) {
   346→   346→      rethrow;
   347→   347→    }
   348→   348→  }

   350→   350→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   351→   351→
   352→   352→  /// Student login using student code + password.
   353→   353→  /// Internally maps to Firebase Auth for push notifications,
   354→   354→  /// multi-device, password reset, security rules, and analytics.
   355→   355→  Future<Map<String, dynamic>> loginStudent({
   356→   356→    required String studentCode,
   357→   357→    required String password,
   358→   358→  }) async {
   359→   359→    try {
   360→   360→      // Step 1: Find user by studentCode
   361→   361→      final snapshot = await _firestore
   362→   362→          .collection(AppConstants.usersCollection)
   363→   363→          .where('studentCode', isEqualTo: studentCode)
   364→   364→          .where('role', isEqualTo: AppConstants.roleStudent)
   365→   365→          .limit(1)
   366→   366→          .get();
   367→   367→
   368→   368→      if (snapshot.docs.isEmpty) {
   369→   369→        throw Exception('Student not found. Please check your student code.');
   370→   370→      }
   371→   371→
   372→   372→      final userDoc = snapshot.docs.first;
   373→   373→      final student = userDoc.data();
   374→   374→
   375→   375→      // Step 2: Verify password
   376→   376→      final storedPasswordHash = student['passwordHash'] as String?;
   377→   377→      final storedPlaintext = student['password'] as String?;
   378→   378→      final inputHash = hashPassword(password);
   379→   379→
   380→   380→      bool passwordMatches = false;
   381→   381→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   382→   382→        passwordMatches = inputHash == storedPasswordHash;
   383→   383→      } else if (storedPlaintext != null) {
   384→   384→        passwordMatches = password == storedPlaintext;
   385→   385→        // Migrate to hash on successful login
   386→   386→        if (passwordMatches) {
   387→   387→          await _firestore
   388→   388→              .collection(AppConstants.usersCollection)
   389→   389→              .doc(userDoc.id)
   390→   390→              .update({'passwordHash': inputHash});
   391→   391→        }
   392→   392→      }
   393→   393→
   394→   394→      if (!passwordMatches) {
   395→   395→        throw Exception('Invalid password. Please try again.');
   396→   396→      }
   397→   397→
   398→   398→      final isActive = student['isActive'] as bool? ?? true;
   399→   399→      if (!isActive) {
   400→   400→        throw Exception('Your account has been deactivated.');
   401→   401→      }
   402→   402→
   403→   403→      // Step 3: Sign in via Firebase Auth using the student's internal email
   404→   404→      final internalEmail = student['authEmail'] as String?;
   405→   405→      if (internalEmail != null && internalEmail.isNotEmpty) {
   406→   406→        try {
   407→   407→          await _auth.signInWithEmailAndPassword(
   408→   408→            email: internalEmail,
   409→   409→            password: password,
   410→   410→          );
   411→   411→        } catch (authError) {
   412→   412→          // Graceful fallback — still allow login even if Firebase Auth fails
   413→   413→        }
   414→   414→      }
   415→   415→
   416→   416→      return {
   417→   417→        'id': userDoc.id,
   418→   418→        'organizationId': student['organizationId'] ?? '',
   419→   419→        'role': AppConstants.roleStudent,
   420→   420→        'authProvider': AuthProviders.studentCode,
   421→   421→        'fullName': student['fullName'] ?? 'Student',
   422→   422→        'studentCode': student['studentCode'],
   423→   423→        'classId': student['classId'],
   424→   424→        'hasCompletedSetup': true,
   425→   425→      };
   426→   426→    } catch (e) {
   427→   427→      rethrow;
   428→   428→    }
   429→   429→  }
