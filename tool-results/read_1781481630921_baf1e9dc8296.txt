    26→  /// Hash a password using SHA-256
    27→  static String hashPassword(String password) {
    28→    final bytes = utf8.encode(password);
    29→    final digest = sha256.convert(bytes);
    30→    return digest.toString();
    31→  }

    38→  Future<Map<String, dynamic>> registerOwner({

    40→    required String password,
    41→    required String fullName,
    42→  }) async {
    43→    final transaction = Sentry.startTransaction(
    44→      'owner_registration',
    45→      'registration',
    46→    );
    47→
    48→    try {
    49→      // ── Step 1: Create Firebase Auth account ──────────────────────────

    53→
    54→      Sentry.addBreadcrumb(Breadcrumb(
    55→        category: 'registration',
    56→        message: 'STEP_1_AUTH_USER_CREATED',
    57→        data: {'uid': user.uid, 'email': email},
    58→      ));
    59→      transaction.setData('uid', user.uid);
    60→
    61→      // Set Sentry user context immediately after auth account creation
    62→      await Sentry.configureScope((scope) {

    66→      final isEmailVerified = user.emailVerified;
    67→
    68→      // ── Step 2: Create user document in Firestore ──────────────────────
    69→      Sentry.addBreadcrumb(const Breadcrumb(
    70→        category: 'registration',
    71→        message: 'STEP_2_USER_DOC_CREATE_START',

    75→
    76→      try {

    78→          collection: AppConstants.usersCollection,
    79→          docId: user.uid,
    80→          data: {
    81→            'organizationId': '', // Placeholder — updated below after org creation
    82→            'role': AppConstants.roleOwner,
    83→            'authProvider': AuthProviders.password,
    84→            'fullName': fullName,
    85→            'email': email,
    86→            'photoUrl': null,
    87→            'phoneNumber': null,
    88→            'isActive': true,
    89→            'isEmailVerified': isEmailVerified,
    90→            'hasCompletedSetup': false,
    91→            'createdAt': FieldValue.serverTimestamp(),
    92→            'updatedAt': FieldValue.serverTimestamp(),
    93→          },
    94→          flow: 'owner_registration',
    95→          step: 'STEP_2_USER_DOC_CREATE',
    96→        );
    97→
    98→        // Doc ID audit trail
    99→        KlasivoSentry.docIdAudit.logUserCreation(
   100→          flow: 'owner_registration',
   101→          collection: AppConstants.usersCollection,
   102→          docIdStrategy: 'uid',
   103→          actualDocId: user.uid,
   104→          authUid: user.uid,

   106→
   107→        createUserDocSpan.status = const SpanStatus.ok();
   108→      } catch (e, st) {
   109→        createUserDocSpan.status = const SpanStatus.internalError();
   110→        rethrow;
   111→      } finally {
   112→        await createUserDocSpan.finish();
   113→      }
   114→
   115→      Sentry.addBreadcrumb(Breadcrumb(
   116→        category: 'registration',
   117→        message: 'STEP_2_USER_DOC_CREATE_SUCCESS',
   118→        data: {'uid': user.uid},
   119→      ));
   120→
   121→      // ── Step 3: Create organization ────────────────────────────────────
   122→      Sentry.addBreadcrumb(const Breadcrumb(
   123→        category: 'registration',
   124→        message: 'STEP_3_ORG_CREATE_START',
   125→      ));
   126→
   127→      final createOrgSpan = transaction.startChild('create_organization');
   128→      final orgService = OrganizationService();
   129→
   130→      String orgId;
   131→      try {
   132→        orgId = await orgService.createOrganization(

   137→      } catch (e, st) {
   138→        createOrgSpan.status = const SpanStatus.internalError();
   139→        await Sentry.captureException(
   140→          e,
   141→          stackTrace: st,
   142→          withScope: (scope) {
   143→            scope.setTag('collection', 'organizations');
   144→            scope.setTag('operation', 'create');
   145→            scope.setTag('ownerId', user.uid);
   146→            scope.setTag('flow', 'owner_registration');
   147→            scope.setTag('step', 'STEP_3_ORG_CREATE');
   148→          },
   149→        );
   150→        rethrow;
   151→      } finally {
   152→        await createOrgSpan.finish();
   153→      }
   154→
   155→      Sentry.addBreadcrumb(Breadcrumb(
   156→        category: 'registration',
   157→        message: 'STEP_3_ORG_CREATE_SUCCESS',
   158→        data: {'orgId': orgId},
   159→      ));
   160→
   161→      // ── Step 4: Patch user doc with real organizationId ────────────────
   162→      Sentry.addBreadcrumb(const Breadcrumb(
   163→        category: 'registration',
   164→        message: 'STEP_4_USER_PATCH_START',
   165→      ));
   166→
   167→      final patchUserSpan = transaction.startChild('update_user_doc');
   168→
   169→      try {
   170→        await SentryFirestoreHelper.docUpdate(
   171→          collection: AppConstants.usersCollection,
   172→          docId: user.uid,
   173→          data: {
   174→            'organizationId': orgId,
   175→            'updatedAt': FieldValue.serverTimestamp(),
   176→          },
   177→          flow: 'owner_registration',
   178→          step: 'STEP_4_USER_PATCH',
   179→        );
   180→        patchUserSpan.status = const SpanStatus.ok();
   181→      } catch (e, st) {

   204→        'organizationId': orgId,

   206→        'authProvider': AuthProviders.password,

   210→        'hasCompletedSetup': false,

   212→    } catch (e) {
   213→      transaction.status = const SpanStatus.internalError();
   214→      rethrow;
   215→    } finally {
   216→      await transaction.finish();
   217→    }

   301→  /// Unified email+password login for owners, teachers, and parents.
   302→  Future<Map<String, dynamic>> loginWithEmail({
   303→    required String email,
   304→    required String password,
   305→  }) async {
   306→    final transaction = KlasivoSentry.transactions.loginFlow('email');
   307→
   308→    try {
   309→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   310→        'method': 'email',
   311→        'email': email,
   312→      });
   313→
   314→      final userCredential =
   315→          await FirebaseService.loginWithEmail(email, password);
   316→      final user = userCredential.user!;
   317→
   318→      KlasivoSentry.breadcrumb.auth('auth_user_authenticated', data: {
   319→        'uid': user.uid,
   320→        'method': 'email',
   321→      });
   322→
   323→      // Set Sentry user context immediately after auth
   324→      await KlasivoSentry.userContext.setUser(
   325→        uid: user.uid,
   326→        email: email,
   327→      );
   328→
   329→      final readUserSpan = transaction.startChild('read_user_doc');
   330→      final userDoc = await _firestore
   331→          .collection(AppConstants.usersCollection)
   332→          .doc(user.uid)
   333→          .get();
   334→      await readUserSpan.finish();
   335→
   336→      if (!userDoc.exists) {
   337→        KlasivoSentry.breadcrumb.auth('login_failed_user_doc_missing', data: {
   338→          'uid': user.uid,
   339→        });
   340→        await Sentry.captureMessage(
   341→          'Login failed: users/${user.uid} document does not exist',
   342→          level: SentryLevel.error,
   343→        );
   344→        throw Exception('User data not found. Please contact your administrator.');
   345→      }
   346→
   347→      final userData = userDoc.data()!;
   348→      final role = userData['role'] as String?;
   349→      final isActive = userData['isActive'] as bool? ?? true;
   350→
   351→      // Update Sentry context with role and org
   352→      await KlasivoSentry.userContext.setRole(role ?? 'unknown');
   353→      final orgId = userData['organizationId'] as String?;
   354→      if (orgId != null && orgId.isNotEmpty) {
   355→        await KlasivoSentry.userContext.setOrganizationId(orgId);
   356→      }
   357→
   358→      if (!isActive) {
   359→        await _auth.signOut();
   360→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   361→          'uid': user.uid,
   362→          'role': role,
   363→        });
   364→        throw Exception('Your account has been deactivated. Contact your administrator.');
   365→      }
   366→
   367→      // Students should NOT use email login — they use student code login
   368→      if (role == AppConstants.roleStudent) {
   369→        await _auth.signOut();
   370→        KlasivoSentry.breadcrumb.auth('login_failed_student_email_login', data: {
   371→          'uid': user.uid,
   372→        });
   373→        throw Exception('Students must login with their student code.');
   374→      }
   375→
   376→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   377→        'uid': user.uid,
   378→        'role': role ?? 'unknown',
   379→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   380→      });
   381→
   382→      transaction.status = const SpanStatus.ok();
   383→
   384→      return {
   385→        'id': user.uid,
   386→        'organizationId': userData['organizationId'] ?? '',
   387→        'role': role ?? AppConstants.roleTeacher,
   388→        'authProvider': AuthProviders.password,
   389→        'fullName': userData['fullName'] ?? 'User',
   390→        'email': userData['email'] ?? email,
   391→        'isEmailVerified': user.emailVerified,
   392→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   393→      };
   394→    } catch (e, st) {
   395→      transaction.status = const SpanStatus.internalError();
   396→      await Sentry.captureException(
   397→        e,
   398→        stackTrace: st,
   399→        withScope: (scope) {
   400→          scope.setTag('flow', 'email_login');
   401→          scope.setTag('step', 'login_with_email');
   402→        },
   403→      );
   404→      rethrow;
   405→    } finally {
   406→      await transaction.finish();
   407→    }
   408→  }

   410→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   411→
   412→  /// Student login using student code + password.
   413→  /// Internally maps to Firebase Auth for push notifications,
   414→  /// multi-device, password reset, security rules, and analytics.
   415→  Future<Map<String, dynamic>> loginStudent({
   416→    required String studentCode,
   417→    required String password,
   418→  }) async {
   419→    final transaction = KlasivoSentry.transactions.loginFlow('student');
   420→
   421→    try {
   422→      KlasivoSentry.breadcrumb.auth('login_started', data: {
   423→        'method': 'student_code',
   424→      });
   425→
   426→      // Step 1: Find user by studentCode
   427→      final findUserSpan = transaction.startChild('find_user_by_code');
   428→      final snapshot = await _firestore
   429→          .collection(AppConstants.usersCollection)
   430→          .where('studentCode', isEqualTo: studentCode)
   431→          .where('role', isEqualTo: AppConstants.roleStudent)
   432→          .limit(1)
   433→          .get();
   434→      await findUserSpan.finish();
   435→
   436→      if (snapshot.docs.isEmpty) {
   437→        KlasivoSentry.breadcrumb.auth('login_failed_student_not_found', data: {
   438→          'method': 'student_code',
   439→        });
   440→        throw Exception('Student not found. Please check your student code.');
   441→      }
   442→
   443→      final userDoc = snapshot.docs.first;
   444→      final student = userDoc.data();
   445→
   446→      // Step 2: Verify password
   447→      final storedPasswordHash = student['passwordHash'] as String?;
   448→      final storedPlaintext = student['password'] as String?;
   449→      final inputHash = hashPassword(password);
   450→
   451→      bool passwordMatches = false;
   452→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   453→        passwordMatches = inputHash == storedPasswordHash;
   454→      } else if (storedPlaintext != null) {
   455→        passwordMatches = password == storedPlaintext;
   456→        // Migrate to hash on successful login
   457→        if (passwordMatches) {
   458→          await _firestore
   459→              .collection(AppConstants.usersCollection)
   460→              .doc(userDoc.id)
   461→              .update({'passwordHash': inputHash});
   462→        }
   463→      }
   464→
   465→      if (!passwordMatches) {
   466→        KlasivoSentry.breadcrumb.auth('login_failed_invalid_password', data: {
   467→          'method': 'student_code',
   468→          'userId': userDoc.id,
   469→        });
   470→        throw Exception('Invalid password. Please try again.');
   471→      }
   472→
   473→      final isActive = student['isActive'] as bool? ?? true;
   474→      if (!isActive) {
   475→        KlasivoSentry.breadcrumb.auth('login_failed_account_deactivated', data: {
   476→          'userId': userDoc.id,
   477→          'role': 'student',
   478→        });
   479→        throw Exception('Your account has been deactivated.');
   480→      }
   481→
   482→      // Step 3: Sign in via Firebase Auth using the student's internal email
   483→      final internalEmail = student['authEmail'] as String?;
   484→      if (internalEmail != null && internalEmail.isNotEmpty) {
   485→        try {
   486→          await _auth.signInWithEmailAndPassword(
   487→            email: internalEmail,
   488→            password: password,
   489→          );
   490→          KlasivoSentry.breadcrumb.auth('student_auth_signed_in', data: {
   491→            'userId': userDoc.id,
   492→          });
   493→        } catch (authError) {
   494→          // Graceful fallback — still allow login even if Firebase Auth fails
   495→          KlasivoSentry.breadcrumb.auth('student_auth_signin_failed_fallback', data: {
   496→            'userId': userDoc.id,
   497→            'error': authError.toString().substring(0, (authError.toString().length).clamp(0, 100)),
   498→          });
   499→        }
   500→      }
   501→
   502→      // Set Sentry user context
   503→      await KlasivoSentry.userContext.setUser(
   504→        uid: userDoc.id,
   505→        email: internalEmail ?? '',
   506→        role: AppConstants.roleStudent,
   507→        organizationId: student['organizationId'] as String?,
   508→      );
   509→
   510→      KlasivoSentry.breadcrumb.auth('login_success', data: {
   511→        'uid': userDoc.id,
   512→        'role': 'student',
   513→        'method': 'student_code',
   514→      });
   515→
   516→      // Log doc ID audit — student uses Firestore doc ID (not auth UID)
   517→      KlasivoSentry.docIdAudit.logUserCreation(
   518→        flow: 'student_code_login',
   519→        collection: AppConstants.usersCollection,
   520→        docIdStrategy: userDoc.id.length > 28 ? 'auto_id' : 'uid',
   521→        actualDocId: userDoc.id,
   522→        authUid: _auth.currentUser?.uid,
   523→      );
   524→
   525→      transaction.status = const SpanStatus.ok();
   526→
   527→      return {
   528→        'id': userDoc.id,
   529→        'organizationId': student['organizationId'] ?? '',
   530→        'role': AppConstants.roleStudent,
   531→        'authProvider': AuthProviders.studentCode,
   532→        'fullName': student['fullName'] ?? 'Student',
   533→        'studentCode': student['studentCode'],
   534→        'classId': student['classId'],
   535→        'hasCompletedSetup': true,
   536→      };
   537→    } catch (e, st) {
   538→      transaction.status = const SpanStatus.internalError();
   539→      await Sentry.captureException(
   540→        e,
   541→        stackTrace: st,
   542→        withScope: (scope) {
   543→          scope.setTag('flow', 'student_login');
   544→          scope.setTag('step', 'login_student');
   545→          scope.setTag('method', 'student_code');
   546→        },
   547→      );
   548→      rethrow;
   549→    } finally {
   550→      await transaction.finish();
   551→    }
   552→  }

   607→      final createUserDocSpan = transaction.startChild('create_user_doc');

  1359→  Future<bool> needsSetup() async {
  1360→    try {
  1361→      final user = _auth.currentUser;
  1362→      if (user == null) return false;
  1363→
  1364→      final userDoc = await _firestore
  1365→          .collection(AppConstants.usersCollection)
  1366→          .doc(user.uid)
  1367→          .get();
  1368→
  1369→      if (!userDoc.exists) return false;
  1370→
  1371→      final data = userDoc.data()!;
  1372→      final role = data['role'] as String?;
  1373→      final hasCompletedSetup = data['hasCompletedSetup'] as bool? ?? true;
  1374→
  1375→      // Owners need workspace naming, parents need child linking
  1376→      if ((role == AppConstants.roleOwner || role == AppConstants.roleParent) &&
  1377→          !hasCompletedSetup) {
  1378→        return true;
  1379→      }
  1380→
  1381→      return false;
  1382→    } catch (e) {
  1383→      return false;
  1384→    }
