    33→    33→  /// Hash a password using SHA-256
    34→    34→  static String hashPassword(String password) {
    35→    35→    final bytes = utf8.encode(password);
    36→    36→    final digest = sha256.convert(bytes);
    37→    37→    return digest.toString();
    38→    38→  }

   165→   165→  /// Unified email+password login for owners, teachers, and parents.
   166→   166→  Future<Map<String, dynamic>> loginWithEmail({
   167→   167→    required String email,
   168→   168→    required String password,
   169→   169→  }) async {
   170→   170→    try {
   171→   171→      final userCredential =
   172→   172→          await FirebaseService.loginWithEmail(email, password);
   173→   173→      final user = userCredential.user!;
   174→   174→
   175→   175→      final userDoc = await _firestore
   176→   176→          .collection(AppConstants.usersCollection)
   177→   177→          .doc(user.uid)
   178→   178→          .get();
   179→   179→
   180→   180→      if (!userDoc.exists) {
   181→   181→        throw Exception('User data not found. Please contact your administrator.');
   182→   182→      }
   183→   183→
   184→   184→      final userData = userDoc.data()!;
   185→   185→      final role = userData['role'] as String?;
   186→   186→      final isActive = userData['isActive'] as bool? ?? true;
   187→   187→
   188→   188→      if (!isActive) {
   189→   189→        await _auth.signOut();
   190→   190→        throw Exception('Your account has been deactivated. Contact your administrator.');
   191→   191→      }
   192→   192→
   193→   193→      // Students should NOT use email login — they use student code login
   194→   194→      if (role == KlasivoRole.student) {
   195→   195→        await _auth.signOut();
   196→   196→        throw Exception('Students must login with their student code.');
   197→   197→      }
   198→   198→
   199→   199→      return {
   200→   200→        'id': user.uid,
   201→   201→        'organizationId': userData['organizationId'] ?? '',
   202→   202→        'role': role ?? KlasivoRole.teacher,
   203→   203→        'authProvider': AuthProviders.password,
   204→   204→        'fullName': userData['fullName'] ?? 'User',
   205→   205→        'email': userData['email'] ?? email,
   206→   206→        'isEmailVerified': user.emailVerified,
   207→   207→        'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   208→   208→      };
   209→   209→    } catch (e) {
   210→   210→      rethrow;
   211→   211→    }
   212→   212→  }

   214→   214→  // ─── Student Login (Student Code + Password) ──────────────────────────────
   215→   215→
   216→   216→  /// Student login using student code + password.
   217→   217→  /// Internally maps to Firebase Auth for push notifications,
   218→   218→  /// multi-device, password reset, security rules, and analytics.
   219→   219→  Future<Map<String, dynamic>> loginStudent({
   220→   220→    required String code,
   221→   221→    required String password,
   222→   222→  }) async {
   223→   223→    try {
   224→   224→      // Step 1: Find user by studentCode
   225→   225→      final snapshot = await _firestore
   226→   226→          .collection(AppConstants.usersCollection)
   227→   227→          .where('studentCode', isEqualTo: code)
   228→   228→          .where('role', isEqualTo: KlasivoRole.student)
   229→   229→          .limit(1)
   230→   230→          .get();
   231→   231→
   232→   232→      if (snapshot.docs.isEmpty) {
   233→   233→        throw Exception('Student not found. Please check your student code.');
   234→   234→      }
   235→   235→
   236→   236→      final userDoc = snapshot.docs.first;
   237→   237→      final student = userDoc.data();
   238→   238→
   239→   239→      // Step 2: Verify password
   240→   240→      final storedPasswordHash = student['passwordHash'] as String?;
   241→   241→      final storedPlaintext = student['password'] as String?;
   242→   242→      final inputHash = hashPassword(password);
   243→   243→
   244→   244→      bool passwordMatches = false;
   245→   245→      if (storedPasswordHash != null && storedPasswordHash.isNotEmpty) {
   246→   246→        passwordMatches = inputHash == storedPasswordHash;
   247→   247→      } else if (storedPlaintext != null) {
   248→   248→        passwordMatches = password == storedPlaintext;
   249→   249→        // Migrate to hash on successful login
   250→   250→        if (passwordMatches) {
   251→   251→          await _firestore
   252→   252→              .collection(AppConstants.usersCollection)
   253→   253→              .doc(userDoc.id)
   254→   254→              .update({'passwordHash': inputHash});
   255→   255→        }
   256→   256→      }
   257→   257→
   258→   258→      if (!passwordMatches) {
   259→   259→        throw Exception('Invalid password. Please try again.');
   260→   260→      }
   261→   261→
   262→   262→      final isActive = student['isActive'] as bool? ?? true;
   263→   263→      if (!isActive) {
   264→   264→        throw Exception('Your account has been deactivated.');
   265→   265→      }
   266→   266→
   267→   267→      // Step 3: Sign in via Firebase Auth using the student's internal email
   268→   268→      final internalEmail = student['authEmail'] as String?;
   269→   269→      if (internalEmail != null && internalEmail.isNotEmpty) {
   270→   270→        try {
   271→   271→          await _auth.signInWithEmailAndPassword(
   272→   272→            email: internalEmail,
   273→   273→            password: password,
   274→   274→          );
   275→   275→        } catch (authError) {
   276→   276→          // Graceful fallback — still allow login even if Firebase Auth fails
   277→   277→        }
   278→   278→      }
   279→   279→
   280→   280→      return {
   281→   281→        'id': userDoc.id,
   282→   282→        'organizationId': student['organizationId'] ?? '',
   283→   283→        'role': KlasivoRole.student,
   284→   284→        'authProvider': AuthProviders.studentCode,
   285→   285→        'fullName': student['fullName'] ?? 'Student',
   286→   286→        'studentCode': student['studentCode'],
   287→   287→        'classId': student['classId'],
   288→   288→        'hasCompletedSetup': true,
   289→   289→      };
   290→   290→    } catch (e) {
   291→   291→      rethrow;
   292→   292→    }
   293→   293→  }

   541→   541→  // ─── Google Sign-In (Unified — Owner / Teacher / Parent) ──────────────────
   542→   542→
   543→   543→  /// Sign in with Google. Works for owners, teachers, and parents.
   544→   544→  /// If the user already exists, logs them in.
   545→   545→  /// If the user is new, creates an account based on [expectedRole]:
   546→   546→  ///   - owner: auto-creates organization (hasCompletedSetup=false)
   547→   547→  ///   - teacher: needs invite code (handled separately via registerTeacherWithGoogle)
   548→   548→  ///   - parent: creates parent account (hasCompletedSetup=false, needs to link child)
   549→   549→  Future<Map<String, dynamic>> loginWithGoogle({
   550→   550→    String? expectedRole,
   551→   551→  }) async {
   552→   552→    return _signInWithGoogle(
   553→   553→      expectedRole: expectedRole,
   554→   554→      isNewUser: false,
   555→   555→    );
   556→   556→  }
   557→   557→
   558→   558→  Future<Map<String, dynamic>> _signInWithGoogle({
   559→   559→    String? expectedRole,
   560→   560→    bool isNewUser = false,
   561→   561→  }) async {
   562→   562→    try {
   563→   563→      final googleUser = await GoogleSignIn().signIn();
   564→   564→      if (googleUser == null) {
   565→   565→        throw Exception('Google sign-in was cancelled.');
   566→   566→      }
   567→   567→
   568→   568→      final googleAuth = await googleUser.authentication;
   569→   569→      final credential = GoogleAuthProvider.credential(
   570→   570→        accessToken: googleAuth.accessToken,
   571→   571→        idToken: googleAuth.idToken,
   572→   572→      );
   573→   573→
   574→   574→      final userCredential = await _auth.signInWithCredential(credential);
   575→   575→      final user = userCredential.user!;
   576→   576→
   577→   577→      // Check if user document already exists
   578→   578→      final userDoc = await _firestore
   579→   579→          .collection(AppConstants.usersCollection)
   580→   580→          .doc(user.uid)
   581→   581→          .get();
   582→   582→
   583→   583→      if (userDoc.exists) {
   584→   584→        // Existing user — log them in
   585→   585→        final userData = userDoc.data()!;
   586→   586→        final role = userData['role'] as String?;
   587→   587→        final isActive = userData['isActive'] as bool? ?? true;
   588→   588→
   589→   589→        if (!isActive) {
   590→   590→          await _auth.signOut();
   591→   591→          throw Exception('Your account has been deactivated.');
   592→   592→        }
   593→   593→
   594→   594→        // Students cannot use Google Sign-In
   595→   595→        if (role == KlasivoRole.student) {
   596→   596→          await _auth.signOut();
   597→   597→          throw Exception('Students must login with their student code.');
   598→   598→        }
   599→   599→
   600→   600→        return {
   601→   601→          'id': user.uid,
   602→   602→          'organizationId': userData['organizationId'] ?? '',
   603→   603→          'role': role ?? expectedRole ?? KlasivoRole.owner,
   604→   604→          'authProvider': AuthProviders.google,
   605→   605→          'fullName': userData['fullName'] ?? user.displayName ?? 'User',
   606→   606→          'email': userData['email'] ?? user.email ?? '',
   607→   607→          'isEmailVerified': user.emailVerified,
   608→   608→          'hasCompletedSetup': userData['hasCompletedSetup'] ?? true,
   609→   609→        };
   610→   610→      }
   611→   611→
   612→   612→      // New user — create account based on expected role
   613→   613→      final fullName = user.displayName ?? 'User';
   614→   614→      final email = user.email ?? '';
   615→   615→      final role = expectedRole ?? KlasivoRole.owner;
   616→   616→      final isEmailVerified = user.emailVerified; // Google emails are pre-verified
   617→   617→
   618→   618→      String? organizationId;
   619→   619→      bool hasCompletedSetup = true;
   620→   620→
   621→   621→      if (role == KlasivoRole.owner) {
   622→   622→        // Create user doc FIRST with placeholder orgId so isTeacherOrOwner() resolves
   623→   623→        await _firestore
   624→   624→            .collection(AppConstants.usersCollection)
   625→   625→            .doc(user.uid)
   626→   626→            .set({
   627→   627→          'organizationId': '', // Placeholder — updated below
   628→   628→          'role': KlasivoRole.owner,
   629→   629→          'authProvider': AuthProviders.google,
   630→   630→          'fullName': fullName,
   631→   631→          'email': email,
   632→   632→          'photoUrl': user.photoURL,
   633→   633→          'phoneNumber': null,
   634→   634→          'isActive': true,
   635→   635→          'isEmailVerified': isEmailVerified,
   636→   636→          'hasCompletedSetup': false,
   637→   637→          'createdAt': FieldValue.serverTimestamp(),
   638→   638→          'updatedAt': FieldValue.serverTimestamp(),
   639→   639→        });
   640→   640→
   641→   641→        // Now create the organization — rules can verify user is an owner
   642→   642→        final orgService = OrganizationService();
   643→   643→        organizationId = await orgService.createOrganization(
   644→   644→          ownerId: user.uid,
   645→   645→          name: "$fullName's Workspace",
   646→   646→        );
   647→   647→
   648→   648→        // Patch the user doc with the real organizationId
   649→   649→        await _firestore
   650→   650→            .collection(AppConstants.usersCollection)
   651→   651→            .doc(user.uid)
   652→   652→            .update({
   653→   653→          'organizationId': organizationId,
   654→   654→          'updatedAt': FieldValue.serverTimestamp(),
   655→   655→        });
   656→   656→
   657→   657→        hasCompletedSetup = false; // Needs to name workspace
   658→   658→      } else if (role == KlasivoRole.parent) {
   659→   659→        organizationId = null; // Set when parent links a child
   660→   660→        hasCompletedSetup = false; // Needs to link child
   661→   661→      }
   662→   662→
   663→   663→      // For non-owner roles, write the user doc normally (owner doc already created above)
   664→   664→      if (role != KlasivoRole.owner) {
   665→   665→        await _firestore
   666→   666→            .collection(AppConstants.usersCollection)
   667→   667→            .doc(user.uid)
   668→   668→            .set({
   669→   669→          'organizationId': organizationId,
   670→   670→          'role': role,
   671→   671→          'authProvider': AuthProviders.google,
   672→   672→          'fullName': fullName,
   673→   673→          'email': email,
   674→   674→          'photoUrl': user.photoURL,
   675→   675→          'phoneNumber': null,
   676→   676→          'isActive': true,
   677→   677→          'isEmailVerified': isEmailVerified,
   678→   678→          'hasCompletedSetup': hasCompletedSetup,
   679→   679→          'createdAt': FieldValue.serverTimestamp(),
   680→   680→          'updatedAt': FieldValue.serverTimestamp(),
   681→   681→        });
   682→   682→      }
   683→   683→
   684→   684→      return {
   685→   685→        'id': user.uid,
   686→   686→        'organizationId': organizationId,
   687→   687→        'role': role,
   688→   688→        'authProvider': AuthProviders.google,
   689→   689→        'fullName': fullName,
   690→   690→        'email': email,
   691→   691→        'isEmailVerified': isEmailVerified,
   692→   692→        'hasCompletedSetup': hasCompletedSetup,
   693→   693→      };
   694→   694→    } catch (e) {
   695→   695→      rethrow;
   696→   696→    }
   697→   697→  }
