   155→  Future<void> _initializeEnterpriseServices() async {
   156→    try {
   157→      final box = Hive.box(AppConstants.authBox);
   158→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   159→      final orgId = box.get('organizationId') as String?;
   160→
   161→      if (isLoggedIn && orgId != null) {
   162→        // Load feature flags for the organization
   163→        final flagService = ref.read(featureFlagServiceProvider);
   164→        await flagService.loadFlags(orgId);
   165→
   166→        // Load custom permissions for the organization
   167→        final permService = ref.read(permissionServiceProvider);
   168→        await permService.loadPermissions(orgId);
   169→
   170→        debugPrint('[MyApp] Enterprise services initialized for org: $orgId');
   171→      }
   172→    } catch (e) {
   173→      debugPrint('[MyApp] Enterprise service initialization failed: $e');
   174→    }

   223→// ─── Auth Change Notifier for GoRouter refresh ──────────────────────────────
   224→
   225→class AuthChangeNotifier extends ChangeNotifier {
   226→  StreamSubscription<User?>? _firebaseSub;
   227→  StreamSubscription? _hiveSub;
   228→
   229→  AuthChangeNotifier() {
   230→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   231→      (_) => notifyListeners(),
   232→      onError: (_) => notifyListeners(),
   233→    );
   234→    _startHiveWatch();
   235→  }
   236→
   237→  void _startHiveWatch() {
   238→    // Use Hive's built-in watch() instead of polling every 500ms
   239→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   240→      notifyListeners();
   241→    });
   242→  }
   243→
   244→  @override
   245→  void dispose() {
   246→    _firebaseSub?.cancel();
   247→    _hiveSub?.cancel();
   248→    super.dispose();
   249→  }
   250→}
   251→
   252→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
   253→  final notifier = AuthChangeNotifier();
   254→  ref.onDispose(() => notifier.dispose());
   255→  return notifier;
   256→});

   258→// ─── GoRouter with Auth Guards & v1.7 Complete Navigation ─────────────────────
   259→
   260→final routerProvider = Provider<GoRouter>((ref) {
   261→  final notifier = ref.watch(authChangeNotifierProvider);
   262→
   263→  return GoRouter(
   264→    initialLocation: '/',
   265→    debugLogDiagnostics: true,
   266→    refreshListenable: notifier,
   267→    redirect: (context, state) {
   268→      final box = Hive.box(AppConstants.authBox);
   269→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   270→      final userRole = box.get('userRole', defaultValue: '') as String;
   271→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: true) as bool;
   272→
   273→      final isOnSplash = state.matchedLocation == '/';
   274→      final isOnAuth = state.matchedLocation.startsWith('/auth');
   275→      final isOnWelcome = state.matchedLocation == '/welcome';
   276→      final isOnDashboard = state.matchedLocation.startsWith('/dashboard') ||
   277→          state.matchedLocation.startsWith('/academic') ||
   278→          state.matchedLocation.startsWith('/people') ||
   279→          state.matchedLocation.startsWith('/inbox') ||
   280→          state.matchedLocation.startsWith('/settings') ||
   281→          state.matchedLocation.startsWith('/teacher') ||
   282→          state.matchedLocation.startsWith('/lms');
   283→      final isOnStudent = state.matchedLocation.startsWith('/student');
   284→      final isOnParent = state.matchedLocation.startsWith('/parent') ||
   285→          state.matchedLocation.startsWith('/auth/parent');
   286→
   287→      // Splash → redirect based on auth state
   288→      if (isOnSplash) {
   289→        if (isLoggedIn && userRole.isNotEmpty) {
   290→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   291→            return '/welcome';
   292→          }
   293→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   294→            return '/dashboard';
   295→          }
   296→          if (userRole == AppConstants.roleStudent) return '/student';
   297→          if (userRole == AppConstants.roleParent) return '/parent';
   298→        }
   299→        return '/auth';
   300→      }
   301→
   302→      // Welcome screen — only accessible to logged-in owners who haven't completed setup
   303→      if (isOnWelcome) {
   304→        if (!isLoggedIn) return '/auth';
   305→        if (userRole != AppConstants.roleOwner) return '/dashboard';
   306→        if (hasCompletedSetup) return '/dashboard';
   307→        return null;
   308→      }
   309→
   310→      // Auth screens → redirect if already logged in
   311→      if (isOnAuth) {
   312→        if (isLoggedIn && userRole.isNotEmpty) {
   313→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   314→            return '/welcome';
   315→          }
   316→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   317→            return '/dashboard';
   318→          }
   319→          if (userRole == AppConstants.roleStudent) return '/student';
   320→          if (userRole == AppConstants.roleParent) return '/parent';
   321→        }
   322→        return null;
   323→      }
   324→
   325→      // Protected screens → require login
   326→      if (isOnDashboard || isOnStudent || isOnParent) {
   327→        if (!isLoggedIn) return '/auth';
   328→        if (userRole.isEmpty) return '/auth';
   329→
   330→        if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   331→          return '/welcome';
   332→        }
   333→
   334→        // ─── Feature Flag Gates ───────────────────────────────────────────
   335→        // LMS routes require lms flag
   336→        if (state.matchedLocation.startsWith('/lms')) {
   337→          final flagService = ref.read(featureFlagServiceProvider);
   338→          if (!flagService.isEnabled(FeatureFlags.lms)) {
   339→            return '/dashboard';
   340→          }
   341→        }
   342→
   343→        // Parent portal requires parentPortal flag
   344→        if (state.matchedLocation.startsWith('/parent') ||
   345→            state.matchedLocation.startsWith('/auth/parent')) {
   346→          final flagService = ref.read(featureFlagServiceProvider);
   347→          if (!flagService.isEnabled(FeatureFlags.parentPortal)) {
   348→            if (userRole == AppConstants.roleParent) return '/auth';
   349→            return '/dashboard';
   350→          }
   351→        }
   352→
   353→        if ((userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) &&
   354→            isOnDashboard) {
   355→          return null;
   356→        }
   357→        if (userRole == AppConstants.roleStudent && isOnStudent) {
   358→          return null;
   359→        }
   360→        if (userRole == AppConstants.roleParent && isOnParent) {
   361→          return null;
   362→        }
   363→
   364→        if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   365→          return '/dashboard';
   366→        }
   367→        if (userRole == AppConstants.roleStudent) return '/student';
   368→        if (userRole == AppConstants.roleParent) return '/parent';
   369→      }
   370→
   371→      return null;
   372→    },
