   275→    Object error,
   276→    StackTrace stackTrace,

   278→  ) {

   280→      error,

   285→      },
   286→    );
   287→  }

   376→class AuthChangeNotifier extends ChangeNotifier {
   377→  StreamSubscription<User?>? _firebaseSub;
   378→  StreamSubscription? _hiveSub;
   379→
   380→  AuthChangeNotifier() {
   381→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
   382→      (_) => notifyListeners(),
   383→      onError: (_) => notifyListeners(),
   384→    );
   385→    _startHiveWatch();
   386→  }

   388→  void _startHiveWatch() {
   389→    // Use Hive's built-in watch() instead of polling every 500ms
   390→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
   391→      notifyListeners();
   392→    });
   393→  }
   394→
   395→  @override
   396→  void dispose() {
   397→    _firebaseSub?.cancel();
   398→    _hiveSub?.cancel();
   399→    super.dispose();
   400→  }
   401→}

   411→/// Custom [NavigatorObserver] that logs route changes as Sentry breadcrumbs
   412→/// for full navigation traceability in error reports.
   413→class SentryNavigationObserver extends NavigatorObserver {
   414→  @override
   415→  void didPush(Route<dynamic> route, Route<dynamic>? previousRoute) {
   416→    _logNavigation('push', route, previousRoute);
   417→  }
   418→
   419→  @override
   420→  void didPop(Route<dynamic> route, Route<dynamic>? previousRoute) {
   421→    _logNavigation('pop', route, previousRoute);
   422→  }
   423→
   424→  @override
   425→  void didReplace({Route<dynamic>? newRoute, Route<dynamic>? oldRoute}) {
   426→    if (newRoute != null) {
   427→      KlasivoSentry.breadcrumb.navigation(
   428→        newRoute.settings.name ?? newRoute.runtimeType.toString(),
   429→        action: 'replace',
   430→        data: {
   431→          if (oldRoute?.settings.name != null)
   432→            'from': oldRoute!.settings.name!,
   433→        },
   434→      );
   435→    }
   436→  }
   437→
   438→  void _logNavigation(
   439→    String action,
   440→    Route<dynamic> route,
   441→    Route<dynamic>? previousRoute,
   442→  ) {
   443→    final name = route.settings.name ?? route.runtimeType.toString();
   444→    KlasivoSentry.breadcrumb.navigation(name, action: action, data: {
   445→      if (previousRoute?.settings.name != null)
   446→        'from': previousRoute!.settings.name!,
   447→    });
   448→  }
   449→}

   459→    refreshListenable: notifier,

   462→    errorBuilder: (context, state) {
   463→      final error = state.error;
   464→      if (error != null) {
   465→        Sentry.captureException(
   466→          error,
   467→          withScope: (scope) {
   468→            scope.setTag('source', 'go_router');
   469→            scope.setTag('route', state.matchedLocation);
   470→            scope.setExtra('uri', state.uri.toString());
   471→          },
   472→        );
   473→        KlasivoSentry.breadcrumb.navigation(
   474→          state.matchedLocation,
   475→          action: 'error',
   476→          data: {'error': error.toString()},
   477→        );
   478→      }
   479→      return Scaffold(
   480→        body: Center(
   481→          child: Column(
   482→            mainAxisAlignment: MainAxisAlignment.center,
   483→            children: [
   484→              const Icon(Icons.error_outline, size: 48, color: KlasivoColors.error),
   485→              const SizedBox(height: KlasivoSpacing.lg),
   486→              Text(
   487→                'Page not found',
   488→                style: KlasivoTypography.headlineMedium.copyWith(
   489→                  color: Theme.of(context).brightness == Brightness.dark
   490→                      ? KlasivoColors.darkTextPrimary
   491→                      : KlasivoColors.lightTextPrimary,
   492→                ),
   493→              ),
   494→              const SizedBox(height: KlasivoSpacing.sm),
   495→              Text(
   496→                'The page you are looking for does not exist.',
   497→                style: KlasivoTypography.bodyMedium.copyWith(
   498→                  color: Theme.of(context).brightness == Brightness.dark
   499→                      ? KlasivoColors.darkTextTertiary
   500→                      : KlasivoColors.lightTextTertiary,
   501→                ),
   502→                textAlign: TextAlign.center,
   503→              ),
   504→              const SizedBox(height: KlasivoSpacing.xl),
   505→              ElevatedButton(
   506→                onPressed: () => context.go('/'),
   507→                child: const Text('Go Home'),
   508→              ),
   509→            ],
   510→          ),
   511→        ),
   512→      );
   513→    },
   514→    redirect: (context, state) {
   515→      final box = Hive.box(AppConstants.authBox);
   516→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
   517→      final userRole = box.get('userRole', defaultValue: '') as String;
   518→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: false) as bool;
   519→      final mustChangePassword = box.get('mustChangePassword', defaultValue: false) as bool;
   520→      final isOnChangePassword = state.matchedLocation == '/change-password';
   521→
   522→      // Force redirect to change password if required
   523→      if (isLoggedIn && mustChangePassword && !isOnChangePassword) {
   524→        return '/change-password';
   525→      }
   526→      // Prevent navigating away from change password if forced
   527→      if (isLoggedIn && mustChangePassword && isOnChangePassword) {
   528→        return null; // Allow staying on change password
   529→      }
   530→      // If no longer required, redirect away from change password
   531→      if (isLoggedIn && !mustChangePassword && isOnChangePassword) {
   532→        // Management roles → /dashboard; student → /student; parent → /parent
   533→        if (KlasivoRole.managementRoles.contains(userRole)) return '/dashboard';
   534→        if (userRole == KlasivoRole.student) return '/student';
   535→        if (userRole == KlasivoRole.parent) return '/parent';
   536→        return '/dashboard'; // Fallback for unknown roles
   537→      }
   538→
   539→      final isOnSplash = state.matchedLocation == '/';
   540→      final isOnAuth = state.matchedLocation.startsWith('/auth');
