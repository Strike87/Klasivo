    27→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    28→
    29→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    30→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    31→class _SensitiveFields {
    32→  static const Set<String> keys = {
    33→    'password',
    34→    'passwordHash',
    35→    'confirmPassword',
    36→    'newPassword',
    37→    'accessToken',
    38→    'refreshToken',
    39→    'idToken',
    40→    'otp',
    41→    'otpCode',
    42→    'inviteCode',
    43→    'secret',
    44→    'apiKey',
    45→    'authEmail', // Internal student auth email — PII
    46→    'studentCode', // Sensitive — could be used to log in
    47→  };
    48→
    49→  /// Returns true if the key contains any sensitive substring.
    50→  static bool isSensitive(String key) {
    51→    final lower = key.toLowerCase();
    52→    for (final s in keys) {
    53→      if (lower.contains(s.toLowerCase())) return true;
    54→    }
    55→    return false;
    56→  }
    57→
    58→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    59→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    60→    return data.map((key, value) {
    61→      if (isSensitive(key)) {
    62→        return MapEntry(key, '[REDACTED]');
    63→      }
    64→      // Recursively sanitize nested maps
    65→      if (value is Map<String, dynamic>) {
    66→        return MapEntry(key, sanitize(value));
    67→      }
    68→      return MapEntry(key, value);
    69→    });
    70→  }
    71→
    72→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    73→  static String sanitizeString(String input) {
    74→    // Redact Bearer tokens
    75→    var result = input.replaceAll(
    76→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    77→      'Bearer [REDACTED]',
    78→    );
    79→    // Redact long hex strings (likely tokens)
    80→    result = result.replaceAll(
    81→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    82→      '[REDACTED_TOKEN]',
    83→    );
    84→    return result;
    85→  }
    86→}

   254→// ─── Firestore Observability Helper ──────────────────────────────────────────

   256→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception
   257→/// capture, and scope tags for every critical write.
   258→///
   259→/// Every Firestore exception will:
   260→///   1. Reach Sentry
   261→///   2. Include collection name
   262→///   3. Include document id
   263→///   4. Include user role
   264→///   5. Include organization id
   265→class SentryFirestoreHelper {
   266→  static final FirebaseFirestore _db = FirebaseFirestore.instance;

   268→  /// Wrapped `.set()` with full observability.
   269→  static Future<void> docSet({
   270→    required String collection,
   271→    required String docId,
   272→    required Map<String, dynamic> data,
   273→    String? flow,
   274→    String? step,
   275→    SetOptions? options,
   276→  }) async {

   278→    KlasivoSentry.breadcrumb.firestore(
   279→      'create',
   280→      collection: collection,
   281→      docId: docId,
   282→      data: {'flow': flow, 'step': step},
   283→    );

   285→    try {
   286→      await _db.collection(collection).doc(docId).set(data, options);
   287→      KlasivoSentry.breadcrumb.firestore(
   288→        'create_success',
   289→        collection: collection,
   290→        docId: docId,
   291→      );

   293→      await _captureFirestoreException(
   294→        e,
   295→        st,
   296→        collection: collection,
   297→        operation: 'set',
   298→        docId: docId,
   299→        flow: flow,
   300→        step: step,
   301→        dataPreview: sanitizedData,

   304→    }

   443→        collection: collection,

   452→  /// Capture a Firestore exception with full context tags.
   453→  static Future<void> _captureFirestoreException(
   454→    Object exception,
   455→    StackTrace stackTrace, {
   456→    required String collection,
   457→    required String operation,
   458→    String? docId,
   459→    String? flow,
   460→    String? step,
   461→    Map<String, dynamic>? dataPreview,
   462→  }) async {
   463→    // Determine if this is a permission-denied error
   464→    String? errorCode;
   465→    if (exception is FirebaseException) {
   466→      errorCode = exception.code;
   467→    }
   468→
   469→    await Sentry.captureException(
   470→      exception,
   471→      stackTrace: stackTrace,
   472→      withScope: (scope) {
   473→        scope.setTag('collection', collection);
   474→        scope.setTag('operation', operation);
   475→        if (docId != null) scope.setTag('documentId', docId);
   476→        if (flow != null) scope.setTag('flow', flow);
   477→        if (step != null) scope.setTag('step', step);
   478→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   479→
   480→        // Attach user role and org from current Sentry context
   481→        final currentUser = FirebaseAuth.instance.currentUser;
   482→        if (currentUser != null) {
   483→          scope.setTag('auth_uid', currentUser.uid);
   484→        }
   485→
   486→        if (dataPreview != null && dataPreview.isNotEmpty) {
   487→          scope.setExtra('data_preview', dataPreview);
   488→        }
   489→      },
   490→    );
   491→  }
   492→}

   494→// ─── Guarded Async Operation ─────────────────────────────────────────────────
   495→
   496→/// Runs an async operation with automatic Sentry breadcrumb, exception capture,
   497→/// and metadata attachment. This is the recommended way to wrap any async
   498→/// operation that should never fail silently.
   499→///
   500→/// Example:
   501→///   final result = await KlasivoSentry.runGuarded(
   502→///     'create_exam',
   503→///     () => examService.createExam(...),
   504→///     category: 'firestore',
   505→///   );
   506→class KlasivoSentryGuard {
   507→  /// Execute [operation] inside a Sentry-guarded wrapper.
   508→  ///
   509→  /// - Automatically adds a breadcrumb before execution.
   510→  /// - On success: adds a success breadcrumb.
   511→  /// - On failure: captures the exception with scope tags and rethrows.
   512→  /// - The [category] defaults to the operation name but can be overridden.
   513→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   514→  ///   and the exception scope.
   515→  static Future<T> runGuarded<T>(
   516→    String operationName,
   517→    Future<T> Function() operation, {
   518→    String? category,
   519→    Map<String, String>? tags,
   520→    Map<String, dynamic>? data,
   521→  }) async {
   522→    final cat = category ?? 'operation';
   523→
   524→    Sentry.addBreadcrumb(Breadcrumb(
   525→      category: cat,
   526→      message: '${operationName}_started',
   527→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   528→      level: SentryLevel.info,
   529→    ));
   530→
   531→    try {
   532→      final result = await operation();
   533→
   534→      Sentry.addBreadcrumb(Breadcrumb(
   535→        category: cat,
   536→        message: '${operationName}_success',
   537→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   538→        level: SentryLevel.info,
   539→      ));
   540→
   541→      return result;
   542→    } catch (e, st) {
   543→      Sentry.addBreadcrumb(Breadcrumb(
   544→        category: cat,
   545→        message: '${operationName}_failed',
   546→        data: _SensitiveFields.sanitize({
   547→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   548→          ...?data,
   549→        }),
   550→        level: SentryLevel.error,
   551→      ));
   552→
   553→      await Sentry.captureException(
   554→        e,
   555→        stackTrace: st,
   556→        withScope: (scope) {
   557→          scope.setTag('operation', operationName);
   558→          tags?.forEach(scope.setTag);
   559→          if (data != null) {
   560→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   561→          }
   562→        },
   563→      );
   564→
   565→      rethrow;
   566→    }
   567→  }
   568→
   569→  /// Execute [operation] with a Sentry transaction span.
   570→  ///
   571→  /// Creates a child span under the given [transaction], measures duration,
   572→  /// and sets the span status based on success/failure.
   573→  static Future<T> runWithSpan<T>(
   574→    ISentrySpan transaction,
   575→    String spanName,
   576→    Future<T> Function() operation, {
   577→    Map<String, String>? tags,
   578→  }) async {
   579→    final span = transaction.startChild(spanName);
   580→
   581→    try {
   582→      final result = await operation();
   583→      span.status = const SpanStatus.ok();
   584→      return result;
   585→    } catch (e, st) {
   586→      span.status = const SpanStatus.internalError();
   587→
   588→      await Sentry.captureException(
   589→        e,
   590→        stackTrace: st,
   591→        withScope: (scope) {
   592→          scope.setTag('span', spanName);
   593→          tags?.forEach(scope.setTag);
   594→        },
   595→      );
   596→
   597→      rethrow;
   598→    } finally {
   599→      await span.finish();
   600→    }
   601→  }
   602→}

   604→// ─── User Context Management ─────────────────────────────────────────────────
   605→
   606→/// Manages Sentry user context — attached on sign-in, updated on role/org
   607→/// changes, cleared on logout.
   608→///
   609→/// Attaches:
   610→///   - uid
   611→///   - email
   612→///   - role (tag)
   613→///   - organizationId (tag)
   614→///   - app version (tag — set once at init)
   615→///   - build number (tag — set once at init)
   616→class SentryUserContext {
   617→  /// Set user context after authentication.
   618→  /// Updates BOTH Sentry and Crashlytics for unified observability.
   619→  static Future<void> setUser({
   620→    required String uid,
   621→    required String email,
   622→    String? role,
   623→    String? organizationId,
   624→  }) async {
   625→    // ── Sentry ──
   626→    await Sentry.configureScope((scope) {
   627→      scope.setUser(SentryUser(
   628→        id: uid,
   629→        email: email,
   630→      ));
   631→      if (role != null) {
   632→        scope.setTag('role', role);
   633→      }
   634→      if (organizationId != null && organizationId.isNotEmpty) {
   635→        scope.setTag('organizationId', organizationId);
   636→      }
   637→    });
   638→
   639→    // ── Crashlytics ──
   640→    try {
   641→      final crashlytics = FirebaseCrashlytics.instance;
   642→      crashlytics.setUserIdentifier(uid);
   643→      if (role != null) {
   644→        crashlytics.setCustomKey('role', role);
   645→      }
   646→      if (organizationId != null && organizationId.isNotEmpty) {
   647→        crashlytics.setCustomKey('organizationId', organizationId);
   648→      }
   649→      crashlytics.setCustomKey('email', email);
   650→    } catch (_) {

   749→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   750→
   751→/// Documents the doc ID strategy for every user creation path.
   752→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   753→///
   754→/// | Path                  | Collection | Doc ID     | Status    |
   755→/// |-----------------------|-----------|------------|-----------|
   756→/// | Owner (email)         | users     | user.uid   | OK        |
   757→/// | Owner (Google)        | users     | user.uid   | OK        |
   758→/// | Teacher (invite)      | users     | user.uid   | OK        |
   759→/// | Teacher (Google)      | users     | user.uid   | OK        |
   760→/// | Parent (email)        | users     | user.uid   | OK        |
   761→/// | Parent (Google)       | users     | user.uid   | OK        |
   762→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   763→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   764→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   765→///
   766→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   767→/// `.doc(user.uid)`, which means:
   768→///   1. The doc ID doesn't match the auth UID
   769→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   770→///   3. QR enrollment is 100% broken in production
   771→class SentryDocIdAudit {
   772→  /// Log a doc ID audit breadcrumb for any user creation path.
   773→  static void logUserCreation({
   774→    required String flow,
   775→    required String collection,
   776→    required String docIdStrategy, // 'uid' or 'auto_id'
   777→    required String actualDocId,
   778→    String? authUid,
   779→  }) {
   780→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   781→    final isAutoId = docIdStrategy == 'auto_id';
   782→
   783→    Sentry.addBreadcrumb(Breadcrumb(
   784→      category: 'doc_id_audit',
   785→      message: 'user_doc_created',
   786→      data: _SensitiveFields.sanitize({
   787→        'flow': flow,
   788→        'collection': collection,
   789→        'docIdStrategy': docIdStrategy,
   790→        'actualDocId': actualDocId,
   791→        if (authUid != null) 'authUid': authUid,
   792→        'idMismatch': isMismatch,
   793→        'autoIdUsed': isAutoId,
   794→        'brokenByRules': isAutoId, // auto-id always fails security rules
   795→      }),
   796→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   797→    ));
   798→
   799→    if (isMismatch || isAutoId) {
   800→      Sentry.captureMessage(
   801→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   802→        '(docId=$actualDocId, authUid=$authUid). '
   803→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   804→        level: SentryLevel.warning,
   805→      );
   806→    }
   807→  }
   808→}

   902→// ─── Centralized Event Sanitizer ────────────────────────────────────────────
   903→
   904→/// Global Sentry event sanitizer used by the `beforeSend` callback in
   905→/// `main.dart`. Replaces the duplicate `_SentrySanitizer` class that was
   906→/// previously defined in main.dart.
   907→///
   908→/// Strips sensitive data from breadcrumbs, extras, and request headers.
   909→class KlasivoSentrySanitizer {
   910→  KlasivoSentrySanitizer._();
   911→
   912→  /// Sanitize a [SentryEvent] — returns the same event with sensitive data
   913→  /// replaced by '[REDACTED]'.
   914→  static SentryEvent sanitizeEvent(SentryEvent event) {
   915→    // Strip sensitive data from breadcrumbs
   916→    if (event.breadcrumbs != null) {
   917→      for (int i = 0; i < event.breadcrumbs!.length; i++) {
   918→        final crumb = event.breadcrumbs![i];
   919→        if (crumb.data != null) {
   920→          final sanitized = <String, dynamic>{};
   921→          crumb.data!.forEach((key, value) {
   922→            sanitized[key] = _SensitiveFields.isSensitive(key)
   923→                ? '[REDACTED]'
   924→                : value;
   925→          });
   926→          event.breadcrumbs![i] = Breadcrumb(
   927→            category: crumb.category,
   928→            message: crumb.message,
   929→            data: sanitized,
   930→            level: crumb.level,
   931→            timestamp: crumb.timestamp,
   932→            type: crumb.type,
   933→          );
   934→        }
   935→      }
   936→    }
   937→
   938→    // Strip sensitive data from extra fields
   939→    if (event.extra != null) {
   940→      final sanitized = <String, dynamic>{};
   941→      event.extra!.forEach((key, value) {
   942→        sanitized[key] = _SensitiveFields.isSensitive(key)
   943→            ? '[REDACTED]'
   944→            : value;
   945→      });
   946→      event.extra = sanitized;
   947→    }
   948→
   949→    // Strip sensitive data from request headers
   950→    if (event.request?.headers != null) {
   951→      final sanitized = <String, String>{};
   952→      event.request!.headers!.forEach((key, value) {
   953→        final lower = key.toLowerCase();
   954→        if (lower.contains('authorization') ||
   955→            lower.contains('cookie') ||
   956→            lower.contains('token')) {
   957→          sanitized[key] = '[REDACTED]';
   958→        } else {
   959→          sanitized[key] = value;
   960→        }
   961→      });
   962→      event.request = SentryRequest(headers: sanitized);
   963→    }
   964→
   965→    return event;
   966→  }
   967→}
