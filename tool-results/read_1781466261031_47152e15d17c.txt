    48→class UserDetailScreen extends ConsumerStatefulWidget {
    49→  final String userId;
    50→
    51→  const UserDetailScreen({Key? key, required this.userId}) : super(key: key);
    52→
    53→  @override
    54→  ConsumerState<UserDetailScreen> createState() => _UserDetailScreenState();
    55→}
    56→
    57→class _UserDetailScreenState extends ConsumerState<UserDetailScreen>
    58→    with SingleTickerProviderStateMixin {
    59→  late TabController _tabController;
    60→
    61→  @override
    62→  void initState() {
    63→    super.initState();
    64→    _tabController = TabController(
    65→      length: _DetailTab.values.length,
    66→      vsync: this,
    67→    );
    68→  }
    69→
    70→  @override
    71→  void dispose() {
    72→    _tabController.dispose();
    73→    super.dispose();
    74→  }
    75→
    76→  @override
    77→  Widget build(BuildContext context) {
    78→    final theme = Theme.of(context);
    79→    final isDark = theme.brightness == Brightness.dark;
    80→    final profileAsync = ref.watch(userRbacProfileProvider(widget.userId));
    81→
    82→    return profileAsync.when(
    83→      loading: () => Scaffold(
    84→        appBar: AppBar(title: const Text('User Details')),
    85→        body: const Center(child: KlasivoLoading()),
    86→      ),
    87→      error: (e, _) => Scaffold(
    88→        appBar: AppBar(title: const Text('User Details')),
    89→        body: Center(child: Text('Error: $e')),
    90→      ),
    91→      data: (profile) {
    92→        final user = profile.user;
    93→
    94→        return Scaffold(
    95→          appBar: AppBar(
    96→            title: Text(user.fullName),
    97→            bottom: TabBar(
    98→              controller: _tabController,
    99→              isScrollable: true,
   100→              tabAlignment: TabAlignment.start,
   101→              labelColor: isDark ? AppColors.primaryLight : AppColors.primary,
   102→              unselectedLabelColor: isDark
   103→                  ? AppColors.lightTextTertiary
   104→                  : AppColors.darkTextTertiary,
   105→              indicatorColor:
   106→                  isDark ? AppColors.primaryLight : AppColors.primary,
   107→              tabs: _DetailTab.values
   108→                  .map((tab) => Tab(
   109→                        icon: Icon(tab.icon, size: 16),
   110→                        text: tab.label,
   111→                      ))
   112→                  .toList(),
   113→            ),
   114→          ),
   115→          body: TabBarView(
   116→            controller: _tabController,
   117→            children: [
   118→              _OverviewTab(profile: profile),
   119→              _ScopeTab(profile: profile),
   120→              _OverridesTab(profile: profile),
   121→              _EffectiveTab(profile: profile),
   122→              _AuditTab(userId: widget.userId),
   123→            ],
   124→          ),
   125→          // ─── Quick Action FAB ────────────────────────────────────
   126→          floatingActionButton: KlasivoPermissionGate(
   127→            permission: Permission.userAssignRole,
   128→            fallback: const SizedBox.shrink(),
   129→            child: FloatingActionButton.extended(
   130→              onPressed: () => _showRoleAssignment(context, profile),
   131→              icon: const Icon(Icons.shield_rounded),
   132→              label: const Text('Change Role'),
   133→            ),
   134→          ),
   135→        );
   136→      },
   137→    );
   138→  }
   139→

   142→      context: context,
   143→      isScrollControlled: true,
   144→      useSafeArea: true,
   145→      shape: const RoundedRectangleBorder(
   146→        borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
   147→      ),
   148→      builder: (context) => RoleAssignmentSheet(
   149→        userId: profile.user.id,
   150→        currentRole: profile.user.role,
   151→        userName: profile.user.fullName,
   152→      ),
   153→    );
   154→  }

   455→class _OverridesTab extends ConsumerWidget {

   460→  @override
   461→  Widget build(BuildContext context, WidgetRef ref) {
   462→    final theme = Theme.of(context);
   463→    final user = profile.user;
   464→    final overrides = Map<String, bool>.from(user.permissionOverrides);
   465→    final diff = PermissionOverrides.getOverrideDiff(user.role, overrides);
   466→
   467→    return ListView(
   468→      padding: const EdgeInsets.all(16),
   469→      children: [
   470→        // Diff summary card
   471→        KlasivoCard(
   472→          accentColor: diff.hasOverrides ? AppColors.warning : AppColors.success,
   473→          child: Padding(
   474→            padding: const EdgeInsets.all(16),
   475→            child: Column(
   476→              crossAxisAlignment: CrossAxisAlignment.start,
   477→              children: [
   478→                Text('Override Impact', style: theme.textTheme.titleMedium),
   479→                const SizedBox(height: 8),
   480→                Text(
   481→                  diff.summary,
   482→                  style: theme.textTheme.bodyMedium?.copyWith(
   483→                    color: diff.hasOverrides ? AppColors.warning : AppColors.success,
   484→                  ),
   485→                ),
   486→                if (diff.hasGrants) ...[
   487→                  const SizedBox(height: 8),
   488→                  Text('+${diff.grantedByOverrides.length} permissions granted',
   489→                      style: theme.textTheme.bodySmall?.copyWith(
   490→                          color: AppColors.success)),
   491→                ],
   492→                if (diff.hasDenials) ...[
   493→                  const SizedBox(height: 4),
   494→                  Text('-${diff.deniedByOverrides.length} permissions denied',
   495→                      style: theme.textTheme.bodySmall?.copyWith(
   496→                          color: AppColors.error)),
   497→                ],
   498→              ],
   499→            ),
   500→          ),
   501→        ),
   502→
   503→        const SizedBox(height: 16),
   504→
   505→        // Edit overrides button
   506→        KlasivoPermissionGate(
   507→          permission: Permission.userAssignRole,
   508→          fallback: const SizedBox.shrink(),
   509→          child: KlasivoButton(
   510→            label: 'Edit Permission Overrides',
   511→            icon: Icons.tune_rounded,
   512→            variant: KlasivoButtonVariant.secondary,
   513→            fullWidth: true,
   514→            onPressed: () {
   515→              Navigator.of(context).push(
   516→                MaterialPageRoute(
   517→                  builder: (_) => PermissionOverrideScreen(
   518→                    userId: user.id,
   519→                    userName: user.fullName,
   520→                    currentRole: user.role,
   521→                    currentOverrides: overrides,
   522→                  ),
   523→                ),
   524→              );
   525→            },
   526→          ),
   527→        ),
   528→
   529→        const SizedBox(height: 16),

   531→        // List of current overrides
   532→        if (overrides.isEmpty)
   533→          KlasivoEmptyState(
   534→            icon: Icons.tune_rounded,
   535→            title: 'No overrides',
   536→            subtitle: 'This user uses their role\'s default permissions',
   537→          )
   538→        else
   539→          ...overrides.entries.map((entry) => _OverrideTile(
   540→                permission: entry.key,
   541→                isGranted: entry.value,
   542→              )),
   543→      ],
   544→    );
   545→  }
   546→}

   570→class _AuditTab extends ConsumerWidget {
   571→  final String userId;
   572→
   573→  const _AuditTab({required this.userId});
   574→
   575→  @override
   576→  Widget build(BuildContext context, WidgetRef ref) {
   577→    final theme = Theme.of(context);
   578→    final isDark = theme.brightness == Brightness.dark;
   579→    final auditAsync = ref.watch(userAuditHistoryProvider(userId));
   580→
   581→    return auditAsync.when(
   582→      loading: () => const Center(child: KlasivoLoading()),
   583→      error: (e, _) => Center(child: Text('Error: $e')),
   584→      data: (logs) {
   585→        if (logs.isEmpty) {
   586→          return const KlasivoEmptyState(
   587→            icon: Icons.history_rounded,
   588→            title: 'No audit history',
   589→            subtitle: 'RBAC changes for this user will appear here',
   590→          );
   591→        }
   592→
   593→        return ListView.separated(
   594→          padding: const EdgeInsets.all(16),
   595→          itemCount: logs.length,
   596→          separatorBuilder: (_, __) => const SizedBox(height: 4),
   597→          itemBuilder: (context, index) {
   598→            final log = logs[index];
   599→            final action = log['action'] as String? ?? '';
   600→            final targetType = log['targetType'] as String? ?? '';
   601→            final details = log['details'] as String? ?? '';
   602→            final userName = log['userName'] as String? ?? 'System';
   603→            final timestamp = log['timestamp'];
   604→
   605→            final timeText = timestamp != null
   606→                ? timeago.format((timestamp as dynamic).toDate() as DateTime)
   607→                : '';
   608→
   609→            return KlasivoCard(
   610→              child: Padding(
   611→                padding:
   612→                    const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
   613→                child: Row(
   614→                  children: [
   615→                    _AuditActionIcon(action: action),
   616→                    const SizedBox(width: 12),
   617→                    Expanded(
   618→                      child: Column(
   619→                        crossAxisAlignment: CrossAxisAlignment.start,
   620→                        children: [
   621→                          Text(
   622→                            _auditDescription(action, targetType, details),
   623→                            style: theme.textTheme.bodyMedium,
   624→                          ),
   625→                          const SizedBox(height: 2),
   626→                          Text(
   627→                            'By $userName · $timeText',
   628→                            style: theme.textTheme.bodySmall?.copyWith(
   629→                              color: isDark
   630→                                  ? AppColors.lightTextTertiary
   631→                                  : AppColors.darkTextTertiary,
   632→                            ),
   633→                          ),
   634→                        ],
   635→                      ),
   636→                    ),
   637→                  ],

   836→// READ-ONLY SCOPE TREE — User Detail view
   837→// ═══════════════════════════════════════════════════════════════════════════════
   838→
   839→/// Read-only hierarchical tree showing the user's scope assignments.
   840→/// Highlights assigned nodes within the full org structure:
   841→///
   842→///   Main Campus ✓
   843→///   └─ Primary ✓
   844→///      ├─ 5A ✓
   845→///      ├─ 5B
   846→///      └─ 5C ✓
   847→class _ReadOnlyScopeTree extends ConsumerWidget {
   848→  final UserListItem user;
   849→
   850→  const _ReadOnlyScopeTree({required this.user});
   851→
   852→  @override
   853→  Widget build(BuildContext context, WidgetRef ref) {
   854→    final treeAsync = ref.watch(scopeTreeProvider);
   855→
   856→    return treeAsync.when(
   857→      loading: () => const Center(child: KlasivoLoading()),
   858→      error: (e, _) => Center(child: Text('Error loading tree: $e')),
   859→      data: (tree) {
   860→        if (tree.isEmpty) {
   861→          return const KlasivoEmptyState(
   862→            icon: Icons.account_tree_rounded,
   863→            title: 'No organizational structure',
   864→            subtitle: 'Create stages and classes first',
   865→          );
   866→        }
   867→
   868→        final assignedCampusIds = user.campusIds.toSet();
   869→        final assignedStageIds = user.stageIds.toSet();
   870→        final assignedClassIds = user.classIds.toSet();
   871→
   872→        // The tree may have campuses as roots, or stages as roots
   873→        // (when no campuses exist in the organization).
   874→        final hasCampusRoots = tree.isNotEmpty && tree.first.type == 'campus';
   875→
   876→        if (hasCampusRoots) {
   877→          // Filter to only campuses that have relevant assignments
   878→          final relevantCampuses = tree.where((campus) {
   879→            final isCampusAssigned = assignedCampusIds.contains(campus.id);
   880→            final hasAssignedChildren = campus.children.any(
   881→              (stage) =>
   882→                  assignedStageIds.contains(stage.id) ||
   883→                  stage.children
   884→                      .any((cls) => assignedClassIds.contains(cls.id)),
   885→            );
   886→            return isCampusAssigned || hasAssignedChildren;
   887→          }).toList();
   888→
   889→          if (relevantCampuses.isEmpty) {
   890→            return const SizedBox.shrink();
   891→          }
   892→
   893→          return Column(
   894→            crossAxisAlignment: CrossAxisAlignment.start,
   895→            children: relevantCampuses
   896→                .map((campus) => _ReadOnlyCampusNode(
   897→                      campus: campus,
   898→                      assignedCampusIds: assignedCampusIds,
   899→                      assignedStageIds: assignedStageIds,
   900→                      assignedClassIds: assignedClassIds,
   901→                    ))
   902→                .toList(),
   903→          );
   904→        }
   905→
   906→        // No campuses — stages are root nodes
   907→        final relevantStages = tree.where((stage) {
   908→          return assignedStageIds.contains(stage.id) ||
   909→              stage.children
   910→                  .any((cls) => assignedClassIds.contains(cls.id));
   911→        }).toList();
   912→
   913→        if (relevantStages.isEmpty) {
   914→          return const SizedBox.shrink();
   915→        }
   916→
   917→        return Column(
   918→          crossAxisAlignment: CrossAxisAlignment.start,
   919→          children: relevantStages
   920→              .asMap()
   921→              .entries
   922→              .map((entry) => _ReadOnlyStageNode(
   923→                    stage: entry.value,
   924→                    assignedStageIds: assignedStageIds,
   925→                    assignedClassIds: assignedClassIds,
   926→                    isLastStage: entry.key == relevantStages.length - 1,
   927→                  ))
   928→              .toList(),
   929→        );
   930→      },
   931→    );
   932→  }
   933→}
