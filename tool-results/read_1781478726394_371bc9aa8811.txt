    21→// ─── Auth Change Notifier for GoRouter refresh ──────────────────────────────
    22→
    23→class AuthChangeNotifier extends ChangeNotifier {
    24→  StreamSubscription<User?>? _firebaseSub;
    25→  StreamSubscription? _hiveSub;
    26→
    27→  AuthChangeNotifier() {
    28→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
    29→      (_) => notifyListeners(),
    30→      onError: (_) => notifyListeners(),
    31→    );
    32→    _startHiveWatch();
    33→  }
    34→
    35→  void _startHiveWatch() {
    36→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
    37→      notifyListeners();
    38→    });
    39→  }
    40→
    41→  @override
    42→  void dispose() {
    43→    _firebaseSub?.cancel();
    44→    _hiveSub?.cancel();
    45→    super.dispose();
    46→  }
    47→}
    48→
    49→final authChangeNotifierProvider = Provider<AuthChangeNotifier>((ref) {
    50→  final notifier = AuthChangeNotifier();
    51→  ref.onDispose(() => notifier.dispose());
    52→  return notifier;
    53→});

    55→// ─── GoRouter Provider ──────────────────────────────────────────────────────
    56→
    57→final klasivoRouterProvider = Provider<GoRouter>((ref) {
    58→  final notifier = ref.watch(authChangeNotifierProvider);
    59→
    60→  return GoRouter(
    61→    initialLocation: '/',
    62→    debugLogDiagnostics: true,
    63→    refreshListenable: notifier,
    64→    redirect: (context, state) {
    65→      final box = Hive.box(AppConstants.authBox);
    66→      final isLoggedIn = box.get('isLoggedIn', defaultValue: false) as bool;
    67→      final userRole = box.get('userRole', defaultValue: '') as String;
    68→      final hasCompletedSetup = box.get('hasCompletedSetup', defaultValue: true) as bool;
    69→
    70→      final isOnSplash = state.matchedLocation == '/';
    71→      final isOnAuth = state.matchedLocation.startsWith('/auth');
    72→      final isOnWelcome = state.matchedLocation == '/welcome';
    73→      final isOnDashboard = state.matchedLocation.startsWith('/dashboard') ||
    74→          state.matchedLocation.startsWith('/academic') ||
    75→          state.matchedLocation.startsWith('/people') ||
    76→          state.matchedLocation.startsWith('/inbox') ||
    77→          state.matchedLocation.startsWith('/settings') ||
    78→          state.matchedLocation.startsWith('/teacher') ||
    79→          state.matchedLocation.startsWith('/lms');
    80→      final isOnStudent = state.matchedLocation.startsWith('/student');
    81→      final isOnParent = state.matchedLocation.startsWith('/parent') ||
    82→          state.matchedLocation.startsWith('/auth/parent');
    83→
    84→      // Splash → redirect based on auth state
    85→      if (isOnSplash) {
    86→        if (isLoggedIn && userRole.isNotEmpty) {
    87→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
    88→            return '/welcome';
    89→          }
    90→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
    91→            return '/dashboard';
    92→          }
    93→          if (userRole == AppConstants.roleStudent) return '/student';
    94→          if (userRole == AppConstants.roleParent) return '/parent';
    95→        }
    96→        return '/auth';
    97→      }
    98→
    99→      // Welcome screen — only accessible to logged-in owners who haven't completed setup
   100→      if (isOnWelcome) {
   101→        if (!isLoggedIn) return '/auth';
   102→        if (userRole != AppConstants.roleOwner) return '/dashboard';
   103→        if (hasCompletedSetup) return '/dashboard';
   104→        return null;
   105→      }
   106→
   107→      // Auth screens → redirect if already logged in
   108→      if (isOnAuth) {
   109→        if (isLoggedIn && userRole.isNotEmpty) {
   110→          if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   111→            return '/welcome';
   112→          }
   113→          if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   114→            return '/dashboard';
   115→          }
   116→          if (userRole == AppConstants.roleStudent) return '/student';
   117→          if (userRole == AppConstants.roleParent) return '/parent';
   118→        }
   119→        return null;
   120→      }
   121→
   122→      // Protected screens → require login
   123→      if (isOnDashboard || isOnStudent || isOnParent) {
   124→        if (!isLoggedIn) return '/auth';
   125→        if (userRole.isEmpty) return '/auth';
   126→
   127→        if (userRole == AppConstants.roleOwner && !hasCompletedSetup) {
   128→          return '/welcome';
   129→        }
   130→
   131→        // ─── Feature Flag Gates ───────────────────────────────────────────
   132→        // LMS routes require lms flag
   133→        if (state.matchedLocation.startsWith('/lms')) {
   134→          final flagService = ref.read(featureFlagServiceProvider);
   135→          if (!flagService.isEnabled(FeatureFlags.lms)) {
   136→            return '/dashboard';
   137→          }
   138→        }
   139→
   140→        // LiveKit routes require livekit flag
   141→        if (state.matchedLocation.startsWith('/live-classes')) {
   142→          final flagService = ref.read(featureFlagServiceProvider);
   143→          if (!flagService.isEnabled(FeatureFlags.livekit)) {
   144→            return '/dashboard';
   145→          }
   146→        }
   147→
   148→        // People Hub requires userManagement flag
   149→        if (state.matchedLocation.startsWith('/people')) {
   150→          final flagService = ref.read(featureFlagServiceProvider);
   151→          if (!flagService.isEnabled(FeatureFlags.userManagement)) {
   152→            return '/dashboard';
   153→          }
   154→        }
   155→
   156→        // Parent portal requires parentPortal flag
   157→        if (state.matchedLocation.startsWith('/parent') ||
   158→            state.matchedLocation.startsWith('/auth/parent')) {
   159→          final flagService = ref.read(featureFlagServiceProvider);
   160→          if (!flagService.isEnabled(FeatureFlags.parentPortal)) {
   161→            if (userRole == AppConstants.roleParent) return '/auth';
   162→            return '/dashboard';
   163→          }
   164→        }
   165→
   166→        if ((userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) &&
   167→            isOnDashboard) {
   168→          return null;
   169→        }
   170→        if (userRole == AppConstants.roleStudent && isOnStudent) {
   171→          return null;
   172→        }
   173→        if (userRole == AppConstants.roleParent && isOnParent) {
   174→          return null;
   175→        }
   176→
   177→        if (userRole == AppConstants.roleTeacher || userRole == AppConstants.roleOwner) {
   178→          return '/dashboard';
   179→        }
   180→        if (userRole == AppConstants.roleStudent) return '/student';
   181→        if (userRole == AppConstants.roleParent) return '/parent';
   182→      }
   183→
   184→      return null;
   185→    },

   755→// ─── LiveKit Wrapper Screens (Sprint 1+2) ───────────────────────────────────
   756→// These wrappers read runtime context from providers and pass to the
   757→// actual LiveKit screens, which require orgId/userId/displayName.
   758→
   759→class _LiveClassesWrapper extends ConsumerWidget {
   760→  const _LiveClassesWrapper();
   761→
   762→  @override
   763→  Widget build(BuildContext context, WidgetRef ref) {
   764→    final box = Hive.box(AppConstants.authBox);
   765→    final orgId = box.get('organizationId', defaultValue: '') as String;
   766→    final userId = box.get('userId', defaultValue: '') as String;
   767→    final displayName = box.get('displayName', defaultValue: '') as String;
   768→    final role = box.get('userRole', defaultValue: '') as String;
   769→    final isTeacher = role == AppConstants.roleTeacher || role == AppConstants.roleOwner;
   770→
   771→    return ScheduledClassesScreen(
   772→      orgId: orgId,
   773→      userId: userId,
   774→      displayName: displayName,
   775→      isTeacher: isTeacher,
   776→    );
   777→  }
   778→}
