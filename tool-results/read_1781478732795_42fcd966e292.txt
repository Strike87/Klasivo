    23→    23→class AuthChangeNotifier extends ChangeNotifier {
    24→    24→  StreamSubscription<User?>? _firebaseSub;
    25→    25→  StreamSubscription? _hiveSub;
    26→    26→
    27→    27→  AuthChangeNotifier() {
    28→    28→    _firebaseSub = FirebaseAuth.instance.authStateChanges().listen(
    29→    29→      (_) => notifyListeners(),
    30→    30→      onError: (_) => notifyListeners(),
    31→    31→    );
    32→    32→    _startHiveWatch();
    33→    33→  }

    35→    35→  void _startHiveWatch() {
    36→    36→    _hiveSub = Hive.box(AppConstants.authBox).watch().listen((event) {
    37→    37→      notifyListeners();
    38→    38→    });
    39→    39→  }
    40→    40→
    41→    41→  @override
    42→    42→  void dispose() {
    43→    43→    _firebaseSub?.cancel();
    44→    44→    _hiveSub?.cancel();
    45→    45→    super.dispose();
    46→    46→  }
    47→    47→}

    50→    50→  final notifier = AuthChangeNotifier();

    52→    52→  return notifier;

   756→   756→// These wrappers read runtime context from providers and pass to the
   757→   757→// actual LiveKit screens, which require orgId/userId/displayName.
   758→   758→
   759→   759→class _LiveClassesWrapper extends ConsumerWidget {
   760→   760→  const _LiveClassesWrapper();
   761→   761→
   762→   762→  @override
   763→   763→  Widget build(BuildContext context, WidgetRef ref) {
   764→   764→    final box = Hive.box(AppConstants.authBox);
   765→   765→    final orgId = box.get('organizationId', defaultValue: '') as String;
   766→   766→    final userId = box.get('userId', defaultValue: '') as String;
   767→   767→    final displayName = box.get('displayName', defaultValue: '') as String;
   768→   768→    final role = box.get('userRole', defaultValue: '') as String;
   769→   769→    final isTeacher = role == AppConstants.roleTeacher || role == AppConstants.roleOwner;
   770→   770→
   771→   771→    return ScheduledClassesScreen(
   772→   772→      orgId: orgId,
   773→   773→      userId: userId,
   774→   774→      displayName: displayName,
   775→   775→      isTeacher: isTeacher,
   776→   776→    );
   777→   777→  }
   778→   778→}
