    27→    27→// ─── Sensitive Field Sanitization ─────────────────────────────────────────────
    28→    28→
    29→    29→/// Keys that must NEVER be sent to Sentry in any breadcrumb, tag, or extra.
    30→    30→/// Covers passwords, tokens, OTP codes, invite codes, and secrets.
    31→    31→class _SensitiveFields {
    32→    32→  static const Set<String> keys = {
    33→    33→    'password',
    34→    34→    'passwordHash',
    35→    35→    'confirmPassword',
    36→    36→    'newPassword',
    37→    37→    'accessToken',
    38→    38→    'refreshToken',
    39→    39→    'idToken',
    40→    40→    'otp',
    41→    41→    'otpCode',
    42→    42→    'inviteCode',
    43→    43→    'secret',
    44→    44→    'apiKey',
    45→    45→    'authEmail', // Internal student auth email — PII
    46→    46→    'studentCode', // Sensitive — could be used to log in
    47→    47→  };
    48→    48→
    49→    49→  /// Returns true if the key contains any sensitive substring.
    50→    50→  static bool isSensitive(String key) {
    51→    51→    final lower = key.toLowerCase();
    52→    52→    for (final s in keys) {
    53→    53→      if (lower.contains(s.toLowerCase())) return true;
    54→    54→    }
    55→    55→    return false;
    56→    56→  }
    57→    57→
    58→    58→  /// Sanitize a map — replace sensitive values with '[REDACTED]'.
    59→    59→  static Map<String, dynamic> sanitize(Map<String, dynamic> data) {
    60→    60→    return data.map((key, value) {
    61→    61→      if (isSensitive(key)) {
    62→    62→        return MapEntry(key, '[REDACTED]');
    63→    63→      }
    64→    64→      // Recursively sanitize nested maps
    65→    65→      if (value is Map<String, dynamic>) {
    66→    66→        return MapEntry(key, sanitize(value));
    67→    67→      }
    68→    68→      return MapEntry(key, value);
    69→    69→    });
    70→    70→  }
    71→    71→
    72→    72→  /// Sanitize a string by removing patterns that look like tokens/secrets.
    73→    73→  static String sanitizeString(String input) {
    74→    74→    // Redact Bearer tokens
    75→    75→    var result = input.replaceAll(
    76→    76→      RegExp(r'Bearer\s+[A-Za-z0-9\-._~+/]+=*'),
    77→    77→      'Bearer [REDACTED]',
    78→    78→    );
    79→    79→    // Redact long hex strings (likely tokens)
    80→    80→    result = result.replaceAll(
    81→    81→      RegExp(r'\b[A-Fa-f0-9]{32,}\b'),
    82→    82→      '[REDACTED_TOKEN]',
    83→    83→    );
    84→    84→    return result;
    85→    85→  }
    86→    86→}

   106→   106→/// Fluent API for creating standardized Sentry breadcrumbs.
   107→   107→///
   108→   108→/// Example:
   109→   109→///   KlasivoSentry.breadcrumb.auth('login_started', data: {'email': email});

   111→   111→class SentryBreadcrumbBuilder {
   112→   112→  /// Auth breadcrumb (login, logout, signup, password reset)
   113→   113→  void auth(String message, {Map<String, dynamic>? data}) {
   114→   114→    Sentry.addBreadcrumb(Breadcrumb(
   115→   115→      category: SentryCategories.auth,
   116→   116→      message: message,
   117→   117→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   118→   118→      level: SentryLevel.info,
   119→   119→    ));
   120→   120→  }

   122→   122→  /// Registration breadcrumb (account creation steps)

   124→   124→    Sentry.addBreadcrumb(Breadcrumb(
   125→   125→      category: SentryCategories.registration,
   126→   126→      message: message,
   127→   127→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   128→   128→      level: SentryLevel.info,

   254→   254→// ─── Firestore Observability Helper ──────────────────────────────────────────

   256→   256→/// Wraps Firestore operations with automatic Sentry breadcrumbs, exception
   257→   257→/// capture, and scope tags for every critical write.
   258→   258→///
   259→   259→/// Every Firestore exception will:
   260→   260→///   1. Reach Sentry
   261→   261→///   2. Include collection name
   262→   262→///   3. Include document id
   263→   263→///   4. Include user role
   264→   264→///   5. Include organization id
   265→   265→class SentryFirestoreHelper {
   266→   266→  static final FirebaseFirestore _db = FirebaseFirestore.instance;
   267→   267→
   268→   268→  /// Wrapped `.set()` with full observability.
   269→   269→  static Future<void> docSet({
   270→   270→    required String collection,
   271→   271→    required String docId,
   272→   272→    required Map<String, dynamic> data,
   273→   273→    String? flow,
   274→   274→    String? step,
   275→   275→    SetOptions? options,
   276→   276→  }) async {
   277→   277→    final sanitizedData = _SensitiveFields.sanitize(data);
   278→   278→    KlasivoSentry.breadcrumb.firestore(
   279→   279→      'create',
   280→   280→      collection: collection,
   281→   281→      docId: docId,
   282→   282→      data: {'flow': flow, 'step': step},
   283→   283→    );
   284→   284→
   285→   285→    try {
   286→   286→      await _db.collection(collection).doc(docId).set(data, options);
   287→   287→      KlasivoSentry.breadcrumb.firestore(
   288→   288→        'create_success',
   289→   289→        collection: collection,
   290→   290→        docId: docId,
   291→   291→      );
   292→   292→    } catch (e, st) {
   293→   293→      await _captureFirestoreException(
   294→   294→        e,
   295→   295→        st,
   296→   296→        collection: collection,
   297→   297→        operation: 'set',
   298→   298→        docId: docId,
   299→   299→        flow: flow,
   300→   300→        step: step,
   301→   301→        dataPreview: sanitizedData,
   302→   302→      );
   303→   303→      rethrow;
   304→   304→    }
   305→   305→  }

   452→   452→  /// Capture a Firestore exception with full context tags.
   453→   453→  static Future<void> _captureFirestoreException(
   454→   454→    Object exception,
   455→   455→    StackTrace stackTrace, {
   456→   456→    required String collection,
   457→   457→    required String operation,
   458→   458→    String? docId,
   459→   459→    String? flow,
   460→   460→    String? step,
   461→   461→    Map<String, dynamic>? dataPreview,
   462→   462→  }) async {
   463→   463→    // Determine if this is a permission-denied error
   464→   464→    String? errorCode;
   465→   465→    if (exception is FirebaseException) {
   466→   466→      errorCode = exception.code;
   467→   467→    }
   468→   468→
   469→   469→    await Sentry.captureException(
   470→   470→      exception,
   471→   471→      stackTrace: stackTrace,
   472→   472→      withScope: (scope) {
   473→   473→        scope.setTag('collection', collection);
   474→   474→        scope.setTag('operation', operation);
   475→   475→        if (docId != null) scope.setTag('documentId', docId);
   476→   476→        if (flow != null) scope.setTag('flow', flow);
   477→   477→        if (step != null) scope.setTag('step', step);
   478→   478→        if (errorCode != null) scope.setTag('firebase_code', errorCode);
   479→   479→
   480→   480→        // Attach user role and org from current Sentry context
   481→   481→        final currentUser = FirebaseAuth.instance.currentUser;
   482→   482→        if (currentUser != null) {
   483→   483→          scope.setTag('auth_uid', currentUser.uid);
   484→   484→        }
   485→   485→
   486→   486→        if (dataPreview != null && dataPreview.isNotEmpty) {
   487→   487→          scope.setExtra('data_preview', dataPreview);
   488→   488→        }
   489→   489→      },
   490→   490→    );
   491→   491→  }
   492→   492→}

   494→   494→// ─── Guarded Async Operation ─────────────────────────────────────────────────
   495→   495→
   496→   496→/// Runs an async operation with automatic Sentry breadcrumb, exception capture,
   497→   497→/// and metadata attachment. This is the recommended way to wrap any async
   498→   498→/// operation that should never fail silently.
   499→   499→///
   500→   500→/// Example:
   501→   501→///   final result = await KlasivoSentry.runGuarded(
   502→   502→///     'create_exam',
   503→   503→///     () => examService.createExam(...),
   504→   504→///     category: 'firestore',
   505→   505→///   );
   506→   506→class KlasivoSentryGuard {
   507→   507→  /// Execute [operation] inside a Sentry-guarded wrapper.
   508→   508→  ///
   509→   509→  /// - Automatically adds a breadcrumb before execution.
   510→   510→  /// - On success: adds a success breadcrumb.
   511→   511→  /// - On failure: captures the exception with scope tags and rethrows.
   512→   512→  /// - The [category] defaults to the operation name but can be overridden.
   513→   513→  /// - Additional [tags] and [data] are attached to both the breadcrumb
   514→   514→  ///   and the exception scope.
   515→   515→  static Future<T> runGuarded<T>(
   516→   516→    String operationName,
   517→   517→    Future<T> Function() operation, {
   518→   518→    String? category,
   519→   519→    Map<String, String>? tags,
   520→   520→    Map<String, dynamic>? data,
   521→   521→  }) async {
   522→   522→    final cat = category ?? 'operation';
   523→   523→
   524→   524→    Sentry.addBreadcrumb(Breadcrumb(
   525→   525→      category: cat,
   526→   526→      message: '${operationName}_started',
   527→   527→      data: data != null ? _SensitiveFields.sanitize(data) : null,
   528→   528→      level: SentryLevel.info,
   529→   529→    ));
   530→   530→
   531→   531→    try {
   532→   532→      final result = await operation();
   533→   533→
   534→   534→      Sentry.addBreadcrumb(Breadcrumb(
   535→   535→        category: cat,
   536→   536→        message: '${operationName}_success',
   537→   537→        data: data != null ? _SensitiveFields.sanitize(data) : null,
   538→   538→        level: SentryLevel.info,
   539→   539→      ));
   540→   540→
   541→   541→      return result;
   542→   542→    } catch (e, st) {
   543→   543→      Sentry.addBreadcrumb(Breadcrumb(
   544→   544→        category: cat,
   545→   545→        message: '${operationName}_failed',
   546→   546→        data: _SensitiveFields.sanitize({
   547→   547→          'error': e.toString().substring(0, (e.toString().length).clamp(0, 200)),
   548→   548→          ...?data,
   549→   549→        }),
   550→   550→        level: SentryLevel.error,
   551→   551→      ));
   552→   552→
   553→   553→      await Sentry.captureException(
   554→   554→        e,
   555→   555→        stackTrace: st,
   556→   556→        withScope: (scope) {
   557→   557→          scope.setTag('operation', operationName);
   558→   558→          tags?.forEach(scope.setTag);
   559→   559→          if (data != null) {
   560→   560→            scope.setExtra('operation_data', _SensitiveFields.sanitize(data));
   561→   561→          }
   562→   562→        },
   563→   563→      );
   564→   564→
   565→   565→      rethrow;
   566→   566→    }
   567→   567→  }
   568→   568→
   569→   569→  /// Execute [operation] with a Sentry transaction span.
   570→   570→  ///
   571→   571→  /// Creates a child span under the given [transaction], measures duration,
   572→   572→  /// and sets the span status based on success/failure.
   573→   573→  static Future<T> runWithSpan<T>(
   574→   574→    ISentrySpan transaction,
   575→   575→    String spanName,
   576→   576→    Future<T> Function() operation, {
   577→   577→    Map<String, String>? tags,
   578→   578→  }) async {
   579→   579→    final span = transaction.startChild(spanName);
   580→   580→
   581→   581→    try {
   582→   582→      final result = await operation();
   583→   583→      span.status = const SpanStatus.ok();
   584→   584→      return result;
   585→   585→    } catch (e, st) {
   586→   586→      span.status = const SpanStatus.internalError();
   587→   587→
   588→   588→      await Sentry.captureException(
   589→   589→        e,
   590→   590→        stackTrace: st,
   591→   591→        withScope: (scope) {
   592→   592→          scope.setTag('span', spanName);
   593→   593→          tags?.forEach(scope.setTag);
   594→   594→        },
   595→   595→      );
   596→   596→
   597→   597→      rethrow;
   598→   598→    } finally {
   599→   599→      await span.finish();
   600→   600→    }
   601→   601→  }
   602→   602→}

   749→   749→// ─── Doc ID Audit Trail ──────────────────────────────────────────────────────
   750→   750→
   751→   751→/// Documents the doc ID strategy for every user creation path.
   752→   752→/// This is critical for the ongoing investigation into missing users/{uid} docs.
   753→   753→///
   754→   754→/// | Path                  | Collection | Doc ID     | Status    |
   755→   755→/// |-----------------------|-----------|------------|-----------|
   756→   756→/// | Owner (email)         | users     | user.uid   | OK        |
   757→   757→/// | Owner (Google)        | users     | user.uid   | OK        |
   758→   758→/// | Teacher (invite)      | users     | user.uid   | OK        |
   759→   759→/// | Teacher (Google)      | users     | user.uid   | OK        |
   760→   760→/// | Parent (email)        | users     | user.uid   | OK        |
   761→   761→/// | Parent (Google)       | users     | user.uid   | OK        |
   762→   762→/// | Student (QR enroll)   | users     | auto-id    | BROKEN    |
   763→   763→/// | Student (Excel import)| users     | ?          | UNKNOWN   |
   764→   764→/// | onUserCreated trigger | users     | READ ONLY  | N/A       |
   765→   765→///
   766→   766→/// The QR enrollment path uses `.doc()` (auto-generated ID) instead of
   767→   767→/// `.doc(user.uid)`, which means:
   768→   768→///   1. The doc ID doesn't match the auth UID
   769→   769→///   2. Security rule `request.auth.uid == userId` ALWAYS FAILS
   770→   770→///   3. QR enrollment is 100% broken in production
   771→   771→class SentryDocIdAudit {
   772→   772→  /// Log a doc ID audit breadcrumb for any user creation path.
   773→   773→  static void logUserCreation({
   774→   774→    required String flow,
   775→   775→    required String collection,
   776→   776→    required String docIdStrategy, // 'uid' or 'auto_id'
   777→   777→    required String actualDocId,
   778→   778→    String? authUid,
   779→   779→  }) {
   780→   780→    final isMismatch = authUid != null && docIdStrategy == 'uid' && actualDocId != authUid;
   781→   781→    final isAutoId = docIdStrategy == 'auto_id';
   782→   782→
   783→   783→    Sentry.addBreadcrumb(Breadcrumb(
   784→   784→      category: 'doc_id_audit',
   785→   785→      message: 'user_doc_created',
   786→   786→      data: _SensitiveFields.sanitize({
   787→   787→        'flow': flow,
   788→   788→        'collection': collection,
   789→   789→        'docIdStrategy': docIdStrategy,
   790→   790→        'actualDocId': actualDocId,
   791→   791→        if (authUid != null) 'authUid': authUid,
   792→   792→        'idMismatch': isMismatch,
   793→   793→        'autoIdUsed': isAutoId,
   794→   794→        'brokenByRules': isAutoId, // auto-id always fails security rules
   795→   795→      }),
   796→   796→      level: (isMismatch || isAutoId) ? SentryLevel.warning : SentryLevel.info,
   797→   797→    ));
   798→   798→
   799→   799→    if (isMismatch || isAutoId) {
   800→   800→      Sentry.captureMessage(
   801→   801→        'Doc ID audit: $flow uses ${isAutoId ? 'auto-id' : 'mismatched uid'} '
   802→   802→        '(docId=$actualDocId, authUid=$authUid). '
   803→   803→        'This path is ${isAutoId ? "blocked by security rules" : "potentially broken"}.',
   804→   804→        level: SentryLevel.warning,
   805→   805→      );
   806→   806→    }
   807→   807→  }
   808→   808→}

   902→   902→// ─── Centralized Event Sanitizer ────────────────────────────────────────────
   903→   903→
   904→   904→/// Global Sentry event sanitizer used by the `beforeSend` callback in
   905→   905→/// `main.dart`. Replaces the duplicate `_SentrySanitizer` class that was
   906→   906→/// previously defined in main.dart.
   907→   907→///
   908→   908→/// Strips sensitive data from breadcrumbs, extras, and request headers.
   909→   909→class KlasivoSentrySanitizer {
   910→   910→  KlasivoSentrySanitizer._();
   911→   911→
   912→   912→  /// Sanitize a [SentryEvent] — returns the same event with sensitive data
   913→   913→  /// replaced by '[REDACTED]'.
   914→   914→  static SentryEvent sanitizeEvent(SentryEvent event) {
   915→   915→    // Strip sensitive data from breadcrumbs
   916→   916→    if (event.breadcrumbs != null) {
   917→   917→      for (int i = 0; i < event.breadcrumbs!.length; i++) {
   918→   918→        final crumb = event.breadcrumbs![i];
   919→   919→        if (crumb.data != null) {
   920→   920→          final sanitized = <String, dynamic>{};
   921→   921→          crumb.data!.forEach((key, value) {
   922→   922→            sanitized[key] = _SensitiveFields.isSensitive(key)
   923→   923→                ? '[REDACTED]'
   924→   924→                : value;
   925→   925→          });
   926→   926→          event.breadcrumbs![i] = Breadcrumb(
   927→   927→            category: crumb.category,
   928→   928→            message: crumb.message,
   929→   929→            data: sanitized,
   930→   930→            level: crumb.level,
   931→   931→            timestamp: crumb.timestamp,
   932→   932→            type: crumb.type,
   933→   933→          );
   934→   934→        }
   935→   935→      }
   936→   936→    }
   937→   937→
   938→   938→    // Strip sensitive data from extra fields
   939→   939→    if (event.extra != null) {
   940→   940→      final sanitized = <String, dynamic>{};
   941→   941→      event.extra!.forEach((key, value) {
   942→   942→        sanitized[key] = _SensitiveFields.isSensitive(key)
   943→   943→            ? '[REDACTED]'
   944→   944→            : value;
   945→   945→      });
   946→   946→      event.extra = sanitized;
   947→   947→    }
   948→   948→
   949→   949→    // Strip sensitive data from request headers
   950→   950→    if (event.request?.headers != null) {
   951→   951→      final sanitized = <String, String>{};
   952→   952→      event.request!.headers!.forEach((key, value) {
   953→   953→        final lower = key.toLowerCase();
   954→   954→        if (lower.contains('authorization') ||
   955→   955→            lower.contains('cookie') ||
   956→   956→            lower.contains('token')) {
   957→   957→          sanitized[key] = '[REDACTED]';
   958→   958→        } else {
   959→   959→          sanitized[key] = value;
   960→   960→        }
   961→   961→      });
   962→   962→      event.request = SentryRequest(headers: sanitized);
   963→   963→    }
   964→   964→
   965→   965→    return event;
   966→   966→  }
   967→   967→}
