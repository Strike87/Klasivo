     1→import 'roles.dart';
     2→
     3→/// A single permission definition in the Klasivo platform.
     4→///
     5→/// Each permission has a unique [id] (dot-separated, e.g. `exam.create`),
     6→/// a [displayName] for the UI, the [module] it belongs to, a [description],
     7→/// and a map of [defaultGrants] that specifies which roles are granted
     8→/// this permission by default.
     9→class Permission {
    10→  final String id;
    11→  final String displayName;
    12→  final String module;
    13→  final String description;
    14→  final Map<KlasivoRole, bool> defaultGrants;
    15→
    16→  const Permission({
    17→    required this.id,
    18→    required this.displayName,
    19→    required this.module,
    20→    required this.description,
    21→    required this.defaultGrants,
    22→  });
    23→
    24→  /// Whether this permission is granted to [role] by default.
    25→  ///
    26→  /// Returns `false` if the role is not in the [defaultGrants] map.
    27→  bool isGrantedByDefault(KlasivoRole role) => defaultGrants[role] ?? false;
    28→}
    29→
    30→/// The central permission engine for the Klasivo platform.
    31→///
    32→/// Defines all permissions organized by module and provides static methods
    33→/// to query permissions by role, module, or individual permission ID.
    34→///
    35→/// ## Hierarchical Role Inheritance
    36→///
    37→/// Roles are organized in a hierarchy from highest to lowest privilege.
    38→/// The [isSeniorRole] method determines hierarchy position, and
    39→/// [getAllPermissions] returns permissions with inherited grants.
    40→///
    41→/// Inheritance rules:
    42→///   - **academicManager** inherits all `teacher` permissions +
    43→///     stage/class management
    44→///   - **stageSupervisor** inherits all `teacher` permissions +
    45→///     stage-level management
    46→///   - **assistantTeacher** gets a subset of `teacher` permissions
    47→///     (no publish, no delete)
    48→///
    49→/// This engine works in concert with the existing [PermissionService] which
    50→/// handles Firestore overrides (explicit allow/deny). The evaluation order is:
    51→///   1. Check explicit deny (Firestore override)
    52→///   2. Check explicit allow (Firestore override)
    53→///   3. Check default grants from this engine (including inheritance)
    54→///   4. Deny by default
    55→class PermissionEngine {
    56→  // Private constructor — this class is not meant to be instantiated.
    57→  PermissionEngine._();
    58→
    59→  // ═══════════════════════════════════════════════════════════════════════════
    60→  // ROLE HIERARCHY
    61→  // ═══════════════════════════════════════════════════════════════════════════
    62→
    63→  /// The role hierarchy, ordered from highest to lowest privilege.
    64→  ///
    65→  /// A role at a lower index is "senior" to one at a higher index.
    66→  /// This is used for permission inheritance — senior roles implicitly
    67→  /// gain the permissions of junior roles below them in the hierarchy.
    68→  ///
    69→  /// Note: [observer] and [student]/[parent] are NOT in the inheritance
    70→  /// chain for teaching/management roles. They have their own scoped
    71→  /// permissions.
    72→  static const List<KlasivoRole> roleHierarchy = [
    73→    KlasivoRole.superAdmin,
    74→    KlasivoRole.tenantOwner,
    75→    KlasivoRole.owner,
    76→    KlasivoRole.admin,
    77→    KlasivoRole.academicManager,
    78→    KlasivoRole.campusManager,
    79→    KlasivoRole.stageSupervisor,
    80→    KlasivoRole.teacher,
    81→    KlasivoRole.assistantTeacher,
    82→    KlasivoRole.student,
    83→    KlasivoRole.parent,
    84→    KlasivoRole.observer,
    85→  ];
    86→
    87→  /// Check if [roleA] is senior to [roleB] (for permission inheritance).
    88→  ///
    89→  /// Returns `true` if [roleA] appears before [roleB] in [roleHierarchy],
    90→  /// meaning [roleA] has higher or equal privilege.
    91→  ///
    92→  /// Example:
    93→  /// ```dart
    94→  /// isSeniorRole(KlasivoRole.admin, KlasivoRole.teacher) // true
    95→  /// isSeniorRole(KlasivoRole.teacher, KlasivoRole.admin) // false
    96→  /// isSeniorRole(KlasivoRole.admin, KlasivoRole.admin)   // true (equal)
    97→  /// ```
    98→  static bool isSeniorRole(KlasivoRole roleA, KlasivoRole roleB) {
    99→    final indexA = roleHierarchy.indexOf(roleA);
   100→    final indexB = roleHierarchy.indexOf(roleB);
   101→    if (indexA == -1 || indexB == -1) return false;
   102→    return indexA <= indexB;
   103→  }
   104→
   105→  /// Check if [roleA] is strictly senior to [roleB] (not equal).
   106→  static bool isStrictlySeniorRole(KlasivoRole roleA, KlasivoRole roleB) {
   107→    final indexA = roleHierarchy.indexOf(roleA);
   108→    final indexB = roleHierarchy.indexOf(roleB);
   109→    if (indexA == -1 || indexB == -1) return false;
   110→    return indexA < indexB;
   111→  }
   112→
   113→  /// Get the direct permissions for a role (without inheritance).
   114→  ///
   115→  /// These are only the permissions explicitly granted in [allPermissions].
   116→  static List<String> getDirectPermissions(KlasivoRole role) {
   117→    return allPermissions
   118→        .where((p) => p.isGrantedByDefault(role))
   119→        .map((p) => p.id)
   120→        .toList();
   121→  }
   122→
   123→  /// Get all permissions for a role, including inherited ones.
   124→  ///
   125→  /// Inheritance rules:
   126→  ///   - **academicManager** inherits all `teacher` permissions +
   127→  ///     stage/class management permissions
   128→  ///   - **stageSupervisor** inherits all `teacher` permissions +
   129→  ///     stage-level permissions
   130→  ///   - **assistantTeacher** inherits a subset of `teacher` permissions
   131→  ///     (excludes publish, delete, and create actions)
   132→  ///
   133→  /// For roles above `teacher` in the hierarchy, all direct permissions
   134→  /// are returned (they already have more permissions than teachers).
   135→  static List<String> getAllPermissions(KlasivoRole role) {
   136→    final directPermissions = getDirectPermissions(role).toSet();
   137→
   138→    // ── Inheritance for academic_manager ─────────────────────────────────
   139→    // Inherits teacher permissions + gets stage/class management
   140→    if (role == KlasivoRole.academicManager) {
   141→      directPermissions.addAll(getDirectPermissions(KlasivoRole.teacher));
   142→      // Academic manager gets additional management permissions
   143→      directPermissions.addAll([
   144→        'stage.manage',
   145→        'stage.view',
   146→        'class.manage',
   147→        'class.view',
   148→        'subject.manage',
   149→        'subject.view',
   150→        'academic.manage',
   151→        'academic.view',
   152→        'curriculum.manage',
   153→      ]);
   154→    }
   155→
   156→    // ── Inheritance for stage_supervisor ─────────────────────────────────
   157→    // Inherits teacher permissions + gets stage-level permissions
   158→    if (role == KlasivoRole.stageSupervisor) {
   159→      directPermissions.addAll(getDirectPermissions(KlasivoRole.teacher));
   160→      // Stage supervisor gets stage-level management permissions
   161→      directPermissions.addAll([
   162→        'stage.manage',
   163→        'stage.view',
   164→        'class.view',
   165→        'class.manage_grades',
   166→        'subject.view',
   167→        'academic.view',
   168→      ]);
   169→    }
   170→
   171→    // ── Inheritance for assistant_teacher ────────────────────────────────
   172→    // Subset of teacher permissions — NO publish, NO delete, NO create
   173→    if (role == KlasivoRole.assistantTeacher) {
   174→      // Remove dangerous permissions that assistant teachers should NOT have
   175→      directPermissions.remove('exam.create');
   176→      directPermissions.remove('exam.publish');
   177→      directPermissions.remove('exam.delete');
   178→      directPermissions.remove('exam.edit');
   179→      directPermissions.remove('assignment.create');
   180→      directPermissions.remove('assignment.delete');
   181→      directPermissions.remove('assignment.edit');
   182→      directPermissions.remove('lms.manage');
   183→      directPermissions.remove('question.create');
   184→      directPermissions.remove('question.delete');
   185→      directPermissions.remove('material.upload');
   186→      directPermissions.remove('material.delete');
   187→      directPermissions.remove('lesson.create');
   188→      directPermissions.remove('lesson.delete');
   189→    }
   190→
   191→    // ── Inheritance for campus_manager ──────────────────────────────────
   192→    // Inherits teacher + academic_manager scoped to their campus
   193→    if (role == KlasivoRole.campusManager) {
   194→      directPermissions.addAll(getDirectPermissions(KlasivoRole.teacher));
   195→      directPermissions.addAll([
   196→        'stage.manage',
   197→        'stage.view',
   198→        'class.manage',
   199→        'class.view',
   200→        'subject.manage',
   201→        'subject.view',
   202→        'academic.view',
   203→        'campus.manage',
   204→        'campus.view',
   205→      ]);
   206→    }
   207→
   208→    return directPermissions.toList()..sort();
   209→  }
   210→
   211→  /// Get all permissions for a role using the string ID (from Firestore).
   212→  ///
   213→  /// Returns an empty list if the role ID is not recognized.
   214→  static List<String> getAllPermissionsByRoleId(String roleId) {
   215→    final role = KlasivoRole.fromId(roleId);
   216→    if (role == null) return [];
   217→    return getAllPermissions(role);
   218→  }
   219→
   220→  /// Check if a role has a specific permission, considering inheritance.
   221→  ///
   222→  /// This is the enhanced version that considers the full permission set
   223→  /// including inherited permissions.
   224→  static bool hasPermissionWithInheritance(
   225→    KlasivoRole role,
   226→    String permissionId,
   227→  ) {
   228→    return getAllPermissions(role).contains(permissionId);
   229→  }

  1273→  /// Check if a role has a specific permission by default.
  1274→  ///
  1275→  /// Throws [ArgumentError] if [permissionId] is not a known permission.
  1276→  static bool hasPermission(KlasivoRole role, String permissionId) {
  1277→    final permission = allPermissions.firstWhere(
  1278→      (p) => p.id == permissionId,
  1279→      orElse: () => throw ArgumentError('Unknown permission: $permissionId'),
  1280→    );
  1281→    return permission.isGrantedByDefault(role);
  1282→  }
  1283→
  1284→  /// Check if a role has a specific permission by default.
  1285→  ///
  1286→  /// Returns `false` if the permission ID is unknown (non-throwing version).
  1287→  static bool maybeHasPermission(KlasivoRole role, String permissionId) {
  1288→    final permission = _permissionById[permissionId];
  1289→    if (permission == null) return false;
  1290→    return permission.isGrantedByDefault(role);
  1291→  }
  1292→
  1293→  /// Get all permission IDs granted to a role by default (direct only).
  1294→  static List<String> getPermissionsForRole(KlasivoRole role) {
  1295→    return allPermissions
  1296→        .where((p) => p.isGrantedByDefault(role))
  1297→        .map((p) => p.id)
  1298→        .toList();
  1299→  }
  1300→
  1301→  /// Get all [Permission] objects granted to a role by default (direct only).
  1302→  static List<Permission> getPermissionObjectsForRole(KlasivoRole role) {
  1303→    return allPermissions.where((p) => p.isGrantedByDefault(role)).toList();
  1304→  }
  1305→
  1306→  /// Get all permissions for a specific module.
  1307→  static List<Permission> getPermissionsForModule(String module) {
  1308→    return allPermissions.where((p) => p.module == module).toList();
  1309→  }
  1310→
  1311→  /// Get all available module names, sorted alphabetically.
  1312→  static List<String> get modules =>
  1313→      allPermissions.map((p) => p.module).toSet().toList()..sort();
  1314→
  1315→  /// Look up a permission by its ID.
  1316→  ///
  1317→  /// Returns `null` if no permission with the given ID exists.
  1318→  static Permission? findById(String permissionId) {
  1319→    return _permissionById[permissionId];
  1320→  }

  1322→  /// Get a summary of a role's effective permissions including inheritance.
  1323→  ///
  1324→  /// Returns a map with:
  1325→  /// - `role`: The role name
  1326→  /// - `direct`: Directly granted permissions
  1327→  /// - `inherited`: Inherited permissions (from other roles)
  1328→  /// - `effective`: All effective permissions (direct + inherited)
  1329→  static Map<String, dynamic> getRolePermissionSummary(KlasivoRole role) {
  1330→    final direct = getPermissionsForRole(role);
  1331→    final effective = getAllPermissions(role);
  1332→    final inherited = effective.where((p) => !direct.contains(p)).toList();
  1333→
  1334→    return {
  1335→      'role': role.id,
  1336→      'directCount': direct.length,
  1337→      'inheritedCount': inherited.length,
  1338→      'effectiveCount': effective.length,
  1339→      'direct': direct,
  1340→      'inherited': inherited,
  1341→      'effective': effective,
  1342→    };
  1343→  }
